local function copy(value, ancestors)
	local valueType = typeof(value)
	if valueType == "Instance" then
		return value.Name, true
	elseif valueType == "EnumItem" then
		return value.Name, true
	elseif type(value) ~= "table" then
		if value == nil or valueType == "boolean" or valueType == "number" or valueType == "string" then
			return value, false
		end
		return nil, true
	end
	ancestors = ancestors or {}
	if ancestors[value] then
		return nil, true
	end
	ancestors[value] = true
	local result = {}
	local changed = false
	for key, item in pairs(value) do
		local keyType = typeof(key)
		if keyType ~= "string" and keyType ~= "number" then
			changed = true
		else
			local copied, itemChanged = copy(item, ancestors)
			if copied ~= nil then
				result[key] = copied
			elseif item ~= nil then
				changed = true
			end
			changed = changed or itemChanged
		end
	end
	ancestors[value] = nil
	return result, changed
end

local function readValue(option)
	if option.Type == "ColorPicker" then
		return { hex = option.Value:ToHex(), transparency = option.Transparency }
	end
	if option.Type == "KeyPicker" then
		return { key = option.Value, mode = option.Mode }
	end
	return option.Value
end

local function isConfigSnapshot(value)
	return type(value) == "table"
		and (type(value.toggles) == "table" or type(value.options) == "table" or type(value.layout) == "table")
end

function ConfigStore.new(library)
	local store = setmetatable({
		configs = {},
		autoloadByGame = {},
		activeGame = nil,
		legacyConfigs = {},
		legacyAutoloadName = nil,
		defaultTheme = nil,
		selectedGames = {},
		sideImages = nil,
		library = library,
	}, ConfigStore)
	if type(readfile) == "function" then
		local readSuccess, contents = pcall(readfile, CONFIG_PATH)
		if readSuccess and type(contents) == "string" then
			local decodeSuccess, data = pcall(HttpService.JSONDecode, HttpService, contents)
			if decodeSuccess and type(data) == "table" then
				store.defaultTheme = type(data.defaultTheme) == "string" and data.defaultTheme or nil
				store.selectedGames = type(data.selectedGames) == "table" and data.selectedGames or {}
				store.sideImages = type(data.sideImages) == "table" and data.sideImages or nil
				store.autoloadByGame = type(data.autoloadByGame) == "table" and data.autoloadByGame or {}
				store.legacyAutoloadName = type(data.autoloadName) == "string" and data.autoloadName or nil
				if type(data.legacyAutoloadName) == "string" then
					store.legacyAutoloadName = data.legacyAutoloadName
				end
				if type(data.legacyConfigs) == "table" then
					store.legacyConfigs = data.legacyConfigs
				end
				local savedConfigs = type(data.configs) == "table" and data.configs or {}
				if data.configSchemaVersion == 2 then
					for gameName, profiles in pairs(savedConfigs) do
						if type(gameName) == "string" and type(profiles) == "table" then
							local validProfiles = {}
							for profileName, snapshot in pairs(profiles) do
								if type(profileName) == "string" and isConfigSnapshot(snapshot) then
									validProfiles[profileName] = snapshot
								end
							end
							if next(validProfiles) then
								store.configs[gameName] = validProfiles
							end
						end
					end
				else
					for profileName, snapshot in pairs(savedConfigs) do
						if type(profileName) == "string" and isConfigSnapshot(snapshot) then
							store.legacyConfigs[profileName] = snapshot
						end
					end
				end
				store.activeGame = store.selectedGames[tostring(game.PlaceId)]
			end
		end
	end
	return store
end

function ConfigStore:Persist()
	if type(writefile) ~= "function" then
		return false, "Executor file APIs are unavailable; configs will only last this session."
	end
	if type(makefolder) == "function" then
		pcall(makefolder, CONFIG_DIRECTORY)
	end
	local payload, normalized = copy({
		configSchemaVersion = 2,
		configs = self.configs,
		autoloadByGame = self.autoloadByGame,
		defaultTheme = self.defaultTheme,
		selectedGames = self.selectedGames,
		sideImages = self.sideImages,
		legacyConfigs = self.legacyConfigs,
		legacyAutoloadName = self.legacyAutoloadName,
	})
	if normalized then
		warn("[YSL Method] Config data contained unsupported or cyclic values; affected values were omitted from the saved file.")
	end
	local encoded, contents = pcall(HttpService.JSONEncode, HttpService, payload)
	if not encoded then
		return false, "Could not serialize config data: " .. tostring(contents)
	end
	local written, writeError = pcall(writefile, CONFIG_PATH, contents)
	if not written then
		return false, tostring(writeError)
	end
	return true
end

function ConfigStore:GetSideImages()

function ConfigStore:SaveSideImages(state)
	local safeState, normalized = copy(state)
	if type(safeState) ~= "table" then
		return false, "Could not save mascot settings: invalid state."
	end
	if normalized then
		warn("[YSL Method] Mascot settings contained unsupported values; affected values were omitted.")
	end
	local previousState = self.sideImages
	self.sideImages = safeState
	local success, err = self:Persist()
	if not success then
		self.sideImages = previousState
		return false, err
	end
	return true
end

function ConfigStore:SetGameContext(gameName)
	if type(gameName) ~= "string" or gameName == "" then
		return false, "Select a game before managing configs."
	end
	self.activeGame = gameName
	local migrated = next(self.legacyConfigs) ~= nil or self.legacyAutoloadName ~= nil
	local gameProfiles = self.configs[gameName]
	if not gameProfiles then
		gameProfiles = {}
		self.configs[gameName] = gameProfiles
	end
	for profileName, snapshot in pairs(self.legacyConfigs) do
		if not gameProfiles[profileName] then
			gameProfiles[profileName] = snapshot
		end
		self.legacyConfigs[profileName] = nil
	end
	if self.legacyAutoloadName then
		if not self.autoloadByGame[gameName] and gameProfiles[self.legacyAutoloadName] then
			self.autoloadByGame[gameName] = self.legacyAutoloadName
		end
		self.legacyAutoloadName = nil
	end
	if next(gameProfiles) == nil then
		self.configs[gameName] = nil
	end
	if migrated then
		return self:Persist()
	end
	return true
end

local function getGameProfiles(store)
	if not store.activeGame then
		return nil
	end
	return store.configs[store.activeGame]
end

function ConfigStore:Save(name)
	local gameProfiles = getGameProfiles(self)
	if not gameProfiles then
		if not self.activeGame then
			return false, "Select a game before saving a config."
		end
		gameProfiles = {}
		self.configs[self.activeGame] = gameProfiles
	end
	name = string.gsub(name or "", "^%s*(.-)%s*$", "%1")
	if name == "" or not string.match(name, "^[%w _%-]+$") then
		return false, "Use letters, numbers, spaces, _ or - for config names."
	end
	local snapshot = {
		toggles = {},
		options = {},
		layout = self.library and self.library:GetLayoutSnapshot() or nil,
		sideImages = self.library and self.library.SideImageController
			and self.library.SideImageController:GetConfigState() or nil,
	}
	for id, toggle in pairs(aztup_toggles) do
		if toggle.Type == "Toggle" then
			snapshot.toggles[id] = toggle.Value
		end
	end
	for id, option in pairs(aztup_options) do
		if option.Type then
			local value, normalized = copy(readValue(option))
			if value ~= nil then
				snapshot.options[id] = { type = option.Type, value = value }
			end
			if normalized then
				warn(string.format(
					"[YSL Method] Config option %s contained unsupported or cyclic values; unsupported branches were omitted.",
					tostring(id)
				))
			end
		end
	end
	local safeSnapshot, normalized = copy(snapshot)
	if normalized then
		warn("[YSL Method] Config snapshot contained unsupported or cyclic values; affected values were omitted.")
	end
	local encoded, encodeError = pcall(HttpService.JSONEncode, HttpService, safeSnapshot)
	if not encoded then
		return false, "Could not serialize config snapshot: " .. tostring(encodeError)
	end
	local previousSnapshot = gameProfiles[name]
	gameProfiles[name] = safeSnapshot
	local success, err = self:Persist()
	if not success then
		gameProfiles[name] = previousSnapshot
		if next(gameProfiles) == nil then
			self.configs[self.activeGame] = nil
		end
	end
	return success, err
end

function ConfigStore:Load(name)
	local gameProfiles = getGameProfiles(self)
	local snapshot = gameProfiles and gameProfiles[name]
	if not snapshot then
		return false, "Config not found."
	end
	local loadErrors = {}
	if self.library then
		self.library.ConfigLoading = true
		self.library.ConfigLoadingOption = nil
		self.library.ConfigLoadErrors = loadErrors
	end
	local function apply(label, callback)
		if self.library then
			self.library.ConfigLoadingOption = label
		end
		local success, err = pcall(callback)
		if not success then
			table.insert(loadErrors, string.format("%s: %s", tostring(label), tostring(err)))
			warn(string.format("[YSL Method] Could not load config value %s: %s", tostring(label), tostring(err)))
		end
	end
	for id, value in pairs(snapshot.toggles or {}) do
		local toggle = aztup_toggles[id]
		if toggle then
			apply(id, function()
				toggle:SetValue(value)
			end)
		end
	end
	if snapshot.sideImages and self.library and self.library.SideImageController then
		apply("side images", function()
			self.library.SideImageController:SetConfigState(snapshot.sideImages)
		end)
	end
	for id, item in pairs(snapshot.options or {}) do
		local option = aztup_options[id]
		if option and type(item) == "table" then
			apply(id, function()
				if item.type == "ColorPicker" then
					assert(type(item.value) == "table" and type(item.value.hex) == "string", "invalid color value")
					option:SetValueRGB(Color3.fromHex(item.value.hex), item.value.transparency)
				elseif item.type == "KeyPicker" then
					assert(type(item.value) == "table", "invalid keybind value")
					option:SetValue({ item.value.key, item.value.mode })
				elseif option.SetValue then
					option:SetValue(item.value)
				end
			end)
		end
	end
	if snapshot.layout and self.library and self.library.SetLayoutSnapshot then
		apply("layout", function()
			self.library:SetLayoutSnapshot(snapshot.layout)
		end)
	end
	if self.library then
		self.library.ConfigLoading = false
		self.library.ConfigLoadingOption = nil
		self.library.ConfigLoadErrors = nil
	end
	if #loadErrors > 0 then
		return false, string.format(
			"Loaded config with %d value error(s). See Output for the affected settings.",
			#loadErrors
		)
	end
	return true
end

function ConfigStore:Delete(name)
	local gameProfiles = getGameProfiles(self)
	if not gameProfiles or not gameProfiles[name] then
		return false
	end
	local previousSnapshot = gameProfiles[name]
	local previousAutoload = self.activeGame and self.autoloadByGame[self.activeGame]
	gameProfiles[name] = nil
	if next(gameProfiles) == nil then
		self.configs[self.activeGame] = nil
	end
	if self.activeGame and self.autoloadByGame[self.activeGame] == name then
		self.autoloadByGame[self.activeGame] = nil
	end
	local success, err = self:Persist()
	if not success then
		self.configs[self.activeGame] = gameProfiles
		gameProfiles[name] = previousSnapshot
		self.autoloadByGame[self.activeGame] = previousAutoload
	end
	return success, err
end

function ConfigStore:SetAutoLoad(name)
	local gameProfiles = getGameProfiles(self)
	if not self.activeGame then
		return false, "Select a game before setting an autoload config."
	end
	if name ~= nil and (not gameProfiles or not gameProfiles[name]) then
		return false, "Config not found."
	end
	local previousName = self.autoloadByGame[self.activeGame]
	self.autoloadByGame[self.activeGame] = name
	local success, err = self:Persist()
	if not success then
		self.autoloadByGame[self.activeGame] = previousName
	end
	return success, err
end

	end
	local key = tostring(placeId)
	if name == nil then
		self.selectedGames[key] = nil
	else
		self.selectedGames[key] = name

function ConfigStore:LoadAutoLoad()

function ConfigStore:List()

local animNamePending = {}

Library.AddMissEntry = Library.AddMissEntry or function(self, entryType, key, name, distance, parent, known)
	local data = self.InfoLoggerData
	-- Honor the blacklist (keys the user muted by right-clicking / via the builder tab's dropdown).
	if data and key ~= nil and data.KeyBlacklistList and data.KeyBlacklistList[key] then
		return
	end
	if not self.AddTextToInfoLogger or not self.InfoLoggerContainer then
		return
	end

	-- Pull the numeric asset id out of the key so we can look up the animation's real Roblox name.
	local assetId = nil
	if key ~= nil then
		local digits = tostring(key):match("(%d+)")
		assetId = digits and tonumber(digits) or nil
	end

	-- Builds the line text. Prefers the resolved Roblox asset name ("punch"); falls back to a name the
	-- caller passed (the timing name for known entries), then just shows the id.
	local function buildText(resolvedName)
		local parts = { "[" .. tostring(entryType or "?") .. "]" }
		if known then
			parts[#parts + 1] = "(known)"
		end
		local shown = resolvedName or (known and name) or nil
		if shown and tostring(shown) ~= "" then
			parts[#parts + 1] = tostring(shown)
		end
		if key ~= nil then
			parts[#parts + 1] = "\194\183 " .. tostring(key)
		end
		if type(distance) == "number" then
			parts[#parts + 1] = string.format("[%.0fm]", distance)
		end
		return table.concat(parts, " ")
	end

	local cached = (assetId and animNameCache[assetId]) or nil

	-- Left-click copies the key (animation id) to paste into the timing builder.
	self:AddTextToInfoLogger(buildText(cached), key ~= nil and tostring(key) or nil)

	local label = (data and data.ContainerLabels and data.ContainerLabels[#data.ContainerLabels]) or nil

	-- Right-click the entry to blacklist its key (mute it) -- AddText only wires left-click, so attach our
	-- own MouseButton2 handler to the label it just appended to ContainerLabels.
	if key ~= nil and label then
		label.InputBegan:Connect(function(input)
			if input.UserInputType == Enum.UserInputType.MouseButton2 then
				data.KeyBlacklistList = data.KeyBlacklistList or {}
				data.KeyBlacklistList[key] = true
				data.KeyBlacklistHistory = data.KeyBlacklistHistory or {}
				data.KeyBlacklistHistory[key] = true
				pcall(function() label:Destroy() end)
				local idx = table.find(data.ContainerLabels, label)
				if idx then
					table.remove(data.ContainerLabels, idx)
				end
				if self.UpdateInfoLoggerSize then
					pcall(function() self:UpdateInfoLoggerSize() end)
				end
				pcall(function() self:Notify("Blacklisted key: " .. tostring(key)) end)
				if aztup_options and aztup_options.BlacklistedKeys and self.KeyBlacklists then
					pcall(function()
						aztup_options.BlacklistedKeys:SetValues(self:KeyBlacklists())
						aztup_options.BlacklistedKeys:Display()
					end)
				end
			end
		end)
	end

	-- Resolve the animation's real Roblox asset name once (cached), then patch the label in place. GetProductInfo
	-- yields + can throttle, so it runs async and best-effort; the id stays shown until the name arrives.
	if assetId and not cached and not animNamePending[assetId] then
		animNamePending[assetId] = true
		task.spawn(function()
			local ok, info = pcall(function()
				return game:GetService("MarketplaceService"):GetProductInfo(assetId, Enum.InfoType.Asset)
			end)
			animNamePending[assetId] = nil
			if ok and type(info) == "table" and type(info.Name) == "string" and info.Name ~= "" then
				animNameCache[assetId] = info.Name
				if label and label.Parent then
					label.Text = buildText(info.Name)
				end
			end
		end)
	end
end

function Library:CreateWindow(...)

	-- The Console (InfoLogger) is built inside CreateWindow and its Data table only has ContainerLabels;
	-- restore the Obsidian blacklist/cycle fields the builder tab + SaveManager expect, so the logger's
	-- blacklist dropdown, "Remove Selected Keys", and config save/load keep working.
	if self.InfoLoggerData then
		self.InfoLoggerData.KeyBlacklistList = self.InfoLoggerData.KeyBlacklistList or {}
		self.InfoLoggerData.KeyBlacklistHistory = self.InfoLoggerData.KeyBlacklistHistory or {}
		self.InfoLoggerData.MissingDataEntries = self.InfoLoggerData.MissingDataEntries or {}
		if self.InfoLoggerData.InfoLoggerCycle == nil then
			self.InfoLoggerData.InfoLoggerCycle = 0
		end
	end
	-- Apply registered colors (accent title label, etc.) + preload the title logo -- the showcase did this,
	-- luma's flow doesn't, so without it the accent name stays FontColor and the logo may not fetch.
	pcall(function() self:UpdateColorsUsingRegistry() end)
	task.spawn(function()
		pcall(function()
			game:GetService("ContentProvider"):PreloadAsync({ "rbxassetid://95969803781339" })
		end)
	end)
	-- luma: instantiate the YSL mascot (side image, anchored above the window). The showcase created
	-- this in its demo-startup, which luma strips -- so do it once here, now that the window's
	-- ScreenGui/WindowHolder exist. The module (@src/ui/side_image) handles preload, drag/anchor,
	-- toggled/dragging visibility, and unload cleanup itself.
	if not self._sideImageInit then
		self._sideImageInit = true
		local ok, controllerOrErr = pcall(function()
			return yslRequire("@src/ui/side_image")(self)
		end)
		if ok then
			self.SideImageController = controllerOrErr
		else
			warn("[Luma] mascot init failed: " .. tostring(controllerOrErr))
		end
	end
	-- luma: instantiate the YSL ArrayList (on-screen HUD of enabled toggles). The module self-registers
	-- self.ArrayListController and hooks ToggleChanged, so we just build it once here after the window.
	if not self._arrayListInit then
		self._arrayListInit = true
		local ok, err = pcall(function()
			yslRequire("@src/ui/arraylist")(self)
		end)
		if not ok then
			warn("[Luma] arraylist init failed: " .. tostring(err))
		end
	end
	if Window and Window.AddTab then
		local __realAddTab = Window.AddTab
		function Window:AddTab(...)
			local Tab = __realAddTab(self, ...)
			if Tab then
				if not Tab.AddDynamicGroupbox then
					function Tab:AddDynamicGroupbox(Name)
						self._lumaDyn = (self._lumaDyn or 0) + 1
						if self._lumaDyn % 2 == 1 then
							return self:AddLeftGroupbox(Name)
						else
							return self:AddRightGroupbox(Name)
						end
					end
				end
				if not Tab.AddFullGroupbox then
					function Tab:AddFullGroupbox(Name) return self:AddGroupbox({ Side = 0, Name = Name }) end
				end
				if not Tab.AddFullTabbox then
					function Tab:AddFullTabbox(Name)
						return self:AddTabbox({ Name = Name, Side = 0 })
					end
				end
				if not Tab.AddDynamicTabbox then
					function Tab:AddDynamicTabbox(Name)
						self._lumaDynT = (self._lumaDynT or 0) + 1
						if self._lumaDynT % 2 == 1 then
							return self:AddLeftTabbox(Name)
						else
							return self:AddRightTabbox(Name)
						end
					end
				end
			end
			return Tab
		end
	end
	return Window
end

	---@class Filesystem
	---@field _path string
	local Filesystem = {}
	Filesystem.__index = Filesystem

	---@return string
	function Filesystem:path()
		if not isfolder(self._path) then
			makefolder(self._path)
		end

		return self._path
	end

	---@param path string
	---@return string
	function Filesystem:append(path)
		return self:path() .. "\\" .. path
	end

	---@param filename string
	---@return boolean
	function Filesystem:file(filename)
		return isfile(self:append(filename))
	end

	---@param filename string
	---@return string
	function Filesystem:read(filename)
		if not self:file(filename) then
			return error("File does not exist or is a folder.", 2)
		end

		return readfile(self:append(filename))
	end

	---@param filename string
	---@return string
	function Filesystem:delete(filename)
		if not self:file(filename) then
			return error("File does not exist or is a folder.", 2)
		end

		return delfile(self:append(filename))
	end

	---@param filename string
	---@param contents string?
	function Filesystem:write(filename, contents)
		writefile(self:append(filename), contents and contents or "")
	end

	---@param raw boolean?
	---@return table
	function Filesystem:list(raw)
		local list = listfiles(self:path())
		if not list then
			return error("File list does not exist.", 2)
		end

		local new = {}

		for idx, path in next, list do
			---@note: Non-raw weird behavior where the path is never detected in the string. Let's manually index remove it.
			new[idx] = raw and path or string.sub(path, #(self:path() .. "\\") + 1, #path)
		end

		return new
	end

	---@param path string
	---@return Filesystem
	function Filesystem.new(path)
		local self = setmetatable({}, Filesystem)
		self._path = path
		return self
	end

	return Filesystem
end)()
end)

__bundle_register("Utility/Profiler", function(require, _LOADED, __bundle_register, __bundle_modules)
return LPH_NO_VIRTUALIZE(function()
	local Profiler = {}

	-- Opt-in: set getgenv().LycorisProfile = true before executing. When off, run/wrap are
	-- pass-throughs with no overhead.
	local enabled = getgenv ~= nil and getgenv().LycorisProfile == true
	local reportInterval = 10.0
	local stats = {}

	local function record(label, start)
		local elapsed = os.clock() - start
		local entry = stats[label]

		if not entry then
			entry = { total = 0, calls = 0, max = 0 }
			stats[label] = entry
		end

		entry.total = entry.total + elapsed
		entry.calls = entry.calls + 1

		if elapsed > entry.max then
			entry.max = elapsed
		end
	end

	---Runs a function, forwarding all return values.
	---@param label string
	---@param functionToProfile function
	function Profiler.run(label, functionToProfile, ...)
		if not enabled then
			return functionToProfile(...)
		end

		local start = os.clock()
		local results = table.pack(functionToProfile(...))
		record(label, start)
		return table.unpack(results, 1, results.n)
	end

	---Wrap function to match Profiler.run signature.
	---@param label string
	---@param functionToProfile function
	---@return function
	function Profiler.wrap(label, functionToProfile)
		if not enabled then
			return functionToProfile
		end

		return function(...)
			return Profiler.run(label, functionToProfile, ...)
		end
	end

	---Print the most expensive labels since the last report, then reset.
	---@param count number?
	function Profiler.report(count)
		local sorted = {}

		for label, entry in next, stats do
			sorted[#sorted + 1] = { label = label, entry = entry }
		end

		table.sort(sorted, function(a, b)
			return a.entry.total > b.entry.total
		end)

		warn(string.format("[Profiler] Top labels over the last %.0fs (total ms / calls / avg ms / max ms):", reportInterval))

		for idx = 1, math.min(count or 15, #sorted) do
			local label = sorted[idx].label
			local entry = sorted[idx].entry

			warn(
				string.format(
					"[Profiler] %2d. %-48s %8.2f / %6d / %6.3f / %6.2f",
					idx,
					label,
					entry.total * 1000,
					entry.calls,
					(entry.total / entry.calls) * 1000,
					entry.max * 1000
				)
			)
		end

		table.clear(stats)
	end

	if enabled then
		task.spawn(function()
			while true do
				task.wait(reportInterval)
				Profiler.report()
			end
		end)
	end

	return Profiler
end)()
end)

---@param newTask any

	if Maid[index] ~= nil then
		return
	end

	local tasks = self._tasks
	local oldTask = tasks[index]

	if oldTask == newTask then
		return
	end

	tasks[index] = newTask

	if oldTask then
		if typeof(oldTask) == "thread" then
			return coroutine.status(oldTask) == "suspended" and task.cancel(oldTask) or nil
		end

		if type(oldTask) == "function" then
			oldTask()
		elseif typeof(oldTask) == "RBXScriptConnection" then
			oldTask:Disconnect()
		elseif typeof(oldTask) == "Instance" and oldTask:IsA("Tween") then
			oldTask:Pause()
			oldTask:Cancel()
			oldTask:Destroy()
		elseif oldTask.Destroy then
			oldTask:Destroy()
		elseif oldTask.Remove then
			oldTask:Remove()
		elseif oldTask.detach then
			oldTask:detach()
		end
	end
end)

---@return any

	self:add(task)
	return task
end)

	if not task then
		return error("task cannot be false or nil", 2)
	end

	local taskId = #self._tasks + 1
	self[taskId] = task

	return taskId
end)

	local tasks = self._tasks
	tasks[taskId] = nil

	local tasks = self._tasks

	for index, task in pairs(tasks) do
		if typeof(task) == "RBXScriptConnection" then
			tasks[index] = nil
			pcall(task.Disconnect, task)
		end
	end

	local index, _task = next(tasks)

	while _task ~= nil do
		tasks[index] = nil

		if typeof(_task) == "thread" then
			if coroutine.status(_task) == "suspended" then
				pcall(task.cancel, _task)
			end
		else
			pcall(function()
				if type(_task) == "function" then
					_task()
				elseif typeof(_task) == "RBXScriptConnection" then
					_task:Disconnect()
				elseif typeof(_task) == "Instance" and _task:IsA("Tween") then

					_task:Cancel()
					_task:Destroy()

					_task:Destroy()
				elseif _task.Remove then
					_task:Remove()
				elseif _task.detach then
					_task:detach()
				end
			end)
		end

	if not idx then

	end

	end

	self._data[name] = action

	table.insert(self._stack, action)

	end

---@note: Mirrors Preprocessor.py's `_inline_timings` scrambling 1:1, including the
---@param raw table
---@return table
function TimingContainer.decode(raw)
	local out = {}
	for k, v in next, raw do
		out[k] = v
	end

	local function decodeHitbox(hb)
		return {
			X = ((((hb.X) + 69) * 12) - 5),
			Y = ((((hb.Y) + 69) * 12) - 5),
			Z = ((((hb.Z) + 69) * 12) - 5),
		}
	end

	if typeof(out.name) == "string" then
		out.name = (function(_s)local _r={} for _i=1,#_s do _r[_i]=string.char(bit32.bxor(string.byte(_s,_i),42)) end return table.concat(_r) end)(out.name)
	end

	if typeof(out.smod) == "string" then
		out.smod = (function(_s)local _r={} for _i=1,#_s do _r[_i]=string.char(bit32.bxor(string.byte(_s,_i),42)) end return table.concat(_r) end)(out.smod)
	end

	if typeof(out._id) == "string" and out._id ~= "" then
		out._id = (function(_s)local _r={} for _i=1,#_s do _r[_i]=string.char(bit32.bxor(string.byte(_s,_i),42)) end return table.concat(_r) end)(out._id)
	end

	if typeof(out.pname) == "string" and out.pname ~= "" then
		out.pname = (function(_s)local _r={} for _i=1,#_s do _r[_i]=string.char(bit32.bxor(string.byte(_s,_i),42)) end return table.concat(_r) end)(out.pname)
	end

	if typeof(out.ename) == "string" and out.ename ~= "" then
		out.ename = (function(_s)local _r={} for _i=1,#_s do _r[_i]=string.char(bit32.bxor(string.byte(_s,_i),42)) end return table.concat(_r) end)(out.ename)
	end

	if typeof(out.imxd) == "number" then
		out.imxd = ((((out.imxd) + 69) * 12) - 5)
	end

	if typeof(out.imdd) == "number" then
		out.imdd = ((((out.imdd) + 69) * 12) - 5)
	end

	if typeof(out.rpd) == "number" and out.rpd ~= 0 then
		out.rpd = ((((out.rpd) + 69) * 12) - 5)
		if typeof(out.rsd) == "number" then
			out.rsd = ((((out.rsd) + 69) * 12) - 5)
		end
	end

	if typeof(out.hitbox) == "table" then
		out.hitbox = decodeHitbox(out.hitbox)
	end

		local actions = {}
		for idx, action in next, out.actions do
			local decoded = {}
			for k, v in next, action do
				decoded[k] = v
			end

			if typeof(decoded._type) == "string" then
				decoded._type = (function(_s)local _r={} for _i=1,#_s do _r[_i]=string.char(bit32.bxor(string.byte(_s,_i),42)) end return table.concat(_r) end)(decoded._type)

			end

			if typeof(decoded.when) == "number" then
				decoded.when = ((((decoded.when) + 69) * 12) - 5)
			end

			if typeof(decoded.hitbox) == "table" then
				decoded.hitbox = decodeHitbox(decoded.hitbox)
			end

			actions[idx] = decoded
		end
		out.actions = actions
	end

	return out
end

---@param name string
---@return Timing?
function TimingContainer:find(name)
	for _, timing in next, self.timings do
		if timing.name ~= name then
			continue
		end

		return timing
	end
end

---@return Timing[]
function TimingContainer:list()
	local timings = {}

	for _, timing in next, self.timings do
		timings[#timings + 1] = timing
	end

	return timings
end

---@return string[]
function TimingContainer:names()
	local names = {}

	for _, timing in next, self.timings do
		names[#names + 1] = timing.name
	end

	table.sort(names)

	return names
end

---@param timing Timing
function TimingContainer:remove(timing)
	local id = timing:id()
	if not id then
		return
	end

	self.timings[id] = nil
end

---@param other TimingContainer
---@return boolean
function TimingContainer:equals(other)
	if self:count() ~= other:count() then
		return false
	end

	for id, timing in next, self.timings do
		local otherTiming = other.timings[id]
		if not otherTiming then
			return false
		end

		if not timing:equals(otherTiming) then
			return false
		end
	end

	return true
end

---@return TimingContainer
function TimingContainer:clone()
	local container = TimingContainer.new(self.module)

	for _, timing in next, self.timings do
		container:push(timing:clone())
	end

	return container
end

---@param timing Timing
function TimingContainer:push(timing)
	local id = timing:id()
	if not id then
		return
	end

	---@note: Timing array keys must all be unique.
	if self.timings[id] then
		return error(string.format("Timing identifier '%s' already exists in container.", id))
	end

	---@note: Every timing must have unique names.
	if self:find(timing.name) then
		return error(string.format("Timing name '%s' already exists in container.", timing.name))
	end

	self.timings[id] = timing
end

function TimingContainer:clear()
	self.timings = {}
end

---@return number
function TimingContainer:count()
	local count = 0

	for _ in next, self.timings do
		count = count + 1
	end

	return count
end

---@param values table
function TimingContainer:load(values)
	for _, value in next, values do
		local timing = self.module.new(self.scrambled and TimingContainer.decode(value) or value)
		if not timing then
			continue
		end

		local id = timing:id()
		if not id then
			continue
		end

		---@note: Timing array keys must all be unique.
		if self.timings[id] then
			return error(string.format("Timing identifier '%s' already exists in container.", id))
		end

		---@note: Every timing must have unique names.
		if self:find(timing.name) then
			return error(string.format("Timing name '%s' already exists in container.", timing.name))
		end

		---@note: Why are the stored timing keys different from what's loaded?

		self.timings[id] = timing
	end
end

---@return table
function TimingContainer:serialize()
	local out = {}

	for _, timing in next, self.timings do
		out[#out + 1] = timing:serialize()
	end

	return out
end

---@param module Timing
---@param scrambled boolean? Whether entries passed to :load() are baked-in truth data that needs one-time decoding.
---@return TimingContainer
function TimingContainer.new(module, scrambled)
	local self = setmetatable({}, TimingContainer)
	self.timings = {}
	self.module = module
	self.scrambled = scrambled or false
	return self
end

return TimingContainer
end)

-- with a set of features (see AutomationFeatures) and a character-creation routine. `construct` builds one;
-- `run` enables the features and starts the state machine; `stop` tears the features back down.

AutomationStruct.__index = AutomationStruct

---@module Features.Automation.Objects.AutomationFeatures

---@module Features.Automation.Objects.General
local General = require("Features/Automation/Objects/General")

---@module Utility.SafeClick

local players = game:GetService("Players")

	return players.LocalPlayer

local function character()

	return lp and lp.Character

---@param opts table
---@return table

	obj.id = opts.id
	obj.state_machine = opts.state_machine
	obj.features = opts.features or {}
	obj.on_run = opts.on_run

		return true
	end

---True while the owning automation is still running (PR's aztup.automation:has_any()).

function AutomationStruct:has_any()
	return self.is_running() == true
end

---Enable every feature, then start the state machine.
function AutomationStruct:run()
	self.active_features = {}

	-- Blow through the start/loading screen so the character actually spawns before we route (every farm
	-- needs this on a fresh join; skipped automatically during character creation).
	General.requestStart(function()
		return self:has_any()
	end)

	for _, name in ipairs(self.features) do
		local feat = AutomationFeatures.create(name, self)
		if feat then
			task.spawn(function()
				local ok, err = pcall(function()
					feat:enable()
				end)
				if not ok then
					warn(string.format("[AutomationStruct] feature %s enable failed: %s", tostring(name), tostring(err)))
				end
			end)
			self.active_features[name] = feat
		end
	end

	if self.on_run then
		task.spawn(self.on_run)
	end

	table.insert(
		self.threads,
		task.spawn(function()
			self.state_machine:start()
		end)
	)
end

---Disable every active feature and cancel any in-flight character-creation helper threads.
function AutomationStruct:stop()
	for _, feat in pairs(self.active_features) do
		pcall(function()
			feat:disable()
		end)
	end
	self.active_features = {}

	if self._creation_threads then
		for _, thread in ipairs(self._creation_threads) do
			pcall(function()
				task.cancel(thread)
			end)
		end
		self._creation_threads = {}
	end
end

---Character creation. Direct port of PR's automation_struct.create_character (luma-native requires).
function AutomationStruct:create_character()
	local options = self.character_creator_handler_opts
	local lp = localPlayer()

	local requests = replicatedStorage:WaitForChild("Requests")
	local characterCreator = requests:WaitForChild("CharacterCreator")
	local pickSpawn = characterCreator:WaitForChild("PickSpawn")
	local changeWeapon = characterCreator:WaitForChild("ChangeWeapon")
	local finishCreation = characterCreator:WaitForChild("FinishCreation")
	local toggleMetaModifier = requests:WaitForChild("ToggleMetaModifier", 20)
	local playerGui = lp:WaitForChild("PlayerGui")
	local characterGui = playerGui:WaitForChild("CharacterCreator")
	local suc

	-- Re-resolve the PickSpawn remote fresh each pick -- the game can replace the CharacterCreator remotes
	-- between invocations, and a cached reference would go stale (InvokeServer on a destroyed remote no-ops).
	local function getPickSpawn()
		local cc = requests:FindFirstChild("CharacterCreator")
		return (cc and cc:FindFirstChild("PickSpawn")) or pickSpawn
	end

	-- Origin selection. The PickSpawn return value is NOT a reliable confirmation -- it echoes the requested
	-- spawn even when the origin never actually applies (the "script thinks it picked but it didn't" bug). The
	-- authoritative current origin is what the creator client itself reads: `DataReplication.GetData().Origin`
	-- (line 1851 of the decompiled creator -- defaults to "Castaway"). So we fire PickSpawn, then verify the
	-- REAL server-synced origin equals our target category (Titus = "Authority Ensign", L2 = "Deepbound"),
	-- retrying until it does or `timeout` seconds pass -> kick. cfg = { category, spawn, timeout, kick_message };
	-- `category` is the origin whose presence in DataReplication.Origin we verify.
	local function clickOriginButton(cfg)
		local deadline = os.clock() + (cfg.timeout or 10)

		local function currentOrigin()
			local origin
			pcall(function()
				origin = require(replicatedStorage:WaitForChild("Info"):WaitForChild("DataReplication")).GetData().Origin
			end)
			return origin
		end

		repeat
			-- Already the right origin (server-synced)? done.
			if cfg.category and currentOrigin() == cfg.category then
				return true
			end
			-- Fire the spawn pick (what the real button does), then re-check the real origin.
			pcall(function()
				getPickSpawn():InvokeServer(cfg.spawn)
			end)
			task.wait(0.25)
			if cfg.category and currentOrigin() == cfg.category then
				return true
			end
		until os.clock() >= deadline

		-- Couldn't get the origin to apply in time. Prefer the automation's on_fail handler (e.g. server-hop +
		-- retry on a fresh server) over kicking, so the farm just tries again instead of dropping the player.
		if cfg.on_fail then
			pcall(cfg.on_fail)
		else
			pcall(function()
				lp:Kick(cfg.kick_message or "Need authority origin")
			end)
		end
		return false
	end

	-- Select the Standard gamemode by CLICKING its button -- NOT just firing ChangeGamemode. The decompiled
	-- creator shows the button handler does MORE than the remote: after ChangeGamemode:InvokeServer it sets the
	-- client's `u40` flag and calls refreshGamemode(), which is what hides the gamemode screen and REVEALS the
	-- origin / creation UI. Firing the remote alone leaves the client stuck on the gamemode screen, so the
	-- origin never applies (that was the "origin never chosen" bug). We confirm by the GameMode frame going
	-- invisible -- exactly what refreshGamemode does once a mode is chosen.
	do
		local deadline = os.clock() + 10
		local function standardElement()
			local gm = characterGui:FindFirstChild("GameMode")
			local opts = gm and gm:FindFirstChild("Options")
			local std = opts and opts:FindFirstChild("Standard")
			return std and std:FindFirstChild("Element")
		end
		local function gamemodeChosen()
			local gm = characterGui:FindFirstChild("GameMode")
			return gm ~= nil and gm.Visible == false
		end
		repeat
			if gamemodeChosen() then
				break
			end
			local el = standardElement()
			if el then
				-- safeClick instead of firesignal (native-crash risk pcall can't catch -- see Utility/SafeClick).
				safeClick(el)
			end
			task.wait(0.25)
			if gamemodeChosen() then
				break
			end
		until os.clock() >= deadline
	end

	-- Pick the origin UP FRONT, before weapon/modifiers/boons/flaws. Deepwoken keys the rest of creation off
	-- the chosen origin, so those picks (and FinishCreation) must happen with an origin already set -- otherwise
	-- the character commits before the origin applies. It's re-asserted again as the last write below to beat
	-- the Standard gamemode's async default-origin overwrite.
	if options.origin_button then
		-- Button path: wait for + click the real UI origin button (kicks the player if it never appears).
		if not clickOriginButton(options.origin_button) then
			return
		end
	elseif options.origin then
		for _ = 1, 10 do
			suc = getPickSpawn():InvokeServer(options.origin)
			task.wait(0.1)
		end
	end

	if options.weapon then
		for _ = 1, 10 do
			suc = changeWeapon:InvokeServer(options.weapon)
			if suc then
				break
			end
			task.wait(0.1)
		end
	end

	if options.modifiers and toggleMetaModifier then
		for _, modifier in ipairs(options.modifiers) do
			toggleMetaModifier:FireServer(modifier)
			if #options.modifiers == 1 then
				break
			end
			task.wait(1)
		end
	end

	local pickBoon = characterCreator:WaitForChild("PickBoon")
	local pickFlaw = characterCreator:WaitForChild("PickFlaw")
	local dataReplication =
		require(replicatedStorage:WaitForChild("Info"):WaitForChild("DataReplication")).GetData()

	for _, flaw in next, dataReplication.Flaws do
		pickFlaw:InvokeServer(tostring(flaw))
	end

	for _, boon in next, dataReplication.Boons do
		pickBoon:InvokeServer(tostring(boon))
	end

	pickBoon:InvokeServer("Autodidact")
	task.wait(1.5)
	pickFlaw:InvokeServer("Obvious")

	if not self:has_any() then
		return
	end

	-- Settle FIRST, BEFORE the final origin lock. The Standard gamemode applies its own default origin
	-- (Authority) asynchronously, and it can land a beat after our picks. Previously this wait happened AFTER
	-- the final origin write, which gave the async default a 2s window to deselect our origin again right
	-- before FinishCreation -- committing the wrong origin (button showing the wrong colour). Waiting here,
	-- before the lock, lets that default land first so our lock below is the genuine last word.
	task.wait(2)

	-- FINAL origin lock -- the very last thing before committing, with NO wait after it, so nothing can
	-- overwrite the origin before FinishCreation. clickOriginButton invokes PickSpawn and only returns once the
	-- server echoes back our spawn (authoritative), so this is the definitive last origin write.
	if options.origin_button then
		if not clickOriginButton(options.origin_button) then
			return
		end
	elseif options.origin then
		for _ = 1, 10 do
			getPickSpawn():InvokeServer(options.origin)
			task.wait(0.1)
		end
	end

	-- Finish creation ONCE, then confirm the "create this character?" prompt. Both helper threads are
	-- tracked on the struct (cancelled by :stop) and time-bounded, so a failed/stopped attempt can't leave a
	-- zombie confirm thread that later fires Choice:FireServer(true) on the NEXT run's prompt -- that is what
	-- intermittently rolled the wrong character. (PR's framework gets a fresh session each cycle, so it never
	-- hits this; luma persists in-session, so we have to guard it.)
	self._creation_threads = self._creation_threads or {}
	table.insert(
		self._creation_threads,
		task.spawn(function()
			pcall(function()
				finishCreation:InvokeServer()
			end)
		end)
	)
	table.insert(
		self._creation_threads,
		task.spawn(function()
			local choicePrompt = playerGui:WaitForChild("ChoicePrompt", 30)
			local choice = choicePrompt and choicePrompt:WaitForChild("Choice", 5)
			if choice and self:has_any() then
				pcall(function()
					choice:FireServer(true)
				end)
			end
		end)
	)

	-- Wait for the character to spawn into Live (30s cap so a failed creation can't hang forever -- the caller
	-- re-checks and retries instead).
	local waitStart = os.clock()
	repeat
		task.wait()
		local c = character()
		if c and c.Parent and c.Parent.Name == "Live" then
			break
		end
	until not self:has_any() or os.clock() - waitStart > 30

	local c = character()
	if c and c.Parent and c.Parent.Name == "Live" then
		c:WaitForChild("HumanoidRootPart")
		c:WaitForChild("Humanoid")
	end
end

return AutomationStruct
end)
__bundle_register("Utility/SafeClick", function(require, _LOADED, __bundle_register, __bundle_modules)
---@note: Click a GuiButton by invoking its MouseButton1Click handlers DIRECTLY (conn.Function()), never via
---@note: firesignal() or conn:Fire(). Both of those are native-crash-prone and a pcall can NOT catch a native
---@note: fault:
---@note:   * conn:Fire() on a getconnections() object dereferences a null pointer on mac/macsploit -- the
---@note:     NULL+0x28 SIGSEGV that OptimizeGame documents and avoids.
---@note:   * firesignal() is in luma's own "these crash" list (see Spoofing.rics).
---@note: getconnections() itself is fine (OptimizeGame uses it); it's the firing that crashes. Every
---@note: getconnections connection exposes .Function (its bound handler) even on macsploit, so calling that
---@note: directly is the safe, cross-executor equivalent of a click.

---@param button Instance A TextButton/ImageButton (anything with MouseButton1Click).
---@return boolean clicked  false if getconnections is unavailable or the button had no handlers.
local function safeClick(button)
	if not button or type(getconnections) ~= "function" then
		return false
	end

	local ok, connections = pcall(getconnections, button.MouseButton1Click)
	if not ok or type(connections) ~= "table" then
		return false
	end

	local fired = false
	for _, connection in ipairs(connections) do
		pcall(function()
			local handler = connection.Function
			if handler then
				handler()
				fired = true
			end
		end)
	end

	return fired
end

return safeClick
end)

-- ---------------------------------------------------------------------------------------------------------
-- Toggle-mapped features. enable() captures the user's current value then forces true; disable() restores.
-- ---------------------------------------------------------------------------------------------------------

---@param toggleName string
local function makeToggleFeature(toggleName)
	return function(_name, _struct)
		local feat = { name = _name, _prev = nil }

		function feat:enable()
			local toggle = Toggles and Toggles[toggleName]
			if not toggle then
				return
			end
			self._prev = toggle.Value
			toggle:SetValue(true)
		end

		function feat:disable()
			local toggle = Toggles and Toggles[toggleName]
			if toggle and self._prev ~= nil then
				toggle:SetValue(self._prev)
			end
			self._prev = nil
		end

		return feat
	end
end

-- ---------------------------------------------------------------------------------------------------------
-- mob_ai_breaker: PR's Titus kill mechanism. Every Heartbeat, slam the root's velocity straight down
-- (Vector3.new(0, -9e9, 0)); every PreRender, restore the pre-slam velocity so the character doesn't
-- actually fall. This desyncs / breaks the boss's AI + physics targeting. NOT an m1 farm.
-- ---------------------------------------------------------------------------------------------------------
local function makeMobAiBreaker(name, struct)
	local feat = { name = name, current_connection = nil, pre_ren = nil }
	local lastVel = nil

	function feat:enable()
		self.current_connection = runService.Heartbeat:Connect(function()
			local r = root()
			if not r then
				return
			end
			lastVel = r.AssemblyLinearVelocity
			r.AssemblyLinearVelocity = Vector3.new(0, -9e9, 0)
		end)

		self.pre_ren = runService.PreRender:Connect(function()
			if lastVel then
				local r = root()
				if r then
					r.AssemblyLinearVelocity = lastVel
				end
				lastVel = nil
			end
		end)
	end

	function feat:disable()
		if self.current_connection then
			pcall(function()
				self.current_connection:Disconnect()
			end)
			self.current_connection = nil
		end
		if self.pre_ren then
			pcall(function()
				self.pre_ren:Disconnect()
			end)
			self.pre_ren = nil
		end
		if lastVel then
			local r = root()
			if r then
				r.AssemblyLinearVelocity = lastVel
			end
			lastVel = nil
		end
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- jesus: anti-water-death. Parents an invisible 2048x2048 floor at Y=-3 under the player and keeps the
-- "NoSwim" effect applied so the game never forces a swim/drown state. Only active in the merit + dungeon
-- places (the phases where the farm skims over water), matching PR's is_eastern/is_etrean gate.
-- ---------------------------------------------------------------------------------------------------------
local function makeJesus(name, struct)
	local feat = { name = name, current_connection = nil, part = nil }

	function feat:enable()
		if not self.part then
			self.part = Instance.new("Part")
			self.part.Transparency = 1
			self.part.Size = Vector3.new(2048, 1, 2048)
			self.part.Anchored = true
			self.part.CanCollide = false
			self.part.Position = Vector3.new(0, 0, 0)
			self.part.Name = "FakeWater"
		end

		self.current_connection = runService.Heartbeat:Connect(function()
			local placeId = game.PlaceId
			if placeId ~= EASTERN_PLACE_ID and placeId ~= TITUS_PLACE_ID then
				if self.part.Parent then
					self.part.Parent = nil
				end
				return
			end
			local r = root()
			if not r then
				return
			end
			if self.part.Parent ~= workspace then
				self.part.Parent = workspace
			end
			self.part.CFrame = CFrame.new(r.Position.X, -3, r.Position.Z)

			pcall(function()
				local er = getER()
				if er and not er:HasEffect("NoSwim") then
					er:CreateEffect("NoSwim")
				end
			end)
		end)
	end

	function feat:disable()
		if self.current_connection then
			pcall(function()
				self.current_connection:Disconnect()
			end)
			self.current_connection = nil
		end
		if self.part then
			pcall(function()
				self.part:Destroy()
			end)
			self.part = nil
		end
		pcall(function()
			local er = getER()
			if er and er.FindEffect then
				local effect = er:FindEffect("NoSwim")
				if effect and effect.Debris then
					effect:Debris(0.1)
				end
			end
		end)
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- no_one_bit: strips the "OneBit" passive (the flaw that one-shots you when you'd otherwise survive a
-- lethal hit) from EffectReplicator.Passives so a freshly-rolled character can't die to it. PR does this
-- via an ssv_parser that just mutates EffectReplicator.Passives[name]; ported directly.
-- ---------------------------------------------------------------------------------------------------------
local function makeNoOneBit(name, struct)
	local feat = { name = name, running = false, removed = false }

	function feat:enable()
		self.running = true
		task.spawn(function()
			while self.running do
				pcall(function()
					local er = getER()
					if er and er.Passives then
						if er.Passives["OneBit"] then
							self.removed = true
						end
						er.Passives["OneBit"] = nil
					end
				end)
				task.wait(1)
			end
		end)
	end

	function feat:disable()
		self.running = false
		if self.removed then
			pcall(function()
				local er = getER()
				if er and er.Passives then
					er.Passives["OneBit"] = true
				end
			end)
			self.removed = false
		end
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- auto_equip_weapon: keeps the weapon drawn. Whenever the "Equipped" effect is absent, fire the
-- CharacterHandler DrawWeapon remote. Faithful to PR's poll loop (its EffectReplicatorHandler:hook is a
-- latency optimization luma has no equivalent for; the 1s poll is the core behavior).
-- ---------------------------------------------------------------------------------------------------------
local function makeAutoEquipWeapon(name, struct)
	local feat = { name = name, running = false }

	local function isUsingUtilityTool()
		local c = char()
		if not c then
			return false
		end
		return c:FindFirstChildOfClass("Tool") ~= nil
	end

	local function drawWeapon()
		if isUsingUtilityTool() then
			return
		end
		local c = char()
		local handler = c and c:FindFirstChild("CharacterHandler")
		local requests = handler and handler:FindFirstChild("Requests")
		local remote = requests and requests:FindFirstChild("DrawWeapon")
		if remote then
			pcall(function()
				remote:FireServer(true)
			end)
		end
	end

	function feat:enable()
		self.running = true
		task.spawn(function()
			while self.running do
				local er = getER()
				if not (er and er:FindEffect("Equipped")) then
					drawWeapon()
				end
				task.wait(1)
			end
		end)
	end

	function feat:disable()
		self.running = false
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- m1_hold: PR's held-m1 feature. While `.held` is true, repeatedly throw a left-click (m1) via luma's
-- InputClient. Tasks flip `struct.active_features.m1_hold.held` on/off around the moments they want to swing.
-- ---------------------------------------------------------------------------------------------------------
local function makeM1Hold(name, struct)
	local feat = { name = name, held = false, running = false }

	function feat:enable()
		self.running = true
		task.spawn(function()
			while self.running do
				if self.held then
					local c = char()
					local r = c and c:FindFirstChild("HumanoidRootPart")
					if r then
						pcall(function()
							-- ignoreChecks + forceGround: PR's m1_hold swings on cooldown with no_aerials on.
							InputClient.left(r.CFrame, true, true)
						end)
					end
				end
				task.wait(0.1)
			end
		end)
	end

	function feat:disable()
		self.running = false
		self.held = false
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- mod_detector: hop away if a Deepwoken Staff member joins the server. PR checks a static member map first;
-- here we do PR's live fallback (GetRankInGroup on the staff group) for every player, which needs no data.
-- ---------------------------------------------------------------------------------------------------------
local STAFF_GROUP_ID = 5212858
local function makeModDetector(name, struct)
	local feat = { name = name, running = false, conn = nil }

	local function check(player)
		if player == players.LocalPlayer then
			return
		end
		task.spawn(function()
			local rank = 0
			pcall(function()
				rank = player:GetRankInGroup(STAFF_GROUP_ID)
			end)
			if rank and rank > 0 and feat.running then
				pcall(function()
					if Library and Library.NotifyWithSound then
						Library:NotifyWithSound(string.format("%s is in the Deepwoken Staff group -- hopping.", player.Name), 15)
					end
				end)
				if struct:has_any() then
					local slot = players.LocalPlayer and players.LocalPlayer:GetAttribute("DataSlot")
					pcall(function()
						ServerHop.hop(slot, true)
					end)
				end
			end
		end)
	end

	function feat:enable()
		self.running = true
		self.conn = players.PlayerAdded:Connect(check)
		task.delay(0.25, function()
			for _, p in ipairs(players:GetPlayers()) do
				check(p)
			end
		end)
	end

	function feat:disable()
		self.running = false
		if self.conn then
			pcall(function()
				self.conn:Disconnect()
			end)
			self.conn = nil
		end
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- fast_swing: remove the attack-cooldown effects (LightAttack/HeavyAttack/etc.) the instant they're added,
-- so swings aren't rate-limited client-side. Direct port of PR's fast_swing (EffectReplicator "added" hook
-- -> effect:Remove(true)).
-- ---------------------------------------------------------------------------------------------------------
local FAST_SWING_CLASSES = { "OffhandAttack", "HeavyAttack", "MediumAttack", "LightAttack", "UsingSpell" }
local function makeFastSwing(name, struct)
	local feat = { name = name, running = false, conn = nil }

	function feat:enable()
		self.running = true
		task.spawn(function()
			local er = getER()
			if not self.running or not er or not er.EffectAdded then
				return
			end
			self.conn = er.EffectAdded:Connect(function(effect)
				if effect and effect.Class and table.find(FAST_SWING_CLASSES, effect.Class) then
					task.defer(function()
						pcall(function()
							effect:Remove(true)
						end)
					end)
				end
			end)
		end)
	end

	function feat:disable()
		self.running = false
		if self.conn then
			pcall(function()
				self.conn:Disconnect()
			end)
			self.conn = nil
		end
	end

	return feat
end

-- ---------------------------------------------------------------------------------------------------------
-- auto_log: Ferryman-style anti-detection. Hop the moment another player comes within the automation's
-- auto-log distance (per-automation AutoLog store, adjustable 0-2000 in the menu; 0 = off). Skipped during
-- character creation (lobby players would softlock the creator).
-- ---------------------------------------------------------------------------------------------------------
local function makeAutoLog(name, struct)
	local feat = { name = name, running = false }

	function feat:enable()
		self.running = true
		task.spawn(function()
			while self.running do
				task.wait(1)
				if not self.running then
					break
				end
				local cc = workspace:FindFirstChild("CharacterCreator")
				local c = char()
				if cc and c and c.Parent == cc then
					-- character creation; don't hop
				else
					local r = c and c:FindFirstChild("HumanoidRootPart")
					local dist = AutoLog.get(struct and struct.id)
					if r and dist > 0 and Finder.pnear(r.Position, dist) then
						if struct:has_any() then
							local slot = players.LocalPlayer and players.LocalPlayer:GetAttribute("DataSlot")
							pcall(function()
								ServerHop.hop(slot, true)
							end)
						end
						return
					end
				end
			end
		end)
	end

	function feat:disable()
		self.running = false
	end

	return feat
end

__bundle_register("Features/Automation/Objects/AutoLog", function(require, _LOADED, __bundle_register, __bundle_modules)
-- AutoLog: per-automation auto-log (server-hop-when-a-player-is-near) distance store. Saved to disk per
-- automation id -- deliberately NOT part of luma's config saves, so the distance is a persistent per-
-- automation setting that survives config switches and rejoins. The menu sliders in AutomationTab read/write
-- it here; the automations (AutomationFeatures.auto_log + AutoTitus/EchoFarm/AutoFerryman watchers) read it.
local AutoLog = {}

---@module Utility.Filesystem
local Filesystem = require("Utility/Filesystem")

---@module Utility.JSON
local JSON = require("Utility/JSON")

local storage = Filesystem.new("Luma/Automation")
local FILE = "autolog.json"

-- Default range preserves the old hardcoded behaviour (1000 studs) for any automation never adjusted.
AutoLog.DEFAULT = 1000

local cache = nil

local function load()
	if cache then
		return cache
	end
	local data
	pcall(function()
		if storage:file(FILE) then
			local ok, decoded = pcall(JSON.decode, storage:read(FILE))
			if ok and type(decoded) == "table" then
				data = decoded
			end
		end
	end)
	cache = data or {}
	return cache
end

---Current auto-log distance for an automation id (DEFAULT if never set).
---@param id string?
---@return number
function AutoLog.get(id)
	if id == nil then
		return AutoLog.DEFAULT
	end
	local v = load()[tostring(id)]
	if type(v) ~= "number" then
		return AutoLog.DEFAULT
	end
	return v
end

---Persist a new auto-log distance for an automation id (writes to disk immediately).
---@param id string
---@param distance number
function AutoLog.set(id, distance)
	if id == nil then
		return
	end
	local c = load()
	c[tostring(id)] = tonumber(distance) or AutoLog.DEFAULT
	pcall(function()
		storage:write(FILE, JSON.encode(c))
	end)
end

return AutoLog
end)
__bundle_register("Utility/JSON", function(require, _LOADED, __bundle_register, __bundle_modules)
return LPH_NO_VIRTUALIZE(function()

	local json = { _version = "0.1.2" }


	local encode

	local escape_char_map = {
		["\\"] = "\\",
		['"'] = '"',
		["\b"] = "b",
		["\f"] = "f",
		["\n"] = "n",
		["\r"] = "r",
		["\t"] = "t",
	}

	local escape_char_map_inv = { ["/"] = "/" }
	for k, v in pairs(escape_char_map) do
		escape_char_map_inv[v] = k
	end

	local function escape_char(c)
		return "\\" .. (escape_char_map[c] or string.format("u%04x", c:byte()))
	end

	local function encode_nil(val)
		return "null"
	end

	local function encode_table(val, stack)
		local res = {}
		stack = stack or {}

		if stack[val] then
			error("circular reference")
		end

		stack[val] = true

		if rawget(val, 1) ~= nil or next(val) == nil then
			local n = 0
			for k in pairs(val) do
				if type(k) ~= "number" then
					error("invalid table: mixed or invalid key types")
				end
				n = n + 1
			end
			if n ~= #val then
				error("invalid table: sparse array")
			end
			for i, v in ipairs(val) do
				table.insert(res, encode(v, stack))
			end
			stack[val] = nil
			return "[" .. table.concat(res, ",") .. "]"
		else
			for k, v in pairs(val) do
				if type(k) ~= "string" then
					error("invalid table: mixed or invalid key types")
				end
				table.insert(res, encode(k, stack) .. ":" .. encode(v, stack))
			end
			stack[val] = nil
			return "{" .. table.concat(res, ",") .. "}"
		end
	end

	local function encode_string(val)
		return '"' .. val:gsub('[%z\1-\31\\"]', escape_char) .. '"'
	end

	local function encode_number(val)
		if val ~= val or val <= -math.huge or val >= math.huge then
			error("unexpected number value '" .. tostring(val) .. "'")
		end
		return string.format("%.14g", val)
	end

	local type_func_map = {
		["nil"] = encode_nil,
		["table"] = encode_table,
		["string"] = encode_string,
		["number"] = encode_number,
		["boolean"] = tostring,
	}

	encode = function(val, stack)
		local t = type(val)
		local f = type_func_map[t]
		if f then
			return f(val, stack)
		end
		error("unexpected type '" .. t .. "'")
	end

	function json.encode(val)
		return (encode(val))
	end


	local parse

	local function create_set(...)
		local res = {}
		for i = 1, select("#", ...) do
			res[select(i, ...)] = true
		end
		return res
	end

	local space_chars = create_set(" ", "\t", "\r", "\n")
	local delim_chars = create_set(" ", "\t", "\r", "\n", "]", "}", ",")
	local escape_chars = create_set("\\", "/", '"', "b", "f", "n", "r", "t", "u")
	local literals = create_set("true", "false", "null")

	local literal_map = {
		["true"] = true,
		["false"] = false,
		["null"] = nil,
	}

	local function next_char(str, idx, set, negate)
		for i = idx, #str do
			if set[str:sub(i, i)] ~= negate then
				return i
			end
		end
		return #str + 1
	end

	local function decode_error(str, idx, msg)
		local line_count = 1
		local col_count = 1
		for i = 1, idx - 1 do
			col_count = col_count + 1
			if str:sub(i, i) == "\n" then
				line_count = line_count + 1
				col_count = 1
			end
		end
		error(string.format("%s at line %d col %d", msg, line_count, col_count))
	end

	local function codepoint_to_utf8(n)
		local f = math.floor
		if n <= 0x7f then
			return string.char(n)
		elseif n <= 0x7ff then
			return string.char(f(n / 64) + 192, n % 64 + 128)
		elseif n <= 0xffff then
			return string.char(f(n / 4096) + 224, f(n % 4096 / 64) + 128, n % 64 + 128)
		elseif n <= 0x10ffff then
			return string.char(f(n / 262144) + 240, f(n % 262144 / 4096) + 128, f(n % 4096 / 64) + 128, n % 64 + 128)
		end
		error(string.format("invalid unicode codepoint '%x'", n))
	end

	local function parse_unicode_escape(s)
		local n1 = tonumber(s:sub(1, 4), 16)
		local n2 = tonumber(s:sub(7, 10), 16)
		if n2 then
			return codepoint_to_utf8((n1 - 0xd800) * 0x400 + (n2 - 0xdc00) + 0x10000)
		else
			return codepoint_to_utf8(n1)
		end
	end

	local function parse_string(str, i)
		local res = ""
		local j = i + 1
		local k = j

		while j <= #str do
			local x = str:byte(j)

			if x < 32 then
				decode_error(str, j, "control character in string")
			elseif x == 92 then
				res = res .. str:sub(k, j - 1)
				j = j + 1
				local c = str:sub(j, j)
				if c == "u" then
					local hex = str:match("^[dD][89aAbB]%x%x\\u%x%x%x%x", j + 1)
						or str:match("^%x%x%x%x", j + 1)
						or decode_error(str, j - 1, "invalid unicode escape in string")
					res = res .. parse_unicode_escape(hex)
					j = j + #hex
				else
					if not escape_chars[c] then
						decode_error(str, j - 1, "invalid escape char '" .. c .. "' in string")
					end
					res = res .. escape_char_map_inv[c]
				end
				k = j + 1
			elseif x == 34 then
				res = res .. str:sub(k, j - 1)
				return res, j + 1
			end

			j = j + 1
		end

		decode_error(str, i, "expected closing quote for string")
	end

	local function parse_number(str, i)
		local x = next_char(str, i, delim_chars)
		local s = str:sub(i, x - 1)
		local n = tonumber(s)
		if not n then
			decode_error(str, i, "invalid number '" .. s .. "'")
		end
		return n, x
	end

	local function parse_literal(str, i)
		local x = next_char(str, i, delim_chars)
		local word = str:sub(i, x - 1)
		if not literals[word] then
			decode_error(str, i, "invalid literal '" .. word .. "'")
		end
		return literal_map[word], x
	end

	local function parse_array(str, i)
		local res = {}
		local n = 1
		i = i + 1
		while 1 do
			local x
			i = next_char(str, i, space_chars, true)
			if str:sub(i, i) == "]" then
				i = i + 1
				break
			end
			x, i = parse(str, i)
			res[n] = x
			n = n + 1
			i = next_char(str, i, space_chars, true)
			local chr = str:sub(i, i)
			i = i + 1
			if chr == "]" then
				break
			end
			if chr ~= "," then
				decode_error(str, i, "expected ']' or ','")
			end
		end
		return res, i
	end

	local function parse_object(str, i)
		local res = {}
		i = i + 1
		while 1 do
			local key, val
			i = next_char(str, i, space_chars, true)
			if str:sub(i, i) == "}" then
				i = i + 1
				break
			end
			if str:sub(i, i) ~= '"' then
				decode_error(str, i, "expected string for key")
			end
			key, i = parse(str, i)
			i = next_char(str, i, space_chars, true)
			if str:sub(i, i) ~= ":" then
				decode_error(str, i, "expected ':' after key")
			end
			i = next_char(str, i + 1, space_chars, true)
			val, i = parse(str, i)
			res[key] = val
			i = next_char(str, i, space_chars, true)
			local chr = str:sub(i, i)
			i = i + 1
			if chr == "}" then
				break
			end
			if chr ~= "," then
				decode_error(str, i, "expected '}' or ','")
			end
		end
		return res, i
	end

	local char_func_map = {
		['"'] = parse_string,
		["0"] = parse_number,
		["1"] = parse_number,
		["2"] = parse_number,
		["3"] = parse_number,
		["4"] = parse_number,
		["5"] = parse_number,
		["6"] = parse_number,
		["7"] = parse_number,
		["8"] = parse_number,
		["9"] = parse_number,
		["-"] = parse_number,
		["t"] = parse_literal,
		["f"] = parse_literal,
		["n"] = parse_literal,
		["["] = parse_array,
		["{"] = parse_object,
	}

	parse = function(str, idx)
		local chr = str:sub(idx, idx)
		local f = char_func_map[chr]
		if f then
			return f(str, idx)
		end
		decode_error(str, idx, "unexpected character '" .. chr .. "'")
	end

	function json.decode(str)
		if type(str) ~= "string" then
			error("expected argument of type string, got " .. type(str))
		end
		local res, idx = parse(str, next_char(str, 1, space_chars, true))
		idx = next_char(str, idx, space_chars, true)
		if idx <= #str then
			decode_error(str, idx, "trailing garbage")
		end
		return res
	end

	return json

__bundle_register("Game/ServerHop", function(require, _LOADED, __bundle_register, __bundle_modules)
---@module Utility.PersistentData
local PersistentData = require("Utility/PersistentData")

---@module Utility.Logger
local Logger = require("Utility/Logger")

local ServerHop = {}

local replicatedStorage = game:GetService("ReplicatedStorage")
local players = game:GetService("Players")
local teleportService = game:GetService("TeleportService")
local runService = game:GetService("RunService")

local LOBBY_PLACE_ID = 4111023553
local DEBUGGING_MODE = true

-- If the current target server hasn't teleported us in this window, blacklist it and move to the next
-- one (full, or the instance is no longer available).
local SERVER_SWITCH_TIMEOUT = 10

local lastRetry = nil
local currentSlot = nil
local currentServerId = nil
local currentServerSince = nil

local function telemetryLog(...)
	if not DEBUGGING_MODE then
		return
	end

	Logger.warn(...)
end

---@return table
local function getBlacklist()
	local blacklist = PersistentData.get("sblacklist") or {}

	for key, value in next, blacklist do
		if tick() - value < 300 then
			continue
		end

		telemetryLog("(%s) Removed old entry (%s) from server blacklist.", tostring(key), tostring(value))

		blacklist[key] = nil
	end

	PersistentData.set("sblacklist", blacklist)

	return blacklist
end

---@param id string The server ID (JobId) to add to the blacklist.
local function addBlacklist(id)
	if not id then
		return
	end

	local blacklist = getBlacklist()

	blacklist[id] = tick()

	PersistentData.set("sblacklist", blacklist)
end

-- ============================================================================================
-- UI-driven server picking. Instead of firing PickSlot/PickServer remotes directly (which the game
-- ignored unless its own UI state was set up), we press the actual menu buttons via firesignal, so
-- the game's own handlers run -- exactly like a real player clicking them.
-- ============================================================================================

---LoadingGui.Overlay.Columns.MainFrame, or nil.
---@return Instance?
local function getMainFrame()
	local playerGui = players.LocalPlayer and players.LocalPlayer:FindFirstChild("PlayerGui")
	local loadingGui = playerGui and playerGui:FindFirstChild("LoadingGui")
	local overlay = loadingGui and loadingGui:FindFirstChild("Overlay")
	local columns = overlay and overlay:FindFirstChild("Columns")
	return columns and columns:FindFirstChild("MainFrame")
end

---Click a GUI button the way the game's own UI would (fire its click signals).
---@param button Instance?
local function pressButton(button)
	if not button then
		return
	end

	pcall(function()
		firesignal(button.MouseButton1Click)
	end)
	pcall(function()
		firesignal(button.Activated)
	end)
end

---Press the slot button (MainFrame.Slots.SlotScroll.<slot>) to select the slot and populate its
---server list.
local function pressSlot()
	local mainFrame = getMainFrame()
	local slots = mainFrame and mainFrame:FindFirstChild("Slots")
	local scroll = slots and slots:FindFirstChild("SlotScroll")
	local slotButton = scroll and currentSlot and scroll:FindFirstChild(currentSlot)
	pressButton(slotButton)
end

---Best-effort player count for a server button, parsed from its text / descendant labels. Unknown
---counts sort as 0 (treated as emptiest) -- it only affects ordering, not correctness.
---@param button Instance
---@return number
local function parsePlayers(button)
	local found = nil

	local function scan(text)
		if type(text) ~= "string" or found then
			return
		end
		local ratio = text:match("(%d+)%s*/%s*%d+") -- "12/40"
		if ratio then
			found = tonumber(ratio)
			return
		end
		local n = text:match("(%d+)")
		if n then
			found = tonumber(n)
		end
	end

	pcall(function()
		scan(button.Text)
	end)

	if not found then
		for _, descendant in next, button:GetDescendants() do
			if descendant:IsA("TextLabel") then
				pcall(function()
					scan(descendant.Text)
				end)
				if found then
					break
				end
			end
		end
	end

	return found or 0
end

---All server buttons in MainFrame.Servers.Servers -- each is a TextButton named by its JobId.
---@return table
local function serverButtons()
	local mainFrame = getMainFrame()
	local serversFrame = mainFrame and mainFrame:FindFirstChild("Servers")
	local scroll = serversFrame and serversFrame:FindFirstChild("Servers")

	local list = {}
	if scroll then
		for _, child in next, scroll:GetChildren() do
			if child:IsA("GuiButton") then
				list[#list + 1] = { id = child.Name, button = child, players = parsePlayers(child) }
			end
		end
	end
	return list
end

---Blacklist the current target so it's never picked again, and drop it so the next tick picks a
---different server.
---@param reason string
local function advanceServer(reason)
	if currentServerId then
		telemetryLog("(%s) %s -- blacklisting and switching servers.", tostring(currentServerId), reason)
		addBlacklist(currentServerId)
	end

	currentServerId = nil
	currentServerSince = nil
end

local function updateServerHopping()
	-- Throttle to once a second.
	if lastRetry and os.clock() - lastRetry <= 1.0 then
		return
	end

	lastRetry = os.clock()

	-- No servers listed yet -- press the slot button to select it and populate the list.
	local buttons = serverButtons()
	if #buttons == 0 then
		pressSlot()
		return
	end

	-- Filter out blacklisted servers, then sort by (parsed) population.
	local blacklist = getBlacklist()
	local candidates = {}
	for _, server in ipairs(buttons) do
		if not blacklist[server.id] then
			candidates[#candidates + 1] = server
		end
	end

	if #candidates == 0 then
		-- Everything currently listed is blacklisted -- re-select the slot to refresh the list.
		pressSlot()
		return
	end

	table.sort(candidates, function(first, second)
		return first.players < second.players
	end)

	-- Drop a target that got blacklisted since we picked it.
	if currentServerId and blacklist[currentServerId] then
		currentServerId = nil
	end

	-- Pick the lowest-population non-blacklisted server.
	if not currentServerId then
		currentServerId = candidates[1].id
		currentServerSince = os.clock()
		telemetryLog("(%s) Trying server (%d players).", tostring(currentServerId), candidates[1].players)
	end

	-- Current target hasn't taken us in time -- blacklist it and move on.
	if currentServerSince and os.clock() - currentServerSince >= SERVER_SWITCH_TIMEOUT then
		advanceServer("Not joined in time")
		return
	end

	-- Press the current target's button to join it (re-find it each tick in case the list rebuilt).
	local target = nil
	for _, server in ipairs(candidates) do
		if server.id == currentServerId then
			target = server
			break
		end
	end

	if target then
		pressButton(target.button)
	else
		currentServerId = nil -- the button disappeared; re-pick next tick
	end
end

local function onTeleportFailure(...)
	-- A full server (or other teleport failure) -- blacklist it and switch.
	advanceServer("Teleport failed")
end

function ServerHop.lobby()
	local shslot = PersistentData.get("shslot")
	if not shslot then
		return error("No server hop slot set in PersistentData.")
	end

	currentSlot = shslot
	currentServerId = nil
	currentServerSince = nil
	lastRetry = nil

	telemetryLog("(%s) Picking a server by pressing the menu buttons.", tostring(shslot))

	teleportService.TeleportInitFailed:Connect(onTeleportFailure)

	runService.PreRender:Connect(updateServerHopping)
end

function ServerHop.rlobby()
	telemetryLog("Returning to lobby.")

	local requests = replicatedStorage:WaitForChild("Requests")
	local returnToMenu = requests:WaitForChild("ReturnToMenu")
	local playerGui = players.LocalPlayer:WaitForChild("PlayerGui")

	while task.wait() do
		returnToMenu:FireServer()

		local prompt = playerGui:FindFirstChild("ChoicePrompt")
		local choicePrompt = prompt and prompt:FindFirstChild("Choice")

		if not choicePrompt then
			continue
		end

		choicePrompt:FireServer(true)
	end
end

---@param slot string The data slot to use.
---@param blacklist boolean If true, blacklist the current slot.
function ServerHop.hop(slot, blacklist)
	PersistentData.set("shslot", slot)

	telemetryLog("(%s, %s) Hopping to new server.", tostring(slot), tostring(blacklist))

	if blacklist then
		local blacklisted = getBlacklist()

		blacklisted[slot] = tick()

		PersistentData.set("sblacklist", blacklisted)
	end

	if game.PlaceId == LOBBY_PLACE_ID then
		return ServerHop.lobby()
	end

	return ServerHop.rlobby()
end

return ServerHop
end)

---@module Utility.Serializer
local Serializer = require("Utility/Serializer")

---@module Utility.Deserializer

---@module Utility.String
local String = require("Utility/String")

---@module Utility.Logger
local Logger = require("Utility/Logger")

local PersistentData = {
	_data = {
		fli = nil,

		shslot = nil,

		sblacklist = {},

		wdata = nil,

		efdata = nil,
	},
}

local memStorageService = game:GetService("MemStorageService")

---@param field string
---@return any
function PersistentData.get(field)
	return PersistentData._data[field]
end

---@param field string
---@param key string
---@param value any
function PersistentData.stf(field, key, value)
	local tbl = PersistentData.get(field)

	if type(tbl) ~= "table" then
		return error(string.format("PersistentData field '%s' is not a table.", tostring(field)))
	end

	tbl[key] = value

	PersistentData.set(field, tbl)
end

---@param field string
---@param value any
function PersistentData.set(field, value)
	PersistentData._data[field] = value

	local marshalSuccess, payload = pcall(Serializer.marshal, PersistentData._data)
	if not marshalSuccess then
		return Logger.warn("(%s) Failed to marshal PersistentData snapshot.", tostring(payload))
	end

	local saveSuccess, saveResult = pcall(memStorageService.SetItem, memStorageService, "LYCORIS_PERSISTENT_DATA", payload)

	if not saveSuccess then
		return Logger.warn("(%s) Failed to set PersistentData snapshot.", tostring(saveResult))
	end

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		Logger.warn("(%s) Successfully set PersistentData snapshot.", tostring(saveResult))
	end
end

function PersistentData.init()
	local hasSuccess, hasResult = pcall(memStorageService.HasItem, memStorageService, "LYCORIS_PERSISTENT_DATA")
	if not hasSuccess then
		return hasResult and Logger.warn("(%s) Failed to check for PersistentData snapshot.", tostring(hasResult))
	end

	local itemSuccess, itemResult = pcall(memStorageService.GetItem, memStorageService, "LYCORIS_PERSISTENT_DATA")
	if not itemSuccess then
		return Logger.warn("(%s) Failed to get PersistentData snapshot", tostring(itemResult))
	end

	if itemResult == nil or itemResult == "" then
		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("PersistentData snapshot is missing or empty.")
		end

		return
	end

	local success, result = pcall(Deserializer.unmarshal_one, String.tba(itemResult))
	if not success then
		return Logger.warn("(%s) Failed to deserialize PersistentData snapshot.", tostring(result))
	end

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		Logger.warn("(%s) Successfully loaded PersistentData snapshot.", tostring(result))
	end

	PersistentData._data = result
end

return PersistentData
end)

	if not sprintSpeed then
		return
	end

	local character = players.LocalPlayer.Character
	if not character then
		return
	end

	local characterHandler = character:FindFirstChild("CharacterHandler")
	local inputClient = characterHandler and characterHandler:FindFirstChild("InputClient")
	if not inputClient then
		return
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end

	local humanoid = character:FindFirstChildWhichIsA("Humanoid")
	if not humanoid then
		return
	end

	local torso = character:FindFirstChild("Torso")
	if not torso then
		return
	end

	local groundSensor = humanController.GroundSensor
	if not groundSensor then
		return
	end

	local stopDodge = KeyHandling.getRemote("StopDodge")
	if not stopDodge then
		return
	end

	local inputData = InputClient.getInputData()
	if not inputData then
		return
	end

	local repDebris = replicatedStorage:FindFirstChild("Debris")
	if not repDebris then
		return
	end

	local sounds = replicatedStorage:FindFirstChild("Sounds")
	local sliding = sounds and sounds:FindFirstChild("Sliding")
	if not sliding then
		return
	end

	local modules = replicatedStorage:FindFirstChild("Modules")
	local vectorMaths = modules and modules:FindFirstChild("VectorMaths")
	local vectorMathModule = vectorMaths and require(vectorMaths)
	if not vectorMathModule then
		return
	end

	local sensedPart = groundSensor.SensedPart

	if not effectReplicatorModule:HasAny("Swimming", "ClientSwim", "Gliding", "Knocked") then
		if not sensedPart then
			return false
		end

		if sensedPart and sensedPart.CollisionGroup == "Ship" then
			effectReplicatorModule:EnsureEffect("OnShip")
		elseif effectReplicatorModule:HasEffect("OnShip") then
			effectReplicatorModule:RemoveEffectsOfClass("OnShip")
		end
	end

	if freefallTrack.IsPlaying then
		freefallTrack:Stop(0.15)
	end

	if not effectReplicatorModule:HasEffect("Falling") then
		return true
	end

	effectReplicatorModule:RemoveEffectsOfClass("Falling")
	effectReplicatorModule:RemoveEffectsOfClass("ShipJump")
	effectReplicatorModule:RemoveEffectsOfClass("WingDashCooldown")

	humanoid.Jump = false

	effectReplicatorModule:CreateEffect("Landed"):Debris(0.05)

	if effectReplicatorModule:HasEffect("SlideJumping") and not effectReplicatorModule:HasEffect("GravityField") then
		effectReplicatorModule:RemoveEffectsOfClass("SlideJumping")

		if sprintSpeed and sprintSpeed.Value >= 9 then
			sprintSpeed.Value = sprintSpeed.Value - 3
		end

		local lookVector = root.CFrame.LookVector * 3
		local slideJump = torso:FindFirstChild("SlideJump")

		if slideJump then
			slideJump:Destroy()
		end

		if effectReplicatorModule:HasEffect("Sprinting") then
			humanoid:LoadAnimation(inputClient.SlideJumpTransition):Play(nil, 0.65)
		else
			humanoid:LoadAnimation(inputClient.LandingAnim):Play(nil, 0.65)
		end

		if slideJumpTrack.IsPlaying and effectReplicatorModule:HasEffect("Equipped") then
			slideJumpTrack:Stop(0.15)
		end

		stopDodge:FireServer(inputData, effectReplicatorModule:HasEffect("LightAttack"), true)

		local lookVectorUnitMulti = lookVector.Unit * 50

		if sensedPart and sensedPart.AssemblyMass > root.AssemblyMass then
			lookVectorUnitMulti = lookVectorUnitMulti
				+ vectorMathModule.getVelocityAtPoint(sensedPart, groundSensor.HitFrame.Position)
		end

		local bodyVelocity = Instance.new("BodyVelocity")
		bodyVelocity.Name = "SlideJump2"
		bodyVelocity.Velocity = lookVectorUnitMulti
		bodyVelocity.MaxForce = Vector3.new(80000, 80000, 80000, 0)
		bodyVelocity:AddTag("AllowedBM")
		bodyVelocity.Parent = torso

		repDebris:Fire(bodyVelocity, 0.2)

		local slidingSound = sliding:Clone()
		slidingSound.Parent = root
		slidingSound:Play()

		repDebris:Fire(slidingSound, 0.25)

		tweenService
			:Create(slidingSound, TweenInfo.new(0.25), {
				Volume = 0,
			})
			:Play()
	end

	task.delay(0.1, function()
		effectReplicatorModule:RemoveEffectsOfClass("WallJumpCD")
		effectReplicatorModule:RemoveEffectsOfClass("BigMomentum")
		task.wait(2)
	end)

	return true
end)

local freefallCheck = LPH_NO_VIRTUALIZE(function()
	if landedCheck() then
		return
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	local character = players.LocalPlayer.Character
	if not character then
		return
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end

	if effectReplicatorModule:HasEffect("OnShip") then
		effectReplicatorModule:RemoveEffectsOfClass("OnShip")
		effectReplicatorModule:EnsureEffect("ShipJump"):Debris(5, true)
	end

	effectReplicatorModule:EnsureEffect("Falling")

	local fallTime = tick() - (effectReplicatorModule:GetEffectEpoch("Falling") or 0)

	if
		effectReplicatorModule:HasAny("DashIt", "Jumped", "UsingAbility", "LightAttack", "Action")
		or root.AssemblyLinearVelocity.Y > 0
	then
		return freefallTrack.IsPlaying and freefallTrack:Stop(0.1)
	end

	if freefallTrack.IsPlaying then
		return
	end

	if fallTime > 0.5 and root.AssemblyLinearVelocity.Y < 0 then
		freefallTrack.Priority = Enum.AnimationPriority.Movement
		freefallTrack:Play(0.3)
	end
end)

InputClient.getHumanController = LPH_NO_VIRTUALIZE(function()
	---@note: Shouldn't be too many connections to PreSimulation.
	for _, connection in next, getconnections(runService.PreSimulation) do
		local func = connection.Function
		if not func then
			continue
		end

		if iscclosure(func) or isexecutorclosure(func) then
			continue
		end

		local upvalues = debug.getupvalues(func)
		if not upvalues then
			continue
		end

		for _, upvalue in next, upvalues do
			if typeof(upvalue) ~= "table" then
				continue
			end

			if upvalue.Jumping == nil then
				continue
			end

			return upvalue
		end
	end

	return nil
end)

InputClient.slide = LPH_NO_VIRTUALIZE(function()
	local character = players.LocalPlayer.Character
	if not character then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end

	local serverSlide = KeyHandling.getRemote("ServerSlide")
	if not serverSlide then
		return
	end

	local effectReplicatorModule = getEffectModule()
	if not effectReplicatorModule then
		return
	end

	local characterHashes = replicatedStorage:FindFirstChild("CharacterHashes")
	local characterHashesModule = characterHashes and require(characterHashes)
	local hashData = characterHashesModule and characterHashesModule.Snap(character)

	local healthPercent = humanoid.Health / humanoid.MaxHealth
	if hashData and hashData["Endurance Runner"] then
		healthPercent = 0.25 + healthPercent * 0.75
	end

	local agility = character:FindFirstChild("PassiveAgility")
	local agilityValue = agility and agility.Value or 0
	local isDanger = effectReplicatorModule:HasEffect("DangerPvP")
	local speed = 20 + agilityValue * (isDanger and 0.4 or 0.5) * healthPercent

	if effectReplicatorModule:HasEffect("ClientDodge") then
		speed = speed + (isDanger and 3 or 10)
	end

	if hashData then
		local kongaRing = hashData["Konga's Clutch Ring"]
		local slick = hashData.Slick

		if kongaRing then
			speed = speed + (isDanger and 5 or 15)
		end

		if slick then
			speed = speed + (isDanger and 0 or 15)
		end

		if not kongaRing and not slick and effectReplicatorModule:FindEffect("BigMomentum") then
			speed = speed + 15
		end
	end

	if effectReplicatorModule:HasEffect("AirBorne") then
		speed = speed + 15
	end

	if effectReplicatorModule:HasEffect("SpeedSlide") then
		speed = speed + 10
	elseif effectReplicatorModule:HasEffect("WardingRadiance") then
		speed = speed + (isDanger and 7.5 or 15)
	end

	local slideEffect = effectReplicatorModule:CreateEffect("ClientSlide2")
	slideEffect:Debris(60)

	if groundslideTrack then
		groundslideTrack:Play()
	end

	serverSlide:FireServer(true)

	local rootCollider = character:FindFirstChild("RootCollider")
	local colliderWeld = rootCollider and rootCollider:FindFirstChild("ColliderWeld")
	if colliderWeld then
		colliderWeld.C0 = CFrame.new(0, -1.9, 0)
	end

	local velocity = root.CFrame.LookVector * speed
	root.AssemblyLinearVelocity = velocity

	local bodyVelocity = Instance.new("BodyVelocity")
	bodyVelocity:AddTag("LocalMover")
	bodyVelocity.Name = "SlideVel"
	bodyVelocity.MaxForce = Vector3.new(100000, 0, 100000)
	bodyVelocity.Velocity = velocity
	bodyVelocity:AddTag("AllowedBM")
	bodyVelocity.Parent = root

	slideBodyVelocity = bodyVelocity
	slideActive = true

	task.spawn(function()
		while slideActive and bodyVelocity.Parent do
			task.wait(0.03)

			local isIce = humanoid.FloorMaterial == Enum.Material.Ice
				and not (
					hashData
					and (
						hashData["Steady Footing"]
						or effectReplicatorModule:HasEffect("WardingRadiance")
					)
				)

			speed = speed - (isIce and 0.25 or 1)
			if speed <= 0 then
				break
			end

			bodyVelocity.Velocity = root.CFrame.LookVector * speed
		end
	end)

	task.delay(3, function()
		if slideActive then
			InputClient.unslide()
		end
	end)
end)

---@param options DodgeOptions
InputClient.dodge = LPH_NO_VIRTUALIZE(function(options)
	-- Dodge is now a DIRECT remote (no longer keyed via KeyHandler), so getRemote("Dodge") returns nil.
	-- Resolve it by name from CharacterHandler.Requests (where the other combat remotes live), falling
	-- back to the global ReplicatedStorage.Requests.
	local dodgeCharacter = players.LocalPlayer and players.LocalPlayer.Character
	local dodgeHandler = dodgeCharacter and dodgeCharacter:FindFirstChild("CharacterHandler")
	local dodgeRequests = dodgeHandler and dodgeHandler:FindFirstChild("Requests")
	local dodge = dodgeRequests and dodgeRequests:FindFirstChild("Dodge")
	if not dodge then
		local globalRequests = replicatedStorage:FindFirstChild("Requests")
		dodge = globalRequests and globalRequests:FindFirstChild("Dodge")
	end
	if not dodge then
		return Logger.warn("Cannot dodge without dodge remote.")
	end

	local stopDodge = KeyHandling.getRemote("StopDodge")
	if not stopDodge then
		return Logger.warn("Cannot dodge without stop dodge remote.")
	end

	local inputData = InputClient.getInputData()
	if not inputData then
		return Logger.warn("Cannot dodge without input data.")
	end

	local effectReplicatorModule = getEffectModule()
	if not effectReplicatorModule then
		return Logger.warn("Cannot dodge without effect replicator module.")
	end

	if options.direct and not options.full then
		-- New Dodge signature: FireServer(dodgeType, optionsTable). The game passes
		-- { ancient, spin_attack, in_air }; the old (type, nil, nil, false) form is now wrong args.
		dodge:FireServer("roll", { ancient = false, spin_attack = effectReplicatorModule:HasEffect("SpinAttack"), in_air = false })

		task.delay(0.15, function()
			stopDodge:FireServer(inputData, effectReplicatorModule:HasEffect("LightAttack"), true)
		end)

		return
	end

	local keybinds = replicatedStorage:FindFirstChild("KeyBinds")
	local keybindsModule = keybinds and require(keybinds)
	if not keybindsModule or not keybindsModule.ForceActionDown or not keybindsModule.ForceActionUp then
		return Logger.warn("Cannot dodge without keybinds module.")
	end

	keybindsModule.ForceActionDown("Dodge")
	keybindsModule.ForceActionUp("Dodge")

	if options.rollCancel then
		task.wait(options.rollCancelDelay)

		local clientFeint = effectReplicatorModule:CreateEffect("ClientFeint")
		if clientFeint then
			clientFeint:Debris(0.1)
		end
	end
end)

---@note: I sold my soul to the GC gods because there's no other way. This basically does a ton of intensive stuff. Updates are only done when needed.
InputClient.cache = LPH_NO_VIRTUALIZE(function()
	InputClient.sprintFunctionCache = nil
	InputClient.rollFunctionCache = nil
	inputDataCache = nil

	for _, value in next, getgc() do
		if typeof(value) ~= "function" or iscclosure(value) or isexecutorclosure(value) then
			continue
		end

		local functionName = debug.getinfo(value).name
		if not functionName then
			continue
		end

		if functionName ~= "Sprint" and functionName ~= "Roll" then
			continue
		end

		local upvalues = debug.getupvalues(value)

		if not upvalues or #upvalues <= 0 then
			continue
		end

		if functionName == "Sprint" then
			InputClient.sprintFunctionCache = value

			if not LPH_OBFUSCATED or shared.Luma.verbose then
				Logger.warn("Sprint function (%s) cache successful.", tostring(value))
			end
		end

		if functionName == "Roll" then
			InputClient.rollFunctionCache = value

			if not LPH_OBFUSCATED or shared.Luma.verbose then
				Logger.warn("Roll function (%s) cache successful.", tostring(value))
			end
		end

		if InputClient.sprintFunctionCache and InputClient.rollFunctionCache then
			break
		end
	end

	local inputData = {}

	for _, connection in next, getconnections(runService.RenderStepped) do
		local func = connection.Function
		if not func then
			continue
		end

		if iscclosure(func) then
			continue
		end

		local consts = debug.getconstants(func)
		if not manualTableFind(consts, ".lastHBCheck") then
			continue
		end

		local upvalues = debug.getupvalues(func)

		---@note: Only table with boolean values is the input table. Find a better way to filter this?
		for _, upvalue in next, upvalues do
			if typeof(upvalue) ~= "table" or getrawmetatable(upvalue) then
				continue
			end

			if not hasNonBooleans(upvalue) then
				continue
			end

			if getTableLength(upvalue) >= 1 and not validateKeys(upvalue) then
				continue
			end

			inputData = upvalue
			break
		end
	end

	inputDataCache = inputData

	local assets = replicatedStorage:FindFirstChild("Assets")
	local anims = assets and assets:FindFirstChild("Anims")
	local movement = anims and anims:FindFirstChild("Movement")
	local roll = movement and movement:FindFirstChild("Roll")
	local cancelLeft = roll and roll:FindFirstChild("CancelLeft")
	local cancelRight = roll and roll:FindFirstChild("CancelRight")

	local character = players.LocalPlayer.Character
	local characterHandler = character and character:FindFirstChild("CharacterHandler")
	local inputClient = characterHandler and characterHandler:FindFirstChild("InputClient")
	local freefallAnim = inputClient and inputClient:FindFirstChild("FreefallAnim")
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	local slideJumpAnim = inputClient and inputClient:FindFirstChild("SlideJump")
	local jumpAnim = inputClient and inputClient:FindFirstChild("Jump")
	local crouchAnim = movement and movement:FindFirstChild("Crouch")
	local groundslideAnim = movement and movement:FindFirstChild("Groundslide")

	if not slideJumpAnim or not jumpAnim or not humanoid or not cancelLeft or not cancelRight or not freefallAnim then
		return false
	end

	---@note: Tracks are reset on CharacterAdded, so retries must not load a fresh copy each time.
	slideJumpTrack = slideJumpTrack or humanoid:LoadAnimation(slideJumpAnim)
	freefallTrack = freefallTrack or humanoid:LoadAnimation(freefallAnim)
	cancelLeftTrack = cancelLeftTrack or humanoid:LoadAnimation(cancelLeft)
	cancelRightTrack = cancelRightTrack or humanoid:LoadAnimation(cancelRight)
	jumpAnimTrack = jumpAnimTrack or humanoid:LoadAnimation(jumpAnim)

	if crouchAnim then
		crouchTrack = crouchTrack or humanoid:LoadAnimation(crouchAnim)
	end

	if groundslideAnim then
		groundslideTrack = groundslideTrack or humanoid:LoadAnimation(groundslideAnim)
	end

	return InputClient.sprintFunctionCache ~= nil and InputClient.rollFunctionCache ~= nil and inputDataCache ~= nil
end)

-- Re-populate the caches on the MAIN thread (serial phase), bounded + retried. This used to be driven by the
-- global `tostring` hook in Hooking.lua, but that hook is invoked from EVERY Luau VM, so the
-- getgc / getconnections / LoadAnimation work inside cache() could run during Deepwoken's parallel Actor
-- phase and Kernel-panic the client (RSL - panic: ...Kernel). A plain task.spawn keeps all of those main-VM
-- ops in the serial phase, where they're legal. cache() returns true only once the game's Sprint/Roll
-- functions and the inputData table exist, so retry until it succeeds (or a newer character supersedes this
-- run via the session bump), then STOP -- steady-state cost is zero, no periodic getgc scanning.
local cacheSession = 0
InputClient.ensureCached = LPH_NO_VIRTUALIZE(function()
	cacheSession = cacheSession + 1
	local session = cacheSession

	task.spawn(function()
		local attempts = 0
		while cacheSession == session and attempts < 40 do
			local ok, success = pcall(InputClient.cache)
			if ok and success then
				break
			end
			attempts = attempts + 1
			task.wait(0.5)
		end
	end)
end)

local characterAddedConnection = players.LocalPlayer.CharacterAdded:Connect(function()
	InputClient.sprintFunctionCache = nil
	InputClient.rollFunctionCache = nil
	inputDataCache = nil

	freefallTrack = nil
	slideJumpTrack = nil
	cancelLeftTrack = nil
	cancelRightTrack = nil
	jumpAnimTrack = nil
	crouchTrack = nil
	groundslideTrack = nil

	if slideBodyVelocity and slideBodyVelocity.Parent then
		slideBodyVelocity:Destroy()
	end

	slideBodyVelocity = nil
	slideActive = false
	crouchActive = false

	-- Re-grab Sprint/Roll/inputData + animation tracks for the new character, on the main thread.
	InputClient.ensureCached()
end)

-- Prime the cache for a character that already exists at load time (re-exec / late inject), since the
-- CharacterAdded above won't fire for it.
if players.LocalPlayer.Character then
	InputClient.ensureCached()
end

function InputClient.detach()
	cacheSession = cacheSession + 1
	characterAddedConnection:Disconnect()
end

return InputClient
end)

---@note: its parallel Actor VM the process-global hook runs there -- reflection mid-parallel-phase is the illegal

__bundle_register("Game/KeyHandling", function(require, _LOADED, __bundle_register, __bundle_modules)
local KeyHandling = {}

---@module Utility.Logger
local Logger = require("Utility/Logger")

-- KeyHandler bypass, project-rain-oss style. luma used to REIMPLEMENT the game's name-encoder
-- (a full SHA-256) and recompute keys itself -- fragile: the moment the game changes its encoding
-- or keying, every keyed remote silently fails to resolve. Instead we grab the game's OWN encoder
-- and remote table straight out of its live KeyHandler state via getgc, so we always match whatever
-- the game is doing (it's been actively reworking this -- remotes like Dodge/Sprint have left the
-- key system entirely).
--
-- The KeyHandler state table is tagged with a magic constant at index 9 ("heaven key"), and carries
-- the encoded-name -> remote map at [12] and the encoder function at [13]. The heaven_key expression
-- is reproduced verbatim from project-rain-oss so the float value matches the game's byte-for-byte.

local pi = 0.125 + tonumber("3")
local heaven_key = 2754 * (72 % (0.6769230769205024 * 7 * 7 % 5 % 64 % 41 / 66 + 12) / 54 / (4 / 36 + 5.3655660377358485) + 1 * 4 % 52 * 21 / 4 % 35 / 3) / 35 * 38 + (66 + (8 + 4 + 5 + 2 + 19035.52631578947) / 34 + 40 % 41 / 56 % 65 * 57 + 3381) + 420 + 35 * (0.3150684931506849 + 69 + 14 / 56 + 28 + 36) + (pi)

local remotes = nil -- the game's encoded-name -> remote instance table (state[12])
local encf = nil    -- the game's name-encoder function (state[13])
local cache = {}    -- remoteName -> encoded key

---Find the game's KeyHandler state in getgc (identified by the heaven_key tag at index 9) and grab
---its remotes table + encoder.
---@return boolean
local function grabKeyHandler()
	for _, tbl in next, getgc(true) do
		if typeof(tbl) ~= "table" then
			continue
		end
		if getrawmetatable(tbl) then
			continue
		end
		if #tbl ~= 13 or tbl[9] ~= heaven_key then
			continue
		end

		remotes = tbl[12]
		encf = tbl[13]
		return remotes ~= nil and encf ~= nil
	end

	return false
end

---Encode a remote name with the game's own encoder (cached).
---@param remoteName string
---@return any
local function encode(remoteName)
	local encoded = cache[remoteName]
	if encoded == nil then
		local ok, res = pcall(encf, remoteName)
		if not ok then
			return nil
		end
		encoded = res
		cache[remoteName] = encoded
	end
	return encoded
end

KeyHandling.init = LPH_NO_VIRTUALIZE(function()
	local retries = 0

	while true do
		if (remotes and encf) or grabKeyHandler() then
			break
		end

		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("KeyHandler retry (%i attempts) grabbing remotes/enc_f", retries)
		end

		retries = retries + 1

		if retries >= 10 then
			Logger.warn("KeyHandler failed to initialize after 10 attempts.")

			return error("Please report this message to the developers.")
		end

		task.wait(0.5)
	end
end)

---Resolve a keyed remote by its original name via the game's encoder.
---@param remoteName string
---@return Instance|nil
KeyHandling.getRemote = LPH_NO_VIRTUALIZE(function(remoteName)
	if not (remotes and encf) then
		return nil
	end

	local encoded = encode(remoteName)

	return (encoded ~= nil and remotes[encoded]) or nil
end)

---Resolve a keyed remote using ONLY the already-computed key cache -- it never calls the game's encoder.
---This is project-rain-oss's `get_cache`: the process-global FireServer / UnreliableFireServer hooks run on
---every remote fire the game makes, and PR only ever does pure table reads there (cache[name] -> remotes[key]).
---luma used to call getRemote() -> encode() -> pcall(encf, ...) from those hooks, which re-invokes the game's
---grabbed encoder closure on every fire for any name that has left the key system (encode never caches a nil
---result, so it retries forever). That high-frequency game-closure call from the global hook body is the
---always-on divergence from PR. Resolving the keys once up front (KeyHandling.prewarm) and reading the cache
---here keeps the hot path to pure table lookups, exactly like PR. remotes[encoded] stays live, so re-keys are
---still followed; only the encoder call is lifted out of the hook.
---@param remoteName string
---@return Instance|nil
KeyHandling.getCachedRemote = LPH_NO_VIRTUALIZE(function(remoteName)
	if not remotes then
		return nil
	end

	local encoded = cache[remoteName]
	if encoded == nil then
		return nil
	end

	return remotes[encoded] or nil
end)

---Compute and cache the keys for a set of remote names once, from the current (main, synchronized) thread.
---Call this at init so the global fire hooks can resolve those remotes with getCachedRemote without ever
---touching the encoder from the hook context.
---@param names string[]
KeyHandling.prewarm = LPH_NO_VIRTUALIZE(function(names)
	if not (remotes and encf) then
		return
	end

	for _, remoteName in ipairs(names) do
		encode(remoteName)
	end
end)

---Register a remote under its (original) name's key, then it can be renamed by the caller.
---@param remote Instance
KeyHandling.registerRemote = LPH_NO_VIRTUALIZE(function(remote)
	if not (remotes and encf) then
		return
	end

	local encoded = encode(remote.Name)
	if encoded ~= nil then
		remotes[encoded] = remote
	end
end)

return KeyHandling
end)
__bundle_register("Features/Automation/Objects/StateMachine", function(require, _LOADED, __bundle_register, __bundle_modules)
-- StateMachine: a faithful port of project-rain-oss's `src/utility/statemachine.lua`.
-- An event/callback finite state machine. Events declare from->to transitions; firing an event named
-- `foo` (via `fsm:foo(...)`) runs onbefore<foo> / onleave<from> / onenter<to> / onafter<foo> callbacks.
-- Transitions are synchronous unless a leave/enter handler returns StateMachine.ASYNC.
local StateMachine = {}
StateMachine.__index = StateMachine

local NONE = "none"
local ASYNC = "async"

local unpack = table.unpack or unpack

local function callHandler(handler, params)
	if not handler then
		return
	end
	return handler(unpack(params))
end

local function createTransition(name)
	local can, to, from, params

	local function transition(self, ...)
		if self.asyncState == NONE then
			can, to = self:can(name)
			from = self.current
			params = { self, name, from, to, ... }

			if not can then
				self.currentTransitioningEvent = nil
				return false
			end

			self.currentTransitioningEvent = name
			local beforeReturn = callHandler(self["onbefore" .. name], params)
			local leaveReturn = callHandler(self["onleave" .. from], params)

			if not self["onleave" .. from] and self.dexit then
				leaveReturn = self.dexit()
			end

			if beforeReturn == false or leaveReturn == false then
				self.currentTransitioningEvent = nil
				return false
			end

			self.asyncState = name .. "WaitingOnLeave"

			if leaveReturn ~= ASYNC then
				return transition(self, ...)
			end

			return true
		elseif self.asyncState == name .. "WaitingOnLeave" then
			self.current = to

			local enterReturn = callHandler(self["onenter" .. to] or self["on" .. to], params)

			self.asyncState = name .. "WaitingOnEnter"

			if enterReturn ~= ASYNC then
				return transition(self, ...)
			end

			return true
		elseif self.asyncState == name .. "WaitingOnEnter" then
			callHandler(self["onafter" .. name] or self["on" .. name], params)
			callHandler(self["onstatechange"], params)
			self.asyncState = NONE
			self.currentTransitioningEvent = nil
			return true
		else
			if string.find(self.asyncState, "WaitingOnLeave") or string.find(self.asyncState, "WaitingOnEnter") then
				self.asyncState = NONE
				return transition(self, ...)
			end
		end

		self.currentTransitioningEvent = nil
		return false
	end

	return transition
end

local function addToMap(map, event)
	if type(event.from) == "string" then
		map[event.from] = event.to
	else
		for _, from in ipairs(event.from) do
			map[from] = event.to
		end
	end
end

function StateMachine.create(options)
	assert(options.events)

	local fsm = {}
	setmetatable(fsm, StateMachine)

	fsm.options = options
	fsm.current = options.initial or "none"
	fsm.asyncState = NONE
	fsm.events = {}
	fsm.dexit = options.dexit

	for _, event in ipairs(options.events or {}) do
		local name = event.name
		fsm[name] = fsm[name] or createTransition(name)
		fsm.events[name] = fsm.events[name] or { map = {} }
		addToMap(fsm.events[name].map, event)
	end

	for name, callback in pairs(options.callbacks or {}) do
		fsm[name] = callback
	end

	return fsm
end

function StateMachine:is(state)
	return self.current == state
end

function StateMachine:can(e)
	local event = self.events[e]
	local to = event and event.map[self.current] or event.map["*"]
	return to ~= nil, to
end

function StateMachine:cannot(e)
	return not self:can(e)
end

function StateMachine:transition(event)
	if self.currentTransitioningEvent == event then
		return self[self.currentTransitioningEvent](self)
	end
end

function StateMachine:cancelTransition(event)
	if self.currentTransitioningEvent == event then
		self.asyncState = NONE
		self.currentTransitioningEvent = nil
	end
end

StateMachine.NONE = NONE
StateMachine.ASYNC = ASYNC

return StateMachine
end)

Tweening.update = LPH_NO_VIRTUALIZE(function(dt)
	Tweening.active = false

	local current = Tweening.current()
	if not current then
		return
	end

	local part = typeof(current.goal) == "Instance" and current.goal or nil
	if part and not part.Parent then
		return Tweening.stop(current.identifier)
	end

	local localPlayer = players.LocalPlayer
	local character = localPlayer and localPlayer.Character
	if not character then
		return
	end

	if not humanoidRootPart then
		return
	end

	local startPosition = humanoidRootPart.Position
	local targetCFrame = typeof(current.goal) == "Instance" and current.goal.CFrame or current.goal
	local targetPosition = targetCFrame.Position

	local distanceToTarget = (targetPosition - startPosition).Magnitude

	if distanceToTarget <= 0.01 then
		return
	end

	local direction = (targetPosition - startPosition) / distanceToTarget
	local moveDistance = (Configuration.expectOptionValue("TweenStudsPerSecond") or 200) * dt

	local newPosition = nil

	if moveDistance >= distanceToTarget then
		newPosition = targetPosition
	else
		newPosition = startPosition + (direction * moveDistance)
		newPosition = Vector3.new(newPosition.X, targetPosition.Y, newPosition.Z)
	end

	local rotation = (targetCFrame - targetCFrame.Position)

	local _, _, _, m00, m01, m02, m10, m11, m12, m20, m21, m22 = rotation:GetComponents()

	if
		m00 == m00
		and m01 == m01
		and m02 == m02
		and m10 == m10
		and m11 == m11
		and m12 == m12
		and m20 == m20
		and m21 == m21
		and m22 == m22
	then
		humanoidRootPart.CFrame = CFrame.new(newPosition) * rotation
	else
		humanoidRootPart.CFrame = CFrame.new(newPosition)
	end

	local newDistanceToTarget = (targetPosition - newPosition).Magnitude

	current.reached = false

	if newDistanceToTarget >= 1.0 then
		return
	end

	if current.swc then
		return Tweening.stop(current.identifier)
	end

	current.reached = true

		return

end

---@return table?
function Tweening.current()
	return Tweening.queue[#Tweening.queue]
end

---@param identifier string
---@param goal BasePart|CFrame
---@param swc boolean Whether or not to stop when we reach the goal.
function Tweening.goal(identifier, goal, swc)
	local _, data = Tweening.get(identifier)

	if data then
		data.goal = goal
		data.swc = swc
		data.reached = false
	else
		table.insert(Tweening.queue, 1, { identifier = identifier, goal = goal, swc = swc, reached = false })
	end
end

---@param identifier string
function Tweening.stop(identifier)
	local idx, _ = Tweening.get(identifier)
	if not idx then
		return
	end

	table.remove(Tweening.queue, idx)
end

return Tweening
end)
__bundle_register("Utility/Prompt", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Hardened ProximityPrompt firing.
--
-- Why this exists: bare `fireproximityprompt(prompt)` "works for some users and
-- fails for others" on the SAME executor because the failures are game-state, not
-- executor. Deepwoken runs with StreamingEnabled, so a prompt's part may not be
-- streamed in for a given client yet (slower stream = further away / lower memory /
-- lower graphics / fired too soon after a teleport), and firing against a
-- streamed-out part silently no-ops. On top of that the engine enforces
-- RequiresLineOfSight / MaxActivationDistance / Enabled / HoldDuration at trigger
-- time, and whether those bite depends on the player's camera, morph/accessories
-- obstructing the line of sight, and how the executor implements the function.
--
-- Prompt.fire() removes every one of those variables: it requests streaming around
-- the prompt, forces the prompt open (LOS off, huge distance, no hold, enabled),
-- fires, then restores the originals so the game's own UI/logic is untouched.
local Prompt = {}

local players = game:GetService("Players")

---Request the world region around a part to stream in, so the part is resident
---(has physics + a valid trigger) before we fire. No-op on non-streaming games.
---@param part BasePart?
local function ensureStreamedIn(part)
	if not part then
		return
	end
	local localPlayer = players.LocalPlayer
	if not localPlayer then
		return
	end
	pcall(function()
		if localPlayer.RequestStreamingAround then
			localPlayer:RequestStreamingAround(part.Position)
		end
	end)
end

---Fire a ProximityPrompt reliably.
---@param prompt Instance? A ProximityPrompt, or any instance to search under for one.
---@return boolean fired
function Prompt.fire(prompt)
	if not prompt then
		return false
	end

	-- Accept a model/part and dig out the prompt so callers don't have to.
	if not prompt:IsA("ProximityPrompt") then
		local found = prompt:FindFirstChildWhichIsA("ProximityPrompt", true)
		if not found then
			return false
		end
		prompt = found
	end

	local part = prompt:FindFirstAncestorWhichIsA("BasePart") or prompt.Parent
	if typeof(part) == "Instance" and part:IsA("BasePart") then
		ensureStreamedIn(part)
	end

	local fireProx = fireproximityprompt

	local fired = false
	pcall(function()
		-- Snapshot the properties the engine checks at trigger time and force them
		-- into the state that always allows a fire. ProximityPrompt property writes
		-- from the client are local-only (never replicate), so this only affects our
		-- own trigger path and is restored immediately after.
		local origEnabled = prompt.Enabled
		local origLOS = prompt.RequiresLineOfSight
		local origDist = prompt.MaxActivationDistance
		local origHold = prompt.HoldDuration

		prompt.Enabled = true
		prompt.RequiresLineOfSight = false
		prompt.MaxActivationDistance = math.max(origDist, 1000)
		prompt.HoldDuration = 0

		if fireProx then
			fireProx(prompt)
			fired = true
		elseif prompt.InputHoldBegin then
			prompt:InputHoldBegin()
			task.wait(origHold or 0)
			prompt:InputHoldEnd()
			fired = true
		end

		prompt.Enabled = origEnabled
		prompt.RequiresLineOfSight = origLOS
		prompt.MaxActivationDistance = origDist
		prompt.HoldDuration = origHold
	end)

	return fired
end

-- =========================================================================================================
-- Interact-key press (used by the automations instead of firing ProximityPrompts directly).
--
-- Rather than fireproximityprompt(), the automations press the real interact key -- like a human -- after
-- they've tweened to the target. The key is configurable under the Automations panel ("Interact Key",
-- default E). We still stream the target's part in and read its HoldDuration so a hold-prompt registers.
-- =========================================================================================================
local DEFAULT_INTERACT_KEY = Enum.KeyCode.E

local function resolveInteractKey()
	local picker = Options and Options.AutomationInteractKey
	local name = picker and picker.Value
	if type(name) == "string" and name ~= "" and name ~= "None" and name ~= "N/A" then
		local ok, code = pcall(function()
			return Enum.KeyCode[name]
		end)
		if ok and code then
			return code
		end
	end
	return DEFAULT_INTERACT_KEY
end

---Press the configured interact key. `prompt` is optional, used only to stream the target in and read its
---HoldDuration (so a hold-prompt is held long enough); the actual interaction is the key press.
---@param prompt Instance?
---@return boolean pressed
function Prompt.interact(prompt)
	local holdDuration = 0

	if typeof(prompt) == "Instance" then
		local resolved = prompt
		if not resolved:IsA("ProximityPrompt") then
			resolved = resolved:FindFirstChildWhichIsA("ProximityPrompt", true)
		end
		if resolved and resolved:IsA("ProximityPrompt") then
			holdDuration = resolved.HoldDuration or 0
			local part = resolved:FindFirstAncestorWhichIsA("BasePart")
			if part then
				ensureStreamedIn(part)
			end
		elseif prompt:IsA("BasePart") then
			ensureStreamedIn(prompt)
		end
	end

	local key = resolveInteractKey()
	local pressed = false
	pcall(function()
		local vim = Instance.new("VirtualInputManager")
		vim:SendKeyEvent(true, key, false, game)
		task.wait(holdDuration + 0.06)
		vim:SendKeyEvent(false, key, false, game)
		pressed = true
	end)
	return pressed
end

return Prompt
end)

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

---Deposit at the guild bank, then push the overflow into the guild treasury.
local function depositNotesAtGuild()
	placeGuildDoor()
	if not running then
		return
	end

	local guildAttr = localPlayer():GetAttribute("Guild")
	if not guildAttr then
		return
	end
	local guildName = tostring(guildAttr)
		:lower()
		:gsub("%s*%[.-%]%s*", " ")
		:gsub("^%d+", "")
		:gsub("z%w+divider", "")
		:gsub("[^%a%d]", "")

	local banker, treasurer
	for _, npc in next, workspace.NPCs:GetChildren() do
		if banker and treasurer then
			break
		end
		if npc:IsA("Model") and npc:GetAttribute("Guild") and tostring(npc:GetAttribute("Guild")):match(guildName) then
			if npc.Name:match("Banker") then
				banker = npc
			elseif npc.Name:match("Treasurer") then
				treasurer = npc
			end
		end
	end

	local data = PersistentData.get("notesdata")
	if banker and banker:FindFirstChild("HumanoidRootPart") and not (data and data.maxNotes) then
		setStage("guild bank")
		tempTween(banker.HumanoidRootPart.CFrame * CFrame.new(5, 0, 0), 170)
		openBankAndDeposit()
		task.wait(3.5)
		if currentNotes() > 0 then
			PersistentData.stf("notesdata", "maxNotes", true)
		end
	end

	if currentNotes() > 0 and treasurer and treasurer:FindFirstChild("HumanoidRootPart") then
		setStage("guild treasury")
		tempTween(treasurer.HumanoidRootPart.CFrame * CFrame.new(5, 0, 0), 170)
		repeat
			pcall(function()
				Prompt.interact(treasurer:FindFirstChildOfClass("ProximityPrompt"))
			end)
			task.wait(0.5)
		until dialogueOpen() or not running
		dialogueChoice("I'd like to add money to the treasury. [Add to Treasury]")
		task.wait(0.5)
		dialogueChoice(20000)
		task.wait(0.5)
		dialogueExit()
	end
end

local function overworld()
	repeat
		task.wait()
	until character() or not running
	shareKnowledge()
	if not running then
		return
	end
	if currentNotes() > 0 then
		if useGuild() then
			depositNotesAtGuild()
		else
			depositNotesAtBank()
		end
	end
end

-- =========================================================================================================
-- Fragments cash-in (Self [The End]) -- same shape as Echo, no echo counting needed here.
-- =========================================================================================================
local function handleFragments()
	setStage("fragments")
	local self_npc
	local deadline = os.clock() + 40
	repeat
		local npcs = workspace:FindFirstChild("NPCs")
		self_npc = npcs and npcs:FindFirstChild("Self")
		task.wait()
	until self_npc or os.clock() >= deadline or not running
	if not self_npc then
		serverHop()
		return
	end

	local tpLocation = (self_npc:GetPivot() - Vector3.new(0, 15.7, 0))
		* CFrame.Angles(math.rad(180), math.rad(90), math.rad(180))
	if playerCheck(200, tpLocation.Position) then
		serverHop()
		while task.wait() do
		end
		return
	end

	local sendDialogue = replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue")
	local prompt = self_npc:WaitForChild("InteractPrompt", 10)
	local start = tick()
	repeat
		local r = root()
		if r then
			r.CFrame = tpLocation
		end
		if prompt then
			Prompt.interact(prompt)
		end
		pcall(function()
			sendDialogue:FireServer({ choice = "[The End]" })
		end)
		task.wait(0.25)
	until (workspace:FindFirstChild("CharacterCreator") and character() and character().Parent == workspace.CharacterCreator)
		or not (currentArea() and tostring(currentArea()):match("Fragments of Self"))
		or not running
		or tick() - start > 20
end

-- =========================================================================================================
-- State machine.
-- =========================================================================================================
local function isCharacterCreation()
	local cc = workspace:FindFirstChild("CharacterCreator")
	local c = character()
	return cc ~= nil and c ~= nil and c.Parent == cc
end

local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "overworld", from = "_check_area", to = "_overworld" },
		{ name = "fragments", from = "_check_area", to = "_handle_fragments" },
		{ name = "die", from = "_check_area", to = "_die" },
		{ name = "cc", from = "_check_area", to = "_create_character" },
		{ name = "continuation", from = "_create_character", to = "_check_area" },
		{ name = "continuation", from = "_handle_fragments", to = "_check_area" },
		{ name = "continuation", from = "_overworld", to = "_check_area" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")

			local area = currentArea()
			if area and tostring(area):match("Fragments of Self") then
				return self:fragments()
			end

			if isCharacterCreation() then
				return self:cc()
			end

			if area and tostring(area):match("The Depths") then
				return self:die()
			end

			if game.PlaceId == MAIN_GAME_PLACE_ID then
				-- If we're in a guild base, respawn to the Inn first.
				if area and tostring(area):find("Base") then
					General.respawn()
					repeat
						task.wait()
					until (playerGui() and playerGui():FindFirstChild("SpawnPrompt")) or not running
					pcall(function()
						playerGui().SpawnPrompt.Choice:FireServer("Inn")
					end)
					repeat
						task.wait()
					until character() or not running
				end
				return self:overworld()
			end
		end,
		onenter_create_character = function(self)
			if not running then
				return
			end
			setStage("create character")
			struct:create_character()
			self:continuation()
		end,
		onenter_handle_fragments = function(self)
			if not running then
				return
			end
			handleFragments()
			self:continuation()
		end,
		onenter_overworld = function(self)
			if not running then
				return
			end
			setStage("overworld")
			overworld()
			if not running then
				return
			end
			-- Wipe and reroll.
			local slot = currentSlot() or (PersistentData.get("notesdata") and PersistentData.get("notesdata").slot)
			pcall(function()
				Wipe.invoke(slot)
			end)
		end,
		onenter_die = function(self)
			if not running then
				return
			end
			local slot = currentSlot() or (PersistentData.get("notesdata") and PersistentData.get("notesdata").slot)
			pcall(function()
				Wipe.invoke(slot)
			end)
		end,
	},
})

-- =========================================================================================================
-- Struct.
-- =========================================================================================================
local function buildStruct()
	return AutomationStruct.construct({
		id = "auto_notes",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"no_fall",
			"noclip",
			"mod_detector",
			"fly",
			"auto_equip_weapon",
			"auto_log",
		},
		character_creator_handler_used = true,
		character_creator_handler_opts = {
			origin = "Etris",
		},
	})
end

-- Public interface.
-- =========================================================================================================

---Menu Start: wipe the current character first (like Echo Farm), then Luma's resume (notesdata) drops us
---into a fresh farm cycle.
function AutoNotes.invoke()
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end
	local slot = currentSlot()
	if not slot then
		return Logger.notify("Auto Notes: no data slot found.")
	end

	PersistentData.set("notesdata", {
		slot = slot,
		startedAt = os.time(),
	})

	-- Wipe the current character on start; the wipe teleports us out and the resume brings us back in.
	pcall(function()
		Wipe.invoke(slot)
	end)
end

function AutoNotes.start()
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end
	local slot = currentSlot()
	if not slot then
		return Logger.notify("Auto Notes: no data slot found.")
	end

	local data = PersistentData.get("notesdata") or {}
	data.slot = slot
	data.startedAt = data.startedAt or os.time()
	PersistentData.set("notesdata", data)

	running = true
	buildOverlay()
	AntiAFK.start("AutoNotes")

	maid:mark(task.spawn(runElapsed))

	stateMachine.current = "idle"
	stateMachine.asyncState = StateMachine.NONE
	stateMachine.currentTransitioningEvent = nil

	struct = buildStruct()
	struct:run()
end

function AutoNotes.stop()
	if not running then
		return
	end
	running = false
	if struct then
		pcall(function()
			struct:stop()
		end)
		struct = nil
	end
	Tweening.stop(MOVE_ID)
	pcall(function()
		AntiAFK.stop("AutoNotes")
	end)
	maid:clean()
	destroyOverlay()
	PersistentData.set("notesdata", nil)
end

return AutoNotes
end)

			local vim = Instance.new("VirtualInputManager")

			vim:SendKeyEvent(false, keyCode, false, game)
		end)

		local pg = players.LocalPlayer and players.LocalPlayer:FindFirstChildOfClass("PlayerGui")
		if not (pg and pg:FindFirstChild("LoadingGui")) then
			break
		end
		pressKey(Enum.KeyCode.A)
		task.wait(0.2)
	end

	Wipe.pwait()

	telemetryLog("(Depths) Looking for anyone in the Fragments of Self area.")

	local wdata = PersistentData.get("wdata")
	if not wdata then
		return error("No wipe data set in PersistentData.")
	end

	if Finder.pnear(FRAGMENTS_OF_SELF_POS, 500) then
		return ServerHop.hop(wdata.slot, true)
	end

	repeat
		task.wait()
	until wipeSlot:InvokeServer(wdata.slot)

	ServerHop.hop(wdata.slot, false)

		return Wipe.lobby()
	end

	return ServerHop.rlobby()
end

---@module Utility.Logger

local replicatedStorage = game:GetService("ReplicatedStorage")

	if not DEBUGGING_MODE then

	end

	Logger.warn(...)

---@param tool Instance
---@param data any Response to interaction to send with the tool use.
function Interactions.utool(tool, data)
	Interactions.etool(tool)

	telemetryLog("Activating tool '%s' until ChoicePrompt appears.", tool.Name)

	local playerGui = players.LocalPlayer:WaitForChild("PlayerGui")
	local choicePrompt = nil

	while not choicePrompt do
		tool:Activate()

		choicePrompt = playerGui:FindFirstChild("ChoicePrompt")

		task.wait()
	end

	telemetryLog("Firing ChoicePrompt ('%s') with data.", tostring(data))

	local choice = choicePrompt:WaitForChild("Choice")
	choice:FireServer(data)

	telemetryLog("Waiting for ChoicePrompt to disappear...")

	repeat
		task.wait()
	until not choicePrompt.Parent

	telemetryLog("ChoicePrompt has disappeared.")
end

---@param tool Instance
function Interactions.etool(tool)
	local character = players.LocalPlayer.Character
	if not character then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	repeat
		humanoid:EquipTool(tool)

		task.wait()
	until tool.Parent == players.LocalPlayer.Character
end

---@param instance Model
---@param actions table
---@param teleport boolean Should we teleport to the instance?
function Interactions.interact(instance, actions, teleport)
	if not instance or not instance.Parent then
		return
	end

	local interactPrompt = instance:FindFirstChild("InteractPrompt")
	if not interactPrompt or not interactPrompt:IsA("ProximityPrompt") then
		return
	end

	telemetryLog("(%s) Interacting until dialogue appears.", instance.Name)

	local playerGui = players.LocalPlayer:WaitForChild("PlayerGui")
	local dialogueGui = playerGui:WaitForChild("DialogueGui")
	local character = players.LocalPlayer.Character or players.LocalPlayer.CharacterAdded:Wait()

	repeat
		if teleport then
			character:PivotTo(instance:GetPivot())
		end

		Prompt.fire(interactPrompt)

		task.wait()
	until dialogueGui.Enabled

	telemetryLog("(%s) Going through dialogue with %i actions.", instance.Name, #actions)

	local requests = replicatedStorage:WaitForChild("Requests")
	local dialogueEvent = requests:WaitForChild("SendDialogue")

	local dialogueCompleted = false

	dialogueEvent.OnClientEvent:Connect(function()
		dialogueCompleted = true
	end)

	for _, action in next, actions do
		dialogueEvent:FireServer(action)

		dialogueCompleted = false

		telemetryLog(
			"(%s) Sent dialogue action '%s' and now waiting for OnClientEvent to fire.",
			instance.Name,
			action.exit and "[EXIT]" or (action.choice or "N/A")
		)

		if action.exit then
			break
		end

		repeat
			task.wait()
		until dialogueCompleted
	end

	telemetryLog("(%s) Completed dialogue.", instance.Name)
end

return Interactions
end)

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

	end

		if player ~= localPlayer() and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			if (origin - player.Character:GetPivot().Position).Magnitude <= 200 then
				return true
			end
		end
	end
	return false
end
local function playersNear(target, mag)
	for _, player in next, players:GetPlayers() do
		if player ~= localPlayer() and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			if (target - player.Character:GetPivot().Position).Magnitude <= mag then
				return true
			end
		end
	end
	return false
end
local function serverHop()
	pcall(function()
		ServerHop.hop(currentSlot(), true)
	end)
end
local function playerSafeTween(cf, speed)
	if playerNear() then
		serverHop()
		while task.wait() do
		end
		return
	end
	local r = root()
	if not r then
		return
	end
	local y = math.random(8000, 10000)
	r.CFrame = r.CFrame + Vector3.new(0, y, 0)
	tempTween(cf + Vector3.new(0, y, 0), speed)
	if playersNear(cf.Position, 200) then
		serverHop()
		while task.wait() do
		end
		return
	end
	r = root()
	if r then
		r.CFrame = cf
	end
	task.wait(1)
end
local function readyWeapon()
	pcall(function()
		character().CharacterHandler.Requests.DrawWeapon:FireServer(true)
	end)
end
local function m1Hold(state)
	local feat = struct and struct.active_features and struct.active_features["m1_hold"]
	if feat then
		feat.held = state
	end
end
local function healthCheck(percent)
	local hum = humanoid()
	return hum ~= nil and hum.Health <= (hum.MaxHealth * percent)
end
local function inDanger()
	local hum = humanoid()
	return hum ~= nil and hum:GetAttribute("DangerExpiration") and hum:GetAttribute("DangerExpiration") > 0
end
local function getClosestMob(name, mag)
	local closest
	local live = workspace:FindFirstChild("Live")
	if not live then
		return nil
	end
	for _, mob in next, live:GetChildren() do
		if mob:IsA("Model") and mob.Name:find(name) and mob:FindFirstChild("HumanoidRootPart") then
			local dist = (root().Position - mob.HumanoidRootPart.Position).Magnitude
			if not closest then
				if not mag or dist < mag then
					closest = mob
				end
			elseif dist < (root().Position - closest.HumanoidRootPart.Position).Magnitude then
				closest = mob
			end
		end
	end
	return closest
end
local function getClosestCampfire()
	local closest
	local thrown = workspace:FindFirstChild("Thrown")
	if not thrown then
		return nil
	end
	for _, campfire in next, thrown:GetChildren() do
		if campfire.Name:match("Campfire") and campfire:IsA("Model") then
			local part = campfire:FindFirstChildOfClass("Part")
			if part then
				if not closest then
					closest = part
				elseif (root().Position - part.Position).Magnitude < (root().Position - closest.Position).Magnitude then
					closest = part
				end
			end
		end
	end
	return closest
end
local function grabChest()
	local thrown = workspace:FindFirstChild("Thrown")
	if not thrown then
		return
	end
	for _, chest in next, thrown:GetChildren() do
		if chest:FindFirstChild("Lid") then
			tempTween(chest.Lid.CFrame, 170)
			local t = tick()
			repeat
				task.wait(0.5)
				pcall(function()
					Prompt.interact(chest:FindFirstChildWhichIsA("ProximityPrompt"))
				end)
				pcall(function()
					tempTween(chest.Lid.CFrame, 170)
				end)
			until (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) or not running or tick() - t > 3 or playerNear()
			if playerNear() then
				break
			end
			local t2 = tick()
			repeat
				task.wait(1)
			until not (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) or not running or tick() - t2 > 3 or playerNear()
			break
		end
	end
end

-- =========================================================================================================
-- Kill a Moon Knight: attach behind it, m1, dodge its kick (fly up when the kick anim plays), then vacuum
-- nearby LootDrops.
-- =========================================================================================================
local function killTarget(target)
	if playerNear() or not target or not target:IsA("Model") then
		return false
	end
	if target.Name:match(localPlayer().Name) then
		return false
	end

	readyWeapon()
	if target:FindFirstChild("HumanoidRootPart") then
		tempTween(target.HumanoidRootPart.CFrame, 170)
	end
	if playerNear() then
		return false
	end

	local foundKick = false
	local animator = target:FindFirstChild("Animator", true)
	local kickConn
	if animator then
		kickConn = animator.AnimationPlayed:Connect(function(track)
			local id = track.Animation.AnimationId
			if (id:match("106711913879378") or id:match("111024000122473")) and not foundKick then
				foundKick = true
				task.delay(2, function()
					foundKick = false
				end)
			end
		end)
	end

	-- Attach behind the knight (fly up when it winds up a kick).
	local attach = task.spawn(function()
		while running and not playerNear() do
			if not target.Parent or not target:FindFirstChild("HumanoidRootPart") then
				break
			end
			local r = root()
			if r then
				r.AssemblyLinearVelocity = Vector3.zero
				r.CFrame = target.HumanoidRootPart.CFrame
					* CFrame.new(0, foundKick and 60 or 10, 0)
					* CFrame.Angles(math.rad(-90), 0, 0)
			end
			task.wait()
		end
	end)

	-- Track health stalls (bail if no damage for 20s).
	local healthTick = tick()
	local lastHealth = nil
	local status = task.spawn(function()
		while running do
			pcall(function()
				local hp = target.Humanoid.Health
				if lastHealth == nil or lastHealth ~= hp then
					healthTick = tick()
					lastHealth = hp
				end
			end)
			task.wait()
		end
	end)

	m1Hold(true)
	repeat
		task.wait()
	until not running
		or not target
		or target.Parent == nil
		or playerNear()
		or healthCheck(0.35)
		or (tick() - healthTick >= 20)

	pcall(function()
		if kickConn then
			kickConn:Disconnect()
		end
	end)
	pcall(function()
		task.cancel(attach)
	end)
	pcall(function()
		task.cancel(status)
	end)
	m1Hold(false)

	-- Vacuum LootDrops.
	if running and not playerNear() then
		task.wait(2)
		local thrown = workspace:FindFirstChild("Thrown")
		for _, drop in next, (thrown and thrown:GetChildren() or {}) do
			if drop:IsA("Model") and drop:FindFirstChild("LootDrop") then
				if (root().Position - drop.LootDrop.Position).Magnitude <= 100 and not playersNear(drop.LootDrop.Position, 200) then
					tempTween(drop.LootDrop.CFrame, 120)
					local t = tick()
					repeat
						pcall(function()
							firetouchinterest(drop.LootDrop, root(), 0)
							firetouchinterest(drop.LootDrop, root(), 1)
						end)
						task.wait()
					until drop.Parent == nil or playerNear() or not running or tick() - t >= 3
					if drop.Parent ~= nil then
						pcall(function()
							drop:Destroy()
						end)
					end
				end
			end
		end
		task.wait(0.2)
	end

	if healthCheck(0.35) then
		return false
	end
	if playerNear() or not running then
		return true
	end
	return false
end

-- =========================================================================================================
-- Main handle: heal if low, go to Moon's Eyrie, open the door, kill knights + loot.
-- =========================================================================================================
local function handle()
	-- Heal at a campfire if we're low.
	if healthCheck(0.35) then
		setStage("resting at campfire")
		playerSafeTween(
			CFrame.new(-7632.27881, 980.369873, 421.907501, -0.95457083, 0, 0.297984123, 0, 1, 0, -0.297984123, 0, -0.95457083),
			170
		)
		task.wait(1)
		local campfire = getClosestCampfire()
		local er = getEffectReplicator()
		if campfire then
			repeat
				local r = root()
				if r then
					r.CFrame = campfire.CFrame * CFrame.new(0, 1, 0)
				end
				task.wait(1)
				pcall(function()
					Prompt.interact(campfire.Parent.InteractPrompt)
				end)
				task.wait(1)
			until (er and er:FindEffect("Resting")) or not running
			repeat
				task.wait()
			until (humanoid() and humanoid().Health >= humanoid().MaxHealth * 0.5) or not running
			-- Stand up (E).
			local vim = Instance.new("VirtualInputManager")
			pcall(function()
				vim:SendKeyEvent(true, Enum.KeyCode.E, false, game)
				task.wait(0.1)
				vim:SendKeyEvent(false, Enum.KeyCode.E, false, game)
			end)
			task.wait(1)
			vim:Destroy()
		end
	end

	if not running then
		return
	end

	-- Travel to Moon's Eyrie.
	setStage("traveling to Moon's Eyrie")
	local marker = replicatedStorage:FindFirstChild("MarkerWorkspace")
	local location = marker
		and marker.AreaMarkers
		and marker.AreaMarkers:FindFirstChild("Moon's Eyrie")
		and marker.AreaMarkers["Moon's Eyrie"].AreaMarker.CFrame
	if not location then
		return
	end
	playerSafeTween(location, 170)

	local door
	repeat
		task.wait()
		door = workspace:FindFirstChild("MoonseyeDoor")
	until door or not running
	if not running then
		return
	end

	readyWeapon()
	playerSafeTween(door.CFrame * CFrame.new(0, -5, 0), 170)

	-- Open the Moon door.
	setStage("opening Moon door")
	repeat
		pcall(function()
			local moonDoor = workspace.NPCs:FindFirstChild("MoonDoor")
			if moonDoor then
				Prompt.interact(moonDoor:FindFirstChild("InteractPrompt"))
			end
		end)
		task.wait(0.5)
	until (playerGui() and playerGui():FindFirstChild("DialogueGui") and playerGui().DialogueGui.Enabled) or not running
	pcall(function()
		replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue"):FireServer({ choice = "[Interact]" })
	end)

	tempTween(door.CFrame * CFrame.new(0, 0, -100), 170)

	-- Wait for a Moon Knight to appear.
	local waited = 0
	repeat
		task.wait(0.1)
		waited = waited + 0.1
	until waited > 3 or playerNear() or getClosestMob("moonknight", 60) or not running
	if playerNear() or not running then
		return
	end

	setStage("fighting Moon Knight")
	local knight = getClosestMob("moonknight", 60)
	local knightWasNil = knight == nil

	killTarget(knight)

	if running and not playerNear() and not healthCheck(0.35) and not knightWasNil then
		task.wait(0.5)
		grabChest()
	end

	-- A second knight may be further out.
	local knight2 = getClosestMob("moonknight", 3000)
	if knight2 then
		killTarget(knight2)
	end
end

local function toSafety()
	setStage("hiding")
	repeat
		local r = root()
		if r then
			r.CFrame = r.CFrame * CFrame.new(0, math.random(8000, 10000), 0)
		end
		task.wait()
	until not inDanger() or not running
end

-- =========================================================================================================
-- State machine.
-- =========================================================================================================
local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "moons", from = "_check_area", to = "_moonseyrie" },
		{ name = "hide", from = "_check_area", to = "_hide" },
		{ name = "hide", from = "_moonseyrie", to = "_hide" },
		{ name = "hop", from = "_moonseyrie", to = "_hop" },
		{ name = "hop", from = "_hide", to = "_hop" },
		{ name = "continuation", from = "_hide", to = "_moonseyrie" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")
			if game.PlaceId == DEPTHS_PLACE_ID then
				Logger.notify("Auto Mooneyrie: ended up in the Depths -- stopping.")
				return AutoMooneyrie.stop()
			end
			if game.PlaceId == EASTERN_PLACE_ID then
				return self:moons()
			end
		end,
		onenter_moonseyrie = function(self)
			if not running then
				return
			end
			handle()
			if playerNear() or inDanger() then
				return self:hide()
			end
			self:hop()
		end,
		onenter_hide = function(self)
			if not running then
				return
			end
			toSafety()
			self:hop()
		end,
		onenter_hop = function(self)
			if not running then
				return
			end
			serverHop()
			while task.wait() do
			end
		end,
	},
})

-- =========================================================================================================
-- Struct.
-- =========================================================================================================
local function buildStruct()
	return AutomationStruct.construct({
		id = "auto_mooneyrie",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"no_fall",
			"noclip",
			"fly",
			"no_fire",
			"m1_hold",
			"no_stun",
			"fast_swing",
			"auto_equip_weapon",
			"auto_log",
		},
	})
end

-- =========================================================================================================
-- Public interface.
-- =========================================================================================================
function AutoMooneyrie.start()
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end
	if not AutoMooneyrie.inValidPlace() then
		return Logger.notify("Auto Mooneyrie: not in the Eastern Luminant.")
	end
	local slot = currentSlot()
	if not slot then
		return Logger.notify("Auto Mooneyrie: no data slot found.")
	end

	local data = PersistentData.get("mooneyriedata") or {}
	data.slot = slot
	data.startedAt = data.startedAt or os.time()
	PersistentData.set("mooneyriedata", data)

	running = true
	buildOverlay()
	AntiAFK.start("AutoMooneyrie")

	maid:mark(task.spawn(runElapsed))

	stateMachine.current = "idle"
	stateMachine.asyncState = StateMachine.NONE
	stateMachine.currentTransitioningEvent = nil

	struct = buildStruct()
	struct:run()
end

function AutoMooneyrie.stop()
	if not running then
		return
	end
	running = false
	if struct then
		pcall(function()
			struct:stop()
		end)
		struct = nil
	end
	Tweening.stop(MOVE_ID)
	pcall(function()
		AntiAFK.stop("AutoMooneyrie")
	end)
	maid:clean()
	destroyOverlay()
	PersistentData.set("mooneyriedata", nil)
end

return AutoMooneyrie

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

			elapsedLabel.Text = "Elapsed: " .. formatDuration(math.max(0, os.time() - (data.startedAt or os.time())))
		end
		task.wait(0.5)
	end
end

---Blocking tween (PR Tween.new(cf, true, speed).wait()).
local function tempTween(cframe, speed)
	setSpeed(speed or 250)
	Tweening.goal(MOVE_ID, cframe, false)
	local goalPos = cframe.Position
	while running do
		local r = root()
		if r and (r.Position - goalPos).Magnitude <= 4 then
			break
		end
		task.wait()
	end
	Tweening.stop(MOVE_ID)
end

end

local function playerNear()
	local origin = root() and root().Position
	if not origin then
		return false
	end
	for _, player in next, players:GetPlayers() do
		if player ~= localPlayer() and player.Character and player.Character:FindFirstChild("HumanoidRootPart") then
			if (origin - player.Character:GetPivot().Position).Magnitude <= 1000 then
				return true
			end
		end
	end
	return false
end

local function serverHop()
	pcall(function()
		ServerHop.hop(currentSlot(), true)
	end)
end

---PR's player_safe_tween: hop if anyone near, else go up to altitude, travel, drop.
local function playerSafeTween(cf, speed)
	if playerNear() then
		serverHop()
		while task.wait() do
		end
		return
	end
	local r = root()
	if not r then
		return
	end
	local y = math.random(8000, 10000)
	r.CFrame = r.CFrame + Vector3.new(0, y, 0)
	tempTween(cf + Vector3.new(0, y, 0), speed)
	if playerNear() then
		serverHop()
		while task.wait() do
		end
		return
	end
	r = root()
	if r then
		r.CFrame = cf
	end
	task.wait(1)
end

local function m1Hold(state)
	local feat = struct and struct.active_features and struct.active_features["m1_hold"]
	if feat then
		feat.held = state
	end
end

local function readyWeapon()
	pcall(function()
		character().CharacterHandler.Requests.DrawWeapon:FireServer(true)
	end)
end

local function playerGuiHasChoice()
	local lp = localPlayer()
	local gui = lp and lp:FindFirstChildOfClass("PlayerGui")
	return gui and gui:FindFirstChild("ChoicePrompt") ~= nil
end

local function grabChests()
	local looted = false
	local thrown = workspace:FindFirstChild("Thrown")
	if not thrown then
		return false
	end
	for _, chest in next, thrown:GetChildren() do
		if not running then
			break
		end
		if chest:FindFirstChild("Lid") and not collectionService:HasTag(chest, "looted") then
			tempTween(chest.Lid.CFrame, 170)
			local t = tick()
			repeat
				task.wait(1)
				pcall(function()
					Prompt.interact(chest:FindFirstChildWhichIsA("ProximityPrompt"))
				end)
				pcall(function()
					tempTween(chest.Lid.CFrame, 170)
				end)
			until (playerGuiHasChoice()) or not running or tick() - t > 3
			if playerGuiHasChoice() then
				looted = true
			end
			local t2 = tick()
			repeat
				task.wait(1)
			until not playerGuiHasChoice() or not running or tick() - t2 > 3
			if playerGuiHasChoice() then
				looted = true
				pcall(function()
					localPlayer().PlayerGui.ChoicePrompt.Choice:FireServer("EXIT")
				end)
			end
			collectionService:AddTag(chest, "looted")
		end
	end
	return looted
end

-- =========================================================================================================
-- Refill food (Etrean ingredients).
-- =========================================================================================================
local function refill()
	setStage("refill food")
	local char = character()
	if not char then
		return
	end
	local water = char:FindFirstChild("Water")
	if not water then
		return
	end

	local function isSatiated()
		return water.Value >= (water.MaxValue * 0.5)
	end

	local passives = char:GetAttribute("ssv_Passives")
	if passives and tostring(passives):find("Carnivore") then
		return
	end

	local function closestIngredient(namePart)
		local closest
		local folder = workspace:FindFirstChild("Ingredients")
		if not folder then
			return nil
		end
		for _, ing in next, folder:GetChildren() do
			if ing.Name:match(namePart) then
				if not closest then
					closest = ing
				elseif (root().Position - ing.Position).Magnitude < (root().Position - closest.Position).Magnitude then
					closest = ing
				end
			end
		end
		return closest
	end

	local function collect(ing)
		playerSafeTween(ing.CFrame, 200)
		local cf = ing.CFrame
		repeat
			tempTween(cf, 200)
			pcall(function()
				Prompt.interact(ing:FindFirstChildOfClass("ProximityPrompt"))
			end)
			task.wait()
		until ing.Parent == nil or not running
	end

	local function eat(foodName)
		local item = backpack() and backpack():FindFirstChild(foodName)
		if not item then
			return
		end
		humanoid():EquipTool(item)
		item:Activate()
		task.wait(0.5)
		humanoid():UnequipTools()
	end

	local start = tick()
	while running and not isSatiated() and tick() - start <= 45 do
		local ing = closestIngredient("Dentifilo") or closestIngredient("Calabash")
		if not ing then
			break
		end
		local name = ing.Name
		collect(ing)
		eat(name)
		task.wait()
	end
end

-- =========================================================================================================
-- Create instance (open manor, clear adds, ride elevator into the Duke dungeon).
-- =========================================================================================================
local function createInstance()
	setStage("opening manor")
	local bp = backpack()
	if not bp then
		return
	end
	local manorKey = bp:WaitForChild("Manor Key", 10) or (character() and character():WaitForChild("Manor Key", 10))
	if not manorKey then
		Logger.notify("Auto Duke: need a Manor Key.")
		return
	end
	manorKey.Parent = character()
	local handle = manorKey:WaitForChild("Handle", 10)
	if not handle then
		return
	end

	-- The key has to be "charged" (handle turns white) before it works. If it isn't, wait up to 10s for it
	-- to change; if it still isn't white, this slot's key is spent -> hop to a fresh server.
	if handle.Color.R < 0.99 or handle.Color.G < 0.99 or handle.Color.B < 0.99 then
		local start = tick()
		local old = handle.Color
		repeat
			task.wait()
		until handle.Color ~= old or tick() - start > 10 or not running
		if handle.Color.R < 0.99 or handle.Color.G < 0.99 or handle.Color.B < 0.99 then
			return serverHop()
		end
	end

	playerSafeTween(CFrame.new(1460, 74, 3069), 200)

	local started = false
	task.delay(20, function()
		if not started and running then
			serverHop()
		end
	end)

	maid:mark(workspace.Live.ChildAdded:Connect(function(entity)
		if (entity.Name:find("brainsucker") or entity.Name:find("dukecultist")) then
			if (entity:GetPivot().Position - (root() and root().Position or Vector3.zero)).Magnitude <= 400 then
				started = true
			end
		end
	end))

	setStage("entering manor")
	while running and not started do
		if sethiddenproperty and humanoid() then
			pcall(function()
				sethiddenproperty(humanoid(), "MoveDirectionInternal", Vector3.one)
			end)
		end
		pcall(function()
			Prompt.interact(workspace.Doors.ErisiaDungeonGate.Main.InteractPrompt)
		end)
		task.wait()
	end

	readyWeapon()
	m1Hold(true)

	setStage("clearing adds")
	local done = false
	while running and not done do
		done = true
		for _, entity in next, workspace.Live:GetChildren() do
			if entity.Name:find("brainsucker") or entity.Name:find("dukecultist") then
				if (entity:GetPivot().Position - (root() and root().Position or Vector3.zero)).Magnitude <= 900 then
					local hum = entity:FindFirstChildOfClass("Humanoid")
					while running and hum and hum.Health > 0 and entity:FindFirstChild("HumanoidRootPart") do
						if playerNear() then
							serverHop()
							while task.wait() do
							end
							return
						end
						m1Hold(true)
						tweenGoal(entity.HumanoidRootPart.CFrame * CFrame.new(0, 0, 3), 250)
						task.wait()
					end
					done = false
				end
			end
		end
	end
	m1Hold(false)
	Tweening.stop(MOVE_ID)

	task.delay(30, function()
		if running then
			serverHop()
		end
	end)

	setStage("riding elevator")
	playerSafeTween(CFrame.new(1470, 84, 3586), 200)
	for _ = 1, 300 do
		if not running then
			break
		end
		if playerNear() then
			serverHop()
			while task.wait() do
			end
			return
		end
		tempTween(CFrame.new(1470, 84, 3586), 250)
		pcall(function()
			Prompt.interact(workspace.ErisiaElevator.Switch.InteractPrompt)
		end)
		task.wait(0.1)
	end
end

-- =========================================================================================================
-- Kill the Duke.
-- =========================================================================================================
local function killDuke()
	setStage("fighting duke")
	readyWeapon()
	m1Hold(true)

	-- Dodge the golem laser (GolemLaserFire client effect).
	local nopeTfOut = false
	local clientEffect
	repeat
		clientEffect = replicatedStorage:FindFirstChild("ClientEffect", true)
		if not clientEffect then
			task.wait()
		end
	until clientEffect or not running
	if clientEffect then
		maid:mark(clientEffect.OnClientEvent:Connect(function(effectName)
			if effectName == "GolemLaserFire" then
				nopeTfOut = true
				task.delay(0.4, function()
					nopeTfOut = false
				end)
			end
		end))
	end

	-- Phase 1: clear the prime golems / remaining cultists (Z > 2000, i.e. in the arena).
	local done = false
	while running and not done do
		done = true
		for _, entity in next, workspace.Live:GetChildren() do
			if not running then
				break
			end
			if (entity.Name:find("prime_golem") or entity.Name:find("dukecultist")) then
				if (entity:GetPivot().Position - (root() and root().Position or Vector3.zero)).Magnitude <= 900 and entity:GetPivot().Z >= 2000 then
					local hum = entity:FindFirstChildOfClass("Humanoid")
					while running and hum and hum.Health > 0 do
						m1Hold(true)
						if not entity:FindFirstChild("HumanoidRootPart") then
							break
						end
						if not nopeTfOut then
							tweenGoal(
								entity.HumanoidRootPart.CFrame
									* (entity.Name:find("prime_golem") and CFrame.new(0, 5, 0.35) or CFrame.new(0, 2, 3)),
								250
							)
						else
							tweenGoal(entity.HumanoidRootPart.CFrame * CFrame.new(0, 100, 10), 250)
						end
						task.wait()
					end
					done = false
				end
			end
		end
	end
	m1Hold(false)
	if not running then
		return
	end

	-- Phase 2: start the Duke dialogue -- fire the interact prompt on .theduke11 until dialogue opens.
	local sendDialogue = replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue")
	local got = false
	maid:mark(sendDialogue.OnClientEvent:Connect(function(v)
		got = not (v and v.exit)
	end))

	local vim = Instance.new("VirtualInputManager")
	repeat
		local duke = workspace.Live:FindFirstChild(DUKE_NAME)
		if duke and duke:FindFirstChild("HumanoidRootPart") then
			tempTween(duke.HumanoidRootPart.CFrame * CFrame.new(0, 0, 3), 250)
			pcall(function()
				Prompt.interact(duke.InteractPrompt)
			end)
		end
		pcall(function()
			vim:SendKeyEvent(true, Enum.KeyCode.One, false, game)
			task.wait()
			vim:SendKeyEvent(false, Enum.KeyCode.One, false, game)
		end)
		task.wait()
	until got or not running
	-- Spam the accept key while the dialogue is open.
	while running and got do
		pcall(function()
			vim:SendKeyEvent(true, Enum.KeyCode.One, false, game)
			task.wait()
			vim:SendKeyEvent(false, Enum.KeyCode.One, false, game)
		end)
		task.wait()
	end

	-- Wait for the Duke to actually target us (fight starts).
	repeat
		task.wait()
		local duke = workspace.Live:FindFirstChild(DUKE_NAME)
		if duke and duke:FindFirstChild("Target") and duke.Target.Value then
			break
		end
	until not running
	readyWeapon()

	-- Detect the Duke's dangerous animations.
	task.delay(5, function()
		local duke = workspace.Live:FindFirstChild(DUKE_NAME)
		if duke and duke:FindFirstChildOfClass("Humanoid") then
			maid:mark(duke.Humanoid.AnimationPlayed:Connect(function(anim)
				local id = anim.Animation.AnimationId
				if id:find("8275408994") or id:find("8290899374") or id:find("8294560344") then
					nopeTfOut = true
					task.delay(id:find("8294560344") and 7.5 or 2.5, function()
						nopeTfOut = false
					end)
				end
			end))
		end
	end)

	-- Void mode?
	local voidingMode = Toggles and Toggles.DukeVoids and Toggles.DukeVoids.Value == true
	if Toggles and Toggles.VoidMobs then
		Toggles.VoidMobs:SetValue(voidingMode)
	end

	-- Main Duke loop.
	local duke = workspace.Live:FindFirstChild(DUKE_NAME)
	while running and workspace.Live:FindFirstChild(DUKE_NAME) do
		task.wait()
		duke = workspace.Live:FindFirstChild(DUKE_NAME)
		m1Hold(not General.playing_ap_anims(duke))

		if not duke:FindFirstChild("HumanoidRootPart") or not duke:FindFirstChild("Humanoid") or not duke.Parent or duke.Humanoid.Health <= 0 then
			break
		end

		-- Void mode: cast Frost Grab while safe.
		if voidingMode and not nopeTfOut and not General.playing_ap_anims(duke) then
			local frost = backpack() and backpack():FindFirstChild("Mantra:ChokeIce{{Frost Grab}}")
			if frost then
				pcall(function()
					character().CharacterHandler.Requests.ActivateMantra:FireServer(frost)
				end)
			end
		end

		local hrp = duke.HumanoidRootPart
		if hrp.CFrame.Y <= -1000 then
			continue -- void can fail; don't chase into the void
		end

		local x, y, z = hrp.CFrame:GetComponents()
		local xSin = math.sin(tick() * 2) * 30
		local zCos = math.cos(tick() * 2) * 30

		if General.playing_ap_anims(duke, { "rbxassetid://8085349676" }) or nopeTfOut then
			local critical = KeyHandling.getRemote("CriticalClick")
			if critical then
				pcall(function()
					critical:FireServer({ S = false, NOAERIALS = false, Space = false, Right = false, W = false, Left = true }, false)
				end)
			end
			tweenGoal(CFrame.new(x + xSin, y < 70 and math.random(175, 250) or math.random(0, 10), z + zCos), 250)
		elseif General.playing_ap_anims(duke) then
			tweenGoal(CFrame.new(x + xSin, y < 70 and math.random(175, 250) or math.random(0, 10), z + zCos), 250)
		else
			tweenGoal(hrp.CFrame * CFrame.new(0, -6, 0) * CFrame.Angles(math.rad(90), 0, 0), 250)
		end
	end
	m1Hold(false)
	Tweening.stop(MOVE_ID)
	if not running then
		return
	end

	-- Wait out danger, then loot the chests and respawn out.
	setStage("looting")
	task.delay(60 * 5, function()
		if not running then
			return
		end
		local er = getEffectReplicator()
		repeat
			task.wait()
			tempTween(CFrame.new(2495, 139, 1926), 160)
		until not (er and er:FindEffect("Danger")) or not running
		General.respawn()
	end)

	while running do
		local ok, res = pcall(grabChests)
		if ok and res then
			break
		end
		task.wait(1)
	end

	local er = getEffectReplicator()
	repeat
		task.wait()
		tempTween(CFrame.new(2495, 139, 1926), 160)
	until not (er and er:FindEffect("Danger")) or not running
	General.respawn()
end

-- =========================================================================================================
-- State machine.
-- =========================================================================================================
local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "refill", from = "_check_area", to = "_refill" },
		{ name = "initiate", from = "_check_area", to = "_initiate" },
		{ name = "initiate", from = "_refill", to = "_initiate" },
		{ name = "kill", from = "_check_area", to = "_kill" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")

			-- Somehow ended up in the Depths -> stop (PR kicks here).
			if game.PlaceId == DEPTHS_PLACE_ID then
				Logger.notify("Auto Duke: ended up in the Depths -- stopping.")
				return AutoDuke.stop()
			end

			local dungeon = localPlayer() and localPlayer():GetAttribute("Dungeon")
			if dungeon and tostring(dungeon):match("Duke") then
				return self:kill()
			end

			if game.PlaceId == EASTERN_PLACE_ID then
				-- Wrong region -- teleport back to the Etrean Luminant (manor is there).
				Teleports.etrean()
				return
			end

			if game.PlaceId == MAIN_GAME_PLACE_ID then
				self:refill()
				self:initiate()
				return
			end
		end,
		onenter_refill = function(self)
			if not running then
				return
			end
			refill()
			self:initiate()
		end,
		onenter_initiate = function(self)
			if not running then
				return
			end
			createInstance()
		end,
		onenter_kill = function(self)
			if not running then
				return
			end
			killDuke()
		end,
	},
})

-- =========================================================================================================
-- Struct.
-- =========================================================================================================
local function buildStruct()
	return AutomationStruct.construct({
		id = "auto_duke",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"no_fall",
			"noclip",
			"no_kill_bricks",
			"mod_detector",
			"fly",
			"m1_hold",
			"auto_wisp",
			"auto_equip_weapon",
			"auto_log",
		},
	})
end

-- =========================================================================================================
-- Public interface.
-- =========================================================================================================
function AutoDuke.start()
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end
	if not AutoDuke.inValidPlace() then
		return Logger.notify("Auto Duke: not in the Etrean/Eastern Luminant.")
	end
	local slot = currentSlot()
	if not slot then
		return Logger.notify("Auto Duke: no data slot found.")
	end

	local data = PersistentData.get("dukedata") or {}
	data.slot = slot
	data.startedAt = data.startedAt or os.time()
	PersistentData.set("dukedata", data)

	running = true
	buildOverlay()
	AntiAFK.start("AutoDuke")

	maid:mark(task.spawn(runElapsed))

	stateMachine.current = "idle"
	stateMachine.asyncState = StateMachine.NONE
	stateMachine.currentTransitioningEvent = nil

	struct = buildStruct()
	struct:run()
end

function AutoDuke.stop()
	if not running then
		return
	end
	running = false
	if struct then
		pcall(function()
			struct:stop()
		end)
		struct = nil
	end
	Tweening.stop(MOVE_ID)
	pcall(function()
		AntiAFK.stop("AutoDuke")
	end)
	maid:clean()
	destroyOverlay()
	PersistentData.set("dukedata", nil)
end

return AutoDuke
end)
