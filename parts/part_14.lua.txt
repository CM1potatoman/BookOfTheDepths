local FakeVoid = {}

local Signal = require("Utility/Signal")

local Logger = require("Utility/Logger")

local Configuration = require("Utility/Configuration")

local runService = game:GetService("RunService")

local fakeVoidMaid = Maid.new()

local preSimulation = Signal.new(runService.PreSimulation)
local workspaceDescendantAdded = Signal.new(workspace.DescendantAdded)
local workspaceDescendantRemoving = Signal.new(workspace.DescendantRemoving)

local TRIGGER_HEIGHT = 100
local DEPTHS_ROOF_Y = -600
local DEPTHS_RAY_LENGTH = 1000
local ROOF_OFFSET = 3
local TELEPORT_COOLDOWN = 1.0

local killPlanes = {}

local lastTeleport = 0
local initialized = false

local function isKillPlane(instance)
	return instance:IsA("BasePart") and (instance.Name == "KillPlane" or instance.Name:match("Chasm") ~= nil)
end

local function getBounds(part)
	local cframe = part.CFrame
	if cframe.Y ~= cframe.Y or math.abs(cframe.Y) == math.huge then
		return nil
	end

	local size = part.Size
	local right, up, look = cframe.RightVector, cframe.UpVector, cframe.LookVector
	local extents = Vector3.new(
		math.abs(right.X) * size.X + math.abs(up.X) * size.Y + math.abs(look.X) * size.Z,
		math.abs(right.Y) * size.X + math.abs(up.Y) * size.Y + math.abs(look.Y) * size.Z,
		math.abs(right.Z) * size.X + math.abs(up.Z) * size.Y + math.abs(look.Z) * size.Z
	) / 2

	return { min = cframe.Position - extents, max = cframe.Position + extents }
end

local function onWorkspaceDescendantAdded(descendant)
	if not isKillPlane(descendant) then
		return
	end

	killPlanes[descendant] = getBounds(descendant) or false
end

local function onWorkspaceDescendantRemoving(descendant)
	killPlanes[descendant] = nil
end

local function getKillPlaneY(position)
	local fpdh = workspace:GetAttribute("OriginalFPDH") or workspace.FallenPartsDestroyHeight
	local best = (typeof(fpdh) == "number" and fpdh == fpdh) and fpdh or nil

	for part, cached in next, killPlanes do
		local data = getBounds(part) or cached
		if not data then
			continue
		end

		killPlanes[part] = data

		if position.X < data.min.X or position.X > data.max.X then
			continue
		end

		if position.Z < data.min.Z or position.Z > data.max.Z then
			continue
		end

		if not best or data.max.Y > best then
			best = data.max.Y
		end
	end

	return best
end

local function getDepthsRoofY(position)
	local map = workspace:FindFirstChild("Map")
	if not map then
		return DEPTHS_ROOF_Y
	end

	local params = RaycastParams.new()
	params.FilterDescendantsInstances = { map }
	params.FilterType = Enum.RaycastFilterType.Include

	local result = workspace:Raycast(
		Vector3.new(position.X, DEPTHS_ROOF_Y, position.Z),
		Vector3.new(0, -DEPTHS_RAY_LENGTH, 0),
		params
	)

	return result and result.Position.Y + ROOF_OFFSET or DEPTHS_ROOF_Y
end

local function updateFakeVoid()
	if not Configuration.expectToggleValue("FakeVoid") then
		return
	end

	if os.clock() - lastTeleport <= TELEPORT_COOLDOWN then
		return
	end

	local character = players.LocalPlayer and players.LocalPlayer.Character
	local rootPart = character and character:FindFirstChild("HumanoidRootPart")
	if not rootPart then
		return
	end

	local position = rootPart.Position

	local killPlaneY = getKillPlaneY(position)

	if not killPlaneY or position.Y - killPlaneY > TRIGGER_HEIGHT then
		return
	end

	local targetY = getDepthsRoofY(position)
	if targetY - killPlaneY <= TRIGGER_HEIGHT then
		return
	end

	lastTeleport = os.clock()

	Logger.warn(
		"(FakeVoid) Teleporting from Y %.1f to Y %.1f (kill Y %.1f).",
		position.Y,
		targetY,
		killPlaneY
	)

	rootPart.CFrame = rootPart.CFrame + Vector3.new(0, targetY - position.Y, 0)
	rootPart.AssemblyLinearVelocity = rootPart.AssemblyLinearVelocity * Vector3.new(1, 0, 1)
end

function FakeVoid.init()
	if initialized then
		return
	end
	initialized = true

	for _, descendant in next, workspace:GetDescendants() do
		onWorkspaceDescendantAdded(descendant)
	end

	fakeVoidMaid:add(
		workspaceDescendantAdded:connect("FakeVoid_WorkspaceDescendantAdded", onWorkspaceDescendantAdded)
	)
	fakeVoidMaid:add(
		workspaceDescendantRemoving:connect("FakeVoid_WorkspaceDescendantRemoving", onWorkspaceDescendantRemoving)
	)
	fakeVoidMaid:add(preSimulation:connect("FakeVoid_PreSimulation", updateFakeVoid))

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		local count = 0

		for _ in next, killPlanes do
			count = count + 1
		end

		Logger.warn("FakeVoid initialized with %d kill planes.", count)
	end
end

function FakeVoid.detach()
	-- Guard: a startup toggle callback (config load firing OnChanged(false) before FakeVoid was ever
	-- started) must NOT run a spurious detach. The ungated Logger.warn below leaked the "[Luma Recode]"
	-- fingerprint into LogService for release users (where every other log is gated off) and got them
	-- flagged by the anticheat's log scan.
	if not initialized then
		return
	end
	initialized = false

	fakeVoidMaid:clean()
	table.clear(killPlanes)

	if not LPH_OBFUSCATED or (shared.Luma and shared.Luma.verbose) then
		Logger.warn("FakeVoid detached.")
	end
end

return FakeVoid
end)

---@param groupbox table
function AutomationTab.initAutoBuilderSection(groupbox)
	-- The tab only holds the link box + load button. The loaded build's stats, talents and mantras
	-- (preshrine on the left, postshrine on the right) live in a separate window that Load opens.
	local linkInput = groupbox:AddInput("AutoBuilderLink", {
		Text = "Builder Link",
		Placeholder = "deepwoken.co/builder?id=...",
	})

	groupbox:AddButton("Load Build", function()
		local build = AutoBuilder.load(linkInput.Value)
		local classified = AutoBuilder.classified

		if build and classified then
			AutoBuilderWindow.show(build, classified)
		end
	end)

	-- A Mantra Hand always contains a special "Roll 2" card that grants 2 Talents instead of a Mantra.
	-- With this on, the builder always takes Roll 2 on mantra hands rather than picking a mantra.
	groupbox:AddToggle("AutoBuilderOnlyRoll2", {
		Text = "Only Roll 2 For Mantras",
		Default = false,
		Tooltip = "On a Mantra Hand, always take the 'Roll 2' card (2 Talents instead of a Mantra). Talent choices are unaffected.",
	})

	-- Start/Stop button. Reads "Start Preshrine"/"Start Postshrine" per the selected target (double-
	-- click to confirm), and flips to "Stop" once picking is running.
	local startButton
	local function startLabel()
		return AutoBuilder.target == "post" and "Start Postshrine" or "Start Preshrine"
	end

	local function applyState()
		if AutoBuilder.picking then
			startButton.Text = "Stop"
			startButton.Label.Text = "Stop"
			startButton.DoubleClick = false
			startButton.Func = AutoBuilder.stopPicking
		else
			local label = startLabel()
			startButton.Text = label
			startButton.Label.Text = label
			startButton.DoubleClick = true
			startButton.Func = AutoBuilder.startPicking
		end
	end

	startButton = groupbox:AddButton({
		Text = startLabel(),
		DoubleClick = true,
		Func = AutoBuilder.startPicking,
	})

	-- Keep the button in sync with target changes (from the window) and start/stop (incl. a hop
	-- resume that flips picking on without this button being pressed).
	AutoBuilder.onStateChanged = applyState

	-- Reflect any state that a server-hop resume may have already set before the menu was built.
	applyState()
end

	addTweenSpeedSlider(groupbox, "TitusTweenSpeed")

	-- Relic names to bank when we hold >= 25 of them (leave empty to never bank).
	local bankList = groupbox:AddDropdown("TitusBankItems", {
		Text = "Bank Items (>=25)",
		-- Pre-populated with the real relic item names (exact Backpack Tool names, same list PR banks) so
		-- they always match countItem(). The old empty list forced you to hand-type a name, and any typo /
		-- wrong case / missing apostrophe silently never matched -- so it never banked. Use the input +
		-- "Add Bank Item" below only for relics outside this standard set.
		Values = { "Moonseye Tome", "Idol of Yun'Shul", "Armorer's Needle", "Smith's Alloy" },
		Default = { "Moonseye Tome", "Idol of Yun'Shul", "Armorer's Needle", "Smith's Alloy" },
		SaveValues = true,
		Multi = true,
		AllowNull = true,
	})

	local bankInput = groupbox:AddInput("TitusBankItemInput", {
		Text = "Bank Item Name",
		Placeholder = "Exact relic name.",
	})

	groupbox:AddButton("Add Bank Item", function()
		local itemName = bankInput.Value
		if #itemName <= 0 then
			return Logger.notify("Please enter a valid relic name.")
		end
		-- Snapshot the current selection so adding a custom relic doesn't wipe what's already ticked.
		local selected = {}
		for selectedName in next, bankList.Value do
			selected[selectedName] = true
		end

		local values = bankList.Values
		if not table.find(values, itemName) then
			table.insert(values, itemName)
		end
		bankList:SetValues(values)

		-- Keep the previous selection and also tick the item we just added.
		selected[itemName] = true
		bankList:SetValue(selected)
		bankList:Display()
	end)

	groupbox:AddButton("Remove Selected Bank Items", function()
		local values = bankList.Values
		for selected, _ in next, bankList.Value do
			local index = table.find(values, selected)
			if index then
				table.remove(values, index)
			end
		end
		bankList:SetValues(values)
		bankList:SetValue({})
		bankList:Display()
	end)
end

-- Minimal modal confirm popup (luma's Library has no dialog primitive): dims the screen, shows a title +
-- message with Confirm / Cancel, and calls back with a boolean. Parented on top of Library.ScreenGui.
local function confirmDialog(titleText, bodyText, onResult)
	local backdrop = Library:Create("Frame", {
		Name = "LumaConfirmDialog",
		BackgroundColor3 = Color3.new(0, 0, 0),
		BackgroundTransparency = 0.45,
		BorderSizePixel = 0,
		Position = UDim2.fromScale(0, 0),
		Size = UDim2.fromScale(1, 1),
		ZIndex = 999990,
		Active = true,
		Parent = Library.ScreenGui,
	})

	local dialog = Library:Create("Frame", {
		AnchorPoint = Vector2.new(0.5, 0.5),
		Position = UDim2.fromScale(0.5, 0.5),
		Size = UDim2.fromOffset(360, 160),
		BackgroundColor3 = Library.MainColor,
		BorderColor3 = Library.OutlineColor,
		BorderMode = Enum.BorderMode.Inset,
		ZIndex = 999991,
		Active = true,
		Parent = backdrop,
	})
	Library:AddToRegistry(dialog, { BackgroundColor3 = "MainColor", BorderColor3 = "OutlineColor" })

	Library:Create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(10, 12),
		Size = UDim2.new(1, -20, 0, 22),
		Font = Enum.Font.GothamBold,
		Text = titleText,
		TextColor3 = Library.AccentColor,
		TextSize = 18,
		TextXAlignment = Enum.TextXAlignment.Center,
		ZIndex = 999992,
		Parent = dialog,
	})

	Library:Create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.fromOffset(14, 44),
		Size = UDim2.new(1, -28, 0, 58),
		Font = Enum.Font.Gotham,
		Text = bodyText,
		TextColor3 = Library.FontColor,
		TextSize = 15,
		TextWrapped = true,
		TextXAlignment = Enum.TextXAlignment.Center,
		TextYAlignment = Enum.TextYAlignment.Top,
		ZIndex = 999992,
		Parent = dialog,
	})

	local done = false
	local function finish(result)
		if done then
			return
		end
		done = true
		pcall(function()
			backdrop:Destroy()
		end)
		Library:SafeCallback(onResult, result)
	end

	local function makeButton(text, xScale, isAccent)
		return Library:Create("TextButton", {
			AnchorPoint = Vector2.new(0.5, 1),
			Position = UDim2.new(xScale, 0, 1, -14),
			Size = UDim2.fromOffset(150, 32),
			BackgroundColor3 = isAccent and Library.AccentColor or Library.MainColor,
			BorderColor3 = Library.OutlineColor,
			BorderMode = Enum.BorderMode.Inset,
			Font = Enum.Font.Gotham,
			Text = text,
			TextColor3 = Library.FontColor,
			TextSize = 15,
			AutoButtonColor = true,
			ZIndex = 999993,
			Parent = dialog,
		})
	end

	makeButton("Confirm", 0.28, true).MouseButton1Click:Connect(function()
		finish(true)
	end)
	makeButton("Cancel", 0.72, false).MouseButton1Click:Connect(function()
		finish(false)
	end)
end

---@param detectionBox table
---@param thresholdsBox table
function CombatTab.initAntiBreakerSection(detectionBox, thresholdsBox)
	local toggle = detectionBox:AddToggle("AntiBreakerEnabled", {
		Text = "Anti-AP Breaker",
		Default = false,
	})

	toggle:AddKeyPicker(
		"AntiBreakerKeybind",
		{ Default = "N/A", SyncToggleState = true, Text = "Anti-AP Breaker" }
	)

	-- Confirm before enabling: Anti-AP Breaker degrades your AP, so require explicit acceptance. We snap
	-- the toggle back off immediately and only re-enable it if the user picks Confirm.
	local programmatic = false
	local dialogOpen = false
	toggle:OnChanged(function(state)
		if programmatic or not state then
			return
		end
		programmatic = true
		toggle:SetValue(false)
		programmatic = false
		if dialogOpen then
			return
		end
		dialogOpen = true
		confirmDialog("Anti-AP Breaker", "THIS WILL MAKE AP SIGNIFICANTLY WORSE", function(confirmed)
			dialogOpen = false
			if confirmed then
				programmatic = true
				toggle:SetValue(true)
				programmatic = false
			end
		end)
	end)

	-- Notifies you when a nearby player is flagged as trying to AP-break you (ported from Mira).
	detectionBox:AddToggle("AntiBreakerNotify", {
		Text = "Ap Break Notifications",
		Default = false,
	})
end

---@param callback function
---@return function
local function timingOverrideCallback(callback)
	return function()
		local overrideData = Library.OverrideData
		if not overrideData then
			return false
		end

		local selectedTo = Configuration.expectOptionValue("TimingOverrideList")
		if not selectedTo or #selectedTo == 0 then
			return false
		end

		local override = overrideData[selectedTo]
		if not override then
			return false
		end

		callback(override)
	end
end

local function refreshTimingOverrideList()
	local overrideData = Library.OverrideData
	if not overrideData then
		return
	end

	local timingList = Options["TimingOverrideList"]
	if not timingList then
		return
	end

	local values = {}

	for rname, _ in next, overrideData do
		values[#values + 1] = rname
	end

	timingList:SetValues(values)
end

---@param groupbox table
function CombatTab:initTimingsSection(groupbox)
	self.tol = groupbox:AddDropdown("TimingOverrideList", {
		Text = "Timing Override List",
		Values = {},
		SaveValues = false,
		Multi = false,
		AllowNull = true,
		Searchable = true,
		Callback = function()
			self.dashInsteadOfParryRate:SetRawValue(0)
			self.failureRate:SetRawValue(0)
			self.ignoreAnimationEndRate:SetRawValue(0)
			self.reactToFeintsRate:SetRawValue(0)

			local overrideData = Library.OverrideData
			if not overrideData then
				return false
			end

			local selectedTo = Configuration.expectOptionValue("TimingOverrideList")
			if not selectedTo or #selectedTo == 0 then
				return false
			end

			local override = overrideData[selectedTo]
			if not override then
				return false
			end

			self.dashInsteadOfParryRate:SetRawValue(override.dipr or 0)
			self.failureRate:SetRawValue(override.fr or 0)
			self.ignoreAnimationEndRate:SetRawValue(override.iaer or 0)
			self.reactToFeintsRate:SetRawValue(override.rtfr or 0)
		end,
	})

	groupbox:AddDivider()

	local names = {}
	local plist = { SaveManager.as, SaveManager.es, SaveManager.ss, SaveManager.ps }

	---@todo: There should be a specific method for this; I mean as far as I'm aware THERE IS ONE. But, I'm too lazy.
	for _, pair in next, plist do
		for _, timing in next, pair:list() do
			names[#names + 1] = timing.name
		end
	end

	table.sort(names)

	self.ti = groupbox:AddDropdown("TimingOverrideTimingList", {
		Text = "Timing List",
		Values = names,
		Multi = false,
		AllowNull = true,
		Searchable = true,
	})

	groupbox:AddButton("Add Timing Override", function()
		if Library.OverrideData[self.ti.Value] then
			return Logger.notify("A timing override with that name already exists.")
		end

		local override = {
			fr = self.failureRate.Value,
			dipr = self.dashInsteadOfParryRate.Value,
			iaer = self.ignoreAnimationEndRate.Value,
			rtfr = self.reactToFeintsRate.Value,
		}

		Library.OverrideData[self.ti.Value] = override

		refreshTimingOverrideList()

		self.tol:SetValue(self.ti.Value)
		self.tol:Display()
	end)

	groupbox:AddButton("Remove Selected Override", function()
		Library.OverrideData[self.tol.Value] = nil

		refreshTimingOverrideList()

		self.tol:SetValue(nil)
		self.tol:Display()
	end)

	refreshTimingOverrideList()
end

__bundle_register("Features/Combat/SilentAim", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Silent Aim (ported from Project Rain).
--
-- Deepwoken resolves where your attacks/abilities go by asking the CLIENT two things:
--   1. Requests.GetCameraToMouse (RemoteFunction) -- "what CFrame is your camera->mouse aim?"
--   2. Requests.UpdateMouse (RemoteEvent, fired every frame) -- "where is your mouse pointing in world?"
-- We override (1) to hand back a CFrame aimed at the target under the FOV, and we fire (2) ourselves with
-- the target's predicted position. The game's own UpdateMouse is suppressed in Game/Hooking.lua's namecall
-- hook while this is on, so only our redirected position reaches the server.
--
-- When the toggle is off (or no target is in the FOV) the GetCameraToMouse override returns the exact same
-- value the stock client returns -- CFrame.lookAt(camera, real mouse hit) -- so it is behaviourally
-- identical to vanilla and safe to leave installed.

---@module Utility.Configuration
local Configuration = require("Utility/Configuration")

---@module Game.Latency
local Latency = require("Game/Latency")

---@module Game.PlayerScanning
local PlayerScanning = require("Game/PlayerScanning")

---@module GUI.Library
local Library = require("GUI/Library")

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local RunService = game:GetService("RunService")
local GuiService = game:GetService("GuiService")

local SilentAim = {}

local localPlayer = Players.LocalPlayer

local TARGET_REFRESH = 1 / 8

local running = false
local installed = false
local renderConn = nil
local characterConn = nil
local currentTarget = nil
local lastTargetRefresh = 0
local lastSentPosition = nil

local fovCircle = nil
local outlineCircle = nil

local cachedEffectModule = nil

---@return any
local function getEffectModule()
	if cachedEffectModule then
		return cachedEffectModule
	end
	local effectReplicator = ReplicatedStorage:FindFirstChild("EffectReplicator")
	cachedEffectModule = effectReplicator and require(effectReplicator) or nil
	return cachedEffectModule
end

---@return boolean
local function hasCameraLock()
	local module = getEffectModule()
	if not module then
		return false
	end
	local ok, result = pcall(function()
		return module:HasEffect("CameraLock")
	end)
	return (ok and result) and true or false
end

---Strip NaN out of numbers/Vectors/CFrames so a bad target position can't desync us or trip a kick.
---@param value any
---@return any
local function safeNAN(value)
	if value == value then
		return value
	end

	if typeof(value) == "Vector3" then
		return Vector3.new(safeNAN(value.X), safeNAN(value.Y), safeNAN(value.Z))
	elseif typeof(value) == "CFrame" then
		local position = safeNAN(value.Position)
		local rotation = value.Rotation
		if rotation ~= rotation then
			rotation = CFrame.identity
		end
		return CFrame.new(position) * rotation
	elseif typeof(value) == "number" then
		return value == value and value or 0
	end

	return value
end

---@param entity Model
---@return BasePart?
local function getPrimary(entity)
	return entity.PrimaryPart or entity:FindFirstChild("HumanoidRootPart")
end

---Lead a target by one round-trip, scaled by the prediction slider.
---@param target Model
---@return Vector3?
local function predictedPosition(target)
	local primary = getPrimary(target)
	if not primary then
		return nil
	end
	local prediction = (Configuration.expectOptionValue("SilentAimPrediction") or 0) / 100
	local lead = primary.AssemblyLinearVelocity * (Latency.rtt() or 0) * prediction
	return primary.Position + lead
end

---The CFrame handed back to the server for GetCameraToMouse, aimed at the target.
---@param target Model
---@return CFrame?
local function getGCTM(target)
	local camera = workspace.CurrentCamera
	local aimPos = predictedPosition(target)
	if not camera or not aimPos then
		return nil
	end
	return safeNAN(CFrame.lookAt(camera.CFrame.Position, aimPos))
end

---Pick the entity in workspace.Live whose box falls inside the FOV circle around the cursor.
---@return Model?
local function getTarget()
	local live = workspace:FindFirstChild("Live")
	if not live then
		return nil
	end

	local camera = workspace.CurrentCamera
	local mouse = localPlayer and localPlayer:GetMouse()
	if not camera or not mouse then
		return nil
	end

	local targets = Configuration.expectOptionValue("SilentAimTargets") or {}
	local fovRadius = Configuration.expectOptionValue("SilentAimFOV") or 90

	local mouseX = mouse.X
	local mouseY = mouse.Y + GuiService:GetGuiInset().Y
	local character = localPlayer and localPlayer.Character

	for _, entity in next, live:GetChildren() do
		local isMob = entity.Name:sub(1, 1) == "."
		local isPlayer = not isMob

		if isMob and not targets.Mobs then
			continue
		end
		if isPlayer and not targets.Players then
			continue
		end
		if entity == character then
			continue
		end

		local primary = getPrimary(entity)
		if not primary then
			continue
		end

		if isPlayer then
			local player = Players:GetPlayerFromCharacter(entity)
			if player and PlayerScanning.isAlly(player) and not targets.Guildmates then
				continue
			end
		end

		-- 8 corners of a rough character box projected to screen; first corner inside the FOV wins.
		local size = Vector3.new(4, 5, 6)
		local base = primary.CFrame - (size / 2)
		local mins = base.Position
		local maxs = (base + size).Position

		local points = {
			Vector3.new(mins.X, mins.Y, mins.Z),
			Vector3.new(mins.X, maxs.Y, mins.Z),
			Vector3.new(maxs.X, maxs.Y, mins.Z),
			Vector3.new(maxs.X, mins.Y, mins.Z),
			Vector3.new(maxs.X, maxs.Y, maxs.Z),
			Vector3.new(mins.X, maxs.Y, maxs.Z),
			Vector3.new(mins.X, mins.Y, maxs.Z),
			Vector3.new(maxs.X, mins.Y, maxs.Z),
		}

		for _, point in next, points do
			local screenPoint, onScreen = camera:WorldToViewportPoint(point)
			if
				onScreen
				and screenPoint.X > mouseX - fovRadius
				and screenPoint.X < mouseX + fovRadius
				and screenPoint.Y > mouseY - fovRadius
				and screenPoint.Y < mouseY + fovRadius
			then
				return entity
			end
		end
	end

	return nil
end

---Install (or re-install after a respawn) the GetCameraToMouse override. Transparent passthrough when the
---toggle is off or no target is found.
local function installCameraHook()
	local requests = ReplicatedStorage:FindFirstChild("Requests")
	local getCameraToMouse = requests and requests:FindFirstChild("GetCameraToMouse")
	if not getCameraToMouse then
		return false
	end

	getCameraToMouse.OnClientInvoke = function()
		if Configuration.expectToggleValue("SilentAim") then
			local target = getTarget()
			if target then
				local cf = getGCTM(target)
				if cf then
					return cf, hasCameraLock()
				end
			end
		end

		-- Vanilla behaviour: aim at the real mouse hit (matches the stock client's handler exactly).
		local camera = workspace.CurrentCamera
		local mouse = localPlayer and localPlayer:GetMouse()
		local origin = camera and camera.CFrame.Position or Vector3.zero
		local hit = (mouse and mouse.Hit and mouse.Hit.Position) or origin
		return safeNAN(CFrame.lookAt(origin, hit)), hasCameraLock()
	end

	return true
end

local function ensureDrawings()
	if fovCircle then
		return
	end
	if not (Drawing and Drawing.new) then
		return
	end

	local ok = pcall(function()
		fovCircle = Drawing.new("Circle")
		fovCircle.Visible = false
		fovCircle.ZIndex = 2
		fovCircle.NumSides = 100
		fovCircle.Thickness = 1
		fovCircle.Filled = true
		fovCircle.Transparency = 0.3

		outlineCircle = Drawing.new("Circle")
		outlineCircle.Visible = false
		outlineCircle.ZIndex = 1
		outlineCircle.NumSides = 100
		outlineCircle.Thickness = 2
		outlineCircle.Filled = false
	end)

	if not ok then
		fovCircle = nil
		outlineCircle = nil
	end
end

local function destroyDrawings()
	for _, drawing in next, { fovCircle, outlineCircle } do
		pcall(function()
			drawing.Visible = false
			drawing:Remove()
		end)
	end
	fovCircle = nil
	outlineCircle = nil
end

local function onRender()
	local mouse = localPlayer and localPlayer:GetMouse()
	if not mouse then
		return
	end

	-- FOV circle (optional; only on executors with a Drawing library).
	ensureDrawings()
	if fovCircle and outlineCircle then
		local showFov = Configuration.expectToggleValue("SilentAimShowFOV")
		local radius = Configuration.expectOptionValue("SilentAimFOV") or 90
		local position = Vector2.new(mouse.X, mouse.Y + GuiService:GetGuiInset().Y)

		fovCircle.Visible = showFov
		outlineCircle.Visible = showFov
		fovCircle.Radius = radius
		outlineCircle.Radius = radius
		fovCircle.Filled = Configuration.expectToggleValue("SilentAimFillFOV")
		fovCircle.Transparency = 1 - (Configuration.expectOptionValue("SilentAimFOVTransparency") or 0.3)
		fovCircle.Position = position
		outlineCircle.Position = position
	end

	if tick() - lastTargetRefresh > TARGET_REFRESH then
		currentTarget = getTarget()
		lastTargetRefresh = tick()
	end

	local aimPos = mouse.Hit and mouse.Hit.Position
	local accent = Library.AccentColor or Color3.fromRGB(255, 80, 80)

	if currentTarget and getPrimary(currentTarget) then
		if fovCircle then
			fovCircle.Color = accent
		end
		aimPos = predictedPosition(currentTarget) or aimPos
	elseif fovCircle then
		fovCircle.Color = Library:GetDarkerColor(accent)
	end

	if not aimPos then
		return
	end

	-- Redirect UpdateMouse (the game's own fire is dropped by the namecall hook while Silent Aim is on).
	if not lastSentPosition or (aimPos - lastSentPosition).Magnitude > 0.1 then
		local requests = ReplicatedStorage:FindFirstChild("Requests")
		local updateMouse = requests and requests:FindFirstChild("UpdateMouse")
		if updateMouse then
			local packet = buffer.create(24)
			buffer.writef64(packet, 0, aimPos.X)
			buffer.writef64(packet, 8, aimPos.Y)
			buffer.writef64(packet, 16, aimPos.Z)
			pcall(function()
				updateMouse:FireServer(packet)
			end)
		end
		lastSentPosition = aimPos
	end
end

function SilentAim.enable()
	if running then
		return
	end
	running = true
	currentTarget = nil
	lastSentPosition = nil

	-- Install the OnClientInvoke override once the character + Requests exist.
	task.spawn(function()
		local timeout = tick() + 30
		repeat
			task.wait()
		until (localPlayer.Character and ReplicatedStorage:FindFirstChild("Requests")) or tick() > timeout

		if not running then
			return
		end

		local ok = pcall(installCameraHook)
		installed = installed or ok
	end)

	-- The game re-sets GetCameraToMouse.OnClientInvoke on respawn, so re-install after each CharacterAdded.
	if characterConn then
		pcall(function()
			characterConn:Disconnect()
		end)
	end
	characterConn = localPlayer.CharacterAdded:Connect(function()
		if not running then
			return
		end
		task.spawn(function()
			task.wait(1)
			if running then
				pcall(installCameraHook)
			end
		end)
	end)

	if renderConn then
		pcall(function()
			renderConn:Disconnect()
		end)
	end
	renderConn = RunService.PreRender:Connect(function()
		if not running then
			return
		end
		pcall(onRender)
	end)
end

function SilentAim.disable()
	running = false
	currentTarget = nil
	lastSentPosition = nil

	if renderConn then
		pcall(function()
			renderConn:Disconnect()
		end)
		renderConn = nil
	end
	if characterConn then
		pcall(function()
			characterConn:Disconnect()
		end)
		characterConn = nil
	end

	destroyDrawings()

	-- The GetCameraToMouse override stays installed but is a transparent passthrough while the toggle is
	-- off (returns the real mouse aim), so leaving it is identical to the stock client.
end

return SilentAim
end)
__bundle_register("Features/Combat/AutoDustlunge", function(require, _LOADED, __bundle_register, __bundle_modules)
local KeyHandling = require("Game/KeyHandling")

local InputClient = require("Game/InputClient")

local Latency = require("Game/Latency")

local Configuration = require("Utility/Configuration")

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")
local UserInputService = game:GetService("UserInputService")

local AutoDustlunge = {}

local HITBOX_SIZE = Vector3.new(2, 6, 10)
local HITBOX_SHIFT_BACK = -5
local TICK_INTERVAL = 1 / 5

local holdKeyPicker = nil
local running = false
local simulationPart = nil

function AutoDustlunge.setHoldKeyPicker(keyPicker)
	holdKeyPicker = keyPicker
end

local function cleanupSimulation()
	if simulationPart then
		simulationPart:Destroy()
		simulationPart = nil
	end
end

local function getEffectModule()
	local effectReplicator = ReplicatedStorage:FindFirstChild("EffectReplicator")
	return effectReplicator and require(effectReplicator)
end

local function setSprinting(state)
	local character = Players.LocalPlayer.Character
	local characterHandler = character and character:FindFirstChild("CharacterHandler")
	local requests = characterHandler and characterHandler:FindFirstChild("Requests")
	local sprintRemote = requests and requests:FindFirstChild("ServerSprint")
	if sprintRemote then
		sprintRemote:FireServer(state)
	end
end

local function runTick()
	local effectReplicatorModule = getEffectModule()
	if not effectReplicatorModule or effectReplicatorModule:FindEffect("GrabCD") then
		cleanupSimulation()
		return
	end

	local character = Players.LocalPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end

	local usedCFrame = root.CFrame * CFrame.new(0, 0, HITBOX_SHIFT_BACK)

	if not simulationPart or simulationPart.Parent ~= workspace then
		cleanupSimulation()
		simulationPart = Instance.new("Part")
		simulationPart.Name = "AutoDustlungeHitbox"
		simulationPart.Anchored = true
		simulationPart.CanCollide = false
		simulationPart.CanQuery = false
		simulationPart.CanTouch = false
		simulationPart.Material = Enum.Material.ForceField
		simulationPart.CastShadow = false
	end

	simulationPart.Transparency = Configuration.expectToggleValue("AutoDustlungeDebug") and 0.5 or 1.0
	simulationPart.Shape = Enum.PartType.Block
	simulationPart.Size = HITBOX_SIZE
	simulationPart.CFrame = usedCFrame
	simulationPart.Parent = workspace

	local live = workspace:FindFirstChild("Live")
	local instances = {}
	if live then
		for _, child in next, live:GetChildren() do
			if child ~= character then
				table.insert(instances, child)
			end
		end
	end

	if #instances <= 0 then
		return
	end

	local params = OverlapParams.new()
	params.FilterDescendantsInstances = instances
	params.FilterType = Enum.RaycastFilterType.Include

	local ok, parts = pcall(function()
		return workspace:GetPartsInPart(simulationPart, params)
	end)

	if not ok or not parts or #parts <= 0 then
		return
	end

	local targetCharacter = parts[1]:FindFirstAncestorWhichIsA("Model")
	local targetHumanoid = targetCharacter and targetCharacter:FindFirstChild("Humanoid")
	local targetRoot = targetCharacter and targetCharacter:FindFirstChild("HumanoidRootPart")
	if not targetHumanoid or not targetRoot then
		return
	end

	local facingDot = (targetRoot.Position - root.Position):Dot(root.CFrame.LookVector)
	if facingDot <= 0 then
		return
	end

	local sprinting = effectReplicatorModule:FindEffect("Sprinting") ~= nil
	if sprinting then
		setSprinting(false)
		task.wait(1 / 20)
	end

	local scriptCrouched = false
	if not effectReplicatorModule:FindEffect("Crouching") then
		InputClient.crouch()
		scriptCrouched = true
	end

	local start = tick()
	local pingWindow = (Latency.rtt() or 0) * 2
	repeat
		task.wait()
	until effectReplicatorModule:FindEffect("Crouching") or tick() - start > pingWindow

	if tick() - start > pingWindow then
		if scriptCrouched then
			InputClient.uncrouch()
		end
		if sprinting then
			setSprinting(true)
		end
		return
	end

	task.spawn(function()
		local remote = KeyHandling.getRemote("LeftClick")
		if not remote then
			return
		end

		remote:FireServer(false, Players.LocalPlayer:GetMouse().Hit, {
			S = false,
			NOAERIALS = true,
			Space = false,
			Right = UserInputService:IsMouseButtonPressed(Enum.UserInputType.MouseButton2),
			W = false,
			Left = true,
			ctrl = false,
		})
	end)

	task.wait(1 / 20)

	if scriptCrouched then
		InputClient.uncrouch()
	else
		local vim = Instance.new("VirtualInputManager")
		vim:SendKeyEvent(true, Enum.KeyCode.LeftControl, false, game)
		vim:SendKeyEvent(false, Enum.KeyCode.LeftControl, false, game)
	end

	if sprinting then
		setSprinting(true)
	end
end

local function loop()
	while running do
		if not holdKeyPicker or not holdKeyPicker:GetState() then
			cleanupSimulation()
		else
			local ok, err = pcall(runTick)
			if not ok then
				cleanupSimulation()
			end
		end

		task.wait(TICK_INTERVAL)
	end
end

function AutoDustlunge.enable()
	if running then
		return
	end

	cleanupSimulation()
	running = true
	task.spawn(loop)
end

function AutoDustlunge.disable()
	running = false
	cleanupSimulation()
end

return AutoDustlunge
end)

---@module Utility.Logger

local originalNewIndex = nil

-- Parallel-hook guard (see Game/Hooking.lua): on executors without parallel_hooks_allowed, install via a
-- read-only metatable swap so this __newindex hook can't be dispatched from an Actor parallel phase, which
-- is an illegal cross-phase op the engine Kernel panics on (RSL - panic: ...Kernel).
local parallelHooksAllowed = parallel_hooks_allowed
local hookmetamethod = (not parallelHooksAllowed)
	and function(target, method, hook)
		local metatable = getrawmetatable(target)
		setreadonly(metatable, false)
		local original = metatable[method]
		metatable[method] = hook
		setreadonly(metatable, true)
		return original
	end
	or hookmetamethod

local function installHook()
	if hooked then
		return true
	end

	if type(hookmetamethod) ~= "function" then
		return false
	end

	local ok, result = pcall(function()
		originalNewIndex = hookmetamethod(game, "__newindex", function(self, key, value)
			if active
				and key == "AutoRotate"
				and value == false
				and typeof(self) == "Instance"
				and self:IsA("Humanoid")
			then
				return
			end

			return originalNewIndex(self, key, value)
		end)
	end)

	if not ok then
		Logger.longNotify("Force Auto Rotate failed to hook: " .. tostring(result))
		return false
	end

	hooked = true
	return true
end


function ForceAutoRotate.init()
	if not installHook() then
		return
	end
	active = true
end

function ForceAutoRotate.detach()
	active = false
end

return ForceAutoRotate
end)

local function startAmbientOverride()
	local em = getEffectModule()
	if not em then return end

	if reverbEffect then
		pcall(function() reverbEffect:Remove() end)
		reverbEffect = nil
	end
	if volumeEffect then
		pcall(function() volumeEffect:Remove() end)
		volumeEffect = nil
	end
	if reverbLoop then
		pcall(function() reverbLoop:Disconnect() end)
		reverbLoop = nil
	end

	task.wait()

	pcall(function()
		reverbEffect = em:CreateEffect("ReverbOverride", {})
	end)

	pcall(function()
		volumeEffect = em:CreateEffect("AmbientVolume", { Value = 1 })
	end)

	reverbLoop = runService.Heartbeat:Connect(LPH_NO_VIRTUALIZE(function()
		if rhythmActive then
			if soundService.AmbientReverb ~= Enum.ReverbType.Alley then
				soundService.AmbientReverb = Enum.ReverbType.Alley
			end
		end
	end))

	local numberValue = Instance.new("NumberValue")
	numberValue.Value = 1
	numberValue.Changed:Connect(function()
		if volumeEffect then
			volumeEffect.Value = numberValue.Value
		end
	end)

	tweenService:Create(numberValue, TweenInfo.new(1), { Value = 0 }):Play()

	task.delay(1.5, function()
		numberValue:Destroy()
	end)
end

local function stopAmbientOverride(immediate)
	if reverbLoop then
		pcall(function() reverbLoop:Disconnect() end)
		reverbLoop = nil
	end

	local currentVolumeEffect = volumeEffect
	local currentReverbEffect = reverbEffect
	volumeEffect = nil
	reverbEffect = nil

	if currentReverbEffect then
		pcall(function() currentReverbEffect:Remove() end)
	end

	soundService.AmbientReverb = Enum.ReverbType.NoReverb

	if not currentVolumeEffect then return end

	if immediate then
		pcall(function() currentVolumeEffect:Remove() end)
		return
	end

	local success, numberValue = pcall(function()
		local nv = Instance.new("NumberValue")
		nv.Value = 0
		return nv
	end)

	if success and numberValue then
		local conn
		conn = numberValue.Changed:Connect(function()
			pcall(function()
				currentVolumeEffect.Value = numberValue.Value
			end)
		end)

		tweenService:Create(numberValue, TweenInfo.new(1), { Value = 1 }):Play()

		task.delay(1.1, function()
			pcall(function() conn:Disconnect() end)
			pcall(function() numberValue:Destroy() end)
			pcall(function() currentVolumeEffect:Remove() end)
		end)
	else
		pcall(function() currentVolumeEffect:Remove() end)
	end
end

local function detectEntities()
	local hrp = getHRP()
	if not hrp then return end

	local character = getCharacter()
	local myPos = hrp.Position
	local range = CONFIG.DETECTION_RANGE

	local liveFolder = workspace:FindFirstChild("Live")
	if not liveFolder then return end

	local clientEffectDirect = replicatedStorage:FindFirstChild("Requests")
		and replicatedStorage.Requests:FindFirstChild("ClientEffectDirect")
	if not clientEffectDirect then return end

	for _, entity in ipairs(liveFolder:GetChildren()) do
		local entityHRP = entity:FindFirstChild("HumanoidRootPart")

		if entityHRP and entity ~= character and not entity:HasTag("InTacet") then
			if not entityTimestamps[entity.Name] then
				entityTimestamps[entity.Name] = tick()
			elseif tick() - entityTimestamps[entity.Name] >= 1 then
				entityTimestamps[entity.Name] = tick()
			else
				continue
			end

			local effectiveRange = range
			if entity:HasTag("Disguised") then
				effectiveRange = effectiveRange * 0.1
			end

			local dist = (myPos - entityHRP.Position).Magnitude
			if dist < effectiveRange then
				local color = Color3.new(1, 1, 1)

				local hum = entity:FindFirstChildOfClass("Humanoid")
				if hum and not entity:HasTag("NoVisibleDamage") then
					local healthPercent = hum.Health / hum.MaxHealth
					if healthPercent < 0.33 then
						color = Color3.fromRGB(255, 0, 0)
					elseif healthPercent < 0.66 then
						color = Color3.fromRGB(255, 179, 0)
					end
				end

				clientEffectDirect:Fire("sonarsmall", {
					pos = entityHRP.Position,
					col = color
				})
			end
		end
	end
end

local function canDeactivate()
	if not rhythmStartTime then return true end
	return (tick() - rhythmStartTime) >= CONFIG.MIN_DURATION
end

local function stopRhythm()
	if not rhythmActive then return end

	rhythmActive = false
	rhythmStartTime = nil
	pendingStop = false
	debounce = true

	stopAnimation()
	currentSound = nil
	deactivateLighting()
	stopAmbientOverride()

	if pulseLoop then
		task.cancel(pulseLoop)
		pulseLoop = nil
	end

	task.spawn(function()
		task.wait(0.2)
		debounce = false
	end)

	task.spawn(function()
		task.wait(1)
		if rhythmCC then
			rhythmCC.Enabled = false
		end
	end)
end

local function startRhythm()
	if debounce then return end
	if rhythmActive then return end

	rhythmActive = true
	debounce = true
	rhythmStartTime = tick()
	pendingStop = false

	playAnimation()

	task.wait(0.1)
	playSound()

	task.wait(0.2)

	if not isCrouching() then
		rhythmActive = false
		rhythmStartTime = nil
		pendingStop = false
		stopAnimation()

		if currentSound then
			tweenService:Create(currentSound, TweenInfo.new(0.2), { Volume = 0 }):Play()
			task.wait(0.2)
			stopSound()
		end

		debounce = false
		return
	end

	setupRhythmCC()
	activateLighting()
	startAmbientOverride()

	entityTimestamps = {}
	detectEntities()

	pulseLoop = task.spawn(function()
		while rhythmActive do
			task.wait(0.2)
			detectEntities()
		end
	end)

	for _ = 1, 10 do
		task.spawn(createExpandingSonar)
		task.wait(0.1)
	end

	task.wait(0.5)

	task.spawn(function()
		task.wait(0.2)
		debounce = false
	end)
end

local function onKeyPress(input, gameProcessed)
	if gameProcessed then return end
	if not Configuration.expectToggleValue("RhythmSpoofer") then return end
	if hasRhythmTalent() then return end

	local keybind = getRhythmKeybind()
	if input.KeyCode ~= keybind then return end

	if rhythmActive then
		if canDeactivate() then
			stopRhythm()
		else
			pendingStop = true
		end
	else
		if canUseRhythm() then
			startRhythm()
		end
	end
end

local monitorRhythmCheck = LPH_NO_VIRTUALIZE(function()
	if not rhythmActive then return end
	if not Configuration.expectToggleValue("RhythmSpoofer") then
		stopRhythm()
		return
	end

	if canDeactivate() then
		local canUse = canUseRhythm()
		if pendingStop or not canUse then
			stopRhythm()
		end
	end
end)

function RhythmSpoofer.init()
	rhythmMaid:add(userInputService.InputBegan:Connect(onKeyPress))
	rhythmMaid:add(runService.Heartbeat:Connect(monitorRhythmCheck))

	rhythmMaid:add(localPlayer.CharacterAdded:Connect(function()
		currentAnimTrack = nil
		currentSound = nil
		stopAmbientOverride(true)
		if rhythmActive then
			stopRhythm()
		end
	end))

	getRhythmKeybind()
end

function RhythmSpoofer.detach()
	if rhythmActive then
		stopRhythm()
	end
	rhythmMaid:clean()

	if rhythmCC then
		rhythmCC:Destroy()
		rhythmCC = nil
	end
end

return RhythmSpoofer
end)

local function formatTime(ms)
	local totalSeconds = math.floor((ms or 0) / 1000)
	local minutes = math.floor(totalSeconds / 60)
	local seconds = totalSeconds % 60
	return string.format("%d:%02d", minutes, seconds)

local function fetchAlbumArt(trackId, url)
	if not trackId or not url or artCache[trackId] or fetchingArt[trackId] then
		return
	end

	fetchingArt[trackId] = true

	task.spawn(function()
		local filename = trackId .. ".png"
		local contentId = nil

		local ok = pcall(function()
			if not artStorage:file(filename) then
				local response = request({ Url = url, Method = "GET" })
				if not response or not response.Success or not response.Body then
					return
				end
				artStorage:write(filename, response.Body)
			end

			contentId = getcustomasset(artStorage:append(filename))
		end)

		if ok and contentId then
			artCache[trackId] = contentId
			if nowPlaying and nowPlaying.trackId == trackId then
				artLabel.Image = contentId
			end
		end

		fetchingArt[trackId] = nil
	end)

local function base64Encode(input)

		local b1, b2, b3 = string.byte(input, i, i + 2)
		b2 = b2 or 0

	end

local function sha256(msg)
	local band, bxor, bnot, rshift, rrotate = bit32.band, bit32.bxor, bit32.bnot, bit32.rshift, bit32.rrotate

	local h0, h1, h2, h3, h4, h5, h6, h7 =
		0x6a09e667, 0xbb67ae85, 0x3c6ef372, 0xa54ff53a, 0x510e527f, 0x9b05688c, 0x1f83d9ab, 0x5be0cd19

	local msgLen = #msg
	local bitLen = msgLen * 8

	msg = msg .. "\128"
	while #msg % 64 ~= 56 do
		msg = msg .. "\0"
	end

	for i = 7, 0, -1 do
		msg = msg .. string.char(band(rshift(bitLen, i * 8), 0xFF))
	end

	for chunkStart = 1, #msg, 64 do
		local w = {}
		for i = 0, 15 do
			local o = chunkStart + i * 4
			local b1, b2, b3, b4 = string.byte(msg, o, o + 3)
			w[i + 1] = b1 * 16777216 + b2 * 65536 + b3 * 256 + b4
		end

		for i = 17, 64 do
			local w15, w2 = w[i - 15], w[i - 2]
			local s0 = bxor(rrotate(w15, 7), rrotate(w15, 18), rshift(w15, 3))
			local s1 = bxor(rrotate(w2, 17), rrotate(w2, 19), rshift(w2, 10))
			w[i] = (w[i - 16] + s0 + w[i - 7] + s1) % 4294967296
		end

		local a, b, c, d, e, f, g, h = h0, h1, h2, h3, h4, h5, h6, h7

		for i = 1, 64 do
			local S1 = bxor(rrotate(e, 6), rrotate(e, 11), rrotate(e, 25))
			local ch = bxor(band(e, f), band(bnot(e), g))
			local temp1 = (h + S1 + ch + SHA256_K[i] + w[i]) % 4294967296
			local S0 = bxor(rrotate(a, 2), rrotate(a, 13), rrotate(a, 22))
			local maj = bxor(band(a, b), band(a, c), band(b, c))
			local temp2 = (S0 + maj) % 4294967296

			h, g, f = g, f, e
			e = (d + temp1) % 4294967296
			d, c, b = c, b, a
			a = (temp1 + temp2) % 4294967296
		end

		h0 = (h0 + a) % 4294967296
		h1 = (h1 + b) % 4294967296
		h2 = (h2 + c) % 4294967296
		h3 = (h3 + d) % 4294967296
		h4 = (h4 + e) % 4294967296
		h5 = (h5 + f) % 4294967296
		h6 = (h6 + g) % 4294967296
		h7 = (h7 + h) % 4294967296
	end

	local out = {}
	for _, v in next, { h0, h1, h2, h3, h4, h5, h6, h7 } do
		out[#out + 1] = string.char(
			band(rshift(v, 24), 0xFF),
			band(rshift(v, 16), 0xFF),
			band(rshift(v, 8), 0xFF),
			band(v, 0xFF)
		)
	end

	return table.concat(out)
end

local function extractCode(pastedUrl)
	return pastedUrl:match("[?&]code=([^&#]+)")
end

function SpotifyClient.completeAuth(pastedUrl)
	if not state.codeVerifier then
		return false, "Click 'Copy Auth URL' first -- authorization needs to start from there."
	end

	local code = extractCode(pastedUrl)
	if not code then
		return false, "Couldn't find a 'code' parameter in that URL."
	end

	local response = request({
		Url = TOKEN_URL,
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/x-www-form-urlencoded",
		},
		Body = encodeForm({
			grant_type = "authorization_code",
			code = code,
			redirect_uri = REDIRECT_URI,
			client_id = state.clientId,
			code_verifier = state.codeVerifier,
		}),
	})

	state.codeVerifier = nil

	if not response or not response.Success then
		return false, "Token exchange request failed."
	end

	local ok, decoded = pcall(JSON.decode, response.Body)
	if not ok or not decoded or not decoded.refresh_token then
		return false, "Token exchange returned an unexpected response."
	end

	state.refreshToken = decoded.refresh_token
	state.accessToken = decoded.access_token
	state.accessTokenExpiresAt = os.clock() + (decoded.expires_in or 3600) - 30
	SpotifyClient.save()

	return true
end

local function refreshAccessToken()
	if state.refreshToken == "" then
		return false
	end

	local response = request({
		Url = TOKEN_URL,
		Method = "POST",
		Headers = {
			["Content-Type"] = "application/x-www-form-urlencoded",
		},
		Body = encodeForm({
			grant_type = "refresh_token",
			refresh_token = state.refreshToken,
			client_id = state.clientId,
		}),
	})

	if not response or not response.Success then
		return false
	end

	local ok, decoded = pcall(JSON.decode, response.Body)
	if not ok or not decoded or not decoded.access_token then
		return false
	end

	state.accessToken = decoded.access_token
	state.accessTokenExpiresAt = os.clock() + (decoded.expires_in or 3600) - 30

	if decoded.refresh_token then
		state.refreshToken = decoded.refresh_token
		SpotifyClient.save()
	end

	return true
end

local function ensureToken()
	if state.accessToken and os.clock() < state.accessTokenExpiresAt then
		return true
	end

	return refreshAccessToken()
end

local function authorizedRequest(method, path, body)
	if not ensureToken() then
		dbg("%s %s: ensureToken() failed, no valid access token", method, path)
		return nil
	end

	local response = request({
		Url = API_BASE .. path,
		Method = method,
		Headers = {
			["Authorization"] = "Bearer " .. state.accessToken,
			["Content-Type"] = "application/json",
		},
		Body = body or "{}",
	})

	if not response then
		dbg("%s %s: request() returned nil", method, path)
	elseif not response.Success then
		dbg("%s %s: failed, status=%s body=%s", method, path, tostring(response.StatusCode), tostring(response.Body))
	else
		dbg("%s %s: ok, status=%s", method, path, tostring(response.StatusCode))
	end

	return response
end

function SpotifyClient.getCurrentlyPlaying()
	local response = authorizedRequest("GET", "/currently-playing")
	if not response or not response.Success or response.StatusCode == 204 or not response.Body or response.Body == "" then
		return nil
	end

	local ok, decoded = pcall(JSON.decode, response.Body)
	if not ok or not decoded or not decoded.item then
		return nil
	end

	local images = decoded.item.album and decoded.item.album.images or {}
	local artists = {}
	for _, artist in next, decoded.item.artists or {} do
		table.insert(artists, artist.name)
	end

	return {
		trackId = decoded.item.id,
		trackName = decoded.item.name,
		artistName = table.concat(artists, ", "),
		albumArtUrl = images[1] and images[1].url,
		progressMs = decoded.progress_ms or 0,
		durationMs = decoded.item.duration_ms or 0,
		isPlaying = decoded.is_playing or false,
	}
end

local function makePermissiveEnv()
	local base = getfenv(1)

	local dummy
	local dummyMeta = {
		__index = function()
			return dummy
		end,
		__newindex = function() end,
		__call = function()
			return dummy
		end,
		__concat = function()
			return dummy
		end,
	}
	for _, op in ipairs({ "__add", "__sub", "__mul", "__div", "__mod", "__pow", "__unm" }) do
		dummyMeta[op] = function()
			return dummy
		end
	end
	dummy = setmetatable({}, dummyMeta)

	return setmetatable({}, {
		__index = function(_, key)
			local real = base[key]
			if real ~= nil then
				return real
			end
			return dummy
		end,
		__newindex = function() end,
	})
end

---@param source string

local function validateModuleSource(source)
	local chunk, compileErr = loadstring(source)
	if not chunk then
		local clean = tostring(compileErr):gsub('^%[string "[^"]*"%]:', "line ")
		return false, "Syntax error: " .. clean
	end

	pcall(function()
		setfenv(chunk, makePermissiveEnv())
	end)

	local ok, result = pcall(chunk)
	if not ok then
		local clean = tostring(result):gsub('^%[string "[^"]*"%]:', "line ")
		return false, "Error while loading: " .. clean
	end

	if type(result) ~= "function" then
		return false, "Module must `return function(self, timing) ... end` (got " .. type(result) .. ")."
	end

	return true
end

function TimingBuilderTab.ensurePersistence()
	if SaveManager.llcn then
		return
	end

	local name = "MyTimings"
	local exists = false
	for _, file in ipairs(SaveManager.list()) do
		if file == name then
			exists = true
			break
		end
	end

	if not exists then
		SaveManager.create(name)
	end

	SaveManager.load(name)
	SaveManager.autoload(name)
end

---@param box table
---@param getActive fun():Timing?
---@param getWritable fun():Timing?
---@return table
local function buildActionsEditor(box, getActive, getWritable)
	getWritable = getWritable or getActive
	local ctrl = {}
	local list, nameInput, typeDrop, delayInput, hbLength, hbWidth, hbHeight

	local function anc(apply)
		return function(value)
			---@note: Only mutate when an action is actually selected. Checking the selection before
			---getWritable() also avoids spuriously cloning an internal timing into the config just
			---from touching a field while nothing is selected (e.g. while configuring a new action).
			if not list.Value then
				return
			end
			local timing = getWritable()
			if not timing then
				return
			end
			local action = timing.actions:find(list.Value)
			if not action then
				return
			end
			apply(action, value)
			SaveManager.markDirty()
		end
	end

	list = box:AddDropdown(nil, {
		Text = "Actions",
		Values = {},
		AllowNull = true,
		Searchable = true,
		Callback = function(value)
			local timing = getActive()
			if not timing or not value then
				return
			end
			local action = timing.actions:find(value)
			if not action then
				return
			end
			nameInput:SetRawValue(action.name)
			typeDrop:SetRawValue(action._type)
			delayInput:SetRawValue(tostring(action._when or 0))
			hbWidth:SetRawValue(action.hitbox.X)
			hbHeight:SetRawValue(action.hitbox.Y)
			hbLength:SetRawValue(action.hitbox.Z)
		end,
	})

	typeDrop = box:AddDropdown(nil, {
		Text = "Action Type",
		Values = ACTION_TYPES,
		Default = 1,
		Callback = anc(function(action, value)
			action._type = value
		end),
	})

	delayInput = box:AddInput(nil, {
		Text = "Action Delay (ms)",
		Numeric = true,
		Finished = true,
		Callback = anc(function(action, value)
			action._when = tonumber(value) or 0
		end),
	})

	hbLength = box:AddSlider(nil, {
		Text = "Action Hitbox Length",
		Min = 0,
		Max = 300,
		Default = 0,
		Rounding = 0,
		Callback = anc(function(action, value)
			action.hitbox = Vector3.new(action.hitbox.X, action.hitbox.Y, value)
		end),
	})

	hbWidth = box:AddSlider(nil, {
		Text = "Action Hitbox Width",
		Min = 0,
		Max = 300,
		Default = 0,
		Rounding = 0,
		Callback = anc(function(action, value)
			action.hitbox = Vector3.new(value, action.hitbox.Y, action.hitbox.Z)
		end),
	})

	hbHeight = box:AddSlider(nil, {
		Text = "Action Hitbox Height",
		Min = 0,
		Max = 300,
		Default = 0,
		Rounding = 0,
		Callback = anc(function(action, value)
			action.hitbox = Vector3.new(action.hitbox.X, value, action.hitbox.Z)
		end),
	})

	box:AddDivider()

	nameInput = box:AddInput(nil, {
		Text = "Action Name",
	})

	box:AddButton("Create Action", function()
		local timing = getWritable()
		if not timing then
			return Logger.longNotify("Select or create a timing first.")
		end
		local name = nameInput.Value
		if not name or #name <= 0 then
			return Logger.longNotify("Enter an action name.")
		end
		if timing.actions:find(name) then
			return Logger.longNotify("The action '%s' already exists.", name)
		end
		---@note: Seed the new action from whatever is currently in the editor fields so "set the
		---type/delay/hitbox, name it, create" produces an action with those values -- instead of a
		---blank action while the values silently apply to the previously-selected action.
		local action = Action.new()
		action.name = name
		action._type = typeDrop.Value or "Parry"
		action._when = tonumber(delayInput.Value) or 0
		action.hitbox = Vector3.new(
			tonumber(hbWidth.Value) or 0,
			tonumber(hbHeight.Value) or 0,
			tonumber(hbLength.Value) or 0
		)
		timing.actions:push(action)
		SaveManager.markDirty()
		---@note: reload() clears the selection and resets the fields, so the editor returns to a
		---clean "new action" state and the next create won't inherit this one's values.
		ctrl.reload(timing)
		list:Display()
	end)

	box:AddButton("Duplicate Action", function()
		local timing = getWritable()
		if not timing then
			return Logger.longNotify("Select or create a timing first.")
		end
		local action = timing.actions:find(list.Value)
		if not action then
			return Logger.longNotify("Select an action to duplicate.")
		end
		local name = nameInput.Value
		if not name or #name <= 0 then
			return Logger.longNotify("Enter a name for the copy.")
		end
		if timing.actions:find(name) then
			return Logger.longNotify("The action '%s' already exists.", name)
		end
		local copy = action:clone()
		copy.name = name
		timing.actions:push(copy)
		SaveManager.markDirty()
		ctrl.reload(timing)
		list:SetValue(name)
		list:Display()
	end)

	box:AddButton("Remove Action", function()
		local timing = getWritable()
		if not timing then
			return
		end
		local action = timing.actions:find(list.Value)
		if not action then
			return Logger.longNotify("Select an action to remove.")
		end
		timing.actions:remove(action)
		SaveManager.markDirty()
		ctrl.reload(timing)
	end)

	---@param timing Timing?
	function ctrl.reload(timing)
		list:SetValues(timing and timing.actions:names() or {})
		list:SetValue(nil)
		list:Display()
		nameInput:SetRawValue("")
		typeDrop:SetRawValue("Parry")
		delayInput:SetRawValue("0")
		hbLength:SetRawValue(0)
		hbWidth:SetRawValue(0)
		hbHeight:SetRawValue(0)
	end

	return ctrl
end

---@param tab table
---@param getActive fun():Timing?
---@param getWritable fun():Timing?
---@return table
local function buildTimingForm(tab, getActive, getWritable)
	getWritable = getWritable or getActive

	form.typeMarker = { Type = "Dropdown", Value = "Animation" }

	local function bind(apply)

			self.hbHeight:SetRawValue(timing.hitbox.Y)
			self.fhb:SetRawValue(timing.fhb)
			self.srpn:SetRawValue(timing.srpn)
			self.ndfb:SetRawValue(timing.ndfb)
			self.nvfb:SetRawValue(timing.nvfb)
			self.nbfb:SetRawValue(timing.nbfb)
			self.ibi:SetRawValue(timing.ibi)
			self.iapf:SetRawValue(timing.iapf)
			self.uif:SetRawValue(timing.uif)
			self.nfd:SetRawValue(timing.nfd)
			self.rsc:SetRawValue(timing.rsc)
			self.rsaid:SetRawValue(timing.rsaid or "")
			self.rsaw:SetRawValue(timing.rsaw)
			self.rpue:SetRawValue(timing.rpue)
			self.rsd:SetRawValue(tostring(timing._rsd or 0))
			self.rpd:SetRawValue(tostring(timing._rpd or 0))

			if typeKey == "Animation" then
				self.ha:SetRawValue(timing.ha)
				self.iae:SetRawValue(timing.iae)
				self.mat:SetRawValue(tostring(timing.mat or 2000))
				self.ieae:SetRawValue(timing.ieae)
				self.phd:SetRawValue(timing.phd)
				self.phds:SetRawValue(timing.phds)
				self.pfh:SetRawValue(timing.pfh)
				self.dp:SetRawValue(timing.dp)
				self.pfht:SetRawValue(timing.pfht)
			elseif typeKey == "Effect" then
				self.ilp:SetRawValue(timing.ilp)
				self.flp:SetRawValue(timing.flp)
			elseif typeKey == "Part" then
				self.uhc:SetRawValue(timing.uhc)
			elseif typeKey == "Sound" then
				self.alp:SetRawValue(timing.alp)
			end
		end

		self.actions.reload(timing)
		Library:UpdateDependencyBoxes()
	end

	return form

local LUA_KEYWORDS = {}
for _, kw in ipairs({
	"and", "break", "do", "else", "elseif", "end", "false", "for", "function", "goto",
	"if", "in", "local", "nil", "not", "or", "repeat", "return", "then", "true", "until",
	"while", "continue", "self",
}) do
	LUA_KEYWORDS[kw] = true
end

local HL_KEYWORD = "#f3a8d6"
local HL_STRING = "#a8e6b0"
local HL_NUMBER = "#f5c07a"
local HL_COMMENT = "#8b7fa0"

local function escapeRich(s)
	return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function span(color, text)
	return '<font color="' .. color .. '">' .. escapeRich(text) .. "</font>"
end

local function highlightLua(code)
	local out = {}
	local i, n = 1, #code

	while i <= n do
		local c = code:sub(i, i)
		local two = code:sub(i, i + 1)

		if two == "--" then
			local eqs = code:sub(i + 2):match("^%[(=*)%[")
			if eqs then
				local close = "]" .. eqs .. "]"
				local endpos = code:find(close, i + 2, true)
				local chunk = endpos and code:sub(i, endpos + #close - 1) or code:sub(i)
				out[#out + 1] = span(HL_COMMENT, chunk)
				i = i + #chunk
			else
				local nl = code:find("\n", i, true)
				local chunk = nl and code:sub(i, nl - 1) or code:sub(i)
				out[#out + 1] = span(HL_COMMENT, chunk)
				i = i + #chunk
			end
		elseif c == '"' or c == "'" then
			local j = i + 1
			while j <= n do
				local cj = code:sub(j, j)
				if cj == "\\" then
					j = j + 2
				elseif cj == c or cj == "\n" then
					j = j + 1
					break
				else
					j = j + 1
				end
			end
			local chunk = code:sub(i, j - 1)
			out[#out + 1] = span(HL_STRING, chunk)
			i = i + #chunk
		elseif c == "[" and code:sub(i):match("^%[=*%[") then
			local eqs = code:sub(i):match("^%[(=*)%[")
			local close = "]" .. eqs .. "]"
			local endpos = code:find(close, i + #eqs + 2, true)
			local chunk = endpos and code:sub(i, endpos + #close - 1) or code:sub(i)
			out[#out + 1] = span(HL_STRING, chunk)
			i = i + #chunk
		elseif c:match("%d") or (c == "." and code:sub(i + 1, i + 1):match("%d")) then
			local num = code:sub(i):match("^0[xX]%x+")
				or code:sub(i):match("^%d*%.?%d+[eE][%+%-]?%d+")
				or code:sub(i):match("^%d*%.?%d+")
				or c
			out[#out + 1] = span(HL_NUMBER, num)
			i = i + #num
		elseif c:match("[%a_]") then
			local word = code:sub(i):match("^[%w_]+")
			if LUA_KEYWORDS[word] then
				out[#out + 1] = span(HL_KEYWORD, word)
			else
				out[#out + 1] = escapeRich(word)
			end
			i = i + #word
		else
			out[#out + 1] = escapeRich(c)
			i = i + 1
		end
	end

	return table.concat(out)
end

---@param tab table
---@param height number
---@return table
local function createCodeEditor(tab, height)
	local GUTTER_W = 34
	local PAD = 6
	local PLACEHOLDER = "-- select or create a module"

	local holder = Library:Create("Frame", {

		Color = Library.OutlineColor,
		Thickness = 1,
		Transparency = 0.5,
		Parent = holder,
	})
	Library:AddToRegistry(stroke, { Color = "OutlineColor" })

	local gutterHolder = Library:Create("Frame", {
		BackgroundColor3 = Library.BackgroundColor,
		BackgroundTransparency = 0.4,
		BorderSizePixel = 0,
		ClipsDescendants = true,
		Position = UDim2.new(0, 0, 0, PAD),
		Size = UDim2.new(0, GUTTER_W, 1, -PAD * 2),
		ZIndex = 6,
		Parent = holder,
	})

	local gutter = Library:Create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 0, 0, 0),
		Size = UDim2.new(1, -4, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		Font = Enum.Font.Code,
		TextSize = 13,
		TextColor3 = Library.OutlineColor,
		TextXAlignment = Enum.TextXAlignment.Right,
		TextYAlignment = Enum.TextYAlignment.Top,
		Text = "1",
		ZIndex = 6,
		Parent = gutterHolder,
	})

	local scroll = Library:Create("ScrollingFrame", {
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Position = UDim2.new(0, GUTTER_W + 2, 0, PAD),
		Size = UDim2.new(1, -(GUTTER_W + 2 + PAD), 1, -PAD * 2),
		CanvasSize = UDim2.new(0, 0, 0, 0),
		AutomaticCanvasSize = Enum.AutomaticSize.XY,
		ScrollingDirection = Enum.ScrollingDirection.XY,
		ScrollBarThickness = 4,
		ZIndex = 6,
		Parent = holder,
	})

	-- Custom text-selection highlight lives here (ZIndex 5), BEHIND the colored text (6) so the
	-- text reads on top of the selection color. Roblox doesn't render native selection on a
	-- transparent-text box, so we draw it ourselves from box.SelectionStart / box.CursorPosition.
	local selectionFrames = {}

	local highlight = Library:Create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 2, 0, 0),
		Size = UDim2.new(0, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.XY,
		RichText = true,
		Font = Enum.Font.Code,
		TextSize = 13,
		TextColor3 = Color3.fromRGB(150, 150, 162),
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Top,
		Text = escapeRich(PLACEHOLDER),
		ZIndex = 6,
		Parent = scroll,
	})

	local box = Library:Create("TextBox", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 2, 0, 0),
		Size = UDim2.new(0, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.XY,
		MultiLine = true,
		ClearTextOnFocus = false,
		TextEditable = true,
		TextWrapped = false,
		Font = Enum.Font.Code,
		TextSize = 13,
		TextColor3 = Library.FontColor,
		TextTransparency = 1,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Top,
		Text = "",
		PlaceholderText = "",
		ZIndex = 7,
		Parent = scroll,
	})

	local TextService = game:GetService("TextService")
	local HUGE = Vector2.new(1e6, 1e6)
	local LINE_H = 15
	pcall(function()
		LINE_H = TextService:GetTextSize("Ag", 13, Enum.Font.Code, HUGE).Y
	end)

	local caret = Library:Create("Frame", {
		BackgroundColor3 = Library.FontColor,
		BorderSizePixel = 0,
		Size = UDim2.new(0, 1, 0, LINE_H),
		Position = UDim2.new(0, 2, 0, 0),
		Visible = false,
		ZIndex = 8,
		Parent = scroll,
	})
	Library:AddToRegistry(caret, { BackgroundColor3 = "FontColor" })

	local function updateCaret()
		if not box:IsFocused() then
			caret.Visible = false
			return
		end
		local pos = box.CursorPosition
		if pos < 1 then
			caret.Visible = false
			return
		end
		local before = box.Text:sub(1, pos - 1)
		local line = 0
		for _ in before:gmatch("\n") do
			line = line + 1
		end
		local seg = before:match("[^\n]*$") or ""
		local width = 0
		pcall(function()
			width = TextService:GetTextSize(seg, 13, Enum.Font.Code, HUGE).X
		end)
		caret.Position = UDim2.new(0, 2 + width, 0, line * LINE_H)
		caret.Visible = true
		caret.BackgroundTransparency = 0
	end

	task.spawn(function()
		while caret.Parent do
			task.wait(0.5)
			if box:IsFocused() and caret.Visible then
				caret.BackgroundTransparency = caret.BackgroundTransparency == 0 and 1 or 0
			end
		end
	end)

	-- Draw the text-selection highlight ourselves: Roblox doesn't render native selection on a
	-- transparent-text box, so build a rectangle per selected line from SelectionStart/CursorPosition.
	local function renderSelection()
		for _, f in ipairs(selectionFrames) do
			f:Destroy()
		end
		selectionFrames = {}

		local sStart, cPos = box.SelectionStart, box.CursorPosition
		if sStart == -1 or cPos == -1 or sStart == cPos then
			return
		end
		local a = math.min(sStart, cPos) -- first selected position (1 = before char 1)
		local b = math.max(sStart, cPos) -- one past the last selected position

		local text = box.Text
		local pos = 1 -- 1-based position at the start of the current line
		for line in (text .. "\n"):gmatch("(.-)\n") do
			local len = #line
			local lineEndPos = pos + len -- position just after this line's last char
			-- Intersect selection [a, b) with this line's positions [pos, lineEndPos] (+1 covers the newline).
			local selA = math.max(a, pos)
			local selB = math.min(b, lineEndPos + 1)
			if selB > selA then
				local col1 = math.min(selA - pos, len)
				local col2 = math.min(selB - pos, len)
				local x1, x2 = 0, 0
				pcall(function()
					x1 = TextService:GetTextSize(line:sub(1, col1), 13, Enum.Font.Code, HUGE).X
					x2 = TextService:GetTextSize(line:sub(1, col2), 13, Enum.Font.Code, HUGE).X
				end)
				-- Line index (0-based) = number of newlines before this line's start position.
				local lineIdx = 0
				for _ in text:sub(1, pos - 1):gmatch("\n") do
					lineIdx = lineIdx + 1
				end
				local w = x2 - x1
				if b > lineEndPos then
					w = w + 4 -- selection continues past this line: show a small trailing block for the newline
				end
				local sel = Library:Create("Frame", {
					BackgroundColor3 = Library.AccentColor,
					BackgroundTransparency = 0.55,
					BorderSizePixel = 0,
					Position = UDim2.new(0, 2 + x1, 0, lineIdx * LINE_H),
					Size = UDim2.new(0, math.max(w, 2), 0, LINE_H),
					ZIndex = 5,
					Parent = scroll,
				})
				Library:AddToRegistry(sel, { BackgroundColor3 = "AccentColor" })
				selectionFrames[#selectionFrames + 1] = sel
			end
			pos = lineEndPos + 1 -- skip the newline
		end
	end

	box:GetPropertyChangedSignal("CursorPosition"):Connect(function()
		updateCaret()
		renderSelection()
	end)
	box:GetPropertyChangedSignal("SelectionStart"):Connect(renderSelection)
	box.Focused:Connect(updateCaret)
	box.FocusLost:Connect(function()
		caret.Visible = false
		renderSelection()
	end)

	local function refresh()
		local text = box.Text

		if text == "" then
			highlight.TextColor3 = Color3.fromRGB(150, 150, 162)
			highlight.Text = escapeRich(PLACEHOLDER)
		else
			highlight.TextColor3 = Library.FontColor
			local ok, colored = pcall(highlightLua, text)
			highlight.Text = ok and colored or escapeRich(text)
		end

		local lines = 1
		for _ in text:gmatch("\n") do
			lines = lines + 1
		end
		local nums = {}
		for k = 1, lines do
			nums[k] = tostring(k)
		end
		gutter.Text = table.concat(nums, "\n")
	end

	box:GetPropertyChangedSignal("Text"):Connect(function()
		refresh()
		updateCaret()
		renderSelection()
	end)
	scroll:GetPropertyChangedSignal("CanvasPosition"):Connect(function()
		gutter.Position = UDim2.new(0, 0, 0, -scroll.CanvasPosition.Y)
	end)

	refresh()

	pcall(function()
		tab:Resize()
	end)

	return {
		get = function()
			return box.Text
		end,
		set = function(text)
			box.Text = text or ""
			refresh()
		end,
		-- Toggle whether the box can be typed into. Used to make the template PREVIEW read-only
		-- while keeping module editing writable. When locked, also hide the custom caret.
		setEditable = function(editable)
			box.TextEditable = editable and true or false
			if not editable then
				caret.Visible = false
			end
		end,
	}
end

---@return string? encoded, number timingCount, number moduleCount
local function buildExportString()
	local payload = {
		v = 1,
		timings = {
			version = 1,
			animation = SaveManager.as.config:serialize(),
			effect = SaveManager.es.config:serialize(),
			part = SaveManager.ps.config:serialize(),
			sound = SaveManager.ss.config:serialize(),
		},
	}

	local modules, globals = ModuleManager.exportSources()
	payload.modules = modules
	payload.globals = globals

	local ok, encoded = pcall(function()
		return JSON.encode(payload)
	end)
	if not ok or type(encoded) ~= "string" then
		return nil
	end

	local timingCount = #payload.timings.animation
		+ #payload.timings.effect
		+ #payload.timings.part
		+ #payload.timings.sound
	local moduleCount = countKeys(modules) + countKeys(globals)

	return "LUMA1:" .. base64Encode(encoded), timingCount, moduleCount
end

-- Base export folder in the executor workspace; each export drops a timestamped subfolder here so a
-- history is kept.
local EXPORT_ROOT = "Luma/TimingBuilderExport"

---Ensure a folder path exists (creating each level, since some executors don't make nested folders).
---@param path string
local function ensureFolder(path)
	local built = nil
	for segment in string.gmatch(path, "[^/]+") do
		built = built and (built .. "/" .. segment) or segment
		if not isfolder(built) then
			pcall(makefolder, built)
		end
	end
end

---Make a module name safe to use as a flat filename (path separators -> underscores).
---@param name string
---@return string
local function safeFileName(name)
	return (tostring(name):gsub("[\\/]", "_"))
end

---Write the new/changed modules + the timing patch JSON into a folder in the executor workspace.
---Returns (folderPath, moduleFileCount, err).
---@return string?, number?, string?
local function exportToWorkspace()
	if type(writefile) ~= "function" or type(makefolder) ~= "function" or type(isfolder) ~= "function" then
		return nil, nil, "This executor has no filesystem functions."
	end

	local modules, globals = ModuleManager.exportSources()

	-- The timing patch: just the timings portion of the export payload.
	local timingPatch = {
		version = 1,
		animation = SaveManager.as.config:serialize(),
		effect = SaveManager.es.config:serialize(),
		part = SaveManager.ps.config:serialize(),
		sound = SaveManager.ss.config:serialize(),
	}

	local okEncode, timingsJson = pcall(JSON.encode, timingPatch)
	if not okEncode or type(timingsJson) ~= "string" then
		return nil, nil, "Failed to encode the timing patch."
	end

	-- Timestamped subfolder per export so previous exports are preserved as a history.
	local stamp = os.date("!%Y-%m-%d_%H-%M-%S")
	local exportFolder = EXPORT_ROOT .. "/" .. stamp

	ensureFolder(EXPORT_ROOT)

	-- If two exports land in the same second, suffix with a counter so we never clobber one.
	if isfolder(exportFolder) then
		local n = 2
		while isfolder(exportFolder .. "_" .. n) do
			n = n + 1
		end
		exportFolder = exportFolder .. "_" .. n
	end

	ensureFolder(exportFolder)

	local okWrite, resultOrErr = pcall(function()
		writefile(exportFolder .. "/timings.json", timingsJson)

		local fileCount = 0

		if next(modules) then
			ensureFolder(exportFolder .. "/modules")
			for name, source in pairs(modules) do
				writefile(exportFolder .. "/modules/" .. safeFileName(name) .. ".lua", source)
				fileCount = fileCount + 1
			end
		end

		if next(globals) then
			ensureFolder(exportFolder .. "/globals")
			for name, source in pairs(globals) do
				writefile(exportFolder .. "/globals/" .. safeFileName(name) .. ".lua", source)
				fileCount = fileCount + 1
			end
		end

		return fileCount
	end)

	if not okWrite then
		return nil, nil, tostring(resultOrErr)
	end

	return exportFolder, resultOrErr, nil
end

local function resetTimingData()
	pcall(function() SaveManager.as.config:clear() end)
	pcall(function() SaveManager.es.config:clear() end)
	pcall(function() SaveManager.ps.config:clear() end)
	pcall(function() SaveManager.ss.config:clear() end)
	if SaveManager.llcn then
		pcall(function() SaveManager.write(SaveManager.llcn) end)
	end
	pcall(function() ModuleManager.resetUserModules() end)
	pcall(function() ModuleManager.refresh() end)

	if TimingBuilderTab._refreshEditList then
		TimingBuilderTab._refreshEditList()
	end
	if TimingBuilderTab._refreshModulesList then
		TimingBuilderTab._refreshModulesList()
	end

	Logger.notify("Timing data reset to the original baked-in set.")
end

---Count all distinct timings the script knows (baked-in internal + config overrides/additions),
---deduped by id per type so a config override of an internal timing isn't double-counted.
---@return number
local function countKnownTimings()
	local total = 0
	for _, typeKey in ipairs(TYPE_ORDER) do
		local pair = TYPES[typeKey].pair()
		if pair then
			local seen = {}
			for _, timing in ipairs(pair.config:list()) do
				seen[timing:id()] = true
			end
			for _, timing in ipairs(pair.internal:list()) do
				seen[timing:id()] = true
			end
			for _ in next, seen do
				total = total + 1
			end
		end
	end
	return total
end

---Count all modules the script knows (user files + baked-in), matching the Modules tab list.
---@return number
local function countKnownModules()
	local ok, files = pcall(ModuleManager.listFiles)
	if ok and type(files) == "table" then
		return #files
	end
	return 0
end

---@param tab table
function TimingBuilderTab.initCreate(tab)
	local state = { type = "Animation", draft = AnimationTiming.new(), suppressTypeReset = false, dupEntries = {} }
	local form, nameInput, idInput, typeDrop, dupDrop

	-- Build the "Duplicate From" list: every existing timing (config overrides + baked-in), by type.
	local function rebuildDuplicateList()
		state.dupEntries = {}
		local names = {}
		for _, typeKey in ipairs(TYPE_ORDER) do
			local pair = TYPES[typeKey].pair()
			if pair then
				local seen = {}
				for _, timing in ipairs(pair.config:list()) do
					local display = string.format("[%s] %s", typeKey, timing.name)
					state.dupEntries[display] = { timing = timing, type = typeKey }
					names[#names + 1] = display
					seen[timing:id()] = true
				end
				for _, timing in ipairs(pair.internal:list()) do
					if not seen[timing:id()] then
						local display = string.format("[%s] %s", typeKey, timing.name)
						state.dupEntries[display] = { timing = timing, type = typeKey }
						names[#names + 1] = display
					end
				end
			end
		end
		table.sort(names)
		if dupDrop then
			dupDrop:SetValues(names)
		end
	end

	typeDrop = tab:AddDropdown(nil, {
		Text = "Timing Type",
		Values = TYPE_ORDER,
		Default = 1,
		Callback = function(value)
			-- Skip the reset when the type is being set programmatically by a duplicate load.
			if state.suppressTypeReset then
				return
			end
			state.type = value
			state.draft = TYPES[value].new()
			nameInput:SetRawValue("")
			idInput:SetRawValue("")
			form:load(state.draft, value)
		end,
	})

	nameInput = tab:AddInput(nil, {
		Text = "Timing Name",
	})

	idInput = tab:AddInput(nil, {
		Text = "Timing ID",
		Placeholder = "Animation/Sound ID, or Part/Effect name",
	})

	-- Duplicate an existing timing: clones all of its fields into the draft. Picks up the source's
	-- type, suffixes the name with " Copy", and clears the ID so you set a new (unique) one.
	dupDrop = tab:AddDropdown(nil, {
		Text = "Duplicate From",
		Values = {},
		Default = nil,
		AllowNull = true,
		Searchable = true,
		Callback = function(value)
			if not value then
				return
			end
			local entry = state.dupEntries[value]
			if not entry then
				return
			end

			state.type = entry.type
			state.suppressTypeReset = true
			pcall(function()
				typeDrop:SetValue(entry.type)
			end)
			state.suppressTypeReset = false

			state.draft = entry.timing:clone()
			nameInput:SetRawValue((entry.timing.name or "") .. " Copy")
			idInput:SetRawValue("")
			form:load(state.draft, entry.type)

			Logger.notify("Loaded a copy of '%s'. Set a new Timing ID before creating.", entry.timing.name)
		end,
	})

	tab:AddButton("Create Timing", function()
		local info = TYPES[state.type]
		local name = nameInput.Value
		local id = idInput.Value
		if not name or #name <= 0 then
			return Logger.longNotify("Enter a timing name.")
		end
		if not id or #id <= 0 then
			return Logger.longNotify("Enter a timing ID.")
		end
		local pair = info.pair()
		if pair:index(id) then
			return Logger.longNotify("A timing with ID '%s' already exists.", id)
		end
		if pair:find(name) then
			return Logger.longNotify("A timing named '%s' already exists.", name)
		end

		state.draft.name = name
		state.draft[info.idProp] = id

		local ok, err = pcall(function()
			pair.config:push(state.draft)
		end)
		if not ok then
			return Logger.longNotify("Failed to create timing: %s", tostring(err))
		end

		SaveManager.markDirty()
		Logger.notify("Created %s timing '%s'.", state.type, name)
		TimingBuilderTab.refreshEditList()
		rebuildDuplicateList()

		state.draft = info.new()
		nameInput:SetRawValue("")
		idInput:SetRawValue("")
		form:load(state.draft, state.type)
	end)

	tab:AddButton("Refresh Duplicate List", rebuildDuplicateList)

	tab:AddDivider()

	form = buildTimingForm(tab, function()
		return state.draft
	end)
	form:load(state.draft, "Animation")

	rebuildDuplicateList()
end

---@param tab table
function TimingBuilderTab.initEdit(tab)
	local state = { selected = nil, entry = nil, entries = {} }
	local form, list, renameInput, idInput

	local function rebuildList()
		state.entries = {}
		local names = {}

		for _, typeKey in ipairs(TYPE_ORDER) do
			local pair = TYPES[typeKey].pair()
			if pair then
				local seen = {}
				for _, timing in ipairs(pair.config:list()) do
					local display = string.format("[%s] %s", typeKey, timing.name)
					state.entries[display] = { pair = pair, timing = timing, type = typeKey }
					names[#names + 1] = display
					seen[timing:id()] = true
				end
				for _, timing in ipairs(pair.internal:list()) do
					if not seen[timing:id()] then
						local display = string.format("[%s] %s", typeKey, timing.name)
						state.entries[display] = { pair = pair, timing = timing, type = typeKey }
						names[#names + 1] = display
					end
				end
			end
		end

		table.sort(names)

		if list then
			list:SetValues(names)
			list:SetValue(nil)
			list:Display()
		end

		if TimingBuilderTab._refreshStats then
			TimingBuilderTab._refreshStats()
		end
	end

	local function getWritable()
		local timing = state.selected
		if not timing then
			return nil
		end
		local entry = state.entry
		if not entry then
			return timing
		end
		local pair = entry.pair
		local id = timing:id()
		if pair.config.timings[id] == timing then
			return timing
		end
		local existing = pair.config.timings[id]
		if existing then
			state.selected = existing
			return existing
		end
		local copy = timing:clone()
		local ok = pcall(function()
			pair.config:push(copy)
		end)
		if not ok then
			return timing
		end
		state.selected = copy
		SaveManager.markDirty()
		Logger.notify("Editing '%s' (copied into your config so changes save).", copy.name)
		return copy
	end

	list = tab:AddDropdown(nil, {
		Text = "Timing",
		Values = {},
		AllowNull = true,
		Searchable = true,
		Callback = function(display)
			state.selected = nil
			state.entry = nil
			if not display then
				if form then
					form:load(nil, "Animation")
				end
				return
			end
			local entry = state.entries[display]
			if not entry then
				return
			end

			-- If editing a baked-in timing previously created a config override (getWritable clones on
			-- first edit), select THAT override so its saved changes show -- otherwise state.entries
			-- still points at the stale internal object and the edits look lost on re-select.
			local timing = entry.timing
			local override = entry.pair and entry.pair.config.timings[timing:id()]
			if override then
				timing = override
			end

			state.selected = timing
			state.entry = entry
			renameInput:SetRawValue(timing.name)
			idInput:SetRawValue(timing:id())
			form:load(timing, entry.type)
		end,
	})

	tab:AddButton("Refresh List", rebuildList)

	renameInput = tab:AddInput(nil, {
		Text = "Rename To",
		Finished = true,
	})

	tab:AddButton("Rename Timing", function()
		if not state.entry then
			return Logger.longNotify("Select a timing.")
		end
		local newName = renameInput.Value
		if not newName or #newName <= 0 then
			return Logger.longNotify("Enter a new name.")
		end
		if state.entry.pair:find(newName) then
			return Logger.longNotify("A timing named '%s' already exists.", newName)
		end
		local timing = getWritable()
		if not timing then
			return
		end
		timing.name = newName
		SaveManager.markDirty()
		Logger.notify("Renamed to '%s'.", newName)
		rebuildList()
	end)

	idInput = tab:AddInput(nil, {
		Text = "ID To",
		Finished = true,
	})

	tab:AddButton("Change ID", function()
		if not state.entry then
			return Logger.longNotify("Select a timing.")
		end
		local newId = idInput.Value
		if not newId or #newId <= 0 then
			return Logger.longNotify("Enter a new ID.")
		end
		local pair = state.entry.pair
		local oldId = state.entry.timing:id()
		if newId == oldId then
			return Logger.longNotify("ID is unchanged.")
		end
		if pair:index(newId) then
			return Logger.longNotify("A timing with ID '%s' already exists.", newId)
		end
		local timing = getWritable()
		if not timing then
			return
		end
		local idProp = TYPES[state.entry.type].idProp
		---@note: The config container is keyed by id, so rehome the timing: remove it under the old
		---id, set the new id field, then re-push it under the new id.
		pair.config:remove(timing)
		timing[idProp] = newId
		local ok, err = pcall(function()
			pair.config:push(timing)
		end)
		if not ok then
			-- Restore the old id and re-add so a failed change can't drop the timing.
			timing[idProp] = oldId
			pcall(function()
				pair.config:push(timing)
			end)
			return Logger.longNotify("Failed to change ID: %s", tostring(err))
		end
		SaveManager.markDirty()
		Logger.notify("Changed ID to '%s'.", newId)
		rebuildList()
	end)

	tab:AddButton({
		Text = "Delete Timing",
		DoubleClick = true,
		Func = function()
			local entry = state.entry
			if not entry then
				return Logger.longNotify("Select a timing.")
			end
			local id = entry.timing:id()
			local configTiming = entry.pair.config.timings[id]
			if not configTiming then
				return Logger.longNotify("Internal timings are baked in; edit one to make a deletable copy first.")
			end
			entry.pair.config:remove(configTiming)
			SaveManager.markDirty()
			Logger.notify("Deleted '%s'.", entry.timing.name)
			state.selected = nil
			state.entry = nil
			form:load(nil, "Animation")
			rebuildList()
		end,
	})

	tab:AddDivider()

	form = buildTimingForm(tab, function()
		return state.selected
	end, getWritable)
	form:load(nil, "Animation")

	TimingBuilderTab._refreshEditList = rebuildList
	rebuildList()
end

function TimingBuilderTab.refreshEditList()
	if TimingBuilderTab._refreshEditList then
		TimingBuilderTab._refreshEditList()
	end
end

---@param tab table
function TimingBuilderTab.initModules(tab)
	local list, nameInput, templateDrop, editor

	local function rebuild()
		local files = ModuleManager.listFiles()
		if list then
			list:SetValues(files)
			list:SetValue(nil)
			list:Display()
		end

		if TimingBuilderTab._refreshStats then
			TimingBuilderTab._refreshStats()
		end
	end

	list = tab:AddDropdown(nil, {
		Text = "Modules",
		Values = {},
		AllowNull = true,
		Searchable = true,
		Callback = function(name)
			if not name then
				editor.set("")
				return
			end
			-- Selecting a real module -> editable.
			editor.setEditable(true)
			editor.set(ModuleManager.readFile(name) or "-- (empty)")
		end,
	})

	tab:AddButton("Refresh List", function()
		rebuild()
		ModuleManager.refresh()
	end)

	tab:AddButton("Save Code", function()
		local name = list.Value
		if not name then
			return Logger.longNotify("Select a module to save.")
		end
		local ok, err = validateModuleSource(editor.get())
		if not ok then
			return Logger.longNotify("Not saved -- %s", err)
		end
		if ModuleManager.writeFile(name, editor.get()) then
			Logger.notify("Saved module '%s'.", name)
			ModuleManager.refresh()
		else
			Logger.longNotify("Failed to save module '%s'.", name)
		end
	end)

	tab:AddButton({
		Text = "Delete Module",
		DoubleClick = true,
		Func = function()
			local name = list.Value
			if not name then
				return Logger.longNotify("Select a module to delete.")
			end
			if ModuleManager.deleteFile(name) then
				Logger.notify("Deleted module '%s'.", name)
				editor.set("")
				rebuild()
				ModuleManager.refresh()
			else
				Logger.longNotify("Failed to delete module '%s'.", name)
			end
		end,
	})

	tab:AddDivider()

	nameInput = tab:AddInput(nil, {
		Text = "New Module Name",
	})

	templateDrop = tab:AddDropdown(nil, {
		Text = "Template",
		Values = TEMPLATE_NAMES,
		Default = 1,
		Searchable = true,
		Callback = function(value)
			if editor then
				-- Template preview is read-only; create a module from it to edit it.
				editor.set(TEMPLATE_CODE[value] or "")
				editor.setEditable(false)
			end
		end,
	})

	tab:AddButton("Create Module", function()
		local name = nameInput.Value
		if not name or #name <= 0 then
			return Logger.longNotify("Enter a module name.")
		end
		name = name:gsub("%.lua$", "")
		if ModuleManager.fileExists(name) then
			return Logger.longNotify("Module '%s' already exists.", name)
		end
		local template = TEMPLATE_CODE[templateDrop.Value] or ""
		if not ModuleManager.writeFile(name, template) then
			return Logger.longNotify("Failed to create module '%s'.", name)
		end
		Logger.notify("Created module '%s'.", name)
		rebuild()
		ModuleManager.refresh()
		list:SetValue(name)
		list:Display()
		-- Now editing the newly-created module -> editable again.
		editor.setEditable(true)
		editor.set(template)
	end)

	tab:AddDivider()

	editor = createCodeEditor(tab, 320)
	TimingBuilderTab._refreshModulesList = rebuild
	rebuild()
end

-- Parallel-hook guard (mirrors Project Rain's hooking.lua). Deepwoken runs client logic in a parallel
-- Actor VM. On executors that don't advertise parallel-hook support, the native hookmetamethod can let
-- our game __namecall/__newindex hooks be invoked from inside the scheduler's PARALLEL phase -- running
-- main-VM Lua there is an illegal cross-phase operation and the engine Kernel panics on it deliberately
-- (RBXCRASH: FatalRuntimeError (RSL - panic: ...Kernel)). When parallel_hooks_allowed is absent, install
-- the hook via a plain read-only metatable swap instead, so it only ever lives in this VM's metatable
-- and can never be dispatched mid-parallel-phase. When the executor advertises support, use the native
-- hookmetamethod unchanged.
local parallelHooksAllowed = parallel_hooks_allowed
local hookmetamethod = (not parallelHooksAllowed)
	and function(target, method, hook)
		local metatable = getrawmetatable(target)
		setreadonly(metatable, false)
		local original = metatable[method]
		metatable[method] = hook
		setreadonly(metatable, true)
		return original
	end
	or hookmetamethod

---@param type number
local handleFlowState = LPH_NO_VIRTUALIZE(function(type)
	local effectReplicator = getEffectReplicator()
	if not effectReplicator then
		return
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	local localPlayer = playersService.LocalPlayer
	if not localPlayer then
		return
	end

	local backpack = localPlayer:FindFirstChild("Backpack")
	if not backpack then
		return
	end

	local flowStateTool = backpack:FindFirstChild("Talent:Flow State")
	if not flowStateTool then
		return
	end

	local flowStateRemote = flowStateTool:FindFirstChild("ActivateRt")
	if not flowStateRemote then
		return
	end

	local flowStateCooldown = false
	local shUpperCooldown = false
	local shDashCooldown = false
	local shGapCloserCooldown = false

	for _, effect in next, effectReplicatorModule.Effects do
		local index = rawget(effect, "index")
		if not index then
			continue
		end

		local class = index.Class
		-- typeof (not `type`): the parameter above is named `type`, which shadows Lua's global type()
		-- function. Calling type(class) here was calling the input-type NUMBER as a function -> it threw,
		-- and since onInterceptedInput isn't pcall'd in the fire hooks, the error aborted the real M1 fire.
		if typeof(class) ~= "string" then
			continue
		end

		if class == "ToolCD" and index.Value == "Talent: Flow State" then
			flowStateCooldown = true
		end

		if class:match("SHUpper") and class:match("CD") then
			shUpperCooldown = true
		end

		if class:match("SHDash") and class:match("CD") then
			shDashCooldown = true
		end

		if class:match("SHGap") and class:match("CD") then
			shGapCloserCooldown = true
		end
	end

	Logger.warn(
		"(%s, %s, %s, %s, %s) Current cooldown state...",
		shUpperCooldown and "SHUpperCD" or "SHUpperReady",
		shDashCooldown and "SHDashCD" or "SHDashReady",
		shGapCloserCooldown and "SHGapCloserCD" or "SHGapCloserReady",
		effectReplicatorModule:FindEffect("SlideAttackCD") and "SlideAttackCD" or "SlideAttackReady",
		flowStateCooldown and "FlowStateCD" or "FlowStateReady"
	)

	if flowStateCooldown then
		return Logger.warn("Flow state is on cooldown.")
	end

	Logger.warn("(%i) Attempting to detect Silentheart moves for input type.", type)

	if
		effectReplicatorModule:FindEffect("Sliding")
		and not effectReplicatorModule:FindEffect("SlideAttackCD")
		and type == INPUT_LEFT_CLICK
	then
		Logger.warn("Detected 'Ankle Cutter' move.")

		flowStateRemote:FireServer()
	end

	if effectReplicatorModule:FindEffect("SHDash") and not shDashCooldown and type == INPUT_LEFT_CLICK then
		Logger.warn("Detected 'Mayhem' move.")

		flowStateRemote:FireServer()
	end

	if
		effectReplicatorModule:FindEffect("AerialCD")
		and effectReplicatorModule:FindEffect("LightAttack")
		and not shGapCloserCooldown
		and type == INPUT_LEFT_CLICK
	then
		Logger.warn("Detected 'Relentless Hunt' move.")

		flowStateRemote:FireServer()
	end

	if
		(
			effectReplicatorModule:FindEffect("Sliding")
			or effectReplicatorModule:FindEffect("ClientCrouch")
			or effectReplicatorModule:FindEffect("Crouching")
		)
		and not shUpperCooldown
		and type == INPUT_RIGHT_CLICK
	then
		Logger.warn("Detected 'Rising Star' move.")

		flowStateRemote:FireServer()
	end
end)

---@param type number
local handleActionRolling = LPH_NO_VIRTUALIZE(function(type)
	local actionRollTypes = Configuration.expectOptionValue("ActionRollingActions") or {}
	local actionRollInputs = {
		[INPUT_LEFT_CLICK] = actionRollTypes["Roll On M1"],
		[INPUT_CRITICAL] = actionRollTypes["Roll On Critical"],
		[INPUT_CAST] = actionRollTypes["Roll On Cast"],
		[INPUT_BLOCK] = actionRollTypes["Roll On Parry"],
	}

	if not actionRollInputs[type] then
		return
	end

	if type ~= INPUT_CAST then
		local effectReplicator = getEffectReplicator()
		if not effectReplicator then
			return
		end

		local effectReplicatorModule = require(effectReplicator)
		if not effectReplicatorModule then
			return
		end

		if not effectReplicatorModule:HasEffect("Equipped") then
			return
		end
	end

	if type == INPUT_CAST then
		local lMantraActivated = StateListener.lMantraActivated
		if not lMantraActivated then
			return
		end

		if lMantraActivated.Name:match("Wisp") then
			return
		end
	end

	if
		lastActionRoll
		and os.clock() - lastActionRoll < (Configuration.expectOptionValue("ActionRollCooldown") or 2.0)
	then
		return
	end

	lastActionRoll = os.clock()

	if type ~= INPUT_BLOCK then
		task.wait(0.1)
	end

	local options = DodgeOptions.new()
	options.rollCancel = true
	options.rollCancelDelay = Configuration.expectOptionValue("ActionRollCancelDelay") or 0.1
	options.actionRolling = true

	Logger.warn("(%i) Performing action roll dodge.", type)

	if type == INPUT_CAST then
		TaskSpawner.spawn("ActionRolling_CastDodge", InputClient.dodge, options)
	else
		InputClient.dodge(options)
	end
end)

		return Logger.warn("Unable to feint for delayed feint.")

	if not laLatency then
		return Logger.warn("No animation latency for delayed feint.")

	Logger.warn("Delaying for %.2f seconds before sending the feint remote.", delayed)

	task.wait(delayed)
end)

---@param type number
---@param state number
local onInterceptedInput = LPH_NO_VIRTUALIZE(function(type, state)
	-- Every handler is pcall'd: onInterceptedInput runs INLINE inside the FireServer / UnreliableFireServer
	-- hook bodies, so an error thrown by any feature handler would propagate out of the hook and abort the
	-- player's real input fire (you'd silently lose the M1/parry/cast). Containing it here means a broken
	-- feature degrades to a no-op instead of eating your inputs. pcall is yieldable in Luau, so handlers that
	-- task.wait (ActionRolling, DelayedFeints) still work.
	if Configuration.expectToggleValue("AutoFlowState") and state == INPUT_TYPE_BEFORE and type ~= INPUT_CAST then
		pcall(handleFlowState, type)
	end

	if Configuration.expectToggleValue("ActionRolling") and state == INPUT_TYPE_AFTER then
		pcall(handleActionRolling, type)
	end

	if
		Configuration.expectToggleValue("DelayedFeints")
		and state == INPUT_TYPE_BEFORE
		and type == INPUT_RIGHT_CLICK
		and StateListener.lAnimationValidTrack
	then
		pcall(handleDelayedFeints)
	end
end)

---@param value Color3
local modifyAmbienceColor = LPH_NO_VIRTUALIZE(function(value)
	local ambienceColor = Configuration.expectOptionValue("AmbienceColor")
	local shouldUseOriginalAmbienceColor = Configuration.expectToggleValue("OriginalAmbienceColor")

	if not shouldUseOriginalAmbienceColor and ambienceColor then
		return ambienceColor
	end

	local brightness = Configuration.expectOptionValue("OriginalAmbienceColorBrightness") or 0.0
	local red, green, blue = value.R, value.G, value.B

	red = math.min(red * 255 + brightness, 255)
	green = math.min(green * 255 + brightness, 255)
	blue = math.min(blue * 255 + brightness, 255)

	return Color3.fromRGB(red, green, blue)
end)

---@return any
local onNameCall = LPH_NO_VIRTUALIZE(function(self, ...)
	if not LeaderboardClient.calling and checkcaller() then
		return oldNameCall(self, ...)
	end

	if banRemotes[self] then
		return Logger.warn("(%s) Anticheat is referencing a ban remote.", self.Name)
	end

	if typeof(self) ~= "Instance" then
		return oldNameCall(self, ...)
	end

	local secondArg = ...
	local method = getnamecallmethod()
	local name = self.Name

	-- Silent Aim: drop the game's own UpdateMouse fire so only our redirected mouse position reaches the
	-- server. Our own UpdateMouse fire is checkcaller() == true and short-circuits at the top of this hook,
	-- so it is never caught here -- only the game's (checkcaller() == false) calls reach this point.
	if name == "UpdateMouse" and method == "FireServer" and Configuration.expectToggleValue("SilentAim") then
		return
	end

	if method == "Raycast" then
		local callingScript = getcallingscript()

		if callingScript and callingScript.Name == "ClientPhysics" then
			local _, info = findClientPhysicsLevel()

			if info and info.name == "" then
				return false
			end
		end
	end

	if method == "Create" then
		---@note: Fix object if we're using a lighting template.
		local lightingTemplate = select(3, ...)

		if typeof(lightingTemplate) == "table" then
			if
				lightingTemplate["FogStart"]
				and lightingTemplate["FogEnd"]
				and Configuration.expectToggleValue("NoFog")
			then
				lightingTemplate["FogStart"] = 9e9
				lightingTemplate["FogEnd"] = 9e9
			end

			if lightingTemplate["Ambient"] and Configuration.expectToggleValue("ModifyAmbience") then
				lightingTemplate["Ambient"] = modifyAmbienceColor(lightingTemplate["Ambient"])
			end

			if lightingTemplate["Density"] and Configuration.expectToggleValue("NoFog") then
				lightingTemplate["Density"] = 0
			end

			---@note: lightingTemplate is mutated in place (same table reference), so
			return oldNameCall(self, ...)
		end
	end

	if Configuration.expectToggleValue("InfoSpoofing") then
		if typeof(secondArg) == "string" and method == "GetAttribute" and not Spoofing.force then
			local localPlayer = playersService.LocalPlayer
			local character = localPlayer and localPlayer.Character
			local humanoid = game.FindFirstChild(character, "Humanoid")
			local foreign = true

			if character and humanoid and (self.Parent == character or self.Parent == humanoid) then
				foreign = false
			end

			if foreign and not Configuration.expectToggleValue("SpoofOtherPlayers") then
				return oldNameCall(self, ...)
			end

			if secondArg == "FirstName" then
				return foreign and "Linoria V2" or Configuration.expectOptionValue("SpoofedFirstName")
			end

			if secondArg == "LastName" then
				return foreign and "On Top" or Configuration.expectOptionValue("SpoofedLastName")
			end

			if secondArg == "CharacterName" then
				local characterName = Configuration.expectOptionValue("SpoofedFirstName")
					.. " "
					.. Configuration.expectOptionValue("SpoofedLastName")

				return foreign and "Linoria V2 On Top" or characterName
			end

			if secondArg == "Guild" then
				return foreign and "discord.gg/lyc" or Configuration.expectOptionValue("SpoofedGuild")
			end

			if secondArg == "GuildRich" then
				return foreign and "discord.gg/lyc" or Configuration.expectOptionValue("SpoofedGuildName")
			end
		end
	end

	if name == "ActivateMantra" and method == "FireServer" then
		local blockMantra = false
		if not checkcaller() then
			local ok, result = pcall(Defense.shouldBlockInput, "Mantras")
			blockMantra = ok and result == true
		end

		if blockMantra then
			return
		end

		StateListener.lMantraActivated = secondArg

		onInterceptedInput(INPUT_CAST, INPUT_TYPE_BEFORE)

		if setnamecallmethod then
			setnamecallmethod(method)
		end

		local result = oldNameCall(self, ...)

		onInterceptedInput(INPUT_CAST, INPUT_TYPE_AFTER)

		return result
	end

	if name == "ClientEffectBatch" and method == "FireServer" then
		return
	end

	return oldNameCall(self, ...)
end)

---@return any
local onUnreliableFireServer = LPH_NO_VIRTUALIZE(function(self, ...)
	if banRemotes[self] then
		return Logger.warn("(%s) Anticheat is calling a unreliable ban remote.", self.Name)
	end

	-- getCachedRemote (pure cache read, never the encoder) -- PR resolves these in setup and only pure-reads in
	-- the global fire hooks. The keys are prewarmed once at init (Hooking.init -> KeyHandling.prewarm).
	local leftClickRemote = KeyHandling.getCachedRemote("LeftClick")
	local criticalRemote = KeyHandling.getCachedRemote("CriticalClick")
	local feintClickRemote = KeyHandling.getCachedRemote("FeintClick")
	local offhandAttackRemote = KeyHandling.getCachedRemote("OffhandAttack")

	local inputType = nil

	if criticalRemote and self == criticalRemote then
		inputType = INPUT_CRITICAL
	end

	if leftClickRemote and self == leftClickRemote then
		inputType = INPUT_LEFT_CLICK
	end

	if (feintClickRemote and self == feintClickRemote) or (offhandAttackRemote and self == offhandAttackRemote) then
		inputType = INPUT_RIGHT_CLICK
	end

	if inputType then
		-- SelfFire.active (a plain boolean) instead of checkcaller(): this is a process-global hook that runs in
		-- the game's parallel Actor VMs too, and checkcaller() there is reflection mid-parallel-phase = Kernel,2090.
		-- luma raises SelfFire around its own attack re-fires (InputClient), so `not SelfFire.active` == "the game
		-- fired this", exactly what checkcaller() meant -- but as a reflection-free state read, the way PR does it.
		local blockMove = BLOCK_INPUT_MOVE_MAP[inputType]
		if not SelfFire.active then
			local ok, blocked = pcall(Defense.shouldBlockInput, blockMove)
			if ok and blocked then
				return
			end
		end

		if inputType == INPUT_LEFT_CLICK and not SelfFire.active then
			local ok, handled = pcall(Defense.tryAutoMixup)
			if ok and handled then
				return
			end
		end

		onInterceptedInput(inputType, INPUT_TYPE_BEFORE)

		local result = oldUnreliableFireServer(self, ...)

		onInterceptedInput(inputType, INPUT_TYPE_AFTER)

		return result
	end

	return oldUnreliableFireServer(self, ...)
end)

---@return any
local onFireServer = LPH_NO_VIRTUALIZE(function(self, ...)
	if banRemotes[self] then
		return Logger.warn("(%s) Anticheat is calling a ban remote.", self.Name)
	end

	local blockRemote = KeyHandling.getCachedRemote("Block")

	local inputType = nil

	if blockRemote and self == blockRemote then
		inputType = INPUT_BLOCK
	end

	if inputType then
		onInterceptedInput(inputType, INPUT_TYPE_BEFORE)

		local result = oldFireServer(self, ...)

		onInterceptedInput(inputType, INPUT_TYPE_AFTER)

		return result
	end

	return oldFireServer(self, ...)
end)

---@return number
local getWalkSpeedMultiplier = LPH_NO_VIRTUALIZE(function()
	if os.clock() - lastWalkSpeedRefresh > WALK_SPEED_REFRESH then
		local min = Configuration.expectOptionValue("WalkSpeedMultiplierMinimum") or 1.0
		local max = Configuration.expectOptionValue("WalkSpeedMultiplierMaximum") or 1.0

		if min > max then
			min, max = max, min
		end

		walkSpeedMultiplier = walkSpeedRandom:NextNumber(min, max)
		lastWalkSpeedRefresh = os.clock()
	end

	return walkSpeedMultiplier
end)

---@return any
local onNewIndex = LPH_NO_VIRTUALIZE(function(self, index, value, ...)
	if checkcaller() then
		return oldNewIndex(self, index, value, ...)
	end

	if typeof(self) ~= "Instance" then
		return oldNewIndex(self, index, value, ...)
	end

	---@note: Only these keys are ever rewritten below, so bail cheaply on anything
	if
		index ~= "Text"
		and index ~= "ActiveController"
		and index ~= "MouseIconEnabled"
		and index ~= "Ambient"
		and index ~= "WalkSpeed"
	then
		return oldNewIndex(self, index, value, ...)
	end

	---@note: Scale the game's own walkspeed writes for our local humanoid. Because
	if index == "WalkSpeed" and Configuration.expectToggleValue("WalkSpeedMultiplier") and typeof(value) == "number" then
		local localPlayer = playersService.LocalPlayer
		local character = localPlayer and localPlayer.Character

		if character and self.Parent == character and isA(self, "Humanoid") then
			return oldNewIndex(self, index, value * getWalkSpeedMultiplier(), ...)
		end
	end

			if self.Name == "Slot" and self.Parent and self.Parent.Name == "CharacterInfo" then
				return oldNewIndex(self, index, Configuration.expectOptionValue("SpoofedSlotString"))
			end

			if self.Name == "GameVersion" then
				return oldNewIndex(self, index, Configuration.expectOptionValue("SpoofedGameVersion"))
			end

			if self.Name == "Date" then
				return oldNewIndex(self, index, Configuration.expectOptionValue("SpoofedDateString"))
			end
		end
	end

	if Configuration.expectToggleValue("NoClip") and Configuration.expectToggleValue("Fly") then

			if self.Parent then
				local airController = self:FindFirstChild("AirController")
				return oldNewIndex(self, index, airController or value)
			end
		end
	end

	end

		if self == lighting and Configuration.expectToggleValue("ModifyAmbience") then
			return oldNewIndex(self, index, modifyAmbienceColor(value))
		end
	end

	return oldNewIndex(self, index, value, ...)
end)

---@return any
-- EXACTLY Project Rain's tostring hook: the crash-string trap and NOTHING else. `tostring` is hooked on the
-- process-global base function, so this runs inside EVERY Luau VM -- including Deepwoken's parallel Actor
-- VMs. The old body walked the Lua stack (a debug.getinfo loop over InputClient frames) and ran
-- InputClient.cache() (getgc / getconnections / LoadAnimation) from right here; whenever a parallel Actor
-- happened to call tostring, all of that main-VM reflection executed mid-parallel-phase -- the illegal
-- cross-phase operation the engine Kernel-panics on (RBXCRASH: FatalRuntimeError (RSL - panic: ...Kernel)).
-- That is THE crash: it fired randomly tens of minutes in, was independent of InfoSpoofing, and survived
-- every hook-SET change because the panic lived in the hook BODY, not the set. PR's tostring is this exact
-- 3-line pass-through. InputClient caching now runs on the main thread off CharacterAdded (InputClient.lua
-- -> InputClient.ensureCached), never again from a cross-VM hook.
local onToString = LPH_NO_VIRTUALIZE(function(str, ...)
	if str == "EEKEWAEJIWAJDOIWAJDIOJAWDIOJAWODJOAIW" then
		Logger.warn("Crash was attempted.")
		return coroutine.yield()
	end

	return oldToString(str, ...)
end)

local maintainNoFall = LPH_NO_VIRTUALIZE(function()
	local module = noFallModule
	if not module then
		return
	end

	local on = Configuration.expectToggleValue("NoFallDamage") and true or false

	if on then
		pcall(function()
			if not module:FindEffect("NoFall") then
				module:CreateEffect("NoFall")
			end
		end)
	elseif noFallLastOn then
		pcall(function()
			local effect = module:FindEffect("NoFall")
			if effect then
				effect:Debris(0.1)
			end
		end)
	end

	noFallLastOn = on
end)

---@return string
local generateRandomName = LPH_NO_VIRTUALIZE(function()
	local guid = httpService:GenerateGUID(false):lower():gsub("[%-]*", "")
	local length = math.random(8, 16)
	local built = guid:sub(1, length)
	local capitalizedLast = false

	for i = 1, length do
		local prefix = built:sub(1, i - 1)
		local char = built:sub(i, i)
		local suffix = built:sub(i + 1)

		local asNumber = tonumber(char)
		if asNumber then
			char = string.char(103 + asNumber)
		end

		if i == 1 or (math.random(1, 6) == 1 and not capitalizedLast and i < length) then
			char = char:upper()
			capitalizedLast = true
		else
			capitalizedLast = false
		end

		built = prefix .. char .. suffix
	end

	return built:sub(1, length)
end)

local SPRINT_KEYS = {
	[Enum.KeyCode.W] = true,
	[Enum.KeyCode.A] = true,
	[Enum.KeyCode.S] = true,
	[Enum.KeyCode.D] = true,
}
local autoSprintConnection = nil
local autoSprintActive = false

local onSprintInput = LPH_NO_VIRTUALIZE(function(input, gameProcessed)
	if gameProcessed or not SPRINT_KEYS[input.KeyCode] or autoSprintActive then
		return
	end
	if not Configuration.expectToggleValue("AutoSprint") then
		return
	end
	autoSprintActive = true

	local sprintDelay = 0
	if Configuration.expectToggleValue("AutoSprintDelay") then
		sprintDelay = Configuration.expectOptionValue("AutoSprintDelayTime") or 0
	end

	task.delay(sprintDelay, function()
		if not autoSprintActive or not Configuration.expectToggleValue("AutoSprint") then
			return
		end

		local localPlayer = playersService.LocalPlayer
		local character = localPlayer and localPlayer.Character
		local humanoid = character and character:FindFirstChildWhichIsA("Humanoid")
		local characterHandler = character and character:FindFirstChild("CharacterHandler")
		local requests = characterHandler and characterHandler:FindFirstChild("Requests")
		local stopSprint = requests and requests:FindFirstChild("StopSprint")
		if not humanoid or not stopSprint then
			return
		end

		local effectReplicator = getEffectReplicator()
		local effectReplicatorModule = effectReplicator and require(effectReplicator)

		for _, connection in next, getconnections(stopSprint.OnClientEvent) do
			local func = connection.Function
			local okInfo, info = pcall(debug.getinfo, func)
			if not okInfo or not info or not tostring(info.source):find("InputClient") then
				continue
			end

			local okEnv, env = pcall(getfenv, func)
			local sprintFn = okEnv and env and env.Sprint
			if type(sprintFn) ~= "function" then
				continue
			end

			while humanoid.MoveDirection.Magnitude >= 0.1 and Configuration.expectToggleValue("AutoSprint") do
				local sprinting = effectReplicatorModule and effectReplicatorModule:FindEffect("Sprinting")
				local crouching = effectReplicatorModule and effectReplicatorModule:FindEffect("ClientCrouch")
				if not sprinting and (not crouching or Configuration.expectToggleValue("AutoSprintOnCrouch")) then
					pcall(sprintFn, true)
				end
				task.wait()
			end
		end
	end)

	local localPlayer = playersService.LocalPlayer
	local humanoid = localPlayer and localPlayer.Character and localPlayer.Character:FindFirstChildWhichIsA("Humanoid")
	if not humanoid then
		autoSprintActive = false
		return
	end

	local moveConn
	local destroyConn
	local function release()
		autoSprintActive = false
		if moveConn then
			moveConn:Disconnect()
		end
		if destroyConn then
			destroyConn:Disconnect()
		end
	end

	moveConn = humanoid:GetPropertyChangedSignal("MoveDirection"):Connect(function()
		if humanoid.MoveDirection.Magnitude <= 0.1 then
			release()
		end
	end)
	destroyConn = humanoid.Destroying:Connect(release)
end)

function Hooking.earlyInit()
	if game.PlaceId == LOBBY_PLACE_ID then
		return
	end

	task.spawn(pcall, function()
		local clientManager = playersService.LocalPlayer:WaitForChild("PlayerScripts"):WaitForChild("ClientActor"):WaitForChild("ClientManager")
		clientManager.Enabled = false
	end)

	oldFireServer = hookfunction(Instance.new("RemoteEvent").FireServer, onFireServer)
	oldUnreliableFireServer = hookfunction(Instance.new("UnreliableRemoteEvent").FireServer, onUnreliableFireServer)
	oldToString = hookfunction(tostring, onToString)

	-- EXACTLY Project Rain's game-hook set: only __namecall and __newindex (plus the FireServer / tostring
	-- hooks above). PR hooks NOTHING else on `game` -- no __index, no tick, no warn/print/GetLogHistory --
	-- because any of those dispatched from Deepwoken's parallel Actor VM is the illegal cross-phase op the
	-- engine Kernel-panics on (RSL - panic: ...Kernel). __index especially fires on every property read, so
	-- it was the hottest path to trip it. Dropping them -- what PR does -- is what stops the crash; the
	-- earlier "confine __index via a metatable swap" still crashed. Cost (same trade PR makes): InfoSpoofing
	-- loses its __index read-intercept (the __namecall/__newindex spoofs + Spoofing.rics value-bounce still
	-- work), AutoSprint-via-tick stops, and the print/warn/GetLogHistory log-hiding is gone.
	if hookmetamethod then
		oldNameCall = hookmetamethod(game, "__namecall", onNameCall)
		oldNewIndex = hookmetamethod(game, "__newindex", onNewIndex)
		usingMetaMethodHooks = true
	else
		oldNameCall = hookfunction(getrawmetatable(game).__namecall, onNameCall)
		oldNewIndex = hookfunction(getrawmetatable(game).__newindex, onNewIndex)
		usingMetaMethodHooks = false
	end
end

---@note: Hooking.earlyInit() must have already run before this.
function Hooking.init()
	if game.PlaceId == LOBBY_PLACE_ID then
		return
	end

	local requests = replicatedStorage:WaitForChild("Requests", 30)
	if not requests then
		return error("Requests folder not found.")
	end

	---@note: Dynamically get the ban remotes.
	local banRemoteCount = 0

	for _, request in next, requests:GetChildren() do
		local hasChangedConnection = #getconnections(request.Changed)
		if hasChangedConnection <= 0 then
			continue
		end

		banRemoteCount = banRemoteCount + 1
		banRemotes[request] = true
	end

	-- Resolve the input-remote keys ONCE here, on the main (synchronized) thread, so the process-global
	-- FireServer / UnreliableFireServer hooks can identify them with getCachedRemote (pure table reads) and
	-- never run the game's encoder from the hook body. KeyHandling.init has already run by this point.
	KeyHandling.prewarm({ "LeftClick", "CriticalClick", "FeintClick", "OffhandAttack", "Block" })

	local nulledBanRemotes = {}

	for _, instance in next, getnilinstances() do
		if instance.Name ~= "NulledBanRemote" then
			continue
		end

		nulledBanRemotes[#nulledBanRemotes + 1] = instance
	end

	if #nulledBanRemotes ~= 2 then
		if banRemoteCount ~= 2 then
			return error("Anticheat has less or more than two ban remotes.")
		end
	end

	local effectReplicator = replicatedStorage:WaitForChild("EffectReplicator", 30)
	if not effectReplicator then
		return error("EffectReplicator not found.")
	end

	local effectReplicatorModule = require(effectReplicator)

	-- NOTE: luma no longer hooks HasEffect. It was a process-global hookfunction whose body called checkcaller()
	-- (executor reflection) on EVERY HasEffect call -- and the game calls HasEffect constantly, including from its
	-- parallel Actor VMs. Reflection dispatched mid-parallel-phase is the illegal cross-phase op the engine
	-- Kernel-panics on ("RSL - panic ...Kernel,2090") -- the exact crash the old getgc-in-tostring hook caused.
	-- PR never hooks HasEffect; it does NoAttackingClientChecks by REMOVING the attack effects (see
	-- StateListener.onEffectReplicated -> effect:Remove for ATTACK_EFFECT_CLASSES), which luma already does too.
	noFallModule = effectReplicatorModule
	if noFallConnection then
		pcall(function()
			noFallConnection:Disconnect()
		end)
	end
	noFallConnection = runService.Heartbeat:Connect(function()
		maintainNoFall()
	end)

	if autoSprintConnection then
		pcall(function()
			autoSprintConnection:Disconnect()
		end)
	end
	autoSprintConnection = userInputService.InputBegan:Connect(onSprintInput)

	---@note: KeyHandler is now fully emulated (Project Rain style) instead of being

	local replicatedStorage = game:GetService("ReplicatedStorage")
	local modules = replicatedStorage:WaitForChild("Modules", 30)
	local clientManager = modules and modules:WaitForChild("ClientManager", 30)
	local keyHandler = clientManager and clientManager:WaitForChild("KeyHandler", 30)
	if not keyHandler then
		return error("KeyHandler not found.")
	end

	local keyHandlerModule = require(keyHandler)

	---@note: Future require()s of the module now build our emulated table.
	hookedKeyHandlerModule = keyHandlerModule

	oldKeyHandlerModule = hookfunction(
		keyHandlerModule,
		LPH_NO_VIRTUALIZE(function()
			return { emulatedGetKey, emulatedCreateKey, "Luma On Top" }
		end)
	)

	---@note: Replace the already-created get/create pair in memory so existing
	for _, candidate in next, getgc(true) do
		if typeof(candidate) ~= "table" then
			continue
		end

		if getrawmetatable(candidate) then
			continue
		end

		if #candidate ~= 2 then
			continue
		end

		if typeof(candidate[1]) ~= "function" or typeof(candidate[2]) ~= "function" then
			continue
		end

		local getFn = candidate[1]
		local createFn = candidate[2]

		local getOk, getInfo = pcall(debug.getinfo, getFn)
		local createOk, createInfo = pcall(debug.getinfo, createFn)
		if not getOk or not createOk or not getInfo or not createInfo then
			continue
		end

		if not getInfo.source:find("KeyHandler") or not createInfo.source:find("KeyHandler") then
			continue
		end

		hookedGetFn = getFn
		hookedCreateFn = createFn

		oldGetKey = hookfunction(getFn, emulatedGetKey)
		oldCreateKey = hookfunction(createFn, emulatedCreateKey)

		candidate[1] = emulatedGetKey
		candidate[2] = emulatedCreateKey

		break
	end

	if not hookedGetFn then
		Logger.warn("KeyHandler get/create pair not found in getgc; module hook only.")
	end

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		Logger.warn("Client-side anticheat has been penetrated.")
	end
end

function Hooking.detach()
	local localPlayer = playersService.LocalPlayer

	if autoSprintConnection then
		pcall(function()
			autoSprintConnection:Disconnect()
		end)
		autoSprintConnection = nil
	end
	autoSprintActive = false

	if hookedKeyHandlerModule and oldKeyHandlerModule then
		hookfunction(hookedKeyHandlerModule, oldKeyHandlerModule)
	end

	if hookedGetFn and oldGetKey then
		hookfunction(hookedGetFn, oldGetKey)
	end

	if hookedCreateFn and oldCreateKey then
		hookfunction(hookedCreateFn, oldCreateKey)
	end

	if oldToString then
		hookfunction(tostring, oldToString)
	end

	if oldNameCall then
		if usingMetaMethodHooks and hookmetamethod then
			hookmetamethod(game, "__namecall", oldNameCall)
		else
			hookfunction(getrawmetatable(game).__namecall, oldNameCall)
		end
	end

	if oldNewIndex then
		if usingMetaMethodHooks and hookmetamethod then
			hookmetamethod(game, "__newindex", oldNewIndex)
		else
			hookfunction(getrawmetatable(game).__newindex, oldNewIndex)
		end
	end

	if oldFireServer then
		hookfunction(Instance.new("RemoteEvent").FireServer, oldFireServer)
	end

	if oldUnreliableFireServer then
		hookfunction(Instance.new("UnreliableRemoteEvent").FireServer, oldUnreliableFireServer)
	end

	local effectReplicator = getEffectReplicator()
	local effectReplicatorModule = effectReplicator and require(effectReplicator)


	-- Stop maintaining NoFall and clear any effect we left behind.
	if noFallConnection then
		pcall(function()
			noFallConnection:Disconnect()
		end)
		noFallConnection = nil
	end
	if effectReplicatorModule then
		pcall(function()
			local effect = effectReplicatorModule:FindEffect("NoFall")
			if effect then
				effect:Debris(0.1)
			end
		end)
	end
	noFallModule = nil
	noFallLastOn = false

	usingMetaMethodHooks = false

	Logger.warn("Pulled out of client-side anticheat.")
end

return Hooking
end)

local function installRawSetter(item)
    if type(item) == "table" and type(item.SetValue) == "function" and item.SetRawValue == nil then
        item.SetRawValue = function(self, value)
            local callback = self.Callback
            local changed = self.Changed

            self.Changed = nil
            local ok, result = pcall(self.SetValue, self, value)
            self.Callback = callback
            self.Changed = changed
            if not ok then error(result, 2) end

local function patchRegistry(registry)
    if type(registry) ~= "table" then

    local currentMeta = getmetatable(registry) or {}
    local previousNewIndex = currentMeta.__newindex
    currentMeta.__newindex = function(target, key, value)
        installRawSetter(value)
        if type(previousNewIndex) == "function" then
            return previousNewIndex(target, key, value)
        elseif type(previousNewIndex) == "table" then
            previousNewIndex[key] = value
        else
            rawset(target, key, value)
        end
    end
    setmetatable(registry, currentMeta)
end
