		return false
	end

	return (entityRoot.Position - localRootPart.Position):Dot(localRootPart.CFrame.LookVector) < 0
end)

---@param self Defender
---@param entity Model
---@return Target?
Defender.target = LPH_NO_VIRTUALIZE(function(self, entity)
	return Targeting.find(entity)
end)

---@param self Defender
---@param timing AnimationTiming
---@return number
Defender.fsecs = LPH_NO_VIRTUALIZE(function(self, timing)
	local player = players:GetPlayerFromCharacter(self.entity)
	local sd = (player and player:GetAttribute("AveragePing") or 50.0) / 2000
	-- self._gateDelay is the anti-breaker auth/quarantine delay (0 on the normal path); adding it
	-- keeps the hitbox extrapolation leading correctly when a cue arrives late under a flood.
	return (timing.pfht or 0.15) + (sd + Latency.rdelay()) + (self._gateDelay or 0)
end)

---@param self Defender
---@param ref Model|Part
---@param timing Timing
---@param info RepeatInfo
Defender.srpue = LPH_NO_VIRTUALIZE(function(self, ref, timing, info)
	local cache = {
		["name"] = timing.name,
		["imdd"] = timing.imdd,
		["imxd"] = timing.imxd,
		["rsd"] = timing:rsd(),
		["rpd"] = timing:rpd(),
	}

	local target = self:target(ref)
	local part = target and target.root or CFrame.new()

	if ref:IsA("Part") then
		part = ref
	end

	local options = HitboxOptions.new(part, timing)
	options.spredict = not timing.duih
	options.ptime = self:fsecs(timing)
	options.entity = ref:IsA("Model") and target or nil
	options.hmid = info.hmid
	options:ucache()

	self:mark(Task.new(string.format("RPUE_%s_%i", timing.name, 0), function()
		return cache["rsd"] - info.irdelay - Latency.sdelay()
	end, timing.punishable, timing.after, self.rpue, self, ref, timing, info, cache, options))

	if Environment.isTester() then
		self:notify(
			timing,
			"Added RPUE '%s' (%.2fs, then every %.2fs) with ping '%.2f' (changing) subtracted.",
			cache["name"],
			cache["rsd"],
			cache["rpd"],
			Latency.rtt()
		)
	else
		self:notify(
			timing,
			"Added RPUE '%s' ([redacted], then every [redacted]) with ping '%.2f' (changing) subtracted.",
			cache["name"],
			Latency.rtt()
		)
	end
end)

---@param self Defender
---@param ref Model|Part
---@param timing Timing
---@param info RepeatInfo
---@param cache table Cache table for RPUE to prevent unnecessary recalculations.
---@param options HitboxOptions
Defender.rpue = LPH_NO_VIRTUALIZE(function(self, ref, timing, info, cache, options)
	local distance = self:distance(ref)
	if not distance then
		return Logger.warn("Stopping RPUE '%s' because the distance is not valid.", cache.name)
	end

	if not self:rc(info) then
		return Logger.warn("Stopping RPUE '%s' because the repeat condition is not valid.", cache.name)
	end

	local target = ref:IsA("Model") and self:target(ref) or true
	local success = false
	local reasons = {}

	options.part = ref:IsA("Part") and ref or (target and target.root)
	options.cframe = not options.part and CFrame.new()

	if timing.duih and target then
		success = self:hc(options, info)
		reasons[#reasons + 1] = string.format("hitbox (%s)", tostring(options:hitbox()))
	end

	local within = (distance >= cache.imdd and distance <= cache.imxd)

	if timing then
		success = within
	end

	if not within then
		reasons[#reasons + 1] = string.format("distance range (%.2f < %.2f > %.2f)", cache.imdd, distance, cache.imxd)
	end

	info.index = info.index + 1

	self:mark(Task.new(string.format("RPUE_%s_%i", timing.name, info.index), function()
		return cache.rpd - info.irdelay - Latency.sdelay()
	end, timing.punishable, timing.after, self.rpue, self, ref, timing, info, cache, options))

	if not target then
		return Logger.warn("Skipping RPUE '%s' because the target is not valid.", cache.name)
	end

	if not success then
		return Logger.warn("Skipping RPUE '%s' (%s)", cache.name, #reasons > 1 and table.concat(reasons, ", ") or "N/A")
	end

	self:notify(timing, "Action type 'RPUE Parry' is being executed.")

	self:parry(timing, nil)
end)

---@param self Defender
---@param options ValidationOptions
---@return boolean
Defender.valid = LPH_NO_VIRTUALIZE(function(self, options)
	local integer = Random.new():NextNumber(1.0, 100.0)
	local rate = Configuration.expectOptionValue("FailureRate") or 0.0
	local timing = options.timing

	local function internalNotifyFunction(...)
		if not options.notify then
			return
		end

		return self:notify(...)
	end

	local overrideData = Library:GetOverrideData(timing.name)

	if overrideData then
		rate = overrideData.fr
	end

	if (Configuration.expectToggleValue("AllowFailure") or overrideData) and integer <= rate then
		return internalNotifyFunction(timing, "(%i <= %i) Intentionally did not run.", integer, rate)
	end

	if self.hits and next(self.hits) then
		return internalNotifyFunction(timing, "Already landed a hit on the attacker; skipping to not contest M1 priority.")
	end

	local selectedFilters = Configuration.expectOptionValue("AutoDefenseFilters") or {}

	if selectedFilters["Disable While Holding Block"] and StateListener.hblock() then
		return internalNotifyFunction(timing, "User is pressing down on a key binded to Block.")
	end

	if selectedFilters["Disable While Using Payback"] and StateListener.pback() then
		return internalNotifyFunction(timing, "User is using the 'Payback' move.")
	end

	local chatInputBarConfiguration = textChatService:FindFirstChildOfClass("ChatInputBarConfiguration")

	if
		selectedFilters["Disable When Textbox Focused"]
		and (userInputService:GetFocusedTextBox() or chatInputBarConfiguration.IsFocused)
	then
		return internalNotifyFunction(timing, "User is typing in a text box.")
	end

	if selectedFilters["Disable While Using Sightless Beam"] and StateListener.csb() then
		return internalNotifyFunction(timing, "User is using the 'Sightless Beam' move.")
	end

	if selectedFilters["Disable When Window Not Active"] and not iswindowactive() then
		return internalNotifyFunction(timing, "Window is not active.")
	end

	if selectedFilters["Disable During Chime Countdown"] and StateListener.ccd() then
		return internalNotifyFunction(timing, "Chime countdown is active.")
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return internalNotifyFunction(timing, "No effect replicator found.")
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return internalNotifyFunction(timing, "No effect replicator module found.")
	end

	if
		selectedFilters["Dont Parry If In Payback"]
		and (
			(effectReplicatorModule:FindEffect("PaybackHealTally") and tick() - StateListener.lastPaybackDelay > 60)
			or effectReplicatorModule:FindEffect("DelayedPayback")
		)
	then
		return internalNotifyFunction(timing, "User is in Payback.")
	end

	local actionType = options.action and options.action._type or "N/A"

	if
		not Configuration.expectToggleValue("BlatantRoll")
		or (actionType ~= "Dodge" and actionType ~= "Forced Full Dodge")
	then
		if not self.afeinted and not options.sstun and StateListener.astun() then
			return internalNotifyFunction(timing, "User is in action stun.")
		end

		if effectReplicatorModule:FindEffect("Knocked") then
			return internalNotifyFunction(timing, "User is knocked.")
		end
	end

	if not timing.iapf and actionType == "Parry" then
		if effectReplicatorModule:FindEffect("AutoParry") then
			return internalNotifyFunction(timing, "User has auto parry frames.")
		end
	end

	if timing.tag == "M1" and selectedFilters["Filter Out M1s"] then
		return internalNotifyFunction(timing, "Attacker is using a 'M1' attack.")
	end

	if timing.tag == "Mantra" and selectedFilters["Filter Out Mantras"] then
		return internalNotifyFunction(timing, "Attacker is using a 'Mantra' attack.")
	end

	if timing.tag == "Critical" and selectedFilters["Filter Out Criticals"] then
		return internalNotifyFunction(timing, "Attacker is using a 'Critical' attack.")
	end

	if timing.tag == "Undefined" and selectedFilters["Filter Out Undefined"] then
		return internalNotifyFunction(timing, "Attacker is using an 'Undefined' attack.")
	end

	if timing.rsc and StateListener.recentlyCast(timing.rsaid, timing.rsaw) then
		return internalNotifyFunction(timing, "User recently played the same animation themselves (%s).", timing.rsaid)
	end

	return true
end)

---@note: Solara fallback.
local function checkParts(parts, filter)
	for _, part in next, parts do
		for _, fpart in next, filter do
			if part ~= fpart and not part:IsDescendantOf(fpart) then
				continue
			end

			return true
		end
	end

	return false
end

	return self.uids
end

---@param self Defender
---@param identifier number? If the identifier is nil, then we will auto-generate one for each visualization.
---@param cframe CFrame
---@param size Vector3
---@param color Color3
---@param shape Enum.PartType
Defender.visualize = LPH_NO_VIRTUALIZE(function(self, identifier, cframe, size, color, shape)
	local id = identifier or self:uid(10)
	local vpart = self.hmaid[id] or Instance.new("Part")

	pcall(function()
		vpart.Parent = workspace
	end)

	if vpart.Parent then
		vpart.Anchored = true
		vpart.CanCollide = false
		vpart.CanQuery = false
		vpart.CanTouch = false
		vpart.Material = Enum.Material.ForceField
		vpart.CastShadow = false
		vpart.Size = size
		vpart.CFrame = cframe
		vpart.Color = color
		vpart.Shape = shape
		vpart.Transparency = Configuration.expectToggleValue("EnableVisualizations") and 0.2 or 1.0
	end

	if self.hmaid[id] then
		return
	end

	self.hmaid[id] = vpart

	debrisService:AddItem(vpart, MAX_VISUALIZATION_TIME)
end)

---@param self Defender
---@param cframe CFrame
---@param fd boolean
---@param soffset number
---@param size Vector3
---@param filter Instance[]
---@param shape Enum.PartType
---@return boolean?, CFrame?
Defender.hitbox = LPH_NO_VIRTUALIZE(function(self, cframe, fd, soffset, size, filter, shape)
	local shouldManualFilter = getexecutorname
		and (getexecutorname():match("Solara") or getexecutorname():match("Xeno"))

	local overlapParams = OverlapParams.new()
	overlapParams.FilterDescendantsInstances = shouldManualFilter and {} or filter
	overlapParams.FilterType = shouldManualFilter and Enum.RaycastFilterType.Exclude or Enum.RaycastFilterType.Include

	local character = players.LocalPlayer.Character
	if not character then
		return nil, nil
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then
		return nil, nil
	end

	local usedCFrame = cframe

	if fd then
		usedCFrame = usedCFrame * CFrame.new(0, 0, -(size.Z / 2))
	end

	if soffset and soffset ~= 0 then
		usedCFrame = usedCFrame * CFrame.new(0, 0, soffset)
	end

	local simulationPart = Instance.new("Part")
	simulationPart.Size = size
	simulationPart.Material = Enum.Material.ForceField
	simulationPart.Shape = shape
	simulationPart.CFrame = usedCFrame

	if shape == Enum.PartType.Cylinder then
		simulationPart.CFrame = usedCFrame * CFrame.Angles(0, 0, math.rad(90))
	end

	local parts = workspace:GetPartsInPart(simulationPart, overlapParams)

	return shouldManualFilter and checkParts(parts, filter) or #parts > 0, usedCFrame
end)

---@param self Defender
---@param from Model? | BasePart?
---@param pair TimingContainerPair
---@param name string
---@param key string
---@return Timing?
Defender.initial = LPH_NO_VIRTUALIZE(function(self, from, pair, name, key)
	local timing = pair:index(key)

	local distance = self:distance(from)
	if not distance then
		return nil
	end

	if timing then
		self:known(self.__type, key, timing.name, distance, from and tostring(from.Parent) or nil)

		local md = timing.imdd

		if md <= 0.01 then
			md = 0.0
		end

		if distance < md or distance > timing.imxd then
			return nil
		end
	end

	---@note: Ignore return value.
	if not timing then
		self:miss(self.__type, key, name, distance, from and tostring(from.Parent) or nil)
		return false
	end

	return timing
end)

---@param self Defender
---@param options HitboxOptions
---@param info RepeatInfo
---@return boolean
Defender.duih = LPH_NO_VIRTUALIZE(function(self, options, info)
	local clone = options:clone()
	clone.hmid = info.hmid
	clone:ucache()

	while task.wait() do
		if not self:rc(info) then
			return false
		end

		if not self:hc(clone, nil) then
			continue
		end

		return true
	end
end)

---@param self Defender
---@param options HitboxOptions
---@param info RepeatInfo? Pass this in if you want to use the delay until in hitbox.
---@return boolean
Defender.hc = LPH_NO_VIRTUALIZE(function(self, options, info)
	local action = options.action
	local timing = options.timing

	local function innerVisualize(...)
		if not options.visualize then
			return
		end

		return self:visualize(...)
	end

	local character = players.LocalPlayer.Character
	if not character then
		return false
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then
		return false
	end

	if action and action.ihbc then
		return true
	end

	if info then
		return self:duih(options, info)
	end

	local hitbox = options:hitbox()
	local eposition = options.spredict and options:extrapolate() or nil
	local position = options:pos()
	local pt = timing.htype or Enum.PartType.Block

	local result, usedCFrame = self:hitbox(position, timing.fhb, timing.hso, hitbox, options.filter, pt)

	if usedCFrame then
		innerVisualize(options.hmid, usedCFrame, hitbox, options:ghcolor(result), pt)
		innerVisualize(options.hmid and options.hmid + 1 or nil, root.CFrame, root.Size, options:ghcolor(result), pt)
	end

	if not options.spredict or result then
		return result
	end

	local closest = EntityHistory.pclosest(players.LocalPlayer, tick() - (Latency.rtt() * PREDICTION_LENIENCY_MULTI))
	if not closest then
		return false
	end

	local store = OriginalStore.new()

	store:run(root, "CFrame", closest, function()
		result, usedCFrame = self:hitbox(eposition, timing.fhb, timing.hso, hitbox, options.filter, pt)
	end)

	if usedCFrame then
		innerVisualize(options.hmid and options.hmid + 1 or nil, usedCFrame, hitbox, options:gphcolor(result), pt)
		innerVisualize(options.hmid and options.hmid + 1 or nil, closest, root.Size, options:gphcolor(result), pt)
	end

	return result
end)

---@param self Defender
---@param timing Timing
---@param action Action
---@param started number
Defender.handle = LPH_NO_VIRTUALIZE(function(self, timing, action, started)
	local actionType = action._type

	if actionType == "End Block" then
		QueuedBlocking.stop("Defender_StartBlock")
	end

	if
		Configuration.expectToggleValue("AutoFeint")
		and not timing.duih
		and action:when() > 0
		and self.__type == "Animation"
		and actionType ~= "End Block"
		and autoFeintAgainstAllowed(timing)
	then
		self:afeint(timing, action, started, false)
	end

	if actionType ~= "End Block" then
		if not self:valid(ValidationOptions.new(action, timing)) then
			return
		end
	end

	local redactedMap = {
		["Start Slide"] = "1",
		["End Slide"] = "2",
		["Teleport Up"] = "3",
		["Forced Full Dodge"] = "4",
		["Jump"] = "5",
		["Start Block"] = "6",
		["End Block"] = "7",
		["Parry"] = "8",
		["Dodge"] = "9",
		["Start Crouch"] = "10",
		["End Crouch"] = "11",
	}

	if LRM_UserNote then
		self:notify(timing, "Action type '%s' is being executed.", redactedMap[actionType] or actionType)
	else
		self:notify(timing, "Action type '%s' is being executed.", actionType)
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	if actionType == "Start Block" then
		return QueuedBlocking.invoke(QueuedBlocking.BLOCK_TYPE_NORMAL, "Defender_StartBlock", 20.0)
	end

	local options = DodgeOptions.new()
	options.rollCancel = Configuration.expectToggleValue("RollCancel") and actionType ~= "Forced Full Dodge"
	options.rollCancelDelay = Configuration.expectOptionValue("RollCancelDelay") or 0.0
	options.direct = Configuration.expectToggleValue("BlatantRoll")
	options.full = actionType == "Forced Full Dodge"

	local inIFrame = effectReplicatorModule:FindEffect("Immortal")
		or effectReplicatorModule:FindEffect("Dodge")
		or effectReplicatorModule:FindEffect("Ghost")
	
	if (actionType == "Dodge" or actionType == "Forced Full Dodge") and Configuration.expectToggleValue("ParryOnly") then
		actionType = "Parry"
		self:notify(timing, "Action 'Dodge' changed to 'Parry'")
	end

	if actionType == "Dodge" then
		if Configuration.expectToggleValue("UseIFrames") and inIFrame then
			return
		end

		return InputClient.dodge(options)
	end

	if actionType == "Forced Full Dodge" then
		return InputClient.dodge(options)
	end

	---@note: End block is handled elsewhere.
	if actionType == "End Block" then
		return
	end

	if actionType == "Start Crouch" then
		return InputClient.crouch(true)
	end

	if actionType == "End Crouch" then
		return InputClient.crouch(false)
	end

	if actionType == "Start Slide" then
		local serverSlide = KeyHandling.getRemote("ServerSlide")
		if not serverSlide then
			return
		end

		return serverSlide:FireServer(true)
	end

	if actionType == "End Slide" then
		local serverSlideStop = KeyHandling.getRemote("ServerSlideStop")
		if not serverSlideStop then
			return
		end

		return serverSlideStop:FireServer(false)
	end

	if actionType == "Start Crouch" then
		local character = players.LocalPlayer.Character
		if not character then
			return
		end

		local characterHandler = character:FindFirstChild("CharacterHandler")
		if not characterHandler then
			return
		end

		local requests = characterHandler:FindFirstChild("Requests")
		if not requests then
			return
		end

		local serverCrouch = requests:FindFirstChild("ServerCrouch")
		if not serverCrouch then
			return
		end

		return serverCrouch:FireServer(true)
	end

	if actionType == "End Crouch" then
		local character = players.LocalPlayer.Character
		if not character then
			return
		end

		local characterHandler = character:FindFirstChild("CharacterHandler")
		if not characterHandler then
			return
		end

		local requests = characterHandler:FindFirstChild("Requests")
		if not requests then
			return
		end

		local serverCrouch = requests:FindFirstChild("ServerCrouch")
		if not serverCrouch then
			return
		end

		return serverCrouch:FireServer(false)
	end

	if actionType == "Jump" then
		local humanController = InputClient.getHumanController()
		if not humanController then
			return
		end

		if effectReplicatorModule:HasAny("Swimming", "Jumped", "NoJump", "Landed") then
			return
		end

		if not humanController:Jump() then
			return
		end

		return InputClient.ejump()
	end

	if actionType == "Teleport Up" then
		local character = players.LocalPlayer.Character
		if not character then
			return
		end

		local humanoidRootPart = character:FindFirstChild("HumanoidRootPart")
		if not humanoidRootPart then
			return
		end

		if Finder.pnear(humanoidRootPart.Position, 500) then
			return self:notify(timing, "Action 'Teleport Up' blocked because there are players nearby.")
		end

		humanoidRootPart.CFrame = CFrame.new(humanoidRootPart.Position + Vector3.new(0, 75, 0))
	end

	self:parry(timing, action)
end)

---@param self Defender
---@param timing Timing
---@param action Action
Defender.SimulateInput = LPH_NO_VIRTUALIZE(function(self, timing, action)
	if not Configuration.expectToggleValue("EnableSimulation") then
		return
	end

	if not action or not action._type then
		return
	end

	local actionType = action._type

	local function holdTime()
		local min = Configuration.expectOptionValue("SimulationHoldMinimum") or 0.1
		local max = Configuration.expectOptionValue("SimulationHoldMaximum") or 0.15

		if min > max then
			min, max = max, min
		end

		return Random.new():NextNumber(min, max)
	end

	local function simulateRightClick(hold)
		if type(mouse2press) == "function" and type(mouse2release) == "function" then
			mouse2press()
			task.delay(hold, function()
				mouse2release()
			end)
			return
		end

		keypress(0x02)
		task.delay(hold, function()
			keyrelease(0x02)
		end)
	end

	local selectedFilters = Configuration.expectOptionValue("AutoDefenseFilters") or {}

	if selectedFilters["Disable While Holding Block"] and StateListener.hblock() then
		return
	end

	local chatInputBarConfiguration = textChatService:FindFirstChildOfClass("ChatInputBarConfiguration")
	if
		selectedFilters["Disable When Textbox Focused"]
		and (userInputService:GetFocusedTextBox() or (chatInputBarConfiguration and chatInputBarConfiguration.IsFocused))
	then
		return
	end

	if selectedFilters["Disable While Using Sightless Beam"] and StateListener.csb() then
		return
	end

	if selectedFilters["Disable When Window Not Active"] and not iswindowactive() then
		return
	end

	if selectedFilters["Disable During Chime Countdown"] and StateListener.ccd() then
		return
	end

	if timing and timing.tag == "M1" and selectedFilters["Filter Out M1s"] then
		return
	end

	if timing and timing.tag == "Mantra" and selectedFilters["Filter Out Mantras"] then
		return
	end

	if timing and timing.tag == "Critical" and selectedFilters["Filter Out Criticals"] then
		return
	end

	if timing and timing.tag == "Undefined" and selectedFilters["Filter Out Undefined"] then
		return
	end

	if actionType == "Parry" then
		keypress(0x46)
		task.delay(holdTime(), function()
			keyrelease(0x46)
		end)

	elseif actionType == "Dodge" or actionType == "Forced Full Dodge" then
		keypress(0x51)

		if Configuration.expectToggleValue("RollCancel") then
			task.delay(0.01, function()
				simulateRightClick(holdTime())
			end)
		end

		task.delay(holdTime(), function()
			keyrelease(0x51)
		end)

	elseif actionType == "Vent" then
		keypress(0x47)
		task.delay(holdTime(), function()
			keyrelease(0x47)
		end)

	elseif actionType == "Feint" then
		simulateRightClick(holdTime())
	end
end)

---@param self Defender
---@param timing Timing
---@param action Action
Defender.parry = LPH_NO_VIRTUALIZE(function(self, timing, action)
	local options = DodgeOptions.new()
	options.rollCancel = Configuration.expectToggleValue("RollCancel")
	options.rollCancelDelay = Configuration.expectOptionValue("RollCancelDelay") or 0.0
	options.direct = Configuration.expectToggleValue("BlatantRoll")

	local rate = (Configuration.expectOptionValue("DashInsteadOfParryRate") or 0.0)
	local overrideData = Library:GetOverrideData(timing.name)
	if overrideData then
		rate = overrideData.dipr
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local effectReplicatorModule = effectReplicator and require(effectReplicator)

	local inIFrame = effectReplicatorModule:FindEffect("Immortal")
		or effectReplicatorModule:FindEffect("DodgeFrame")
		or effectReplicatorModule:FindEffect("ParryFrame")
		or effectReplicatorModule:FindEffect("Ghost")

	local dashReplacement = Random.new():NextNumber(1.0, 100.0) <= rate

	if action and action._type ~= "Parry" then
		dashReplacement = false
	end

	if not Configuration.expectToggleValue("AllowFailure") and not overrideData then
		dashReplacement = false
	end

	if timing.umoa or timing.actions:count() ~= 1 then
		dashReplacement = false
	end
	
	if Configuration.expectToggleValue("ParryOnly") then
		dashReplacement = false
	end

	local function internalNotify(...)
		if timing.rpue and timing.srpn then
			return
		end

		return self:notify(...)
	end

	-- Dodge Behind Back: instead of parrying, roll behind the attacker, roll cancel and m1 (the same
	-- roll-behind sequence as Auto Mixup).
	if
		Configuration.expectToggleValue("DodgeBehindBack")
		and StateListener.cdodge()
		and Random.new():NextNumber(1.0, 100.0) <= (Configuration.expectOptionValue("DodgeBehindBackProbability") or 0.0)
	then
		local attacker = self.entity
		local attackerRoot = attacker and attacker:FindFirstChild("HumanoidRootPart")

		-- Don't roll behind an attacker that's too far away (>10 studs) -- the roll-behind only makes
		-- sense in close range.
		local localCharacter = players.LocalPlayer and players.LocalPlayer.Character
		local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
		local tooFar = not localRoot
			or not attackerRoot
			or (attackerRoot.Position - localRoot.Position).Magnitude > 10

		if attackerRoot and not tooFar then
			local Defense = require("Features/Combat/Defense")
			if Defense and Defense.rollCancelBehindBack then
				self:SimulateInput(timing, { _type = "Dodge" })
				task.spawn(function()
					Defense.rollCancelBehindBack(attackerRoot)
				end)
				return internalNotify(timing, "Dodge Behind Back: rolled behind and punished.")
			end
		end
	end

	if Configuration.expectToggleValue("UseIFrames") and inIFrame then
		return internalNotify(timing, "Action 'Parry' blocked because there are already existing IFrames.")
	end
	
	if self.hits and next(self.hits) then
		return internalNotify(timing, "Action 'Parry' blocked because opponent is in hitstun.")
	end

	if
		Configuration.expectToggleValue("RollIfBehind")
		and StateListener.cdodge()
		and not timing.ndfb
		and self:behind()
	then
		self:SimulateInput(timing, { _type = "Dodge" })
		InputClient.dodge(options)

		return internalNotify(timing, "Action type 'Parry' rolled because attack is behind.")
	end

	if StateListener.cparry() then
		if timing.ndfb or not StateListener.cdodge() or not dashReplacement then
			if self.__type == "Animation" and self.track then
				runService.Heartbeat:Wait()

				if self:stopped(self.track, timing) then
					return internalNotify(timing, "Feint Detected")
				end
			end

			self:SimulateInput(timing, { _type = "Parry" })
			return QueuedBlocking.invoke(QueuedBlocking.BLOCK_TYPE_DEFLECT, "Defender_Deflect", nil)
		end

		internalNotify(timing, "Action type 'Parry' replaced to 'Dodge' type.")

		self:SimulateInput(timing, { _type = "Dodge" })
		return InputClient.dodge(options)
	end

	local canBlock = Configuration.expectToggleValue("DeflectBlockFallback")
		and not timing.nbfb
		and StateListener.cblock()

	local canVent = StateListener.cvent()
		and Configuration.expectToggleValue("VentFallback")
		and not timing.nvfb
		and not timing.rpue
		and self.__type ~= "Part"

	local canDodge = StateListener.cdodge()
		and Configuration.expectToggleValue("RollOnParryCooldown")
		and not timing.ndfb

	if canDodge then
		self:SimulateInput(timing, { _type = "Dodge" })
		InputClient.dodge(options)

		return internalNotify(timing, "Action type 'Parry' fallback to 'Dodge' type.")
	end

	if canVent then
		self:SimulateInput(timing, { _type = "Vent" })
		InputClient.vent()

		return internalNotify(timing, "Action type 'Parry' fallback to 'Vent' type.")
	end

	if canBlock then
		self:SimulateInput(timing, { _type = "Start Block" })
		QueuedBlocking.invoke(QueuedBlocking.BLOCK_TYPE_NORMAL, "Defender_BlockFallback", timing.bfht)

		return internalNotify(timing, "Action type 'Parry' fallback to 'Block' type.")
	end

	return internalNotify(timing, "Action 'Parry' blocked because no fallbacks are available.")
end)

---@param self Defender
---@return boolean
Defender.blocking = LPH_NO_VIRTUALIZE(function(self)
	for _, marker in next, self.markers do
		if not marker then
			continue
		end

		return true
	end

	for _, task in next, self.tasks do
		if not task:blocking() then
			continue
		end

		return true
	end
end)

---@param task Task
function Defender:mark(task)
	---@note: Anti-breaker occurrence stamping. Only set while the armed anti-flow has an active
	---occurrence context (see AnimatorDefender:apProcess); nil on the normal path, so tasks are
	---untouched and the standard parry path stays byte-identical.
	if self._apStampTrack ~= nil then
		if task.sourceTrack == nil then
			task.sourceTrack = self._apStampTrack
		end
		if task.sourceOccurrence == nil then
			task.sourceOccurrence = self._apStampOccurrence
		end
	end

	self.tasks[#self.tasks + 1] = task
	Defender.armed[self] = true
end

function Defender:clhook()
	for key, old in next, self.rhook do
		if not self[key] then
			continue
		end

		self[key] = old
	end

	self.rhook = {}
end

---@param self Defender
Defender.clean = LPH_NO_VIRTUALIZE(function(self)
	for idx, task in next, self.tasks do
		if task.forced then
			continue
		end

		task:cancel()

		self.tasks[idx] = nil

		if task.identifier ~= "End Block" and task.identifier ~= "Start Block" then
			continue
		end

		QueuedBlocking.stop("Defender_StartBlock")
	end

	self:clhook()

	self.tmaid:clean()

	self.markers = {}

	self.hmaid:clean()

	self.afeinted = false

	if next(self.tasks) == nil then
		Defender.armed[self] = nil
	end
end)

end)

---@param self Defender
---@param timing Timing
---@param action Action
---@param started number Timestamp of when the auto feint task started.
---@param initial boolean Whether this is the initial auto feint check.
Defender.afeint = LPH_NO_VIRTUALIZE(function(self, timing, action, started, initial)
    local function innerNotify(...)
        if initial then return end
        return self:notify(...)
    end

    local lfaction = StateListener.lAnimFaction
    if not lfaction then
        return innerNotify(timing, "Auto feint blocked: No local action.")
    end

    if not autoFeintOwnAllowed() then
        return innerNotify(timing, "Auto feint blocked: Our own attack type isn't selected.")
    end

    local latimestamp = StateListener.lAnimTimestamp
    if latimestamp and (os.clock() - latimestamp) > (lfaction:when() * 0.8) then
        return innerNotify(timing, "Auto feint blocked: Attack is too late to feint")
    end

    if not StateListener.cfeint() then
        return innerNotify(timing, "Auto feint blocked: Unable to feint.")
    end

    local options = ValidationOptions.new(action, timing)
    options.sstun = true
    options.notify = false
    options.visualize = false

    if not self:valid(options) then
        return innerNotify(timing, "Auto feint failed: Action invalid.")
    end

    if not self.ifeinted then
        self:notify(timing, "Legacy auto feint executed.")
    end

    if not initial then
        self.afeinted = true
    else
        self.ifeinted = true
    end

    InputClient.feint()
end)

---@param self Defender

---@param action Action

	local actionType = action._type
    local isParryOrDodge = (actionType == "Parry" or actionType == "Dodge")

	local rdelay = self.offset or Latency.rdelay()

	if not LPH_OBFUSCATED and actionType == "Parry" and self.__type == "Animation" then
		local rtt = rdelay + Latency.sdelay()

		Logger.warn(
			"(%s) [FeintDiag] Scheduling '%s' at clock=%.3f: when=%.3fs rtt=%.3fs -> fireDelay=%.3fs",
			timing.name,
			action.name,
			os.clock(),
			action:when(),
			rtt,
			action:when() - rtt
		)
	end

	local atask = Task.new(action._type, function()
		return action:when() - rdelay - Latency.sdelay()
	end, timing.punishable, timing.after, self.handle, self, timing, action, os.clock())

	if timing.forced then
		atask.forced = true
	end

	self:mark(atask)

	local actionWhen = action:when()

	if Configuration.expectToggleValue("UsePrediction") and isParryOrDodge and actionWhen > counterLead("Prediction") then
		self:mark(Task.new("Prediction", function()
			return action:when() - rdelay - Latency.sdelay() - counterLead("Prediction")
		end, timing.punishable, timing.after, self.prediction, self, timing, action, os.clock()))
	end

	if Configuration.expectToggleValue("UseEqualizer") and isParryOrDodge and actionWhen > counterLead("Equalizer") then
		self:mark(Task.new("Equalizer", function()
			return action:when() - rdelay - Latency.sdelay() - counterLead("Equalizer")
		end, timing.punishable, timing.after, self.Equalizer, self, timing, action, os.clock()))
	end

	if Configuration.expectToggleValue("UsePunishment") and isParryOrDodge and actionWhen > counterLead("Punishment") then
		self:mark(Task.new("Punishment", function()
			return action:when() - rdelay - Latency.sdelay() - counterLead("Punishment")
		end, timing.punishable, timing.after, self.punishment, self, timing, action, os.clock()))
	end

	if Configuration.expectToggleValue("UseUnbidden") and isParryOrDodge and actionWhen > counterLead("Unbidden") then
        self:mark(Task.new("Unbidden", function()
            return action:when() - rdelay - Latency.sdelay() - counterLead("Unbidden")
        end, timing.punishable, timing.after, self.unbidden, self, timing, action, os.clock()))
    end

	if Configuration.expectToggleValue("UseIllusoryCounter") and isParryOrDodge and actionWhen > counterLead("IllusoryCounter") then
		self:mark(Task.new("IllusoryCounter", function()
			return action:when() - rdelay - Latency.sdelay() - counterLead("IllusoryCounter")
		end, timing.punishable, timing.after, self.illusorycounter, self, timing, action, os.clock()))
	end

	if
		Configuration.expectToggleValue("AutoFeint")
		and not timing.duih
		and action:when() > 0
		and self.__type == "Animation"
		and action._type ~= "End Block"
		and autoFeintAgainstAllowed(timing)
	then
		self:mark(Task.new(action._type, function()
			return (action:when() - rdelay - Latency.sdelay()) / 2
		end, timing.punishable, timing.after, self.afeint, self, timing, action, os.clock(), true))
	end
	
	if Environment.isTester() then
		self:notify(
			timing,
			"Added action '%s' (%.2fs) with ping '%.2f' (changing) subtracted.",
			action.name,
			action:when(),
			Latency.rtt()
		)
	else
		self:notify(
			timing,
			"Added action '%s' ([redacted]) with ping '%.2f' (changing) subtracted.",
			action.name,
			Latency.rtt()
		)
	end
end)

---@param self Defender
---@param timing Timing
Defender.actions = LPH_NO_VIRTUALIZE(function(self, timing)
	for _, action in next, timing.actions:get() do
		self:action(timing, action)
	end
end)

---@param key string
---@param new function
---@return boolean, function
function Defender:hook(key, new)
	if self.rhook[key] then
		Logger.warn("Cannot hook '%s' because it is already hooked.", key)
		return false, nil
	end

	local old = self[key]

	if typeof(old) ~= "function" then
		Logger.warn("Cannot hook '%s' because it is not a function.", key)
		return false, nil
	end

	self[key] = new

	self.rhook[key] = old

	Logger.warn("Hooked '%s' with new function.", key)

	return true, old
end

function Defender:detach()
	self:clean()
	self.maid:clean()

	self.hmaid:clean()

	Defender.armed[self] = nil

	self = nil
end

---@return Defender
function Defender.new()
	local self = setmetatable({}, Defender)
	self.tasks = {}
	self.rhook = {}
	self.tmaid = Maid.new()
	self.maid = Maid.new()
	self.hmaid = Maid.new()
	self.uids = 0
	self.markers = {}
	self.lvisualization = os.clock()
	self.afeinted = false
	self.ifeinted = false
	return self
end

return Defender
end)

	end

		self.data[self.index] = self.value
	end)

	self.stored = false
end)

	self:restore()
	self.data = nil
	self.index = nil
	self.value = nil
	self.stored = false

	local plain = text:match("^rbxassetid://(%d+)$")

	if idOccurrences ~= 1 then

	end

	local real = text:match("[?&]id=(%d+)")

---@field manimations table<number, Animation>

---@field sct table<AnimationTrack, boolean> Tracks that need to have their speed changed.

AnimatorDefender.__index = AnimatorDefender
AnimatorDefender.__type = "Animation"

local players = game:GetService("Players")
local replicatedStorage = game:GetService("ReplicatedStorage")

local MAX_REPEAT_TIME = 5.0
local HISTORY_STEPS = 5.0
local PREDICT_FACING_DELTA = 3
local GALE_FEINT_GRACE = 0.1
local FEINT_CUE_GRACE_PERIOD = 0.15

-- Roll whether we react to a feint on this timing. Honors the timing's per-move override (rtfr) and
-- otherwise only reacts when Allow Failure is on (default rate 0 = never blindly react).
---@param timing AnimationTiming?
---@return boolean
local function reactRoll(timing)
	local reactRate = Configuration.expectOptionValue("ReactToFeintsRate") or 0.0
	local overrideData = timing and Library:GetOverrideData(timing.name)
	if overrideData and overrideData.rtfr then
		reactRate = overrideData.rtfr
	elseif not Configuration.expectToggleValue("AllowFailure") then
		return false
	end

	return Random.new():NextNumber(1.0, 100.0) <= reactRate
end

local MAX_ANIMATIONS_PER_SECOND = 50
local animationProcessWindowStart = os.clock()
local animationProcessCount = 0

local function underAnimationProcessCap()
	local now = os.clock()
	if now - animationProcessWindowStart >= 1 then
		animationProcessWindowStart = now
		animationProcessCount = 0
	end

	animationProcessCount = animationProcessCount + 1
	return animationProcessCount <= MAX_ANIMATIONS_PER_SECOND
end

---@param self AnimatorDefender
---@param track AnimationTrack
---@param timing AnimationTiming
---@param notify boolean? Whether to notify or not.
---@return boolean
AnimatorDefender.stopped = LPH_NO_VIRTUALIZE(function(self, track, timing, notify)
	local function internalNotifyFunction(...)
		if not notify then
			return
		end

		return self:notify(...)
	end

	if self._feintedTrack == track then
		return true, internalNotifyFunction(timing, "Feint cue was heard for this track.")
	end

	local overrideData = Library:GetOverrideData(timing.name)
	local rate = (Configuration.expectOptionValue("IgnoreAnimationEndRate") or 0.0)

	if overrideData then
		rate = overrideData.iaer
	end

	if
		(Configuration.expectToggleValue("AllowFailure") or overrideData)
		and not timing.rpue
		and Random.new():NextNumber(1.0, 100.0) <= rate
		and StateListener.cdodge()
	then
		return false, internalNotifyFunction(timing, "Intentionally ignoring animation end to simulate human error.")
	end

	if timing.nfd then
		return false
	end

	-- Anti-AP Breaker: a real attack still resolves server-side even when the attacker Stop()s and
	-- replays the animation client-side (the core of their breaker -- stop, then replay at Core
	-- priority with TimePosition 0). While the anti is armed, ignore the animation end so a faked stop
	-- can't abort the parry -- same effect as Allow Failure's ignore-animation-end, forced on at 100%.
	-- RPUE timings keep their normal end handling (they rely on it to know when to stop repeating).
	if not timing.rpue and Configuration.expectToggleValue("AntiBreakerEnabled") then
		return false, internalNotifyFunction(timing, "Anti-AP Breaker: ignoring animation end.")
	end

	if not timing.iae and not track.IsPlaying and self._respectFeint then
		return true, internalNotifyFunction(timing, "Animation stopped playing.")
	end

	if timing.iae and not timing.ieae and not track.IsPlaying and track.TimePosition < track.Length and self._respectFeint then
		return true, internalNotifyFunction(timing, "Animation stopped playing early.")
	end
end)

---@param self AnimatorDefender
---@param info RepeatInfo
---@return boolean
AnimatorDefender.rc = LPH_NO_VIRTUALIZE(function(self, info)
	---@note: There are cases where we might not have a track. If it's not handled properly, it will throw an error.
	if not info.track then
		return Logger.warn(
			"(%s) Did you forget to pass the track? Or perhaps you forgot to place a hook before using this function.",
			info.timing.name
		)
	end

	if self:stopped(info.track, info.timing) then
		return false
	end

	if info.timing.iae and os.clock() - info.start >= ((info.timing.mat / 1000) or MAX_REPEAT_TIME) then
		return self:notify(info.timing, "Max animation timeout exceeded.")
	end

	return true
end)

---@param self AnimatorDefender
---@param options HitboxOptions
---@return boolean
AnimatorDefender.pfh = LPH_NO_VIRTUALIZE(function(self, options)
	local root = self.entity:FindFirstChild("HumanoidRootPart")
	if not root then
		return false
	end

	local localRoot = players.LocalPlayer.Character and players.LocalPlayer.Character:FindFirstChild("HumanoidRootPart")
	if not localRoot then
		return false
	end

	if not options.timing.ffh then
		local yrate = EntityHistory.yrate(self.entity)
		if not yrate then
			return false
		end

		if math.abs(yrate) < PREDICT_FACING_DELTA then
			return
		end
	end

	local clone = options:clone()
	clone.hcolor = Color3.new(0, 1, 1)
	clone.mcolor = Color3.new(1, 1, 0)
	clone:ucache()

	local result = false
	local store = OriginalStore.new()

	store:run(root, "CFrame", CFrame.lookAt(root.Position, localRoot.Position), function()
		result = self:hc(clone, nil)
	end)

	return result
end)

---@param timing Timing
---@param options HitboxOptions
---@return boolean
AnimatorDefender.phd = LPH_NO_VIRTUALIZE(function(self, timing, options)
	for _, cframe in next, EntityHistory.pstepped(self.entity, HISTORY_STEPS, timing.phds) or {} do
		local clone = options:clone()
		clone.cframe = cframe
		clone.hcolor = Color3.new(0.839215, 0.976470, 0.537254)
		clone.mcolor = Color3.new(0.564705, 0, 1)
		clone:ucache()

		if not self:hc(clone, nil) then
			continue
		end

		return true
	end
end)

AnimatorDefender.fpc = LPH_NO_VIRTUALIZE(function(self, timing, options)
	if timing.duih then
		return false
	end

	if (timing.ffh or timing.pfh) and self:pfh(options) then
		return true
	end

	if timing.phd and self:phd(timing, options) then
		return true
	end
end)

---@param self AnimatorDefender
---@param options ValidationOptions
---@return boolean
AnimatorDefender.valid = LPH_NO_VIRTUALIZE(function(self, options)
	if not Defender.valid(self, options) then
		return false
	end

	local timing = options.timing
	local action = options.action

	local function internalNotifyFunction(...)
		if not options.notify then
			return
		end

		return self:notify(...)
	end

	if not self.track then
		return internalNotifyFunction(timing, "No current track.")
	end

	if not self.entity then
		return internalNotifyFunction(timing, "No entity found.")
	end

	local target = self:target(self.entity)
	if not target then
		return internalNotifyFunction(timing, "Not a viable target.")
	end

	local root = self.entity:FindFirstChild("HumanoidRootPart")
	if not root then
		return internalNotifyFunction(timing, "No humanoid root part found.")
	end

	local character = players.LocalPlayer.Character
	if not character then
		return internalNotifyFunction(timing, "No character found.")
	end

	local targetInstance = self.entity:FindFirstChild("Target")

	if
		targetInstance
		and targetInstance.Value ~= character
		and Configuration.expectToggleValue("CheckTargetingValue")
	then
		return internalNotifyFunction(timing, "Not being targeted.")
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return internalNotifyFunction(timing, "No effect replicator found.")
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return internalNotifyFunction(timing, "No effect replicator module found.")
	end

	if
		not timing.imb
		and root:FindFirstChild("MegalodauntBroken")
		and not players:GetPlayerFromCharacter(self.entity)
	then
		return internalNotifyFunction(timing, "Entity is block broken.")
	end

	if self:stopped(self.track, timing, options.notify) then
		return false
	end

	if timing.uif then
		local function feinted()
			return self._galeFeintTracks ~= nil and table.find(self._galeFeintTracks, self.track) ~= nil
		end

		local deadline = os.clock() + GALE_FEINT_GRACE
		while not feinted() and os.clock() < deadline do
			task.wait()
		end

		if feinted() and self._respectFeint then
			Logger.warn("(%s) Skipping action '%s', gale feint detected.", self.entity.Name, timing.name)
			return internalNotifyFunction(timing, "Gale feint detected (inhale cue) -- skipping.")
		end
	end

	local hoptions = HitboxOptions.new(root, timing)
	hoptions.spredict = not timing.duih and not timing.dp
	hoptions.ptime = self:fsecs(timing)
	hoptions.action = action
	hoptions.entity = self.entity
	hoptions.visualize = options.visualize
	hoptions:ucache()

	local info = RepeatInfo.new(timing, Latency.rdelay(), self:uid(10))
	info.track = self.track

	local hc = self:hc(hoptions, timing.duih and info or nil)
	if hc then
		return true
	end

	local pc = self:fpc(timing, hoptions)
	if pc then
		return true
	end

	return self:notify(timing, "Not in hitbox.")
end)

---@param self AnimatorDefender
AnimatorDefender.update = LPH_NO_VIRTUALIZE(function(self)
	for track, data in next, self.pbdata do
		if not Configuration.expectToggleValue("ShowAnimationVisualizer") then
			self.pbdata[track] = nil
			continue
		end

		if not track.IsPlaying then
			self.pbdata[track] = nil
			self.rpbdata[tostring(track.Animation.AnimationId)] = data

			continue
		end

		data:astrack(track.Speed)
	end

	if not Configuration.expectToggleValue("AnimationSpeedChanger") then
		return
	end

	local min = Configuration.expectOptionValue("AnimationSpeedMinimum") or 1.0
	local max = Configuration.expectOptionValue("AnimationSpeedMaximum") or 1.3
	local rng = Random.new()

	for track, _ in next, self.sct do
		if not track.IsPlaying then
			self.sct[track] = nil
			self.tsc[track] = nil
			self.multiplier[track] = nil
			continue
		end

		if self.tsc[track] == tostring(track.Speed) then
			continue
		end

		local mode = rng:NextInteger(1, 2)
		local generated = nil

		if not self.multiplier[track] and not Configuration.expectToggleValue("SwitchBetweenSpeeds") then
			generated = rng:NextNumber(min, max)
		end

		if not self.multiplier[track] and Configuration.expectToggleValue("SwitchBetweenSpeeds") then
			generated = mode == 1 and min or max
		end

		if not self.multiplier[track] and generated then
			self.multiplier[track] = generated
		end

		local multiplier = self.multiplier[track]
		if not multiplier then
			continue
		end

		if multiplier == 1.0 then
			continue
		end

		local adjusted = track.Speed * multiplier

		track:AdjustSpeed(adjusted)

		self.tsc[track] = tostring(track.Speed)
	end
end)

---@param self AnimatorDefender
---@param track AnimationTrack
---@return boolean
function AnimatorDefender:isFullyDead(track)
	local humanoid = self.entity:FindFirstChild("Humanoid")
	if not humanoid then
		return true
	end

	return not table.find(humanoid:GetPlayingAnimationTracks(), track)
end

---@param self AnimatorDefender
---@param track AnimationTrack
AnimatorDefender.asc = LPH_NO_VIRTUALIZE(function(self, track)
	if not Configuration.expectToggleValue("AnimationSpeedChanger") then
		return
	end

	if
		Configuration.expectToggleValue("LimitToAPAnimations")
		and not SaveManager.as:index(canonicalAid(track.Animation.AnimationId))
	then
		return
	end

	self.sct[track] = true
end)

-- ---- Ingress-quarantine ring: constants + forward declarations -------------
local APB_QUARANTINE_SIZE = 256
local APB_QUARANTINE_LANE_SIZE = 64
local APB_QUARANTINE_TTL = 0.30
local APB_QUARANTINE_LANE_BASE = { mantra = 0, ["weapon-critical"] = 1, weapon = 2, other = 3 }
local APB_WEAPON_SOUND_LATENCY = 0.04
local apbRecoverIngress -- XR: recover a quarantined candidate from authoritative evidence

-- Category -> quarantine lane (PM).
local function apbLane(category)
	if category == "mantra" or category == "weapon-critical" then
		return category
	end
	if category == "weapon-m1" or category == "weapon-special" then
		return "weapon"
	end
	return "other"
end
-- Lane + cursor -> flat ring index (tM).
local function apbLaneIndex(lane, cursor)
	return APB_QUARANTINE_LANE_BASE[lane] * APB_QUARANTINE_LANE_SIZE + cursor
end
-- Clear one quarantine slot and drop it from the by-track index (YA).
local function apbClearQuarantineSlot(self, slot)
	local track = slot.track
	local lane = slot.lane
	if track and lane then
		local byTrack = self._apQuarantineByTrack and self._apQuarantineByTrack[lane]
		if byTrack and byTrack[track] == slot then
			byTrack[track] = nil
		end
	end
	if track then
		self._apQuarantineCount = math.max(0, (self._apQuarantineCount or 0) - 1)
	end
	slot.track = nil
	slot.timing = nil
	slot.aid = nil
	slot.category = nil
	slot.lane = nil
	slot.startedAt = nil
end
-- (Re)initialise the quarantine ring for this defender (uI).
local function apbInitQuarantine(self)
	local ring = self._apQuarantine
	if not ring then
		ring = table.create(APB_QUARANTINE_SIZE)
		for i = 1, APB_QUARANTINE_SIZE do
			ring[i] = {}
		end
		self._apQuarantine = ring
	else
		for i = 1, APB_QUARANTINE_SIZE do
			local slot = ring[i]
			slot.track = nil
			slot.timing = nil
			slot.aid = nil
			slot.category = nil
			slot.lane = nil
			slot.startedAt = nil
		end
	end
	self._apQuarantineCursors = { mantra = 1, ["weapon-critical"] = 1, weapon = 1, other = 1 }
	self._apQuarantineByTrack = {
		mantra = setmetatable({}, { __mode = "k" }),
		["weapon-critical"] = setmetatable({}, { __mode = "k" }),
		weapon = setmetatable({}, { __mode = "k" }),
		other = setmetatable({}, { __mode = "k" }),
	}
	self._apQuarantinePruneCursor = 1
	self._apQuarantineCount = 0
	self._apRecoveredOccurrences = setmetatable({}, { __mode = "k" })
end
-- Max-parry distance by tag/category (gU).
local function apbMaxDist(timing, base)
	base = base or tonumber(Configuration.expectOptionValue("AutoParryMaxDist")) or 28
	local imxd = timing and tonumber(timing.imxd)
	local tag = timing and timing.tag
	local cap
	if tag == "M1" or tag == "weapon-m1" then
		cap = math.min(base, 35)
	elseif tag == "Critical" or tag == "weapon-critical" then
		cap = math.min(math.max(base, 35), 45)
	elseif tag == "Mantra" then
		cap = 65
	else
		cap = math.max(base, 35)
	end
	if imxd and imxd > 0 then
		return math.min(imxd, cap)
	end
	return cap
end
-- Does an effect's prop belong to this attacker (ru)?
local function apbOwnsProp(self, prop, data)
	if typeof(data) == "table" and data.char == self.entity then
		return true
	end
	if typeof(prop) ~= "Instance" then
		return false
	end
	if prop:IsDescendantOf(self.entity) then
		return true
	end
	local owner = self._playerOwner or players:GetPlayerFromCharacter(self.entity)
	local name = owner and owner.Name or self.entity.Name
	local node = prop
	for _ = 1, 6 do
		if not node then
			break
		end
		if string.find(node.Name, name, 1, true) then
			return true
		end
		node = node.Parent
	end
	return false
end
-- Evidence family covers candidate category (Qb).
local function apbFamilyCovers(category, evidence)
	local families = evidence.families
	if not families or families[category] == true then
		return true
	end
	local weaponFamilies = { ["weapon-m1"] = true, ["weapon-critical"] = true, ["weapon-special"] = true }
	return evidence.allowWeaponFamilyBridge == true
		and weaponFamilies[category] == true
		and weaponFamilies[evidence.exactFamily] == true
end

-- Sliding-window prune helpers (window seconds, cap entries).
local AAPB_WINDOW = 1.0
local AAPB_CAP = 16
local function apbPruneTimes(list, now)
	local w = 0
	for i = 1, #list do
		if now - list[i] <= AAPB_WINDOW then
			w = w + 1
			list[w] = list[i]
		end
	end
	for i = #list, w + 1, -1 do
		list[i] = nil
	end
	local over = #list - AAPB_CAP
	if over > 0 then
		table.move(list, over + 1, #list, 1)
		for i = #list, AAPB_CAP + 1, -1 do
			list[i] = nil
		end
	end
end
local function apbPruneRecent(list, now)
	local w = 0
	for i = 1, #list do
		if now - list[i].time <= AAPB_WINDOW then
			w = w + 1
			list[w] = list[i]
		end
	end
	for i = #list, w + 1, -1 do
		list[i] = nil
	end
	local over = #list - AAPB_CAP
	if over > 0 then
		table.move(list, over + 1, #list, 1)
		for i = #list, AAPB_CAP + 1, -1 do
			list[i] = nil
		end
	end
end

local function apbLower(s)
	return typeof(s) == "string" and string.lower(s) or ""
end
local function apbNorm(s)
	if typeof(s) ~= "string" then
		return ""
	end
	return (string.gsub(string.lower(s), "[^%w]+", ""))
end
local function apbCount(map)
	local n = 0
	for _ in next, map or {} do
		n = n + 1
	end
	return n
end

-- Action aliases from StoredAction / StoredToolName for mantra evidence matching (nL).
local function apbMantraDisplay(s)
	if typeof(s) ~= "string" then
		return nil
	end
	return string.match(s, "{{(.-)}}") or string.match(s, "^RecalledMantra:(.-)%$")
end
local function apbMantraPrefix(s)
	if typeof(s) ~= "string" or s == "" then
		return nil
	end
	local base = string.match(s, "^(.-){{") or s
	base = string.match(base, "([^:]+)$") or base
	base = string.gsub(base, "%$", "")
	if base == "" or string.lower(base) == "mantra" then
		return nil
	end
	return base
end
local function apbActionAliases(entity)
	local out = {}
	local seen = {}
	local storedTool = entity and entity:GetAttribute("StoredToolName")
	local function add(value)
		if typeof(value) ~= "string" or value == "" then
			return
		end
		local norm = apbNorm(value)
		if norm == "" or seen[norm] then
			return
		end
		seen[norm] = true
		out[#out + 1] = value
	end
	add(entity and entity:GetAttribute("StoredAction"))
	add(apbMantraDisplay(storedTool))
	add(apbMantraPrefix(storedTool))
	return out
end

-- ---- Category (BA) ----------------------------------------------------------
local function apbSmod(timing)
	return timing and apbLower(timing.smod) or ""
end
local function apbName(timing)
	return timing and apbLower(timing.name) or ""
end
local function BA(timing)
	if not timing then
		return "other"
	end
	local smod = apbSmod(timing)
	local name = apbName(timing)
	if timing.tag == "Mantra" then
		return "mantra"
	end
	if timing.tag == "Critical" or smod:find("critical", 1, true) or name:find("critical", 1, true) then
		return "weapon-critical"
	end
	if timing.tag == "M1"
		or smod:find("swing", 1, true)
		or smod:find("weapontest", 1, true)
		or smod:find("weaponrunningattacktest", 1, true)
		or smod:find("weaponaerialattacktest", 1, true)
		or smod:find("weaponuppercuttest", 1, true)
		or smod:find("weaponflourishtest", 1, true) then
		return "weapon-m1"
	end
	if smod:find("silentheart", 1, true) or smod:find("veinbreaker", 1, true) or smod:find("curved", 1, true)
		or smod:find("pyre", 1, true) or smod:find("wraithclaw", 1, true) or smod:find("mayhem", 1, true)
		or smod:find("relentless", 1, true) or smod:find("risingstar", 1, true) or smod:find("runningcrit", 1, true) then
		return "weapon-special"
	end
	if timing.umoa then
		return "module"
	end
	return "other"
end
local function RA(category)
	return category == "weapon-m1" or category == "weapon-critical" or category == "weapon-special"
end

-- Haystack string for needle matching: normalized smod/name/id.
local function apbHaystack(timing, aid)
	return table.concat({ apbNorm(timing and timing.smod), apbNorm(timing and timing.name), apbNorm(timing and timing._id), apbNorm(aid) }, " ")
end
local function apbNeedleHit(haystack, needles)
	for _, needle in next, needles or {} do
		if string.find(haystack, needle, 1, true) then
			return true
		end
	end
	return false
end
---@return table? descriptor, string reason
local function qM(timing, category, aid)
	if not RA(category) then
		return nil, "non-weapon-timing"
	end
	local smodKey = apbNorm(timing and timing.smod)
	local specific = smodKey ~= "" and APB_SMOD_DESCRIPTORS[smodKey]
	if specific then
		return specific, specific.reason or "specific-smod"
	end
	local haystack = apbHaystack(timing, aid)
	for _, d in next, APB_SPECIFIC_DESCRIPTORS do
		if apbNeedleHit(haystack, d.needles) then
			return d, d.reason
		end
	end
	if apbNeedleHit(haystack, { "silentheart" }) then
		if apbNeedleHit(haystack, { "light" }) then
			return { classes = { Light = true }, family = "SilentheartLight", reason = "specific-silentheart-class" }, "specific-silentheart-class"
		elseif apbNeedleHit(haystack, { "medium", "med" }) then
			return { classes = { Medium = true }, family = "SilentheartMedium", reason = "specific-silentheart-class" }, "specific-silentheart-class"
		elseif apbNeedleHit(haystack, { "heavy" }) then
			return { classes = { Heavy = true }, family = "SilentheartHeavy", reason = "specific-silentheart-class" }, "specific-silentheart-class"
		end
		return nil, "silentheart-unknown-class"
	end
	for _, d in next, APB_GENERIC_DESCRIPTORS do
		if apbNeedleHit(haystack, d.needles) then
			return d, d.reason
		end
	end
	if apbNeedleHit(haystack, { "weapontest", "weaponaerialattacktest", "weaponflourishtest", "weaponrunningattacktest", "weaponuppercuttest" }) then
		return APB_WEAPONTEST_DESCRIPTOR, APB_WEAPONTEST_DESCRIPTOR.reason
	end
	return nil, "unmapped"
end

-- ---- Weapon context (CR) via AttackOwnership -------------------------------
local function apbHasWeaponContext(ctx)
	return ctx and (ctx.weaponName or ctx.primaryWeapon or ctx.weaponClass or ctx.weaponType or ctx.weaponDamageType)
end
local function apbWeaponKeyOf(ctx)
	if not apbHasWeaponContext(ctx) then
		return nil
	end
	return table.concat({ tostring(ctx.weaponName or ctx.primaryWeapon or "unknown"), tostring(ctx.weaponClass or "unknown"), tostring(ctx.weaponType or "unknown") }, "|")
end
-- key-set membership (p1): does value normalize-match any key in set?
local function apbSetMatch(set, value)
	if not set or typeof(value) ~= "string" or value == "" then
		return false, false
	end
	local nv = apbNorm(value)
	if nv == "" then
		return false, false
	end
	for k in next, set do
		if nv == apbNorm(k) then
			return true, true
		end
	end
	return false, true
end
local function apbExactNameMatch(descriptor, ctx)
	local hay = table.concat({ apbNorm(ctx.primaryWeapon), apbNorm(ctx.weaponName) }, " ")
	if hay == "" then
		return nil, "missing-weapon-name"
	end
	for _, name in next, descriptor.exactNames or {} do
		local n = apbNorm(name)
		if n ~= "" and string.find(hay, n, 1, true) then
			return true, "exact-name"
		end
	end
	return false, "exact-name-conflict"
end
local function apbFamilyMatch(descriptor, ctx)
	if descriptor.classes and ctx.weaponClass then
		local ok = apbSetMatch(descriptor.classes, ctx.weaponClass)
		if ok then
			return true, "class"
		end
		return false, "class-conflict"
	end
	local okType, chkType = apbSetMatch(descriptor.types, ctx.weaponType)
	if okType then
		return true, "type"
	end
	local okDmg, chkDmg = apbSetMatch(descriptor.damageTypes, ctx.weaponDamageType)
	if okDmg then
		return true, "damage-type"
	end
	if chkType or chkDmg then
		return false, "family-conflict"
	end
	return nil, "missing-family-context"
end
-- Build a descriptor from the attacker's replicated weapon name when no static one maps (gu).
local function apbNamedWeaponDescriptor(timing, aid, ctx)
	if not ctx or ctx.status ~= "available" then
		return nil
	end
	local haystack = apbHaystack(timing, aid)
	local exact = {}
	local seen = {}
	for _, value in next, { ctx.primaryWeapon, ctx.weaponName } do
		local norm = apbNorm(value)
		if #norm >= 5 and string.find(haystack, norm, 1, true) and not seen[norm] then
			seen[norm] = true
			exact[#exact + 1] = value
		end
	end
	if #exact == 0 then
		return nil
	end
	return { exactNames = exact, family = ctx.weaponType or ctx.weaponClass or "NamedWeapon", reason = "replicated-weapon-name" }
end
local function CR(entity, timing, category, aid)
	if not entity then
		return { status = "unknown", reason = "no-entity" }
	end
	local player = players:GetPlayerFromCharacter(entity)
	if not player then
		return { status = "not-applicable", reason = "npc" }
	end
	if not RA(category) then
		return { status = "not-applicable", reason = "non-weapon-timing" }
	end
	local descriptor, reason = qM(timing, category, aid)
	local out = { status = "unknown", reason = reason, timingWeaponFamily = descriptor and descriptor.family or nil }
	local ctx = AttackOwnership.weaponContext(player)
	if not ctx then
		return { status = "unknown", reason = "weapon-context-read-failed", expectedDescriptor = descriptor }
	end
	for k, v in next, ctx do
		out[k] = v
	end
	out.status = "unknown"
	out.reason = ctx.status == "available" and reason or (ctx.reason or "weapon-context-unknown")
	out.expectedDescriptor = descriptor
	if ctx.status ~= "available" or not apbHasWeaponContext(ctx) then
		return out
	end
	if not descriptor then
		descriptor = apbNamedWeaponDescriptor(timing, aid, ctx)
		if not descriptor then
			out.reason = reason or "unmapped"
			return out
		end
		reason = descriptor.reason
		out.expectedDescriptor = descriptor
		out.timingWeaponFamily = descriptor.family
		out.reason = reason
	end
	local exactType, checkedType = apbSetMatch(descriptor.types, ctx.weaponType)
	out.exactWeaponTypeMatch = exactType == true
	out.exactWeaponTypeChecked = checkedType == true
	if descriptor.anyWeapon then
		out.status = "confirmed"
		out.reason = descriptor.reason or "any-weapon-context"
		return out
	end
	local ok, why
	if descriptor.exactNames then
		ok, why = apbExactNameMatch(descriptor, ctx)
	else
		ok, why = apbFamilyMatch(descriptor, ctx)
	end
	if ok == true then
		out.status = "confirmed"
		out.reason = why
	elseif ok == false then
		out.status = "mismatch"
		out.reason = why
	else
		out.reason = why or out.reason
	end
	return out
end
-- Weapon key for the currently equipped weapon (C1).
local function C1(entity)
	local player = players:GetPlayerFromCharacter(entity)
	if not player then
		return nil
	end
	local ok, ctx = pcall(AttackOwnership.weaponContext, player)
	if not ok or not ctx or ctx.status ~= "available" then
		return nil
	end
	return apbWeaponKeyOf(ctx)
end

-- GL: resolve a timing to its ownership descriptor by trying smod, then name, then _id, then aid.
local function GL(timing, aid)
	local candidates = {}
	local function tryKey(value, label)
		if typeof(value) ~= "string" or value == "" or value == "N/A" then
			return nil
		end
		candidates[#candidates + 1] = label .. "=" .. value
		local descriptor = APB_OWNERSHIP_DESCRIPTORS[value]
		if descriptor then
			return descriptor, label .. "=" .. value
		end
	end
	local d, src = tryKey(timing and timing.smod, "smod")
	if not d then
		d, src = tryKey(timing and timing.name, "name")
	end
	if not d then
		d, src = tryKey(timing and timing._id, "_id")
	end
	if not d then
		d, src = tryKey(aid, "aid")
	end
	return d, src or "unmapped", table.concat(candidates, "|")
end
local function bM(entity, timing, aid)
	local player = players:GetPlayerFromCharacter(entity)
	if not player then
		return { status = "not-applicable", reason = "npc" }
	end
	local descriptor, source = GL(timing, aid)
	local result = AttackOwnership.toolOwnership(player, descriptor)
	result.mappingSource = source
	if descriptor == nil then
		result.reason = "unmapped"
	end
	return result
end

-- ---- Mantra store (sM) -----------------------------------------------------
-- Does the attacker's StoredAction / StoredToolName correspond to this timing?
local function apbTimingTokens(timing, aid)
	local out = {}
	local function add(s)
		local n = apbNorm(s)
		if n ~= "" then
			out[n] = true
		end
	end
	add(timing and timing.smod)
	add(timing and timing.name)
	add(timing and timing._id)
	add(aid)
	if typeof(timing and timing.name) == "string" then
		add(timing.name:match("{{(.-)}}"))
	end
	return out
end
local function sM(entity, timing, aid)
	if not entity then
		return { status = "unknown", reason = "no-entity" }
	end
	local storedAction = entity:GetAttribute("StoredAction")
	local storedTool = entity:GetAttribute("StoredToolName")
	local tokens = apbTimingTokens(timing, aid)
	local candidates = {}
	local function check(value)
		local n = apbNorm(value)
		if n == "" then
			return nil
		end
		candidates[#candidates + 1] = tostring(value)
		if tokens[n] then
			return n
		end
	end
	local key = check(storedAction) or check(storedTool)
	if key then
		return { status = "match", reason = "stored-mantra-match", key = key, storedAction = storedAction, storedTool = storedTool }
	end
	if #candidates > 0 then
		local mismatchKey = apbNorm(storedAction)
		if mismatchKey == "" then
			mismatchKey = apbNorm(storedTool)
		end
		return { status = "mismatch", reason = "stored-mantra-mismatch", key = mismatchKey, storedAction = storedAction, storedTool = storedTool }
	end
	return { status = "unknown", reason = "missing-stored-mantra", storedAction = storedAction, storedTool = storedTool }
end

-- ---- Ether tracking (ka/KI/vI) ---------------------------------------------
local function apbReadEther(inst)
	if not inst then
		return nil
	end
	local ok, value = pcall(function()
		return inst.Value
	end)
	if ok and type(value) == "number" then
		return value
	end
	return nil
end
function AnimatorDefender:_apbEtherUpdate(value, now)
	if type(value) ~= "number" then
		return
	end
	local prev = self._apEtherValue
	if prev ~= nil and value < prev - 0.01 then
		self._apEtherLastSpendAt = now
		self._apEtherLastSpendAmount = prev - value
		self._apEtherSpendSerial = (self._apEtherSpendSerial or 0) + 1
		self._apEtherClaimedSerial = nil
		self._apMantraIntentSerial = nil
		self:apMantraSpend(self._apEtherSpendSerial)
	end
	self._apEtherValue = value
	if not self._apEtherResolvedAt then
		self._apEtherResolvedAt = now
	end
end
function AnimatorDefender:_apbWatchEther(ether, now)
	if self._apEther == ether then
		return
	end
	if self._apEtherConn then
		pcall(function()
			self._apEtherConn:Disconnect()
		end)
		self._apEtherConn = nil
	end
	self._apEther = ether
	self._apEtherValue = nil
	self._apEtherResolvedAt = now
	if not ether then
		return
	end
	self:_apbEtherUpdate(apbReadEther(ether), now)
	self._apEtherConn = ether.Changed:Connect(function(v)
		self:_apbEtherUpdate((type(v) == "number") and v or apbReadEther(ether), os.clock())
	end)
	self.apMaid:add(self._apEtherConn)
end
---Resolve whether a mantra is backed by a recent ether spend (apme).
function AnimatorDefender:apme(timing, aid, now)
	local entity = self.entity
	local store = sM(entity, timing, aid)
	local ether = entity and entity:FindFirstChild("Ether")
	self:_apbWatchEther(ether, now)
	if not ether then
		return { status = "unknown", reason = "missing-ether" }
	end
	self:_apbEtherUpdate(apbReadEther(ether), now)
	if self._apEtherValue == nil then
		return { status = "unknown", reason = "unreadable-ether" }
	end
	if self._apEtherLastSpendAt then
		local age = now - self._apEtherLastSpendAt
		if age <= 1 then
			local serial = self._apEtherSpendSerial or 0
			local key = store.key or "unknown"
			local result = { value = self._apEtherValue, age = age, delta = self._apEtherLastSpendAmount, serial = serial, context = store.status, contextReason = store.reason, contextKey = key }
			if self._apEtherClaimedSerial == serial then
				result.status = "claimed"
				result.reason = "ether-spend-already-claimed"
				return result
			end
			if store.status == "mismatch" then
				result.status = "mismatch"
				result.reason = "stored-mantra-mismatch"
				return result
			end
			self._apEtherClaimedSerial = serial
			result.status = "spent"
			result.reason = store.status == "match" and "recent-ether-drop-matched" or "recent-ether-drop-unmatched"
			return result
		end
	end
	if self._apEtherResolvedAt and now - self._apEtherResolvedAt <= 0.2 then
		return { status = "unknown", reason = "fresh-ether-baseline", value = self._apEtherValue, context = store.status, contextReason = store.reason, contextKey = store.key }
	end
	return { status = "no-spend", reason = "no-recent-ether-drop", value = self._apEtherValue, context = store.status, contextReason = store.reason, contextKey = store.key }
end

-- ---- Real-hit geometry (apbh) ----------------------------------------------
function AnimatorDefender:apbh(timing)
	local root = self.entity and self.entity:FindFirstChild("HumanoidRootPart")
	if not root then
		return false, "no-root"
	end
	if not players.LocalPlayer.Character then
		return false, "no-local-character"
	end
	if not self:target(self.entity) then
		return false, "not-target"
	end
	local firstAction = nil
	if timing.actions and timing.actions:count() > 0 then
		for _, a in next, timing.actions:get() do
			firstAction = a
			break
		end
	end
	local ok, options = pcall(function()
		return HitboxOptions.new(root, timing)
	end)
	if not ok or not options then
		return false, "no-options"
	end
	options.spredict = not timing.duih and not timing.dp
	options.ptime = self:fsecs(timing)
	options.action = firstAction
	options.entity = self.entity
	pcall(function()
		options:ucache()
	end)
	if self:hc(options, nil) then
		return true, "direct-hitbox"
	end
	if timing.duih then
		return false, "delay-until-hitbox"
	end
	if (timing.ffh or timing.pfh) and self:pfh(options) then
		return true, "facing-prediction"
	end
	if timing.phd and self:phd(timing, options) then
		return true, "past-position"
	end
	return false, "no-real-hit-evidence"
end

-- ---- History maintenance (capb) + record (rapb) ----------------------------
function AnimatorDefender:capb(now)
	if now < (self._apNextMaintenanceAt or 0) then
		return
	end
	self._apNextMaintenanceAt = now + 0.05
	for track, info in next, self._apActive do
		if not (track.IsPlaying and now - info.start <= 2) then
			if info.stopConn then
				pcall(function()
					info.stopConn:Disconnect()
				end)
				info.stopConn = nil
			end
			if info.suspicious and not info.shortRecorded and now - info.start <= 0.12 then
				self._apShortDurations[#self._apShortDurations + 1] = now
			end
			self._apActive[track] = nil
			self._apActiveCount = math.max(0, (self._apActiveCount or 1) - 1)
		end
	end
	apbPruneTimes(self._apGlobalHistory, now)
	apbPruneTimes(self._apShortDurations, now)
	apbPruneRecent(self._apRecentIds, now)
	for aid, list in next, self._apHistory do
		apbPruneTimes(list, now)
		if #list == 0 then
			self._apHistory[aid] = nil
		end
	end
end
function AnimatorDefender:rapb(track, aid, report, now)
	local category = report and report.category or "other"
	local hist = self._apHistory[aid]
	if not hist then
		hist = {}
		self._apHistory[aid] = hist
	end
	hist[#hist + 1] = now
	self._apGlobalHistory[#self._apGlobalHistory + 1] = now
	self._apRecentIds[#self._apRecentIds + 1] = { time = now, aid = aid, category = category }
	self._apLastKnownTime = now
	apbPruneTimes(hist, now)
	apbPruneTimes(self._apGlobalHistory, now)
	apbPruneRecent(self._apRecentIds, now)
	if report and (report.score or 0) >= 3 then
		if not self._apActive[track] then
			self._apActiveCount = (self._apActiveCount or 0) + 1
		end
		local info = { start = now, suspicious = true, aid = aid }
		self._apActive[track] = info
		info.stopConn = track.Stopped:Connect(function()
			local w = self._apActive[track]
			if not w or w ~= info then
				return
			end
			local t = os.clock()
			if info.suspicious and not info.shortRecorded and (t - info.start) <= 0.12 then
				info.shortRecorded = true
				self._apShortDurations[#self._apShortDurations + 1] = t
				apbPruneTimes(self._apShortDurations, t)
			end
			if info.stopConn then
				pcall(function()
					info.stopConn:Disconnect()
				end)
				info.stopConn = nil
			end
		end)
		self.apMaid:add(info.stopConn)
	end
end
local function apbActiveSuspicious(map)
	local n = 0
	for _, info in next, map do
		if info.suspicious then
			n = n + 1
		end
	end
	return n
end

-- ---- Heuristic classifier (aapb) -- exact ----------------------------------
function AnimatorDefender:aapb(track, timing, aid, now)
	now = now or os.clock()
	self:capb(now)

	local isPlayer = players:GetPlayerFromCharacter(self.entity) ~= nil
	local tr, m4, iCount = 0, 0, 0
	local signals = {}
	local function t(points, reason)
		tr = tr + points
		iCount = iCount + 1
		signals[#signals + 1] = reason
	end
	local function oa(points)
		if points > 0 then
			m4 = m4 + points
		end
	end

	local ownership = bM(self.entity, timing, aid)
	local category = BA(timing)
	local weaponCtx = CR(self.entity, timing, category, aid)
	local mantraEther = nil
	local Gr = false
	local bt = false
	local NA = weaponCtx.status == "mismatch"
	local EA = weaponCtx.status == "confirmed"
	local zo = false
	local Tt = false

	if not isPlayer then
		oa(4)
	end
	if NA then
		t(4, "weapon-context-mismatch")
	elseif EA then
		oa(3)
	end
	if timing.umoa and (EA or ownership.status == "confirmed") then
		oa(2)
	end

	if isPlayer and category == "mantra" then
		mantraEther = self:apme(timing, aid, now)
		if mantraEther.status == "spent" then
			zo = true
			oa(8)
		elseif mantraEther.status == "no-spend" then
			Gr = true
			t(8, "mantra-no-ether-spend")
		elseif mantraEther.status == "claimed" then
			Gr = true
			t(8, "mantra-ether-spend-claimed")
		elseif mantraEther.status == "mismatch" then
			Gr = true
			t(8, "mantra-ether-spend-mismatch")
		end
	end

	if ownership.status == "missing" then
		t(2, "ownership-missing")
	elseif ownership.status == "confirmed" then
		oa(4)
	end

	if timing.rpue then
		Tt = true
		oa(3)
	end
	if timing.duih then
		Tt = true
		if EA or ownership.status == "confirmed" or not isPlayer then
			oa(2)
		end
	end

	local wt = tonumber(track.WeightTarget) or 0
	local wc = tonumber(track.WeightCurrent) or 0
	local weight = wt > 0 and wt or wc
	local speed = tonumber(track.Speed) or 0
	local length = tonumber(track.Length) or 0
	local priority = track.Priority
	local lowWeight = false
	local badPriority = false
	local badSpeed = false
	local shortLen = false
	local blendGap = false
	local extremeSpeed = false

	if wt >= 1 and wc <= 0.01 then
		blendGap = true
		t(2, "high-target-zero-current")
	elseif wt >= 0.5 and wc <= 0.12 then
		blendGap = true
	end
	if wt > 0 and wt <= 0.05 then
		lowWeight = true
		t(1, "very-low-weight")
	elseif wt > 0 and wt <= 0.12 then
		lowWeight = true
		t(1, "low-weight")
	end
	if priority == Enum.AnimationPriority.Core then
		badPriority = true
		t(2, "core-priority")
	elseif priority == Enum.AnimationPriority.Idle then
		badPriority = true
		t(1, "idle-priority")
	end
	if speed >= 10 then
		badSpeed = true
		extremeSpeed = true
		t(4, "extreme-speed")
	elseif speed >= 3 then
		badSpeed = true
		t(2, "fast-speed")
	elseif speed > 0 and speed < 0.5 then
		badSpeed = true
		t(1, "slow-speed")
	end
	if length > 0 and length < 0.12 then
		shortLen = true
		t(2, "very-short-length")
	end
	if not track.IsPlaying then
		shortLen = true
		t(2, "not-playing")
	end

	local j3 = false
	if isPlayer and self:_apIsMobAnimation(track) then
		j3 = true
		t(5, "player-mob-animation")
	end

	local gap = nil
	if self._apLastKnownTime then
		gap = now - self._apLastKnownTime
		if gap < 0.05 then
			bt = true
			t(3, "tiny-gap")
		elseif gap < 0.12 then
			bt = true
			t(2, "short-gap")
		elseif gap < 0.3 then
			bt = true
			t(gap <= 0.28 and 2 or 1, "breaker-gap")
		end
	end

	local sameId = self._apHistory[aid]
	local sameCount = (sameId and #sameId or 0) + 1
	if sameCount >= 5 then
		t(3, "same-id-spam")
	elseif sameCount >= 3 then
		t(2, "same-id-repeat")
	end

	local globalCount = #self._apGlobalHistory + 1
	local seenIds = { [aid] = true }
	local seenCats = { [category] = true }
	local uniqueIds, uniqueCats = 1, 1
	for _, e in next, self._apRecentIds do
		if e.aid and not seenIds[e.aid] then
			seenIds[e.aid] = true
			uniqueIds = uniqueIds + 1
		end
		if e.category and not seenCats[e.category] then
			seenCats[e.category] = true
			uniqueCats = uniqueCats + 1
		end
	end
	local q = false
	if globalCount >= 8 then
		q = true
		t(3, "global-spam")
	elseif globalCount >= 5 then
		q = true
		t(2, "global-burst")
	elseif globalCount >= 3 then
		q = true
		t(2, "burst-pressure")
	end
	if globalCount >= 3 and uniqueIds >= 3 then
		q = true
		t(2, "rotating-id-burst")
	elseif globalCount >= 3 and uniqueIds >= 2 then
		q = true
		t(1, "rotating-id-pair")
	end
	if globalCount >= 3 and uniqueCats >= 2 then
		q = true
		t(2, "rotating-category-burst")
	end
	if bt and globalCount >= 2 and uniqueIds >= 2 then
		q = true
		t(2, "breaker-window-rotation")
	end
	if RA(category) and globalCount >= 3 and (uniqueIds >= 2 or uniqueCats >= 2) then
		q = true
		t(1, "weapon-rotation-pressure")
	end
	local activeSusp = apbActiveSuspicious(self._apActive)
	if activeSusp >= 4 then
		q = true
		t(2, "active-suspicious-limit")
	end
	local shortCount = #self._apShortDurations
	if shortCount >= 3 then
		q = true
		shortLen = true
		t(3, "stopped-track-pattern")
	elseif shortCount >= 1 and globalCount >= 2 then
		q = true
		shortLen = true
		t(2, "recent-short-stop")
	end

	local shapeSuspicious = lowWeight or badPriority or badSpeed or shortLen or blendGap
	local missing = ownership.status == "missing"
	local confirmedOwn = ownership.status == "confirmed"
	local Or = false
	local needsLegitimacy = isPlayer and (timing.tag == "Mantra" or timing.tag == "Critical" or timing.tag == "M1"
		or timing.tag == "Undefined" or timing.umoa or ownership.reason == "unmapped" or NA or RA(category))
	if needsLegitimacy and not confirmedOwn and not EA and (missing or NA or shapeSuspicious or q or bt) then
		Or = true
		if not missing then
			t(q and 3 or 2, "missing-attack-legitimacy")
		end
	end

	local hitPlausible = false
	if not zo then
		local geomOk, geomReason = self:apbh(timing)
		if geomOk then
			Tt = true
			hitPlausible = true
		elseif geomReason == "delay-until-hitbox" then
			Tt = true
		elseif isPlayer then
			t(3, "no-real-hit-evidence")
			t(2, "known-timing-without-hit")
		end
	end

	local realistic = not shapeSuspicious and track.IsPlaying and (weight <= 0 or weight >= 0.12)
		and (speed <= 0 or (speed >= 0.5 and speed <= 2.25))
	local notPlaying = not track.IsPlaying
	local ownedRealOrConfirmed = confirmedOwn or EA
	local moduleCtx = timing.rpue or (timing.duih and (ownedRealOrConfirmed or not isPlayer)) or (timing.umoa and ownedRealOrConfirmed)
	local n3 = moduleCtx and not q
	local realLegit = not Gr and (not isPlayer or zo or ownedRealOrConfirmed or n3)
	local breakerSignal = (bt and not ownedRealOrConfirmed) or (q and not ownedRealOrConfirmed)
	local cleanShape = realistic and not j3 and not extremeSpeed and not NA
	local ownedRealistic = isPlayer and cleanShape and Tt and not notPlaying and realLegit
	local strongReal = not Gr and (zo or not isPlayer or (cleanShape and Tt and (ownedRealOrConfirmed or (n3 and not bt))))
	local anyReal = not Gr and (strongReal or ownedRealistic or (isPlayer and cleanShape and Tt and not notPlaying and realLegit))
	local highSignals = iCount >= 3
	local burstShape = isPlayer and ((bt and (uniqueIds >= 2 or uniqueCats >= 2 or globalCount >= 3)) or (q and (uniqueIds >= 2 or uniqueCats >= 2)) or (shortCount >= 1 and globalCount >= 2) or activeSusp >= 4)
	local stackedBreaker = burstShape and not strongReal and not ownedRealOrConfirmed and not n3 and (Or or highSignals or tr >= 7)
	local likelyReal = ownedRealistic and not stackedBreaker
	local legitContext = anyReal and not stackedBreaker
	local fakeHard = isPlayer and not legitContext and (Gr or j3 or extremeSpeed or (notPlaying and Or and (shapeSuspicious or breakerSignal)) or (NA and (shapeSuspicious or breakerSignal or not Tt)))
	local fakeStacked = isPlayer and not legitContext and (stackedBreaker or (Or and highSignals and tr >= 7 and (shapeSuspicious or breakerSignal or not Tt or shortCount >= 3)))

	local tier = "Uncertain"
	if not isPlayer then
		tier = "Real"
	elseif strongReal then
		tier = "Real"
	elseif likelyReal then
		tier = "Likely Real"
	elseif fakeHard then
		tier = "Fake"
	elseif fakeHard or fakeStacked then
		tier = "Likely Fake"
	elseif legitContext then
		tier = "Likely Real"
	elseif tr > 0 or Or or q then
		tier = "Suspicious"
	end

	local block = fakeHard or fakeStacked
	local decision = block and "BlockFakeOccurrence" or ((strongReal or likelyReal or legitContext) and "AllowReal" or "AllowUncertain")

	return {
		block = block,
		wouldBlock = block,
		decision = decision,
		tier = tier,
		score = tr,
		trust = m4,
		category = category,
		geometricPlausible = hitPlausible or Tt,
		signals = signals,
		raw = {
			category = category,
			hasBurstPattern = q,
			breakerTimer = bt,
			recentGlobal = globalCount,
			plausible = hitPlausible,
			geometricPlausibility = Tt and "plausible" or "none",
			mantraEtherStatus = mantraEther and mantraEther.status or "not-applicable",
			mantraEtherSerial = mantraEther and mantraEther.serial or nil,
			mantraEtherContext = mantraEther and mantraEther.context or "not-applicable",
			mantraEtherContextKey = mantraEther and mantraEther.contextKey or nil,
		},
	}
end

-- ---- Evidence builder (eu) + authenticator wiring --------------------------
local function apbFps()
	local ok, fps = pcall(function()
		return workspace:GetRealPhysicsFPS()
	end)
	if not ok or typeof(fps) ~= "number" or fps <= 0 then
		return 60
	end
	return math.max(15, fps)
end
-- Earliest action time of a timing, in seconds (Bu).
local function apbTimingDuration(timing)
	if not timing or not timing.actions or timing.actions:count() <= 0 then
		return nil
	end
	local best = nil
	for _, action in next, timing.actions:get() do
		local when = tonumber(action._when)
		if when and when >= 0 then
			local secs = when / 1000
			if best == nil or secs < best then
				best = secs
			end
		end
	end
	return best
end

function AnimatorDefender:apCandidateView(track, timing, aid, report, startedAt)
	local category = report.category
	local weaponCtx = CR(self.entity, timing, category, aid)
	local speed = math.abs(tonumber(track.Speed) or 0)
	local length = tonumber(track.Length) or 0
	local expected = speed > 0.0001 and length / speed or length
	local timingDur = apbTimingDuration(timing)
	local legal = tonumber(weaponCtx.weaponAttackDuration) or (expected > 0 and expected) or 2
	local reach = timingDur and math.max(0, timingDur - Latency.rdelay() - Latency.sdelay()) or legal
	local grace = math.max(2 / apbFps(), 0.02)
	local plannedDefense = startedAt + reach
	local deadline = startedAt + math.max(0, reach - grace)
	local raw = report.raw or {}
	local plausible = raw.plausible == true or (raw.geometricPlausibility ~= nil and raw.geometricPlausibility ~= "none")
	return {
		track = track,
		occurrence = self._apTrackOccurrences[track],
		timing = timing,
		aid = aid,
		category = category,
		weaponContext = weaponCtx,
		startedAt = startedAt,
		plannedDefenseAt = plannedDefense,
		deadlineAt = deadline,
		legalInterval = legal,
		longestTiming = math.max(legal, expected, reach or 0),
		expectedDuration = expected,
		evidenceLatencyMax = category == "weapon-critical" and 0.15 or 0.12,
		progressionGrace = grace,
		geometricPlausible = plausible,
		comboContinuous = true,
		actionKey = raw.mantraEtherContextKey,
		timingActionNames = { timing.name, timing.smod },
		classifierBlocked = report.block == true,
		callbackDispatchDeadlineAt = startedAt + grace,
		recoveredIngress = self._apRecoveredOccurrences[track] == self._apTrackOccurrences[track],
	}
end

-- ---- Ingress candidate resolve (rb) + quarantine (wb/sR) + recovery (XR) ----
apbResolveCandidate = function(self, track)
	local animation = track.Animation
	if not animation then
		return nil
	end
	-- Cached + canonicalized id: survives a later id="" blank and normalizes obfuscated ids
	-- (e.g. rbxhttp://...?id=garbage?id=12345&junk), matching the normal path.
	local aid = self:_apResolveAid(track)
	local timing = SaveManager.as:index(aid)
	if not timing or track.Priority == Enum.AnimationPriority.Core or not track.IsPlaying then
		return nil, aid
	end
	local speed = math.abs(tonumber(track.Speed) or 0)
	local weightTarget = tonumber(track.WeightTarget) or 0
	local timePos = tonumber(track.TimePosition) or 0
	if speed < 0.05 or speed > 4 or weightTarget <= 0.12 or weightTarget > 2 or timePos > 0.4 then
		return nil, aid
	end
	local distance = self:distance(self.entity)
	if not distance then
		return nil, aid
	end
	local minDist = timing.imdd
	if minDist <= 0.01 then
		minDist = 0
	end
	if distance < minDist or distance > apbMaxDist(timing) then
		return nil, aid
	end
	return timing, aid
end

apbQuarantine = function(self, track, timing, aid, startedAt)
	local category = BA(timing)
	local lane = apbLane(category)
	local byTrack = self._apQuarantineByTrack[lane]
	local slot = byTrack[track]
	local existed = slot ~= nil
	if not slot then
		local cursor = self._apQuarantineCursors[lane]
		slot = self._apQuarantine[apbLaneIndex(lane, cursor)]
		apbClearQuarantineSlot(self, slot)
		self._apQuarantineCursors[lane] = cursor % APB_QUARANTINE_LANE_SIZE + 1
		self._apQuarantineCount = math.min(APB_QUARANTINE_SIZE, (self._apQuarantineCount or 0) + 1)
		byTrack[track] = slot
	end
	slot.track = track
	slot.timing = timing
	slot.aid = aid
	slot.category = category
	slot.lane = lane
	slot.startedAt = startedAt
	IngressGuard.quarantineCandidate(existed)
	return not existed
end

apbPruneQuarantine = function(self, now, steps)
	local cursor = self._apQuarantinePruneCursor or 1
	for _ = 1, steps do
		local slot = self._apQuarantine[cursor]
		if slot.track and now - slot.startedAt > APB_QUARANTINE_TTL then
			apbClearQuarantineSlot(self, slot)
		end
		cursor = cursor % APB_QUARANTINE_SIZE + 1
	end
	self._apQuarantinePruneCursor = cursor
end

-- Score one quarantine slot against a piece of evidence; nil = incompatible (ages stale slots).
local function apbScoreQuarantineSlot(self, slot, evidence, at, latMin, latMax)
	local track = slot.track
	local startedAt = slot.startedAt
	if not track or not startedAt then
		return nil
	end
	local latency = at - startedAt
	if latency > APB_QUARANTINE_TTL then
		apbClearQuarantineSlot(self, slot)
		return nil
	end
	local actionOk = not evidence.actionKey or not evidence.requireTimingActionMatch
		or AttackAuthenticator.timingActionCompatible(
			{ timingActionNames = { slot.timing.name, slot.timing.smod } },
			evidence.actionKey,
			evidence.actionAliases
		)
	local occ = self._apOccurrenceStartedAt[track]
	if latency < latMin or latency > latMax or not apbFamilyCovers(slot.category, evidence) or not actionOk
		or (occ and occ.startedAt >= startedAt - 0.0001) then
		return nil
	end
	local speed = math.abs(tonumber(track.Speed) or 0)
	local dur = speed > 0.0001 and (tonumber(track.Length) or 0) / speed or 0
	local evDur = tonumber(evidence.duration)
	local score = 0
	if evDur and dur > 0 then
		local err = math.abs(dur - evDur) / math.max(dur, evDur, 0.01)
		if err > 0.75 then
			return nil
		end
		score = (1 - err) * 8
	end
	local weaponCtx = CR(self.entity, slot.timing, slot.category, slot.aid)
	if evidence.weaponKey and weaponCtx.status == "mismatch" then
		return nil
	elseif weaponCtx.status == "confirmed" then
		score = score + 40
	end
	if evidence.exactFamily == slot.category then
		score = score + 100
	elseif evidence.families and evidence.families[slot.category] then
		score = score + 50
	end
	if (tonumber(track.WeightCurrent) or 0) > 0.12 then
		score = score + 20
	end
	if (tonumber(track.TimePosition) or 0) > 0.0001 then
		score = score + 10
	end
	if latency >= (tonumber(evidence.centralLatencyMin) or latMin) and latency <= (tonumber(evidence.centralLatencyMax) or latMax) then
		score = score + 10
	end
	return score + startedAt * 0.000001
end

apbRecoverIngress = function(self, evidence)
	local at = tonumber(evidence.at) or os.clock()
	local latMin = tonumber(evidence.latencyMin) or -0.02
	local latMax = math.min(tonumber(evidence.latencyMax) or APB_QUARANTINE_TTL, APB_QUARANTINE_TTL)
	local best, bestScore = nil, -math.huge
	for i = 1, APB_QUARANTINE_SIZE do
		local slot = self._apQuarantine[i]
		local score = apbScoreQuarantineSlot(self, slot, evidence, at, latMin, latMax)
		if score and score > bestScore then
			best = slot
			bestScore = score
		end
	end
	if not best then
		return nil
	end
	local track = best.track
	local timing = best.timing
	local startedAt = best.startedAt
	apbClearQuarantineSlot(self, best)
	for i = 1, APB_QUARANTINE_SIZE do
		local slot = self._apQuarantine[i]
		if slot.track == track and slot.startedAt and slot.startedAt <= startedAt then
			apbClearQuarantineSlot(self, slot)
		end
	end
	local occ = (self._apTrackOccurrences[track] or 0) + 1
	self._apTrackOccurrences[track] = occ
	self._apOccurrenceStartedAt[track] = { occurrence = occ, startedAt = startedAt }
	self._apRecoveredOccurrences[track] = occ
	IngressGuard.recoveredEvidence()
	self:apProcess(track, occ, timing, true)
	return self.apAuthenticator:getCandidate(track, occ)
end

-- recordAttackEvidence: family-bridge settle-defer + ingress recovery + authenticate.
function AnimatorDefender:apRecordEvidence(evidence)
	if not self.apAuthenticator or not self.apAdvancedEnabled or not self._isFromPlayer then
		return nil, "inactive"
	end
	if evidence.allowWeaponFamilyBridge and not evidence.candidateSetSettled then
		local copy = {}
		for k, v in next, evidence do
			copy[k] = v
		end
		copy.candidateSetSettled = true
		local generation = self.apEvidenceGeneration
		local settle = math.max(2 / apbFps(), 0.02)
		task.delay(settle, function()
			if self.apAdvancedEnabled and self.apTrackingActive and self.apEvidenceGeneration == generation then
				self:apRecordEvidence(copy)
			end
		end)
		return nil, "candidate-settle-pending"
	end
	if evidence.authoritative and not evidence.causalCandidateId and apbRecoverIngress then
		if evidence._ingressRecoveryAdmitted == nil then
			local owner = self._playerOwner or players:GetPlayerFromCharacter(self.entity)
			evidence._ingressRecoveryAdmitted = owner == nil or IngressGuard.admitRecovery(owner)
		end
		if evidence._ingressRecoveryAdmitted then
			local recovered = apbRecoverIngress(self, evidence)
			if recovered then
				evidence.causalCandidateId = recovered.id
				evidence.causalTrack = recovered.track
				evidence.causalOccurrence = recovered.occurrence
			end
		end
	end
	local candidate, reason = self.apAuthenticator:recordEvidence(evidence)
	if not candidate then
		return nil, reason
	end
	task.defer(function()
		if self.apAuthenticator
			and self.apAuthenticator:isAuthenticated(candidate.track, candidate.occurrence)
			and candidate.track.IsPlaying then
			self:apProcess(candidate.track, candidate.occurrence, nil, false)
		end
	end)
	return candidate, reason
end

-- xA: on an ether spend, if StoredAction+StoredToolName present, record mantra intent.
function AnimatorDefender:apMantraSpend(serial)
	if not self.apAdvancedEnabled or self._apMantraIntentSerial == serial then
		return
	end
	if (self._apEtherSpendSerial or 0) ~= serial then
		return
	end
	local action = apbNorm(self.entity:GetAttribute("StoredAction"))
	local tool = self.entity:GetAttribute("StoredToolName")
	if action == "" or action == "criticalattack" or typeof(tool) ~= "string" or tool == "" then
		return
	end
	local spendAt = self._apEtherLastSpendAt
	if not spendAt or os.clock() - spendAt > 1 then
		return
	end
	self._apMantraIntentSerial = serial
	local grace = math.max(2 / apbFps(), 0.04)
	self:apRecordEvidence({
		kind = "replicated-mantra-intent", at = spendAt, authoritative = true, retainForFutureCandidate = true,
		exactFamily = "mantra", families = { mantra = true }, actionKey = action, actionAliases = apbActionAliases(self.entity),
		requireTimingActionMatch = true,
		latencyMin = -1, latencyMax = 0.25, centralLatencyMin = -grace, centralLatencyMax = grace, uncertainty = grace, serial = serial,
	})
end

function AnimatorDefender:apQueueMantraEvidence(candidate, report)
	local raw = report and report.raw
	if not raw or candidate.category ~= "mantra" then
		return
	end
	local serial = tonumber(raw.mantraEtherSerial)
	local key = raw.mantraEtherContextKey
	if not serial or not key or raw.mantraEtherContext ~= "match" or raw.mantraEtherStatus ~= "spent" then
		return
	end
	if self._apQueuedMantraEvidence[serial] then
		return
	end
	local entry = { serial = serial, key = key, track = candidate.track, occurrence = candidate.occurrence, candidateId = candidate.id }
	self._apQueuedMantraEvidence[serial] = entry
	task.delay(0.04, function()
		if self._apQueuedMantraEvidence[serial] ~= entry or not self.apAdvancedEnabled then
			return
		end
		self._apQueuedMantraEvidence[serial] = nil
		local live = self.apAuthenticator:getCandidate(entry.track, entry.occurrence)
		if live ~= candidate or live.status ~= "pending" then
			return
		end
		self:apRecordEvidence({
			kind = "replicated-mantra-ether-action", at = os.clock(), authoritative = true, exactFamily = "mantra",
			families = { mantra = true }, actionKey = key,
			latencyMin = -0.05, latencyMax = 0.25, centralLatencyMin = -0.02, centralLatencyMax = 0.12, uncertainty = 0.04,
			causalCandidateId = entry.candidateId, causalTrack = entry.track, causalOccurrence = entry.occurrence,
		})
	end)
end

-- ---- Weapon-sound producer (Tu/au/SL/V1/Ny/LA) ------------------------------
local function apbAncestorModelOrPart(inst)
	return inst:FindFirstAncestorWhichIsA("Model") or inst:FindFirstAncestorWhichIsA("BasePart")
end
-- nR: is `container` weld/joint-connected to the attacker (owns the thrown weapon)?
local function apbOwnsContainer(entity, container)
	if not entity or not container or not container.Parent then
		return false
	end
	local function related(a, b)
		return a == b or a:IsDescendantOf(b)
	end
	local function jointLinks(joint)
		if not (joint:IsA("JointInstance") or joint:IsA("WeldConstraint")) then
			return false
		end
		local p0, p1 = joint.Part0, joint.Part1
		return p0 ~= nil and p1 ~= nil
			and ((related(p0, entity) and related(p1, container)) or (related(p1, entity) and related(p0, container)))
	end
	if jointLinks(container) then
		return true
	end
	local scanned = 0
	for _, d in next, container:GetDescendants() do
		scanned = scanned + 1
		if jointLinks(d) then
			return true
		end
		if scanned >= 256 then
			break
		end
	end
	return false
end
-- Tu: find the single matching pending weapon candidate for a sound.
function AnimatorDefender:_apFindWeaponSoundCandidate(now, weaponKey)
	local found = nil
	for _, c in next, self.apAuthenticator.candidates or {} do
		local age = now - c.startedAt
		if c.status == "pending" and RA(c.category) and c.weaponKey == weaponKey
			and c.weaponContext and c.weaponContext.status == "confirmed" and c.weaponContext.exactWeaponTypeMatch == true
			and age >= 0 and age <= 0.04 and now <= c.deadlineAt and c.track and c.track.IsPlaying then
			if found then
				return nil
			end
			found = c
		end
	end
	return found
end
function AnimatorDefender:_apWeaponSoundPlayed(sound, root)
	if not self.apAdvancedEnabled or not self.apTrackingActive or not self._isFromPlayer then
		return
	end
	if not sound:IsDescendantOf(root) or not apbOwnsContainer(self.entity, root) then
		return
	end
	local weaponKey = C1(self.entity)
	if not weaponKey then
		return
	end
	local now = os.clock()
	local candidate = self:_apFindWeaponSoundCandidate(now, weaponKey)
	if not candidate then
		return
	end
	local m1 = 0.04
	self:apRecordEvidence({
		kind = "replicated-weapon-sound", at = now, authoritative = true, retainForFutureCandidate = false,
		weaponKey = weaponKey, exactFamily = candidate.category, families = { [candidate.category] = true },
		requireExactWeaponType = true, latencyMin = 0, latencyMax = m1, centralLatencyMin = 0, centralLatencyMax = m1,
		uncertainty = m1, causalCandidateId = candidate.id, causalTrack = candidate.track, causalOccurrence = candidate.occurrence,
	})
end
function AnimatorDefender:_apBindWeaponSound(sound, root)
	if self._apWeaponSounds[sound] then
		return
	end
	self._apWeaponSounds[sound] = root
	self.apMaid:add(sound.Played:Connect(function()
		self:_apWeaponSoundPlayed(sound, root)
	end))
end
function AnimatorDefender:_apBindWeaponRoots(inst)
	if self._apWeaponSoundRoots[inst] then
		return true
	end
	if not inst or not inst.Parent or not apbOwnsContainer(self.entity, inst) then
		return false
	end
	local root = inst:IsA("Model") and inst or apbAncestorModelOrPart(inst)
	if not root or not apbOwnsContainer(self.entity, root) then
		return false
	end
	if self._apWeaponSoundRoots[root] then
		return true
	end
	self._apWeaponSoundRoots[root] = true
	self.apMaid:add(root.DescendantAdded:Connect(function(child)
		if child:IsA("Sound") then
			self:_apBindWeaponSound(child, root)
		end
	end))
	for _, d in next, root:GetDescendants() do
		if d:IsA("Sound") then
			self:_apBindWeaponSound(d, root)
		end
	end
	return true
end
function AnimatorDefender:_apRefreshWeaponSounds()
	if not self.apAdvancedEnabled or not self.entity or not self.entity.Parent then
		return
	end
	self._apNextWeaponSoundRefreshAt = os.clock() + 2
	local thrown = workspace:FindFirstChild("Thrown")
	local attach = thrown and thrown:FindFirstChild("Attach_" .. self.entity.Name)
	if attach then
		self:_apBindWeaponRoots(attach)
	end
end
function AnimatorDefender:_apClearWeaponSounds()
	self._apWeaponSounds = setmetatable({}, { __mode = "k" })
	self._apWeaponSoundRoots = setmetatable({}, { __mode = "k" })
	self._apNextWeaponSoundRefreshAt = 0
end

-- ---- Client-effect / attribute / root evidence producers -------------------
function AnimatorDefender:apce(name, data)
	if not self.apAdvancedEnabled or not self._isFromPlayer or typeof(name) ~= "string" or typeof(data) ~= "table" then
		return
	end
	local prop = data.prop
	if not apbOwnsProp(self, prop, data) then
		return
	end
	if typeof(prop) == "Instance" then
		self:_apBindWeaponRoots(prop)
	end
	local weaponKey = C1(self.entity)
	local now = os.clock()
	if name == "WeaponTrailOff" then
		local cancelled = self.apAuthenticator:cancelFromEvidence({ kind = "replicated-weapon-trail-off", at = now, weaponKey = weaponKey })
		for _, candidate in next, cancelled do
			self:cancelAuthenticatedOccurrence(candidate)
		end
		return
	end
	if name == "WeaponTrail" then
		self:apRecordEvidence({
			kind = "replicated-weapon-trail", at = now, authoritative = true, weaponKey = weaponKey,
			duration = tonumber(data and data.dur),
			families = { ["weapon-m1"] = true, ["weapon-critical"] = true, ["weapon-special"] = true },
			latencyMin = -0.02, latencyMax = 0.12, centralLatencyMin = -0.02, centralLatencyMax = 0.05, uncertainty = 0.02,
		})
	elseif name == "CriticalSwirl" then
		self:apRecordEvidence({
			kind = "replicated-critical-effect", at = now, authoritative = true, weaponKey = weaponKey,
			duration = tonumber(data and data.dur), exactFamily = "weapon-critical",
			families = { ["weapon-critical"] = true }, allowWeaponFamilyBridge = true,
			latencyMin = -0.08, latencyMax = 0.15, centralLatencyMin = -0.02, centralLatencyMax = 0.05, uncertainty = 0.02,
		})
	end
end
function AnimatorDefender:apae(attr)
	if not self.apAdvancedEnabled or not self._isFromPlayer then
		return
	end
	if attr == "StoredToolName" or attr == "PrimaryWeapon" or attr == "Weapon" then
		self:_apRefreshWeaponSounds()
		return
	end
	if attr ~= "StoredAction" then
		return
	end
	if (self._apEtherSpendSerial or 0) > 0 then
		self:apMantraSpend(self._apEtherSpendSerial)
	end
	if apbNorm(self.entity:GetAttribute("StoredAction")) ~= "criticalattack" then
		return
	end
	self:apRecordEvidence({
		kind = "replicated-stored-critical", at = os.clock(), authoritative = true, weaponKey = C1(self.entity),
		exactFamily = "weapon-critical", families = { ["weapon-critical"] = true }, allowWeaponFamilyBridge = true,
		latencyMin = -0.08, latencyMax = 0.15, centralLatencyMin = -0.02, centralLatencyMax = 0.05, uncertainty = 0.02,
	})
end
function AnimatorDefender:apcr(child)
	if not self.apAdvancedEnabled or not self._isFromPlayer then
		return
	end
	local hrp = self.entity and self.entity:FindFirstChild("HumanoidRootPart")
	if not hrp or child.Parent ~= hrp then
		return
	end
	local now = os.clock()
	local paired = false
	if child:IsA("AudioPlayer") or child:IsA("Sound") then
		self._apRootAudioAt = now
		paired = self._apRootAttachmentAt ~= nil and now - self._apRootAttachmentAt <= 0.05
	elseif child:IsA("Attachment") then
		self._apRootAttachmentAt = now
		paired = self._apRootAudioAt ~= nil and now - self._apRootAudioAt <= 0.05
	else
		return
	end
	if not paired then
		return
	end
	local action = apbNorm(self.entity:GetAttribute("StoredAction"))
	local weaponKey = C1(self.entity)
	if action ~= "criticalattack" then
		if action == "" then
			return
		end
		if self._apMantraRootClaimedAction == action and self._apMantraRootClaimedUntil and now <= self._apMantraRootClaimedUntil then
			return
		end
		local recorded = self:apRecordEvidence({
			kind = "replicated-mantra-root-action-pair", at = now, authoritative = true, retainForFutureCandidate = true,
			exactFamily = "mantra", families = { mantra = true }, actionKey = action, actionAliases = apbActionAliases(self.entity),
			requireTimingActionMatch = true,
			latencyMin = -1, latencyMax = 0.25, centralLatencyMin = -0.02, centralLatencyMax = 0.12, uncertainty = 0.05,
		})
		if recorded then
			self._apMantraRootClaimedAction = action
			self._apMantraRootClaimedUntil = recorded.startedAt + math.max(recorded.behaviorWindow or 0, recorded.legalInterval or 0)
		end
		return
	end
	local recorded = self:apRecordEvidence({
		kind = "replicated-critical-root-pair", at = now, authoritative = true, weaponKey = weaponKey,
		exactFamily = "weapon-critical", families = { ["weapon-critical"] = true }, allowWeaponFamilyBridge = true,
		latencyMin = -0.08, latencyMax = 0.15, centralLatencyMin = 0, centralLatencyMax = 0.08, uncertainty = 0.05,
	})
	if recorded then
		return
	end
	if self.apAuthenticator and self.apAuthenticator.floodActive then
		return
	end
	-- Vestigial critical-pending confirmation (the pending set is not populated in this build).
	for track, info in next, self._apCriticalPending do
		if self._apTrackOccurrences[track] == info.occurrence and now - info.startedAt <= 0.08 then
			self._apCriticalPending[track] = nil
			self._apCriticalConfirmed[track] = info
			self._apCriticalConfirmedSources[track] = "replicated-root-event"
			task.defer(function()
				if self._apTrackOccurrences[track] == info.occurrence and track.IsPlaying then
					self:apProcess(track, info.occurrence)
				end
			end)
			return
		end
	end
end

-- ---- Ingress generation sync -----------------------------------------------
function AnimatorDefender:syncIngressGeneration()
	local gen = IngressGuard.generation()
	if self._apIngressGeneration == gen then
		return
	end
	self._apIngressGeneration = gen
	if self.apAuthenticator then
		for _, candidate in next, self.apAuthenticator.candidates or {} do
			if candidate.status == "pending" or candidate.status == "authenticated" then
				self:cancelAuthenticatedOccurrence(candidate)
			end
		end
		self.apAuthenticator:reset("ingress-reset")
	end
	self._apTrackOccurrences = setmetatable({}, { __mode = "k" })
	self._apOccurrenceStartedAt = setmetatable({}, { __mode = "k" })
	self._apScheduledOccurrences = setmetatable({}, { __mode = "k" })
	self._apMantraRootClaimedAction = nil
	self._apMantraRootClaimedUntil = nil
	apbInitQuarantine(self)
	self.apEvidenceGeneration = (self.apEvidenceGeneration or 0) + 1
end

-- ---- Animation-id cache -----------------------------------------------------
-- Capture a track's (canonical) animation id the first moment it's non-empty -- i.e. at the play
-- instant -- and reuse it forever. The blank-id breaker wipes track.Animation.AnimationId to ""
-- AFTER the attack starts, so a later (deferred) re-read would see nothing; the cache preserves the
-- real id we already saw. Falls back to the live/canonical id until a real one is captured. Weak
-- keys let dead tracks clean themselves.
---@param track AnimationTrack
---@return string
function AnimatorDefender:_apResolveAid(track)
	local animation = track and track.Animation
	local live = canonicalAid(tostring(animation and animation.AnimationId or ""))
	if live ~= "" then
		self._apAidCache[track] = live
		return live
	end
	return self._apAidCache[track] or live
end

-- Is this track a mob animation being played by a player (a fake/exploit)? Uses the cached id so a
-- blanked or obfuscated id still matches the mob-animation set, then falls back to the live id --
-- so it strictly ADDS coverage over the old raw-id lookup, never removes any.
---@param track AnimationTrack
---@return boolean
function AnimatorDefender:_apIsMobAnimation(track)
	local cached = self:_apResolveAid(track)
	if cached ~= "" and self.manimations[cached] ~= nil then
		return true
	end
	local animation = track.Animation
	local live = animation and animation.AnimationId
	return live ~= nil and self.manimations[live] ~= nil
end

-- ---- Occurrence bookkeeping -------------------------------------------------
function AnimatorDefender:isOccurrenceCurrent(track, occurrence)
	return occurrence == nil or self._apTrackOccurrences[track] == occurrence
end

-- Claim the defense schedule for one occurrence exactly once.
function AnimatorDefender:claimOccurrenceSchedule(track, occurrence)
	if not occurrence then
		return true
	end
	if self._apScheduledOccurrences[track] == occurrence then
		return false
	end
	self._apScheduledOccurrences[track] = occurrence
	return true
end

-- Cancel any scheduled tasks belonging to a (now counter-evidenced) occurrence.
function AnimatorDefender:cancelAuthenticatedOccurrence(candidate)
	if not candidate then
		return
	end
	for idx, atask in next, self.tasks do
		if atask.sourceTrack == candidate.track and atask.sourceOccurrence == candidate.occurrence then
			local thread = atask.thread
			if not thread or coroutine.status(thread) ~= "dead" then
				atask:cancel()
			end
			self.tasks[idx] = nil
			if atask.identifier == "End Block" or atask.identifier == "Start Block" then
				QueuedBlocking.stop("Defender_StartBlock")
			end
		end
	end
end

-- Occurrence-aware clean: on an "AllowUncertain" allow, preserve tasks belonging to a
-- different, still-current, still-playing track; otherwise fall through to a full clean.
function AnimatorDefender:cleanForOccurrence(track, report)
	local preserve = self.apAdvancedEnabled and report and report.decision == "AllowUncertain"
	if not preserve then
		return self:clean()
	end
	self.keyframes = {}
	for idx, atask in next, self.tasks do
		local keep = false
		if atask.forced and atask.identifier ~= "Start Block" and atask.identifier ~= "End Block" then
			keep = true
		elseif atask.identifier == "Start Crouch" or atask.identifier == "End Crouch"
			or atask.identifier == "Start Slide" or atask.identifier == "End Slide" then
			keep = true
		else
			local src = atask.sourceTrack
			local thread = atask.thread
			if src and src ~= track and self:isOccurrenceCurrent(src, atask.sourceOccurrence)
				and src.IsPlaying and (not thread or coroutine.status(thread) ~= "dead") then
				keep = true
			end
		end
		if not keep then
			atask:cancel()
			self.tasks[idx] = nil
			if atask.identifier == "End Block" or atask.identifier == "Start Block" then
				QueuedBlocking.stop("Defender_StartBlock")
			end
		end
	end
end

-- Lightweight pre-validation gate (pvalidate). When armed, defers to the classifier's block
-- decision; otherwise applies coarse priority/weight/speed sanity checks.
function AnimatorDefender:pvalidate(track, timing, report)
	if self.apAdvancedEnabled and timing then
		return not (report and report.block)
	end
	if track.Priority == Enum.AnimationPriority.Core then
		return false
	end
	if self._isFromPlayer and track.WeightTarget <= 0.05 then
		return false
	end
	if self._isFromPlayer and self:_apIsMobAnimation(track) then
		return false
	end
	if track.Speed >= 1000 then
		return false
	end
	return true
end

-- ---- Crasher-protection ingress gate (rlimit) ------------------------------
-- Returns (admitted, timing?, recovered?). Rate-limits suspicious animation ingress per player
-- and globally; quarantines rate-limited attack candidates so authoritative evidence can recover
-- them, and sheds pure overflow. Only meaningful for player attackers while armed.
function AnimatorDefender:rlimit(track)
	self:syncIngressGeneration()
	if not self._isFromPlayer then
		return true, nil, false
	end
	local owner = self._playerOwner or players:GetPlayerFromCharacter(self.entity)
	if not owner then
		return true, nil, false
	end
	self._playerOwner = owner
	local admitted, reason, underPressure = IngressGuard.admitSuspicious(owner)
	if admitted and not underPressure then
		return true, nil, false
	end
	local priority = track.Priority
	local isAction = priority == Enum.AnimationPriority.Action or priority == Enum.AnimationPriority.Action2
		or priority == Enum.AnimationPriority.Action3 or priority == Enum.AnimationPriority.Action4
	if underPressure and not isAction then
		IngressGuard.reject("universal-noncombat-priority")
		return false, nil, false
	end
	local timing, candidateId = apbResolveCandidate(self, track)
	if timing and candidateId then
		local candidateOk, candidateReason = IngressGuard.admitCandidate(owner, candidateId)
		if candidateOk then
			return true, timing, true
		end
		reason = candidateReason or reason
		if self.apAdvancedEnabled then
			local now = os.clock()
			local quarantinedNew = apbQuarantine(self, track, timing, candidateId, now)
			if quarantinedNew and self.apAuthenticator then
				local retained = self.apAuthenticator:retainedEvidenceForTiming({
					category = BA(timing), startedAt = now, timingActionNames = { timing.name, timing.smod },
				}, now)
				if retained then
					self:apRecordEvidence(retained)
				end
			end
			reason = "advanced-candidate-quarantine"
		end
	else
		reason = "untrusted-overflow"
	end
	IngressGuard.reject(reason)
	return false, nil, false
end

-- ---- Arm / disarm -----------------------------------------------------------
function AnimatorDefender:apArm(enabled)
	self._isFromPlayer = players:GetPlayerFromCharacter(self.entity) ~= nil
	if self._isFromPlayer then
		self._playerOwner = players:GetPlayerFromCharacter(self.entity)
	end
	enabled = enabled == true and self._isFromPlayer
	if self.apAdvancedEnabled == enabled then
		return
	end
	self.apAdvancedEnabled = enabled
	self.apTrackingActive = enabled

	if not enabled then
		if self.apAuthenticator then
			for _, candidate in next, self.apAuthenticator.candidates or {} do
				if candidate.status == "pending" or candidate.status == "authenticated" then
					self:cancelAuthenticatedOccurrence(candidate)
				end
			end
		end
		pcall(function()
			self.apMaid:clean()
		end)
		self:_apClearWeaponSounds()
		if self.apAuthenticator then
			self.apAuthenticator:reset("disarmed")
		end
		apbInitQuarantine(self)
		self._apScheduledOccurrences = setmetatable({}, { __mode = "k" })
		self._apOccurrenceStartedAt = setmetatable({}, { __mode = "k" })
		self.apEvidenceGeneration = (self.apEvidenceGeneration or 0) + 1
		self._apEther = nil
		self._apMantraRootClaimedAction = nil
		self._apMantraRootClaimedUntil = nil
		-- NOTE: loadout tracking is shared across defenders, so a single defender disarming
		-- must NOT disable it globally; it self-cleans on player removal and idles when unused.
		return
	end

	if not self.apAuthenticator then
		self.apAuthenticator = AttackAuthenticator.new()
	end
	AttackOwnership.init()
	AttackOwnership.setLoadoutTrackingActive(true)
	self:syncIngressGeneration()

	-- Ether watcher.
	self:_apbWatchEther(self.entity:FindFirstChild("Ether"), os.clock())

	-- Weapon-sound roots.
	self:_apRefreshWeaponSounds()

	-- ClientEffect remote -> apce (prop-ownership is checked inside apce).
	local requests = replicatedStorage:FindFirstChild("Requests")
	local clientEffect = requests and requests:FindFirstChild("ClientEffect")
	if clientEffect and (clientEffect:IsA("RemoteEvent") or clientEffect:IsA("UnreliableRemoteEvent")) then
		self.apMaid:add(clientEffect.OnClientEvent:Connect(function(effectName, data)
			self:apce(effectName, data)
		end))
	end

	-- Attribute evidence -> apae.
	self.apMaid:add(self.entity.AttributeChanged:Connect(function(attr)
		self:apae(attr)
	end))

	-- Root-pair + ether-instance evidence -> apcr.
	self.apMaid:add(self.entity.DescendantAdded:Connect(function(child)
		if child.Name == "Ether" then
			self:_apbWatchEther(child, os.clock())
		end
		self:apcr(child)
	end))
end

-- ---- AP-break detection notification (ported from Mira) ---------------------
-- Record one piece of AP-break evidence (a blocked fake occurrence). Purely informational:
-- flags the attacker after enough evidence in a short window and notifies the user. Never gates
-- the block decision itself -- that's already made by the classifier/authenticator.
local APB_EVIDENCE_WINDOW = 1.0       -- seconds of evidence history kept
local APB_EVIDENCE_THRESHOLD = 3      -- evidence events within the window to flag the attacker
local APB_FLAG_RESET_WINDOW = 10.0    -- clean seconds before the flag lifts
local APB_NOTIFY_COOLDOWN = 10.0      -- min seconds between notifications for one attacker
function AnimatorDefender:_apbEvidenceEvent()
	local now = os.clock()
	local ev = self._apbEvidence
	local writeIdx = 0
	for i = 1, #ev do
		if now - ev[i] <= APB_EVIDENCE_WINDOW then
			writeIdx = writeIdx + 1
			ev[writeIdx] = ev[i]
		end
	end
	for i = #ev, writeIdx + 1, -1 do
		ev[i] = nil
	end
	ev[writeIdx + 1] = now

	if writeIdx + 1 >= APB_EVIDENCE_THRESHOLD then
		self._apbFlagged = true
		self._apbFlaggedAt = now
		if Configuration.expectToggleValue("AntiBreakerNotify") then
			if not self._apbNotifiedAt or (now - self._apbNotifiedAt) > APB_NOTIFY_COOLDOWN then
				self._apbNotifiedAt = now
				local name = self.entity and self.entity.Name or "Someone"
				pcall(function()
					Library:Notify(string.format("%s is attempting to AP break you.", name), APB_NOTIFY_COOLDOWN)
				end)
			end
		end
	elseif self._apbFlagged and now - (self._apbFlaggedAt or 0) > APB_FLAG_RESET_WINDOW then
		self._apbFlagged = false
	end
end

-- ---- Advanced authenticator flow (process) ---------------------------------
-- @param track AnimationTrack
-- @param occurrence number Monotonic occurrence token for this track (from ingress/recovery).
-- @param timing AnimationTiming? Pre-resolved timing (from rlimit's candidate admission).
-- @param recovered boolean? Whether this occurrence was recovered from quarantine.
function AnimatorDefender:apProcess(track, occurrence, timing, recovered)
	if not self.apAdvancedEnabled or not self.apAuthenticator then
		return
	end
	self:syncIngressGeneration()

	occurrence = occurrence or self._apTrackOccurrences[track]
	local occurrenceAuthed = self.apAuthenticator:isAuthenticated(track, occurrence)
	if not self:isOccurrenceCurrent(track, occurrence) and not occurrenceAuthed then
		return
	end

	local occInfo = self._apOccurrenceStartedAt[track]
	local startedAt = (occInfo and occInfo.occurrence == occurrence and occInfo.startedAt) or os.clock()

	if self._playerOwner == players.LocalPlayer or (players.LocalPlayer.Character and self.entity == players.LocalPlayer.Character) then
		return
	end

	if self._apNextWeaponSoundRefreshAt and os.clock() >= self._apNextWeaponSoundRefreshAt then
		self:_apRefreshWeaponSounds()
	end
	if (self._apQuarantineCount or 0) > 0 then
		apbPruneQuarantine(self, os.clock(), 8)
	end

	-- Cached + canonicalized id: survives a later id="" blank (captured at the play instant) and
	-- resolves obfuscated ids to their real timing (matches the normal path).
	local aid = self:_apResolveAid(track)
	local resolved = timing or SaveManager.as:index(aid)
	if resolved then
		local distance = self:distance(self.entity)
		local minDist = resolved.imdd
		if not minDist or minDist <= 0.01 then
			minDist = 0
		end
		local maxDist = apbMaxDist(resolved, tonumber(Configuration.expectOptionValue("AutoParryMaxDist")) or 28)
		if not distance or distance < minDist or distance > maxDist then
			resolved = nil
		end
	end
	if not resolved then
		return
	end

	IngressGuard.advancedEntered()
	local report = self:aapb(track, resolved, aid, os.clock())
	local view = self:apCandidateView(track, resolved, aid, report, startedAt)
	view.occurrence = occurrence
	local candidate, isNew, retried = self.apAuthenticator:observeCandidate(view)
	if retried and retried ~= candidate then
		task.defer(function()
			if self.apAuthenticator:isAuthenticated(retried.track, retried.occurrence) then
				self:apProcess(retried.track, retried.occurrence)
			end
		end)
	end
	if isNew then
		self:rapb(track, aid, report, startedAt)
		candidate.historyRecorded = true
		self:apQueueMantraEvidence(candidate, report)
	end

	local candidateAuthed = false
	if candidate.status == "authenticated" then
		candidateAuthed = self.apAuthenticator:isAuthenticated(track, occurrence)
	elseif candidate.status == "released" then
		-- provisional quiet-window release: allowed through
	elseif candidate.status == "pending" then
		if not candidate._releaseScheduled then
			candidate._releaseScheduled = true
			task.delay(math.max(0, candidate.provisionalUntil - os.clock()), function()
				if self._apTrackOccurrences[track] ~= occurrence
					or self.apAuthenticator:getCandidate(track, occurrence) ~= candidate then
					return
				end
				if self.apAuthenticator:releaseCandidate(candidate, os.clock()) and track.IsPlaying then
					self:apProcess(track, occurrence)
				end
			end)
		end
		return
	else
		return
	end

	if not Configuration.expectToggleValue("EnableAutoDefense") then
		return
	end
	IngressGuard.autoDefenderReached()

	-- Evidence override: an authenticated occurrence is trusted regardless of heuristics.
	if candidateAuthed then
		report.block = false
		report.wouldBlock = false
		report.decision = "AllowAuthenticatedEvidence"
		report.tier = "Real"
	end

	if not candidate.historyRecorded then
		self:rapb(track, aid, report, os.clock())
		candidate.historyRecorded = true
	end

	if report.block then
		self:_apbEvidenceEvent()
		return
	end
	if not self:pvalidate(track, resolved, report) then
		return
	end

	-- Schedule the defense exactly once for this occurrence.
	if candidateAuthed then
		if not self.apAuthenticator:claimSchedule(track, occurrence, os.clock()) then
			return
		end
	end
	if not self:claimOccurrenceSchedule(track, occurrence) then
		return
	end

	self._evaluationData[track] = { t = tick() }
	self._apStampTrack = track
	self._apStampOccurrence = occurrence
	-- Anti-breaker held this candidate through the ingress gate / quarantine / provisional
	-- window, so the cue is arriving late by (now - startedAt). Feed that elapsed delay into
	-- the auto-defender so both the cue lead (self.offset) and the hitbox extrapolation
	-- (fsecs) compensate. Clamp it so a long quarantine recovery can't over-lead the parry.
	local gateDelay = os.clock() - startedAt
	local gateCap = (resolved.pfht and resolved.pfht * 2) or 0.3
	gateDelay = math.clamp(gateDelay, 0, gateCap)
	self:processValidated(track, aid, gateDelay)
	self._apStampTrack = nil
	self._apStampOccurrence = nil
end

---@param self AnimatorDefender
---@param track AnimationTrack
-- Global cap on concurrent parry evaluations, ported from Project Rain's `tasks >= task_concurrency` guard.
-- Shared across EVERY AnimatorDefender instance (this is an upvalue, not per-self). Each AnimationPlayed
-- fires its handler on a fresh Luau fiber, and the evaluation below (apProcess / processValidated) yields
-- with task.wait through the parry timing -- so without a ceiling, an animation burst (crowded area, many
-- mobs, fast attacks) spawns an unbounded pile of suspended fibers that never drains, until the scheduler
-- can't allocate another stack and the client dies with "failed to spawn a fiber" (RSL - panic: ...Kernel).
-- The thread logs showed this exactly: count spiking into the thousands within a second in busy areas.
-- 25 matches PR's default; legit combat never approaches it, so no parries are lost in normal fights.
local MAX_CONCURRENT_EVALUATIONS = 25
local activeEvaluations = 0

AnimatorDefender.process = LPH_NO_VIRTUALIZE(function(self, track)
	if track.WeightTarget <= 0.050 and track.WeightCurrent <= 0.050 then
		return
	end

	local isLocalPlayer = players.LocalPlayer.Character and self.entity == players.LocalPlayer.Character

	if isLocalPlayer then
		self:asc(track)

		local aid = tostring(track.Animation.AnimationId)
		self:initial(self.entity, SaveManager.as, self.entity.Name, aid)

		return
	end

	do
		local localCharacter = players.LocalPlayer.Character
		local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")
		local entityRoot = self.entity:FindFirstChild("HumanoidRootPart")
		if localRoot and entityRoot then
			if (localRoot.Position - entityRoot.Position).Magnitude > 200 then
				return
			end
		end
	end

	-- Resolve (and cache) the canonical id at the play instant. For the anti path this primes the
	-- per-track id cache with the real id before any deferred logic runs, so a later id="" blank
	-- can't erase it; for the normal path it's the same canonical value as before.
	local aid = self:_apResolveAid(track)

	-- Concurrency cap (ported from Project Rain). Everything above here is a cheap O(1) filter; from this
	-- point on we commit to an evaluation that yields through the parry timing on this handler's fiber. If
	-- too many are already in flight, drop this one instead of piling up another suspended fiber. Decrement
	-- after 1s like PR (evaluations resolve well under that), which bounds the live-fiber count.
	if activeEvaluations >= MAX_CONCURRENT_EVALUATIONS then
		return
	end
	activeEvaluations = activeEvaluations + 1
	task.delay(1, function()
		activeEvaluations = activeEvaluations - 1
	end)

	-- Anti-breaker: for player attackers, route through the crasher-protection ingress gate
	-- and the authenticator flow while the toggle is on. NPCs and the disabled state fall
	-- straight through to the normal path.
	local advanced = players:GetPlayerFromCharacter(self.entity) ~= nil
		and Configuration.expectToggleValue("AntiBreakerEnabled") == true
	self:apArm(advanced)
	if advanced then
		local admitted, admittedTiming, admittedRecovered = self:rlimit(track)
		if not admitted then
			return
		end
		local occurrence = (self._apTrackOccurrences[track] or 0) + 1
		self._apTrackOccurrences[track] = occurrence
		self._apOccurrenceStartedAt[track] = { occurrence = occurrence, startedAt = os.clock() }
		return self:apProcess(track, occurrence, admittedTiming, admittedRecovered)
	end

	self._evaluationData[track] = { t = tick() }
	self:processValidated(track, aid)
end)

---@param self AnimatorDefender
---@param track AnimationTrack
---@param aid string
---@param gateDelay number? Extra delay from ramp/judge gate to compensate in timing.
AnimatorDefender.processValidated = LPH_NO_VIRTUALIZE(function(self, track, aid, gateDelay)
	if self.track and self.track ~= track and not self:isFullyDead(self.track) then
		local currentEval = self._evaluationData[self.track]
		local incomingEval = self._evaluationData[track]
		if currentEval and incomingEval and incomingEval.t < currentEval.t then
			return
		end
	end

	---@type AnimationTiming?
	local timing = self:initial(self.entity, SaveManager.as, self.entity.Name, aid)

	---@note: Record playback data for visualizer/builder regardless of existing timing.
	if Configuration.expectToggleValue("ShowAnimationVisualizer") then
		self.pbdata[track] = PlaybackData.new(self.entity)
	end

	if not timing then
		return
	end

	if not underAnimationProcessCap() then
		return
	end

	if not Configuration.expectToggleValue("EnableAutoDefense") then
		return
	end

	local humanoidRootPart = self.entity:FindFirstChild("HumanoidRootPart")
	if not humanoidRootPart then
		return
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	if timing.tag == "Critical" then
		timing.nbfb = true
		timing.ha = true
	end

	---@note: Clean up previous tasks that are still waiting or suspended because they're in a different track.
	self:clean()

	self.timing = timing
	self.track = track
	self.offset = Latency.rdelay() + (gateDelay or 0)
	self._gateDelay = gateDelay or 0

	self._feintedTrack = nil
	self._galeFeintTracks = nil

	self._respectFeint = not reactRoll(timing)

	local ed = self._evaluationData[track]
	self._trackPlayedAt = ed and ed.t or tick()

	self._trackStoppedAt = nil
	local stoppedSignal = Signal.new(track.Stopped)
	self.tmaid:add(stoppedSignal:connect("AnimatorDefender_TrackStopped", function()
		self._trackStoppedAt = os.clock()
	end))

	self._trackSeq = self._trackSeq + 1
	local mySeq = self._trackSeq

	if timing.umoa then
		return self:module(timing)
	end

	---@note: Start processing the timing. Add the actions if we're not RPUE.
	if not timing.rpue then
		return self:actions(timing)
	end

	local info = RepeatInfo.new(timing, Latency.rdelay(), self:uid(10))
	info.track = track
	self:srpue(self.entity, timing, info)
end)

---@param self AnimatorDefender
---@param reason string
---@param expectedTrack AnimationTrack?
function AnimatorDefender:_cancelFeintedTrack(reason, expectedTrack)
	if
		not self.track
		or (expectedTrack and expectedTrack ~= self.track)
		-- NOTE: do NOT require self.track to still be in GetPlayingAnimationTracks() here.
		-- A feint STOPS the attack track, so by the time the feint sound's Playing flips true the
		-- track has already left the playing set -- gating on that is what made react-to-feints
		-- never fire. `expectedTrack` (captured at cue time) is the real staleness guard.
		or tick() - (self._trackPlayedAt or 0) < FEINT_CUE_GRACE_PERIOD
		or (self.timing and self.timing.tag == "Mantra")
	then
		return
	end

	if LPH_OBFUSCATED then
		Logger.warn(
			"(%s) %s; cancelling running actions for '%s'.",
			self.entity.Name,
			reason,
			self.timing and self.timing.name or "?"
		)
	else
		Logger.warn(
			"(%s) [FeintDiag] clock=%.3f %s; cancelling running actions for '%s'.",
			self.entity.Name,
			os.clock(),
			reason,
			self.timing and self.timing.name or "?"
		)
	end

	self._feintedTrack = self.track
	self:clean()

	QueuedBlocking.stop("Defender_Deflect")
	QueuedBlocking.stop("Defender_BlockFallback")
end

function AnimatorDefender:clean()
	self.keyframes = {}

	Defender.clean(self)
end

	self.tsc = {}
	self.multiplier = {}

	self._evaluationData = {}

	self.apAdvancedEnabled = false
	self.apTrackingActive = false

	self._playerOwner = players:GetPlayerFromCharacter(entity)
	self.apEvidenceGeneration = 0
	self._apTrackOccurrences = setmetatable({}, { __mode = "k" })
	self._apRecoveredOccurrences = setmetatable({}, { __mode = "k" })
	self._apOccurrenceStartedAt = setmetatable({}, { __mode = "k" })

	self._apEtherLastSpendAmount = 0

	self._apEtherConn = nil
	self._apOnEtherSpend = nil
	-- Vestigial critical/occurrence confirmation tables (present for 1:1 structure; the pending
	-- sets are not populated in this build, so their process/apcr reads stay nil).
	self._apCriticalPending = {}
	self._apCriticalConfirmed = {}
	self._apCriticalConfirmedSources = {}
	self._apStampTrack = nil
	self._apStampOccurrence = nil

	self._apbEvidence = {}

	self._apbFlaggedAt = nil
	self._apbNotifiedAt = nil
	-- Quarantine ring + occurrence tables are lazily allocated on first arm (apArm ->
	-- syncIngressGeneration -> apbInitQuarantine), so an unarmed/NPC defender carries no
	-- anti-breaker overhead.
	self.maid:add(function()
		pcall(function()
			self.apMaid:clean()
		end)
	end)

	self._trackStoppedAt = nil

	self._trackPlayedAt = 0

	self._feintedTrack = nil
	self._feintPlayingConn = nil
	self._galeFeintTracks = nil
	self._respectFeint = true

	self.hits = {}
	self.maid:mark(entity.DescendantAdded:Connect(function(child)
		local isHitEffect = child.Name == "PunchBlood"
			or child.Name == "PunchEffect"
			or child.Name == "BloodSpray"
			or (child:IsA("ParticleEmitter") and child.Texture == "rbxassetid://7216855595")

		if not isHitEffect then
			return
		end

		local id = self:uid(1)
		self.hits[id] = true

		task.delay(0.2, function()
			self.hits[id] = nil
		end)
	end))

	self._feintConn = entity.DescendantAdded:Connect(function(child)
		if child.Name == "Feint" and child:IsA("Sound") then
			if self._feintPlayingConn then
				self._feintPlayingConn:Disconnect()
				self._feintPlayingConn = nil
			end

			local trackAtCue = self.track

			if child.IsPlaying then
				self:_cancelFeintedTrack("feint cue heard", trackAtCue)
			else
				self._feintPlayingConn = child:GetPropertyChangedSignal("Playing"):Connect(function()
					if not child.IsPlaying then
						return
					end

					self:_cancelFeintedTrack("feint cue heard", trackAtCue)
				end)

				self.maid:mark(self._feintPlayingConn)
			end

			return
		end

		if child.Name == "REP_SOUND_1241766316" then
			self._galeFeintTracks = animator:GetPlayingAnimationTracks()
			Logger.warn("(%s) Gale feint cue heard; snapshotted %d track(s).", entity.Name, #self._galeFeintTracks)
			return
		end

		local fakeStrike = child.Name == "REP_SOUND_5115545256" and tostring(child.PlaybackSpeed) == "2"
		if child.Name == "REP_SOUND_4954198253" or fakeStrike then
			self:_cancelFeintedTrack("fake strike cue heard", self.track)
		end
	end)
	self.maid:mark(self._feintConn)

	self._trackSeq = 0

	self.maid:mark(animationPlayed:connect(
		"AnimatorDefender_OnAnimationPlayed",
		LPH_NO_VIRTUALIZE(function(track)
			self:process(track)
		end)
	))

	return self
end

return AnimatorDefender

__bundle_register("Game/Timings/PlaybackData", function(require, _LOADED, __bundle_register, __bundle_modules)
---@class PlaybackData
---@field base number Timestamp of when the object was created.
---@field ash table<number, number> Animation speed history. The key is the timestamp delta and the value is the speed at that point.
---@field entity Model Entity to playback.
local PlaybackData = {}
PlaybackData.__index = PlaybackData

---@param from number
---@return number?, number?
function PlaybackData:last(from)
	local latestExceededSpeed = nil
	local latestExceededDelta = nil

	for delta, speed in next, self.ash do
		if from <= delta then
			continue
		end

		if latestExceededDelta and delta <= latestExceededDelta then
			continue
		end

		latestExceededSpeed = speed
		latestExceededDelta = delta
	end

	return latestExceededSpeed, latestExceededDelta
end

---@param speed number
function PlaybackData:astrack(speed)
	local delta = os.clock() - self.base

	if self:last(delta) == speed then
		return
	end

	self.ash[delta] = speed
end

---@param entity Model
---@return PlaybackData
function PlaybackData.new(entity)
	local self = setmetatable({}, PlaybackData)
	self.base = os.clock()
	self.entity = entity

	---@note: Timestamp delta is how many seconds need to pass before being able to reach this speed.
	self.ash = {}

	return self
end

return PlaybackData
end)
__bundle_register("Features/Combat/Objects/AttackOwnership", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Attack ownership / loadout scanner (ported 1:1 from Katalyst's PlayerScanning additions).
-- Caches each player's Backpack tools + equipped-weapon context (parsed from RichStats /
-- PrimaryWeapon attributes) so the anti-breaker can (a) confirm an attacker actually owns the
-- tool a timing maps to, and (b) match replicated weapon evidence to the equipped weapon.
-- Isolated module -- does NOT touch Luma's own PlayerScanning / ModuleManager.

---@module Utility.Signal
local Signal = require("Utility/Signal")

---@module Utility.Maid
local Maid = require("Utility/Maid")

local Players = game:GetService("Players")
local CollectionService = game:GetService("CollectionService")

local Ownership = { friendCache = {}, loadoutCache = {}, loadoutMaids = {} }

local sharedMaid = Maid.new()
local initialized = false
local loadoutTrackingActive = false

local function clearTable(t)
	for k in next, t do
		t[k] = nil
	end
end

local function isTrackedItem(inst)
	return inst and typeof(inst) == "Instance"
		and (inst:IsA("Tool") or inst:GetAttribute("ItemName") ~= nil or inst.Name:match("^Mantra:"))
end

local function disposeCache(player)
	local cache = Ownership.loadoutCache[player]
	if cache then
		cache.backpackMaid:clean()
		cache.characterMaid:clean()
		cache.weaponMaid:clean()
	end
	local maid = Ownership.loadoutMaids[player]
	if maid then
		maid:clean()
	end
	Ownership.loadoutMaids[player] = nil
	Ownership.loadoutCache[player] = nil
end

local function newCache(player)
	local now = os.clock()
	return {
		player = player, tools = {}, itemNames = {}, backpack = nil, character = nil,
		firstSeen = now, lastChanged = now, backpackLoaded = false, weaponInstance = nil,
		weaponContext = nil, backpackMaid = Maid.new(), characterMaid = Maid.new(), weaponMaid = Maid.new(),
	}
end

local function touch(cache)
	cache.lastChanged = os.clock()
end

local function tableCount(t)
	local n = 0
	for _ in next, t or {} do
		n = n + 1
	end
	return n
end

local function reliable(cache)
	if not cache or not cache.backpack then
		return false
	end
	local now = os.clock()
	local stable = now - cache.lastChanged >= 0.35
	local loaded = cache.backpackLoaded or CollectionService:HasTag(cache.backpack, "Loaded")
	local aged = now - cache.firstSeen >= 3
	return stable and (loaded or aged)
end

local function cacheDebug(cache)
	if not cache then
		return "cache=none"
	end
	local now = os.clock()
	return string.format("cache=%s backpack=%s loaded=%s stable=%.2f age=%.2f tools=%d itemNames=%d",
		reliable(cache) and "reliable" or "warming",
		cache.backpack and "yes" or "no",
		cache.backpackLoaded and "yes" or "no",
		now - (cache.lastChanged or now), now - (cache.firstSeen or now),
		tableCount(cache.tools), tableCount(cache.itemNames))
end

local function addTool(cache, inst)
	if not isTrackedItem(inst) then
		return
	end
	cache.tools[inst.Name] = true
	local itemName = inst:GetAttribute("ItemName")
	if typeof(itemName) == "string" and itemName ~= "" then
		cache.itemNames[itemName] = true
	end
end

local function trim(s)
	if typeof(s) ~= "string" then
		return nil
	end
	local v = s:match("^%s*(.-)%s*$")
	if v == "" then
		return nil
	end
	return v
end

local function stripRich(s)
	if typeof(s) ~= "string" then
		return ""
	end
	return (s:gsub("<[^>]*>", ""):gsub("&amp;", "&"):gsub("&lt;", "<"):gsub("&gt;", ">"))
end

local function firstNonColonLine(text)
	for line in string.gmatch(text or "", "[^\r\n]+") do
		local t = trim(line)
		if t and not string.find(t, ":", 1, true) then
			return t
		end
	end
end

local function splitPipe(text)
	local out = {}
	for part in string.gmatch(text or "", "([^|]+)") do
		out[#out + 1] = trim(part)
	end
	return out
end

local function matchNumber(pattern, text)
	local m = text and text:match(pattern)
	return tonumber(m)
end

---Parse a Weapon instance's attributes into a weapon-context table.
local function parseWeapon(weapon)
	if not weapon then
		return nil
	end
	local primary = trim(weapon:GetAttribute("PrimaryWeapon"))
	local richStats = weapon:GetAttribute("RichStats")
	local clean = stripRich(richStats)
	local typeLine = clean:match("[Tt]ype%s*:%s*([^\r\n]+)")
	local typeParts = splitPipe(typeLine)
	local name = primary or firstNonColonLine(clean)
	if not name and weapon.Name ~= "Weapon" then
		name = weapon.Name
	end
	local ctx = {
		instance = weapon,
		primaryWeapon = primary,
		weaponName = name,
		weaponType = typeParts[1],
		weaponClass = typeParts[2],
		weaponDamageType = typeParts[3],
		weaponSwingSpeed = matchNumber("[Ss]wing%s+[Ss]peed%s*:%s*([%d%.]+)", clean),
		weaponAttackDuration = matchNumber("[Aa]ttack%s+[Dd]uration%s*:%s*([%d%.]+)", clean),
		weaponRange = matchNumber("[Rr]ange%s*:%s*([%d%.]+)", clean),
		richStats = typeof(richStats) == "string" and richStats or nil,
	}
	if not ctx.weaponName and not ctx.weaponType and not ctx.weaponClass and not ctx.weaponDamageType then
		return nil
	end
	return ctx
end

local function bindWeapon(cache, weapon)
	if cache.weaponInstance == weapon then
		return
	end
	cache.weaponInstance = weapon
	cache.weaponMaid:clean()
	if not weapon then
		return
	end
	local function reparse()
		cache.weaponContext = parseWeapon(weapon)
		touch(cache)
	end
	cache.weaponMaid:add(Signal.new(weapon:GetAttributeChangedSignal("PrimaryWeapon"))
		:connect("AttackOwnership_WeaponPrimaryChanged", reparse))
	cache.weaponMaid:add(Signal.new(weapon:GetAttributeChangedSignal("RichStats"))
		:connect("AttackOwnership_WeaponRichStatsChanged", reparse))
end

local function refreshWeapon(cache)
	local weapon = cache.backpack and cache.backpack:FindFirstChild("Weapon")
	bindWeapon(cache, weapon)
	cache.weaponContext = parseWeapon(weapon)
end

local function refreshTools(player)
	local cache = Ownership.loadoutCache[player]
	if not cache then
		return
	end
	cache.tools = {}
	cache.itemNames = {}
	local backpack = cache.backpack
	cache.backpackLoaded = (backpack and CollectionService:HasTag(backpack, "Loaded") == true) or false
	if backpack then
		for _, child in next, backpack:GetChildren() do
			addTool(cache, child)
		end
	end
	refreshWeapon(cache)
	if cache.character then
		for _, child in next, cache.character:GetChildren() do
			addTool(cache, child)
		end
	end
end

local function bindContainer(player, maid, container)
	if not container then
		return
	end
	maid:add(Signal.new(container.ChildAdded):connect("AttackOwnership_LoadoutChildAdded", function(child)
		local cache = Ownership.loadoutCache[player]
		if not cache then
			return
		end
		addTool(cache, child)
		if container == cache.backpack and child.Name == "Weapon" then
			refreshTools(player)
		end
		touch(cache)
	end))
	maid:add(Signal.new(container.ChildRemoved):connect("AttackOwnership_LoadoutChildRemoved", function()
		local cache = Ownership.loadoutCache[player]
		if not cache then
			return
		end
		refreshTools(player)
		touch(cache)
	end))
end

local function bindBackpack(player, cache, backpack)
	if cache.backpack == backpack then
		return false
	end
	cache.backpackMaid:clean()
	cache.backpack = backpack
	cache.backpackLoaded = false
	cache.firstSeen = os.clock()
	bindContainer(player, cache.backpackMaid, backpack)
	return true
end

local function bindCharacter(player, cache, character)
	if cache.character == character then
		return false
	end
	cache.characterMaid:clean()
	cache.character = character
	bindContainer(player, cache.characterMaid, character)
	return true
end

local function buildCache(player)
	if player == Players.LocalPlayer then
		return nil
	end
	if Ownership.loadoutCache[player] then
		return Ownership.loadoutCache[player]
	end
	disposeCache(player)
	local maid = Maid.new()
	local cache = newCache(player)
	Ownership.loadoutMaids[player] = maid
	Ownership.loadoutCache[player] = cache

	local function resync()
		if Ownership.loadoutCache[player] ~= cache then
			return
		end
		refreshTools(player)
		touch(cache)
	end

	maid:add(Signal.new(player.ChildAdded):connect("AttackOwnership_PlayerChildAdded", function(child)
		if child.Name == "Backpack" and bindBackpack(player, cache, child) then
			resync()
		end
	end))
	maid:add(Signal.new(player.ChildRemoved):connect("AttackOwnership_PlayerChildRemoved", function(child)
		if child == cache.backpack then
			bindBackpack(player, cache, player:FindFirstChild("Backpack"))
			resync()
		end
	end))
	maid:add(Signal.new(player.CharacterAdded):connect("AttackOwnership_CharacterAdded", function(character)
		if bindCharacter(player, cache, character) then
			resync()
		end
	end))
	maid:add(Signal.new(player.CharacterRemoving):connect("AttackOwnership_CharacterRemoving", function(character)
		if character == cache.character then
			bindCharacter(player, cache, nil)
			resync()
		end
	end))

	bindBackpack(player, cache, player:FindFirstChild("Backpack"))
	bindCharacter(player, cache, player.Character)
	resync()
	return cache
end

local function disposeAll()
	local players = {}
	for p in next, Ownership.loadoutMaids do
		players[p] = true
	end
	for p in next, Ownership.loadoutCache do
		players[p] = true
	end
	for p in next, players do
		disposeCache(p)
	end
end

---Does the cached loadout satisfy the descriptor (names/exact/contains)?
local function matchDescriptor(cache, descriptor)
	local names = descriptor.names or descriptor.exact or {}
	if typeof(names) == "string" then
		names = { names }
	end
	for _, n in next, names do
		if cache.tools[n] or cache.itemNames[n] then
			return true, n
		end
	end
	local contains = descriptor.contains or {}
	if typeof(contains) == "string" then
		contains = { contains }
	end
	for tool in next, cache.tools do
		for _, needle in next, contains do
			if string.find(tool, needle, 1, true) then
				return true, tool
			end
		end
	end
	for item in next, cache.itemNames do
		for _, needle in next, contains do
			if string.find(item, needle, 1, true) then
				return true, item
			end
		end
	end
	return false, names[1] or contains[1]
end

---@return table
function Ownership.toolOwnership(player, descriptor)
	if not player or player == Players.LocalPlayer then
		return { status = "not-applicable", reason = "npc-or-local" }
	end
	if typeof(descriptor) ~= "table" or descriptor.notApplicable then
		return { status = "not-applicable", reason = "unmapped" }
	end
	if not descriptor.names and not descriptor.exact and not descriptor.contains then
		return { status = "not-applicable", reason = "no-tool-descriptor" }
	end
	local cache = Ownership.loadoutCache[player] or buildCache(player)
	if not cache then
		return { status = "unknown", reason = "no-cache", attack = descriptor.label, cache = "cache=none" }
	end
	local matched, label = matchDescriptor(cache, descriptor)
	local attackLabel = descriptor.label or label
	local dbg = cacheDebug(cache)
	if matched then
		return { status = "confirmed", reason = "matched", attack = attackLabel, matched = matched, cache = dbg }
	end
	if reliable(cache) then
		return { status = "missing", reason = "not-found", attack = attackLabel, expected = label, cache = dbg }
	end
	return { status = "unknown", reason = cache.backpack and "loading-or-unstable" or "no-backpack", attack = attackLabel, expected = label, cache = dbg }
end

---@return table
function Ownership.weaponContext(player)
	if not player or player == Players.LocalPlayer then
		return { status = "not-applicable", reason = "npc-or-local", cache = "cache=none" }
	end
	local cache = Ownership.loadoutCache[player] or buildCache(player)
	if not cache then
		return { status = "unknown", reason = "no-cache", cache = "cache=none" }
	end
	local dbg = cacheDebug(cache)
	if not reliable(cache) then
		return { status = "unknown", reason = cache.backpack and "loading-or-unstable" or "no-backpack", cache = dbg }
	end
	if not cache.weaponContext then
		return { status = "unknown", reason = cache.weaponInstance and "unparsed-weapon" or "no-backpack-weapon", cache = dbg }
	end
	local out = {}
	for k, v in next, cache.weaponContext do
		out[k] = v
	end
	out.status = "available"
	out.reason = "backpack-weapon"
	out.cache = dbg
	return out
end

function Ownership.isAlly(player)
	if not player then
		return false
	end
	local guild = Players.LocalPlayer:GetAttribute("Guild")
	return Ownership.friendCache[player] == true or (guild and guild ~= "" and player:GetAttribute("Guild") == guild)
end

local function onFriendStatus(player, status)
	Ownership.friendCache[player] = status == Enum.FriendStatus.Friend
end

local function onPlayerAdded(player)
	if player == Players.LocalPlayer then
		return
	end
	Ownership.friendCache[player] = Players.LocalPlayer:GetFriendStatus(player) == Enum.FriendStatus.Friend
	if loadoutTrackingActive then
		buildCache(player)
	end
end

local function onPlayerRemoving(player)
	disposeCache(player)
	Ownership.friendCache[player] = nil
end

function Ownership.setLoadoutTrackingActive(active)
	active = active == true
	if loadoutTrackingActive == active then
		if not active then
			disposeAll()
		end
		return
	end
	loadoutTrackingActive = active
	if active and initialized then
		for _, player in next, Players:GetPlayers() do
			if player ~= Players.LocalPlayer then
				buildCache(player)
			end
		end
	else
		disposeAll()
	end
end

function Ownership.init()
	if initialized then
		return
	end
	initialized = true
	pcall(function()
		sharedMaid:add(Signal.new(Players.LocalPlayer.FriendStatusChanged)
			:connect("AttackOwnership_FriendStatusChanged", onFriendStatus))
	end)
	pcall(function()
		sharedMaid:add(Signal.new(Players.PlayerAdded):connect("AttackOwnership_PlayerAdded", onPlayerAdded))
	end)
	pcall(function()
		sharedMaid:add(Signal.new(Players.PlayerRemoving):connect("AttackOwnership_PlayerRemoving", onPlayerRemoving))
	end)
	for _, player in next, Players:GetPlayers() do
		pcall(onPlayerAdded, player)
	end
end

function Ownership.detach()
	initialized = false
	loadoutTrackingActive = false
	sharedMaid:clean()
	disposeAll()
	clearTable(Ownership.friendCache)
end

return Ownership
end)
__bundle_register("Features/Combat/Objects/AnimationIngressGuard", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Animation ingress guard (ported 1:1 from Katalyst). Token-bucket rate limiter that
-- throttles how many suspicious / candidate / recovery animation-ingress events the
-- anti-breaker admits per-player and globally, plus a monotonic generation counter and
-- lightweight stats. Shared singleton across all defenders.

local IngressGuard = {}

local PRESSURE_WINDOW = 0.5
local CANDIDATE_ID_RING = 16
local CANDIDATE_ID_DEDUP = 0.06
local RECOVERY_RATE = 20
local RECOVERY_BURST_GLOBAL = 40
local RECOVERY_BURST_GLOBAL_TOKENS = 6
local RECOVERY_BURST_PLAYER = 12

local playerSuspiciousRate = 15
local playerSuspiciousBurst = 2
local globalSuspiciousRate = 22.5
local globalSuspiciousBurst = 4

local owners = setmetatable({}, { __mode = "k" })
local ownerCount = 0
local globalSuspicious = nil
local globalCandidate = nil
local globalRecovery = nil
local globalPressureUntil = 0
local stats = nil
local generationCounter = 0

local function newStats()
	return {
		incoming = 0, admittedSuspicious = 0, admittedCandidate = 0, quarantinedCandidate = 0,
		coalescedQuarantine = 0, admittedRecovery = 0, recoveredEvidence = 0, rejectedUniversal = 0,
		rejectedPlayerBudget = 0, rejectedGlobalBudget = 0, rejectedCandidateBudget = 0,
		rejectedRecoveryBudget = 0, advancedEntered = 0, autoDefenderReached = 0, activeMonitors = 0,
	}
end

local function newBucket(tokens, updatedAt)
	return { tokens = tokens, updatedAt = updatedAt }
end

local function refill(bucket, rate, burst, now)
	local elapsed = now - bucket.updatedAt
	if elapsed > 0 then
		bucket.tokens = math.min(burst, bucket.tokens + elapsed * rate)
		bucket.updatedAt = now
	end
end

---Try to consume one token from a player bucket AND the matching global bucket.
local function consume(playerBucket, playerRate, playerBurst, globalBucket, globalRate, globalBurst, now)
	refill(playerBucket, playerRate, playerBurst, now)
	refill(globalBucket, globalRate, globalBurst, now)
	if playerBucket.tokens < 1 then
		return false, "player-budget"
	end
	if globalBucket.tokens < 1 then
		return false, "global-budget"
	end
	playerBucket.tokens = playerBucket.tokens - 1
	globalBucket.tokens = globalBucket.tokens - 1
	return true
end

local function resetState(now)
	generationCounter = generationCounter + 1
	owners = setmetatable({}, { __mode = "k" })
	ownerCount = 0
	globalSuspicious = newBucket(globalSuspiciousBurst, now)
	globalCandidate = newBucket(6, now)
	globalRecovery = newBucket(RECOVERY_BURST_PLAYER, now)
	globalPressureUntil = 0
	stats = newStats()
end

local function ownerState(player, now)
	local character = player.Character
	local state = owners[player]
	if state and state.character == character then
		return state
	end
	state = {
		character = character,
		suspicious = newBucket(2, now),
		candidate = newBucket(3, now),
		recovery = newBucket(RECOVERY_BURST_GLOBAL_TOKENS, now),
		candidateIds = {},
		candidateIdRing = table.create(16),
		candidateIdCount = 0,
		candidateIdCursor = 1,
		pressureUntil = 0,
	}
	if not owners[player] then
		ownerCount = ownerCount + 1
	end
	owners[player] = state
	return state
end

---Candidate-id dedup ring: reject a repeated candidate id within the dedup window.
local function admitCandidateId(state, candidateId, now)
	local last = state.candidateIds[candidateId]
	if last and now - last < CANDIDATE_ID_DEDUP then
		return false
	end
	if not last then
		if state.candidateIdCount < CANDIDATE_ID_RING then
			state.candidateIdCount = state.candidateIdCount + 1
			state.candidateIdRing[state.candidateIdCount] = candidateId
		else
			local cursor = state.candidateIdCursor
			local evicted = state.candidateIdRing[cursor]
			if evicted then
				state.candidateIds[evicted] = nil
			end
			state.candidateIdRing[cursor] = candidateId
			state.candidateIdCursor = cursor % CANDIDATE_ID_RING + 1
		end
	end
	state.candidateIds[candidateId] = now
	return true
end

function IngressGuard.configure(rate)
	rate = math.clamp(tonumber(rate) or 15, 1, 200)
	if playerSuspiciousRate == rate and globalSuspicious then
		return
	end
	playerSuspiciousRate = rate
	playerSuspiciousBurst = math.max(2, math.min(4, rate * 0.1))
	globalSuspiciousRate = math.min(120, math.max(rate, rate * 1.5))
	globalSuspiciousBurst = math.max(4, math.min(8, globalSuspiciousRate * 0.1))
	local now = os.clock()
	if not globalSuspicious then
		resetState(now)
		return
	end
	refill(globalSuspicious, globalSuspiciousRate, globalSuspiciousBurst, now)
	globalSuspicious.tokens = math.min(globalSuspicious.tokens, globalSuspiciousBurst)
	for _, state in next, owners do
		refill(state.suspicious, playerSuspiciousRate, playerSuspiciousBurst, now)
		state.suspicious.tokens = math.min(state.suspicious.tokens, playerSuspiciousBurst)
	end
end

function IngressGuard.reset()
	resetState(os.clock())
end

function IngressGuard.generation()
	return generationCounter
end

function IngressGuard.removeOwner(player)
	if player and owners[player] then
		owners[player] = nil
		ownerCount = math.max(0, ownerCount - 1)
	end
end

---@return boolean admitted, string? reason, boolean underPressure
function IngressGuard.admitSuspicious(player)
	local now = os.clock()
	stats.incoming = stats.incoming + 1
	local state = ownerState(player, now)
	local ok, reason = consume(state.suspicious, playerSuspiciousRate, playerSuspiciousBurst,
		globalSuspicious, globalSuspiciousRate, globalSuspiciousBurst, now)
	if ok then
		stats.admittedSuspicious = stats.admittedSuspicious + 1
	else
		state.pressureUntil = now + PRESSURE_WINDOW
		if reason == "global-budget" then
			globalPressureUntil = now + PRESSURE_WINDOW
		end
	end
	return ok, reason, now < state.pressureUntil or now < globalPressureUntil
end

---@return boolean admitted, string? reason
function IngressGuard.admitCandidate(player, candidateId)
	local now = os.clock()
	local state = ownerState(player, now)
	if not admitCandidateId(state, candidateId, now) then
		stats.rejectedCandidateBudget = stats.rejectedCandidateBudget + 1
		return false, "repeated-candidate-id"
	end
	local ok, reason = consume(state.candidate, 12, 3, globalCandidate, 24, 6, now)
	if ok then
		stats.admittedCandidate = stats.admittedCandidate + 1
	else
		stats.rejectedCandidateBudget = stats.rejectedCandidateBudget + 1
	end
	return ok, reason
end

---@return boolean admitted, string? reason
function IngressGuard.admitRecovery(player)
	local now = os.clock()
	local state = ownerState(player, now)
	local ok, reason = consume(state.recovery, RECOVERY_RATE, RECOVERY_BURST_GLOBAL_TOKENS,
		globalRecovery, RECOVERY_BURST_GLOBAL, RECOVERY_BURST_PLAYER, now)
	if ok then
		stats.admittedRecovery = stats.admittedRecovery + 1
	else
		stats.rejectedRecoveryBudget = stats.rejectedRecoveryBudget + 1
	end
	return ok, reason
end

function IngressGuard.reject(reason)
	if reason == "global-budget" then
		stats.rejectedGlobalBudget = stats.rejectedGlobalBudget + 1
	elseif reason == "player-budget" then
		stats.rejectedPlayerBudget = stats.rejectedPlayerBudget + 1
	else
		stats.rejectedUniversal = stats.rejectedUniversal + 1
	end
end

function IngressGuard.quarantineCandidate(coalesced)
	stats.quarantinedCandidate = stats.quarantinedCandidate + 1
	if coalesced then
		stats.coalescedQuarantine = stats.coalescedQuarantine + 1
	end
end

function IngressGuard.recoveredEvidence()
	stats.recoveredEvidence = stats.recoveredEvidence + 1
end

function IngressGuard.advancedEntered()
	stats.advancedEntered = stats.advancedEntered + 1
end

function IngressGuard.autoDefenderReached()
	stats.autoDefenderReached = stats.autoDefenderReached + 1
end

function IngressGuard.monitorAdded()
	stats.activeMonitors = stats.activeMonitors + 1
end

function IngressGuard.monitorRemoved()
	stats.activeMonitors = math.max(0, stats.activeMonitors - 1)
end

function IngressGuard.snapshot()
	local out = {}
	for k, v in next, stats do
		out[k] = v
	end
	out.owners = ownerCount
	out.queued = 0
	out.pending = 0
	return out
end

IngressGuard.configure(playerSuspiciousRate)

return IngressGuard
end)
__bundle_register("Features/Combat/Objects/AttackAuthenticator", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Attack authenticator: confirms whether an attack "candidate" (one occurrence of a track)
-- is a REAL attack by matching it against authoritative server-replicated evidence, and
-- detects decoy floods statistically. Ported from Katalyst's AttackAuthenticator.
--
-- A candidate starts "pending". It becomes:
--   authenticated  -> matched by evidence within its timing/latency window (real).
--   released       -> provisional timeout during a quiet period (allowed w/o evidence).
--   rejected       -> released during a flood, or the classifier already blocked it.
--   expired        -> passed its authentication deadline.
--   cancelled      -> killed by counter-evidence (e.g. weapon-trail-off).

local AttackAuthenticator = {}
AttackAuthenticator.__index = AttackAuthenticator

local BASELINE_SAMPLES = 24
local FEATURES = { "density", "overlap", "overdue", "mismatch", "transition", "deficit" }
local FEATURE_DEFAULT = { density = 1, overlap = 1, overdue = 0, mismatch = 0, transition = 0, deficit = 0 }
local FEATURE_FLOOR = { density = 0.75, overlap = 0.75, overdue = 0.5, mismatch = 0.35, transition = 0.35, deficit = 0.75 }

local WEAPON_FAMILIES = { ["weapon-m1"] = true, ["weapon-critical"] = true, ["weapon-special"] = true }

-- Stopwords ignored when tokenizing action names.
local ACTION_STOPWORDS = {
	action = true, attack = true, cast = true, charge = true, fire = true, flame = true, frost = true,
	gale = true, go = true, ice = true, lightning = true, mantra = true, new = true, recall = true,
	release = true, shadow = true, start = true, startup = true, wind = true, windup = true,
}

---@return number[]
local function sortedValues(map)
	local out = {}
	for _, v in next, map do
		out[#out + 1] = v
	end
	table.sort(out)
	return out
end

---@return number?
local function median(list)
	if #list == 0 then
		return nil
	end
	local sorted = sortedValues(list)
	local mid = math.floor(#sorted / 2)
	if #sorted % 2 == 1 then
		return sorted[mid + 1]
	end
	return (sorted[mid] + sorted[mid + 1]) / 2
end

---Median absolute deviation from a given center.
---@return number?
local function medianDeviation(list, center)
	if #list == 0 or center == nil then
		return nil
	end
	local deviations = {}
	for _, v in next, list do
		deviations[#deviations + 1] = math.abs(v - center)
	end
	return median(deviations)
end

local function removeValue(list, value)
	for i = #list, 1, -1 do
		if list[i] == value then
			table.remove(list, i)
			return
		end
	end
end

---Does the evidence's family set cover the candidate's category?
local function familyMatch(candidate, evidence)
	local families = evidence.families
	if not families then
		return true
	end
	if families[candidate.category] == true then
		return true
	end
	return evidence.allowWeaponFamilyBridge == true
		and WEAPON_FAMILIES[candidate.category] == true
		and WEAPON_FAMILIES[evidence.exactFamily] == true
end

local function weaponKeyOf(candidate)
	local ctx = candidate.weaponContext
	if not ctx then
		return "unknown"
	end
	return table.concat({
		tostring(ctx.weaponName or ctx.primaryWeapon or "unknown"),
		tostring(ctx.weaponClass or "unknown"),
		tostring(ctx.weaponType or "unknown"),
	}, "|")
end

local function normalize(s)
	return (string.gsub(string.lower(tostring(s or "")), "[^%w]", ""))
end

---Split a name into meaningful lowercase tokens (>=4 chars, not stopwords).
local function tokenize(s)
	local text = tostring(s or "")
	text = string.gsub(text, "(%l)(%u)", "%1 %2")
	text = string.gsub(text, "(%a)(%d)", "%1 %2")
	text = string.gsub(text, "(%d)(%a)", "%1 %2")
	text = string.lower(text)
	local tokens = {}
	for word in string.gmatch(text, "%w+") do
		if #word >= 4 and not ACTION_STOPWORDS[word] then
			tokens[word] = true
		end
	end
	return tokens
end

---Length of the common prefix (reverse=false) or suffix (reverse=true) of two strings.
local function commonAffix(a, b, reverse)
	local limit = math.min(#a, #b)
	local count = 0
	for i = 1, limit do
		local ia = reverse and (#a - i + 1) or i
		local ib = reverse and (#b - i + 1) or i
		if string.sub(a, ia, ia) == string.sub(b, ib, ib) then
			count = count + 1
		else
			break
		end
	end
	return count
end

---Fuzzy match: does any of the timing's action names correspond to the evidence's action key(s)?
local function timingActionCompatible(candidate, actionKey, aliases)
	local keys = { actionKey }
	for k, v in next, aliases or {} do
		if type(k) == "number" then
			keys[#keys + 1] = v
		elseif v == true then
			keys[#keys + 1] = k
		else
			keys[#keys + 1] = v
		end
	end

	local keyTokens = {}
	for _, key in next, keys do
		for tok in next, tokenize(key) do
			keyTokens[tok] = true
		end
	end

	for _, name in next, candidate.timingActionNames or {} do
		local normName = normalize(name)
		if #normName >= 4 then
			for _, key in next, keys do
				local normKey = normalize(key)
				if #normKey >= 4 then
					if normName == normKey then
						return true
					end
					local shorter = math.min(#normName, #normKey)
					local contains = string.find(normName, normKey, 1, true) ~= nil
						or string.find(normKey, normName, 1, true) ~= nil
					if contains and shorter / math.max(#normName, #normKey) >= 0.6 then
						return true
					end
					local pre = commonAffix(normName, normKey, false)
					local suf = commonAffix(normName, normKey, true)
					if pre >= 3 and suf >= 3 and (pre + suf) / shorter >= 0.75 then
						return true
					end
				end
			end
			for tok in next, tokenize(name) do
				if keyTokens[tok] then
					return true
				end
			end
		end
	end
	return false
end
AttackAuthenticator.timingActionCompatible = timingActionCompatible

---@return boolean loose, boolean tight, number error
local function durationMatch(candidate, evidence)
	if not evidence.duration or not candidate.expectedDuration or candidate.expectedDuration <= 0 then
		return true, false, math.huge
	end
	local err = math.abs(candidate.expectedDuration - evidence.duration)
		/ math.max(candidate.expectedDuration, evidence.duration, 0.01)
	return err <= 0.75, err <= 0.35, err
end

---Has the candidate's track actually progressed (played forward) since it started?
local function progressed(candidate, now, evidence)
	if candidate.recoveredIngress and evidence and evidence.authoritative then
		return true
	end
	local track = candidate.track
	if not track or not track.IsPlaying then
		return false, "track-not-playing"
	end
	local target = tonumber(track.WeightTarget) or 0
	local position = tonumber(track.TimePosition) or 0
	local weight = tonumber(track.WeightCurrent) or 0
	if target <= 0 then
		return false, "track-target-weight-zero"
	end
	local samples = candidate.progressionSamples
	samples[#samples + 1] = { at = now, position = position, weight = weight, target = target, speed = tonumber(track.Speed) or 0 }
	while #samples > 8 do
		table.remove(samples, 1)
	end
	local first = samples[1]
	local movedPosition = position > candidate.initialTimePosition + 0.0001
	local movedWeight = first and weight > first.weight + 0.001
	if movedPosition or movedWeight or now - candidate.startedAt <= candidate.progressionGrace then
		return true
	end
	return false, "track-did-not-progress"
end

---Score a compatible candidate/evidence pairing (higher = better match).
local function scorePairing(candidate, evidence, latency, durationTight)
	local score = 0
	local ctx = candidate.weaponContext or {}
	if evidence.exactFamily and candidate.category == evidence.exactFamily then
		score = 5
	elseif evidence.allowWeaponFamilyBridge and familyMatch(candidate, evidence) then
		score = 3
	elseif evidence.families and evidence.families[candidate.category] then
		score = 2
	end
	if ctx.status == "confirmed" then
		score = score + 4
	elseif ctx.status == "unknown" then
		score = score + 1
	end
	if durationTight then
		score = score + 2
	end
	if candidate.geometricPlausible and not evidence.authoritative then
		score = score + 2
	end
	if (evidence.centralLatencyMin or evidence.latencyMin) <= latency
		and latency <= (evidence.centralLatencyMax or evidence.latencyMax) then
		score = score + 2
	end
	if candidate.comboContinuous then
		score = score + 1
	end
	return score
end

function AttackAuthenticator:_candidateMap(track)
	local map = self.candidatesByTrack[track]
	if not map then
		map = {}
		self.candidatesByTrack[track] = map
	end
	return map
end

function AttackAuthenticator:_removeCandidate(candidate)
	removeValue(self.candidates, candidate)
	local map = self.candidatesByTrack[candidate.track]
	if map then
		map[candidate.occurrence] = nil
		if next(map) == nil then
			self.candidatesByTrack[candidate.track] = nil
		end
	end
end

function AttackAuthenticator:_prune(now)
	for i = #self.candidates, 1, -1 do
		local c = self.candidates[i]
		local hardExpiry = math.max(c.deadlineAt or c.startedAt, c.startedAt + c.behaviorWindow) + c.behaviorWindow
		if c.status == "pending" and now > (c.deadlineAt or now) then
			c.status = "expired"
			c.reason = "authentication-deadline-expired"
		end
		if now > hardExpiry then
			self:_removeCandidate(c)
		end
	end
	for i = #self.evidence, 1, -1 do
		if now - self.evidence[i].at > self.longestBehaviorWindow * 2 then
			table.remove(self.evidence, i)
		end
	end
	for i = #self.mismatchEvents, 1, -1 do
		if now - self.mismatchEvents[i] > self.longestBehaviorWindow * 2 then
			table.remove(self.mismatchEvents, i)
		end
	end
	while #self.candidates > 32 do
		local oldest = self.candidates[1]
		if oldest.status == "pending" then
			oldest.status = "expired"
			oldest.reason = "candidate-capacity"
		end
		self:_removeCandidate(oldest)
	end
	while #self.evidence > 64 do
		table.remove(self.evidence, 1)
	end
	while #self.mismatchEvents > 64 do
		table.remove(self.mismatchEvents, 1)
	end
end

function AttackAuthenticator:_featureThreshold(feature)
	local samples = self.baselines[feature]
	local center = median(samples) or FEATURE_DEFAULT[feature]
	local dev = medianDeviation(samples, center) or 0
	return center + math.max(dev * 3, FEATURE_FLOOR[feature]), dev
end

function AttackAuthenticator:_featureVector(now, candidate)
	local window = candidate.behaviorWindow
	local expected = math.max(1, math.ceil(window / candidate.legalInterval))
	local inWindow = {}
	local playing = 0
	local overdue = 0
	local deficit = 0
	local transition = 0
	local previous = nil
	for _, c in next, self.candidates do
		if now - c.startedAt <= window then
			inWindow[#inWindow + 1] = c
			if c.track and c.track.IsPlaying then
				playing = playing + 1
			end
			if c.status == "pending" then
				deficit = deficit + 1
				if now - c.startedAt > c.evidenceLatencyMax then
					overdue = overdue + 1
				end
			end
			if previous and c.category ~= previous.category
				and c.startedAt - previous.startedAt < math.min(c.legalInterval, previous.legalInterval) * 0.35 then
				transition = transition + 1
			end
			if c.weaponContext and c.weaponContext.status == "mismatch" then
				transition = transition + 1
			end
			previous = c
		end
	end
	local mismatch = 0
	for _, at in next, self.mismatchEvents do
		if now - at <= window then
			mismatch = mismatch + 1
		end
	end
	local overlapDenom = math.max(1, math.min(2, expected))
	local vector = {
		density = #inWindow / expected,
		overlap = playing / overlapDenom,
		overdue = overdue / expected,
		mismatch = mismatch / expected,
		transition = transition / expected,
		deficit = deficit / expected,
	}
	return vector, deficit
end

function AttackAuthenticator:_evaluateFlood(now, candidate)
	local vector, pendingCount = self:_featureVector(now, candidate)
	local abnormal = 0
	local severe = 0
	for _, feature in next, FEATURES do
		local threshold, dev = self:_featureThreshold(feature)
		local severeMargin = math.max(dev * 2, FEATURE_FLOOR[feature])
		if vector[feature] > threshold then
			abnormal = abnormal + 1
		end
		if vector[feature] > threshold + severeMargin then
			severe = severe + 1
		end
	end
	candidate.featureVector = vector
	candidate.abnormalFeatures = abnormal
	if not self.floodActive then
		if abnormal >= 3 then
			self.enterStreak = self.enterStreak + 1
		else
			self.enterStreak = 0
		end
		if self.enterStreak >= 2 or severe >= 4 then
			self.floodActive = true
			self.floodEnteredAt = now
			self.quietSince = nil
		end
	else
		if abnormal <= 1 and pendingCount == 0 and #self.mismatchEvents == 0 then
			self.quietSince = self.quietSince or now
			local quietWindow = math.max(self.longestBehaviorWindow, candidate.behaviorWindow * 2)
			if now - self.quietSince >= quietWindow then
				self.floodActive = false
				self.enterStreak = 0
				self.quietSince = nil
			end
		else
			self.quietSince = nil
		end
	end
	return vector
end

function AttackAuthenticator:_recordBaseline(vector)
	if self.floodActive or not vector then
		return
	end
	for _, feature in next, FEATURES do
		local samples = self.baselines[feature]
		samples[#samples + 1] = vector[feature]
		while #samples > BASELINE_SAMPLES do
			table.remove(samples, 1)
		end
	end
end

---Register a new candidate for an attack occurrence (idempotent per track+occurrence).
---@return table candidate, boolean isNew, table? retriedTransaction
function AttackAuthenticator:observeCandidate(view)
	local map = self:_candidateMap(view.track)
	local existing = map[view.occurrence]
	if existing then
		return existing, false
	end

	local startedAt = view.startedAt
	local legalInterval = math.clamp(tonumber(view.legalInterval) or 0.6, 0.08, 4)
	local behaviorWindow = math.max(legalInterval, tonumber(view.longestTiming) or legalInterval)
	local evidenceLatencyMax = math.clamp(tonumber(view.evidenceLatencyMax) or legalInterval * 0.2, 0.03, legalInterval)
	local settle = math.max(evidenceLatencyMax, legalInterval * 0.2)
	local weaponKey = weaponKeyOf(view)

	local comboContinuous = view.comboContinuous ~= false
	local last = self.lastTransaction
	if last and not last.cancelled then
		local prev = last.candidate
		local gap = startedAt - last.authenticatedAt
		local minGap = math.clamp(math.min(legalInterval, prev.legalInterval) * 0.1, 0.025, 0.12)
		if gap < 0 or (gap < minGap and (weaponKey ~= prev.weaponKey or view.category ~= prev.category)) then
			comboContinuous = false
		end
	end

	self.sequence = self.sequence + 1
	self.longestBehaviorWindow = math.max(self.longestBehaviorWindow, behaviorWindow)

	local candidate = {
		id = self.sequence,
		track = view.track,
		occurrence = view.occurrence,
		timing = view.timing,
		aid = view.aid,
		category = view.category,
		weaponContext = view.weaponContext,
		weaponKey = weaponKey,
		startedAt = startedAt,
		deadlineAt = view.deadlineAt,
		plannedDefenseAt = view.plannedDefenseAt,
		legalInterval = legalInterval,
		behaviorWindow = behaviorWindow,
		evidenceLatencyMax = evidenceLatencyMax,
		progressionGrace = math.clamp(tonumber(view.progressionGrace) or 0.034, 0.016, 0.1),
		provisionalUntil = math.min(view.deadlineAt, startedAt + settle),
		initialTimePosition = tonumber(view.track.TimePosition) or 0,
		progressionSamples = {
			{
				at = startedAt,
				position = tonumber(view.track.TimePosition) or 0,
				weight = tonumber(view.track.WeightCurrent) or 0,
				target = tonumber(view.track.WeightTarget) or 0,
				speed = tonumber(view.track.Speed) or 0,
			},
		},
		expectedDuration = view.expectedDuration,
		geometricPlausible = view.geometricPlausible == true,
		comboContinuous = comboContinuous,
		actionKey = view.actionKey,
		timingActionNames = view.timingActionNames or {},
		classifierBlocked = view.classifierBlocked == true,
		callbackDispatchDeadlineAt = view.callbackDispatchDeadlineAt or view.deadlineAt,
		recoveredIngress = view.recoveredIngress == true,
		status = "pending",
		reason = "awaiting-combat-evidence",
	}

	map[view.occurrence] = candidate
	self.candidates[#self.candidates + 1] = candidate
	self:_prune(startedAt)
	self:_evaluateFlood(startedAt, candidate)
	local retried = self:_retryRetainedEvidence(startedAt)
	return candidate, true, retried
end

function AttackAuthenticator:getCandidate(track, occurrence)
	local map = self.candidatesByTrack[track]
	return map and map[occurrence] or nil
end

---Provisionally release a pending candidate once its window elapses (allow only when quiet).
function AttackAuthenticator:releaseCandidate(candidate, now)
	if not candidate or candidate.status ~= "pending" then
		return false
	end
	now = now or os.clock()
	self:_prune(now)
	self:_evaluateFlood(now, candidate)
	if now > candidate.deadlineAt then
		candidate.status = "expired"
		candidate.reason = "authentication-deadline-expired"
		return false
	end
	if self.floodActive or self.enterStreak > 0 then
		candidate.status = "rejected"
		candidate.reason = "unauthenticated-realistic-flood-candidate"
		return false
	end
	if candidate.classifierBlocked then
		candidate.status = "rejected"
		candidate.reason = "classifier-rejected"
		return false
	end
	candidate.status = "released"
	candidate.reason = "quiet-release"
	self:_recordBaseline(candidate.featureVector)
	return true
end

---Full compatibility gate between a candidate and a piece of evidence.
---@return boolean ok, table|string result
local function evidenceGate(self, candidate, evidence, now)
	if candidate.status ~= "pending" or evidence.claimedBy then
		return false, "candidate-or-evidence-already-claimed"
	end
	if evidence.at > candidate.deadlineAt then
		return false, "safe-deadline-closed"
	end
	local latency = evidence.at - candidate.startedAt
	if latency < evidence.latencyMin or latency > evidence.latencyMax then
		return false, "evidence-latency-outside-distribution"
	end
	if not familyMatch(candidate, evidence) then
		return false, "attack-family-mismatch"
	end
	if candidate.weaponContext and candidate.weaponContext.status == "mismatch" then
		return false, "weapon-context-mismatch"
	end
	if evidence.weaponKey and candidate.weaponKey ~= evidence.weaponKey then
		return false, "equipped-weapon-mismatch"
	end
	if evidence.requireExactWeaponType and not (candidate.weaponContext and candidate.weaponContext.exactWeaponTypeMatch == true) then
		return false, "exact-weapon-type-mismatch"
	end
	if evidence.actionKey then
		if evidence.requireTimingActionMatch then
			if not timingActionCompatible(candidate, evidence.actionKey, evidence.actionAliases) then
				return false, "timing-action-family-mismatch"
			end
		elseif candidate.actionKey ~= evidence.actionKey then
			return false, "action-context-mismatch"
		end
	end
	if evidence.causalCandidateId and candidate.id ~= evidence.causalCandidateId then
		return false, "causal-anchor-mismatch"
	end
	if evidence.causalTrack and candidate.track ~= evidence.causalTrack then
		return false, "causal-track-mismatch"
	end
	if evidence.causalOccurrence and candidate.occurrence ~= evidence.causalOccurrence then
		return false, "causal-occurrence-mismatch"
	end
	if not candidate.comboContinuous then
		return false, "combo-discontinuity"
	end
	if not candidate.geometricPlausible and not evidence.authoritative then
		return false, "geometry-not-plausible"
	end
	local last = self.lastTransaction
	if last and last.candidate ~= candidate and not last.cancelled
		and evidence.at - last.authenticatedAt < math.min(candidate.legalInterval, last.candidate.legalInterval) * 0.25 then
		return false, "combat-transaction-already-open"
	end
	local ok, reason = progressed(candidate, evidence.at, evidence)
	if not ok then
		return false, reason
	end
	local durLoose, durTight, durErr = durationMatch(candidate, evidence)
	if not durLoose then
		return false, "track-duration-mismatch"
	end
	return true, { candidate = candidate, latency = latency, durationTight = durTight, durationError = durErr, score = scorePairing(candidate, evidence, latency, durTight) }
end

---Record a piece of replicated evidence and authenticate the best-matching candidate.
---@return table? candidate, string reason
function AttackAuthenticator:recordEvidence(evidence, now)
	now = now or os.clock()
	self:_prune(now)
	if evidence.id == nil then
		self.evidenceSequence = self.evidenceSequence + 1
		evidence.id = self.evidenceSequence
		evidence.latencyMin = evidence.latencyMin or -0.02
		evidence.latencyMax = evidence.latencyMax or self.longestBehaviorWindow
		evidence.uncertainty = evidence.uncertainty or 0.02
		if evidence.retainForFutureCandidate == nil then
			evidence.retainForFutureCandidate = evidence.authoritative == true
		end
		self.evidence[#self.evidence + 1] = evidence
	end

	local matches = {}
	for _, candidate in next, self.candidates do
		local ok, result = evidenceGate(self, candidate, evidence, now)
		if ok then
			matches[#matches + 1] = result
		end
	end
	table.sort(matches, function(a, b)
		if a.score ~= b.score then
			return a.score > b.score
		elseif a.durationError ~= b.durationError then
			return a.durationError < b.durationError
		end
		return a.latency < b.latency
	end)

	if #matches == 0 then
		if evidence.retainForFutureCandidate then
			return nil, "awaiting-compatible-candidate"
		end
		if not evidence.finalizedUnmatched then
			evidence.finalizedUnmatched = true
			self.mismatchEvents[#self.mismatchEvents + 1] = evidence.at
		end
		return nil, "no-compatible-candidate"
	end

	local best = matches[1]
	local second = matches[2]
	if second then
		local sameShape = best.candidate.category == second.candidate.category
			and best.candidate.weaponKey == second.candidate.weaponKey
			and best.score == second.score
		local nearLatency = math.abs(best.latency - second.latency) <= evidence.uncertainty
		local nearDuration = math.abs(best.durationError - second.durationError) <= 0.05
		if sameShape and nearLatency and nearDuration then
			self.mismatchEvents[#self.mismatchEvents + 1] = evidence.at
			evidence.reason = "ambiguous-compatible-candidates"
			return nil, evidence.reason
		end
	end

	local candidate = best.candidate
	self.transactionSequence = self.transactionSequence + 1
	local transaction = {
		id = self.transactionSequence,
		candidate = candidate,
		evidence = evidence,
		authenticatedAt = evidence.at,
		scheduled = false,
		issued = false,
		cancelled = false,
	}
	evidence.claimedBy = candidate.id
	candidate.status = "authenticated"
	candidate.reason = evidence.kind
	candidate.authenticatedFamily = evidence.exactFamily or candidate.category
	candidate.familyBridged = evidence.allowWeaponFamilyBridge == true and evidence.exactFamily ~= nil and candidate.category ~= evidence.exactFamily
	candidate.transaction = transaction
	candidate.evidenceLatency = best.latency
	self.lastTransaction = transaction
	self:_recordBaseline(candidate.featureVector)
	return candidate, "authenticated"
end

---Find a retained authoritative evidence that would authenticate a candidate-shaped view
---(used by the ingress-recovery path to anchor a quarantined candidate to prior evidence).
---@return table? evidence
function AttackAuthenticator:retainedEvidenceForTiming(view, now)
	now = now or os.clock()
	self:_prune(now)
	for i = #self.evidence, 1, -1 do
		local evidence = self.evidence[i]
		if evidence.authoritative and evidence.retainForFutureCandidate
			and not evidence.claimedBy and not evidence.causalCandidateId then
			local latency = evidence.at - view.startedAt
			local latencyMin = evidence.latencyMin or -0.02
			local latencyMax = evidence.latencyMax or self.longestBehaviorWindow
			local actionOk = not evidence.actionKey or not evidence.requireTimingActionMatch
				or timingActionCompatible(view, evidence.actionKey, evidence.actionAliases)
			if latencyMin <= latency and latency <= latencyMax and familyMatch(view, evidence)
				and actionOk and (not view.deadlineAt or evidence.at <= view.deadlineAt) then
				return evidence
			end
		end
	end
	return nil
end

function AttackAuthenticator:_retryRetainedEvidence(now)
	now = now or os.clock()
	for _, evidence in next, self.evidence do
		if evidence.retainForFutureCandidate and not evidence.claimedBy then
			local result = self:recordEvidence(evidence, now)
			if result then
				return result
			end
		end
	end
end

---Counter-evidence: cancel any authenticated-but-not-yet-issued occurrence (attack aborted).
---@return table[] cancelled
function AttackAuthenticator:cancelFromEvidence(evidence)
	local cancelled = {}
	for _, candidate in next, self.candidates do
		local transaction = candidate.transaction
		local duration = transaction and transaction.evidence and tonumber(transaction.evidence.duration)
		local age = transaction and evidence.at - transaction.authenticatedAt or math.huge
		local withinWindow = not duration or age <= duration * 0.65
		if transaction and not transaction.issued and not transaction.cancelled and withinWindow
			and evidence.at <= candidate.deadlineAt
			and (not evidence.weaponKey or evidence.weaponKey == candidate.weaponKey) then
			transaction.cancelled = true
			transaction.cancelReason = evidence.kind
			candidate.status = "cancelled"
			candidate.reason = evidence.kind
			cancelled[#cancelled + 1] = candidate
		end
	end
	return cancelled
end

function AttackAuthenticator:isAuthenticated(track, occurrence)
	local candidate = self:getCandidate(track, occurrence)
	return candidate ~= nil and candidate.status == "authenticated"
		and candidate.transaction ~= nil and not candidate.transaction.cancelled
end

function AttackAuthenticator:isOccurrenceValid(track, occurrence)
	local candidate = self:getCandidate(track, occurrence)
	if not candidate then
		return true
	end
	if candidate.status == "released" then
		return true
	end
	return candidate.status == "authenticated" and candidate.transaction ~= nil and not candidate.transaction.cancelled
end

---Claim the defense sequence for an authenticated occurrence exactly once.
function AttackAuthenticator:claimSchedule(track, occurrence, now)
	local candidate = self:getCandidate(track, occurrence)
	local transaction = candidate and candidate.transaction
	if not candidate or candidate.status ~= "authenticated" or not transaction or transaction.cancelled then
		return false, "transaction-not-authenticated"
	end
	if transaction.scheduled then
		return false, "defense-sequence-already-scheduled"
	end
	now = now or os.clock()
	if transaction.authenticatedAt > candidate.deadlineAt then
		transaction.cancelled = true
		transaction.cancelReason = "authentication-deadline-expired"
		candidate.status = "expired"
		candidate.reason = transaction.cancelReason
		return false, transaction.cancelReason
	end
	if now > math.max(candidate.deadlineAt, candidate.callbackDispatchDeadlineAt or candidate.deadlineAt) then
		transaction.cancelled = true
		transaction.cancelReason = "scheduling-dispatch-deadline-expired"
		candidate.status = "expired"
		candidate.reason = transaction.cancelReason
		return false, transaction.cancelReason
	end
	transaction.scheduled = true
	transaction.scheduledAt = now
	return true, "defense-sequence-claimed"
end

function AttackAuthenticator:markIssued(track, occurrence)
	local candidate = self:getCandidate(track, occurrence)
	if candidate and candidate.transaction and not candidate.transaction.cancelled then
		local transaction = candidate.transaction
		if transaction.issued then
			return false
		end
		transaction.issued = true
		transaction.issuedAt = os.clock()
		return true
	end
	return false
end

function AttackAuthenticator:reset(reason)
	for _, candidate in next, self.candidates do
		if candidate.status == "pending" or candidate.status == "authenticated" then
			candidate.status = "cancelled"
			candidate.reason = reason or "reset"
		end
	end
	self.candidates = {}
	self.candidatesByTrack = {}
	self.evidence = {}
	self.mismatchEvents = {}
	self.lastTransaction = nil
	self.floodActive = false
	self.enterStreak = 0
	self.quietSince = nil
end

---@return AttackAuthenticator
function AttackAuthenticator.new()
	local self = setmetatable({}, AttackAuthenticator)
	self.candidates = {}
	self.candidatesByTrack = {}
	self.evidence = {}
	self.mismatchEvents = {}
	self.baselines = { density = {}, overlap = {}, overdue = {}, mismatch = {}, transition = {}, deficit = {} }
	self.floodActive = false
	self.enterStreak = 0
	self.quietSince = nil
	self.floodEnteredAt = nil
	self.lastTransaction = nil
	self.sequence = 0
	self.evidenceSequence = 0
	self.transactionSequence = 0
	self.longestBehaviorWindow = 0.5
	return self
end

return AttackAuthenticator

---@note: First instance is the entrance door. The second instance is the exit door.
---@param name string
---@return Instance?, Instance?
local function getGuildDoors(name)
	for _, instance in next, workspace:GetChildren() do
		if not instance.Name:match("GuildDoor") then
			continue
		end

		local guildName = instance:GetAttribute("GuildName")
		if not guildName then
			continue
		end

		if not guildName:lower():match(name:lower()) then
			continue
		end

		return instance, workspace:FindFirstChild(string.gsub(instance.Name, "GuildDoor", "GuildExitDoor"))
	end

	return nil
end

---@param character Model

	pcall(firetouchinterest, hrp, part, 1)
	pcall(firetouchinterest, hrp, part, 0)
	pcall(firetouchinterest, hrp, part, 1)

---@note: Ported from Project Rain. The game rubber-bands a character it thinks
---@param character Model
local function spoofMoveDirection(character)
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	pcall(sethiddenproperty, humanoid, "MoveDirectionInternal", Vector3.one)
end

---@note: Ported from Project Rain -- stops geometry from blocking the run.
local function ensureNoClip()
	if noClipSetByTeleport then
		return
	end

	if not Toggles or not Toggles.NoClip then
		return
	end

	if Toggles.NoClip.Value then
		return
	end

	noClipSetByTeleport = true
	Toggles.NoClip:SetValue(true)
end

local function restoreNoClip()
	if not noClipSetByTeleport then
		return
	end

	noClipSetByTeleport = false

	if Toggles and Toggles.NoClip then
		Toggles.NoClip:SetValue(false)
	end
end

local function resetTeleportState()
	teleportFrameCount = 0
	teleporterCache = {}
	streamRequestTimes = {}

	restoreNoClip()
end

---@param label string
---@param position Vector3
local function requestStreamAround(label, position)
	local timestamp = os.clock()
	local lastRequestTime = streamRequestTimes[label]
	if lastRequestTime and timestamp - lastRequestTime < STREAM_REQUEST_INTERVAL then
		return
	end

	streamRequestTimes[label] = timestamp
	tmaid:add(
		TaskSpawner.spawn(
			label,
			players.LocalPlayer.RequestStreamAroundAsync,
			players.LocalPlayer,
			position,
			0.1
		)
	)
end

---@param realm string
---@param destination string
---@return boolean
local function isRealmName(realm, destination)
	for _, alias in next, REALM_ALIASES[destination] or { destination } do
		if realm == alias then
			return true
		end
	end

	return false
end

---@param path table
---@return Instance?
local function getWorkspacePath(path)
	local current = workspace
	for _, name in next, path do
		current = current:FindFirstChild(name)
		if not current then
			return nil
		end
	end

	return current
end

---@param realm string
---@return BasePart?
local function getRealmTeleporter(realm)
	local cachedTeleporter = teleporterCache[realm]
	if cachedTeleporter and cachedTeleporter.Parent and isRealmName(cachedTeleporter:GetAttribute("Realm"), realm) then
		return cachedTeleporter
	end

	for _, path in next, REALM_TELEPORTER_PATHS do
		local teleporter = getWorkspacePath(path)
		if not teleporter or not teleporter:IsA("BasePart") then
			continue
		end

		if not isRealmName(teleporter:GetAttribute("Realm"), realm) then
			continue
		end

		teleporterCache[realm] = teleporter
		return teleporter
	end

	return nil
end

---@return BasePart?

	local warpPoints = workspace:FindFirstChild("WarpPoints")
	local voidheartWarpPoint = warpPoints and warpPoints:FindFirstChild("Voidheart")
	if voidheartWarpPoint and voidheartWarpPoint:IsA("BasePart") then
		return voidheartWarpPoint
	end

	local voidheart = workspace:FindFirstChild("Voidheart")
	local voidheartVoidWarp = voidheart and voidheart:FindFirstChild("VoidheartVoidWarp", true)
	if voidheartVoidWarp and voidheartVoidWarp:IsA("BasePart") then
		return voidheartVoidWarp
	end

	return nil
end

---@return BasePart?
local function getVoidheartTeleporter()

	if cachedTeleporter and cachedTeleporter.Parent then
		return cachedTeleporter
	end

	local teleporter = scanVoidheartTeleporter()
	if teleporter then
		teleporterCache.Voidheart = teleporter
	end

	return teleporter
end

local onTeleportLoop = LPH_NO_VIRTUALIZE(function()
	local dest = Teleport.destination
	if not dest then
		return
	end

	local localPlayer = players.LocalPlayer
	local character = localPlayer.Character
	if not character then
		return
	end

	---@note: Ported from Project Rain. We move onto the destination every frame
	local luminantTarget = LUMINANT_TARGETS[dest]
	if luminantTarget then
		requestStreamAround(string.format("Teleport_%sStream", dest), luminantTarget.Position)

		ensureNoClip()
		spoofMoveDirection(character)

		local hrp = character:FindFirstChild("HumanoidRootPart")
		if not hrp then
			return
		end

		hrp.AssemblyLinearVelocity = Vector3.one * (math.random() / 100)
		hrp.CFrame = luminantTarget * CFrame.new(math.random() / 100, math.random() / 100, math.random() / 100)

		local realmTeleport = getValleyRealmTeleport()
		if realmTeleport then
			fireTouchOn(character, realmTeleport)
		end

		return
	end

	if dest == "Depths" then
		tmaid:add(
			TaskSpawner.spawn(
				"Teleport_DepthsStream",
				players.LocalPlayer.RequestStreamAroundAsync,
				players.LocalPlayer,
				Vector3.new(39911.3672, 39980.9375, 39708.3203),
				0.1
			)
		)

		local modOffice = workspace:FindFirstChild("ModOffice")
		local officePit = modOffice and modOffice:FindFirstChild("OfficePit")
		if not officePit then
			return
		end

		character:PivotTo(officePit.CFrame)
	end

	if dest == "Voidheart" then
		local voidheartTeleporter = getVoidheartTeleporter()
		requestStreamAround(
			"Teleport_VoidheartStream",
			voidheartTeleporter and voidheartTeleporter.Position or Vector3.new(-20000.0, 19713.9609, -20000.0)
		)

		if not voidheartTeleporter then
			return
		end

		local vhCFrame = voidheartTeleporter.CFrame
		teleportFrameCount = teleportFrameCount + 1
		local cycleFrame = (teleportFrameCount - 1) % (SPAM_FRAMES + PAUSE_FRAMES)

		if cycleFrame < SPAM_FRAMES then
			character:PivotTo(vhCFrame)
			fireTouchOn(character, voidheartTeleporter)
			return
		end

		if cycleFrame == SPAM_FRAMES + PAUSE_FRAMES - 1 then
			local hrp = character:FindFirstChild("HumanoidRootPart")
			if hrp and (hrp.Position - vhCFrame.Position).Magnitude <= RUBBERBAND_THRESHOLD then
				return Teleport.stop()
			end
		end
	end

	if dest == "TrialOfOne" then
		tmaid:add(
			TaskSpawner.spawn(
				"Teleport_TrialStream",
				players.LocalPlayer.RequestStreamAroundAsync,
				players.LocalPlayer,
				Vector3.new(-959.787659, 146.996887, -6659.63037),
				0.1
			)
		)

		local oneEntrance = workspace:FindFirstChild("OneEntrance")
		if not oneEntrance then
			return
		end

		local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
		if not effectReplicator then
			return
		end

		local effectReplicatorModule = require(effectReplicator)
		if not effectReplicatorModule then
			return
		end

		if effectReplicatorModule:FindEffect("Knocked") then
			return Teleport.stop()
		end

		character:PivotTo(oneEntrance.CFrame)
	end

	---@note: Ported from Project Rain. Instead of dropping below the base bounds
	if dest == "GuildDoor" then
		local guildName = Configuration.expectOptionValue("GuildDoorName")
		if not guildName or #guildName <= 0 then
			return
		end

		local entranceDoor, exitDoor = getGuildDoors(guildName)
		if not entranceDoor or not exitDoor then
			return
		end

		local hrp = character:FindFirstChild("HumanoidRootPart")
		if not hrp then
			return
		end

		if (hrp.Position - entranceDoor.Position).Magnitude < 20 then
			hrp.AssemblyLinearVelocity = Vector3.zero
			return Teleport.stop()
		end

		requestStreamAround("Teleport_GuildBaseStream", exitDoor.Position)

		ensureNoClip()
		spoofMoveDirection(character)

		hrp.CFrame = exitDoor.CFrame

		local interactPrompt = exitDoor:FindFirstChild("InteractPrompt")
			or exitDoor:FindFirstChildWhichIsA("ProximityPrompt", true)

		if interactPrompt then
			Prompt.fire(interactPrompt)
		end

		return
	end
end)

---@param target_cframe CFrame
---@param speed number? (Optional, defaults to 190)
function Teleport.Safetweenstart(target_cframe, speed)
    Teleport.Safetweenstop()
    task.wait()

	local function internalSafeNAN(val)
        if val == val then return val end
        if typeof(val) == "Vector3" then
            return Vector3.new(val.X == val.X and val.X or 0, val.Y == val.Y and val.Y or 0, val.Z == val.Z and val.Z or 0)
        elseif typeof(val) == "CFrame" then
            local pos = val.Position
            local rot = val.Rotation
            return CFrame.new(pos.X == pos.X and pos.X or 0, pos.Y == pos.Y and pos.Y or 0, pos.Z == pos.Z and pos.Z or 0) * (rot == rot and rot or CFrame.identity)
        elseif typeof(val) == "number" then
            return val == val and val or 0
        end
        return val
    end

    speed = speed or 190
    local heightOffset = 8000
    local target_pos = target_cframe.Position
    local travel_pos = target_pos + Vector3.new(0, heightOffset, 0)
    local local_player = players.LocalPlayer

    local conn
    conn = runService.PreSimulation:Connect(function(dt)
        local character = local_player.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        if not root then return end

        local current_pos = root.Position
        local offset = travel_pos - current_pos
        local distance = offset.Magnitude

        if distance <= 1 or distance ~= distance then return end

        root.AssemblyLinearVelocity = Vector3.zero
        
        root.CFrame = CFrame.new(root.CFrame.X, travel_pos.Y, root.CFrame.Z)

        local dir = offset.Unit
        local step = speed * dt

        if step > distance then step = distance end

        root.CFrame = internalSafeNAN(root.CFrame + internalSafeNAN(dir * step))
    end)

    task.spawn(function()
        local character = local_player.Character
        local root = character and character:FindFirstChild("HumanoidRootPart")
        
    
        while conn and root and (Vector2.new(travel_pos.X, travel_pos.Z) - Vector2.new(root.Position.X, root.Position.Z)).Magnitude > 15 do
            task.wait()
            root = character and character:FindFirstChild("HumanoidRootPart")
        end

        if root and conn then
            root.CFrame = target_cframe
        end

        if conn then
            conn:Disconnect()
            conn = nil
        end
    end)

    Teleport.activeTween = {
        stop = function()
            if conn then
                conn:Disconnect()
                conn = nil
            end
            
            local character = players.LocalPlayer.Character
            local root = character and character:FindFirstChild("HumanoidRootPart")
            if root then
                root.CFrame = CFrame.new(root.Position.X, target_cframe.Y, root.Position.Z)
            end
        end
    }
end

function Teleport.Safetweenstop()
    if Teleport.activeTween then
        Teleport.activeTween.stop()
        Teleport.activeTween = nil
    end
end

---@param destination string
function Teleport.start(destination)
	if not destination then
		return error("Destination must be provided for teleporting.")
	end

	resetTeleportState()
	Teleport.destination = destination
end

function Teleport.stop()
	resetTeleportState()
	Teleport.destination = nil
end

function Teleport.init()
	local localPlayer = players.LocalPlayer
	local playerGui = localPlayer:WaitForChild("PlayerGui")
	local descendantAddedSignal = Signal.new(playerGui.DescendantAdded)
	tmaid:mark(descendantAddedSignal:connect("Teleport_PlayerGui_DescendantAdded", onPlayerGuiDescendantAdded))

	local renderStepped = Signal.new(runService.RenderStepped)
	tmaid:mark(renderStepped:connect("Teleport_RenderStepped", onTeleportLoop))
end

function Teleport.detach()
	Teleport.Safetweenstop()
	tmaid:clean()
end

return Teleport
end)

local GAMEDATA_URL = "https://deepwoken.co/api/proxy/get?type=all"

-- The proxy rejects requests without a browser-looking Origin/Referer/User-Agent.
local USER_AGENT = "Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 "
	.. "(KHTML, like Gecko) Chrome/124.0.0.0 Safari/537.36"

-- The most recently loaded + parsed build, and which stat/talent set the script targets.
---@type table?
AutoBuilder.current = nil
---@type "pre"|"post"
AutoBuilder.target = "pre"

-- When true, the picking loop clicks talent/mantra cards as they appear.
AutoBuilder.picking = false

-- The id of the currently loaded build (used to persist + resume across server hops).
AutoBuilder.loadedId = nil

---The player's current data slot, or nil.
---@return any
function AutoBuilder.currentSlot()
	local localPlayer = players.LocalPlayer
	return localPlayer and localPlayer:GetAttribute("DataSlot")
end

-- Persist the active state (link + target + slot) so a server hop resumes it on the same character.
local function persist()
	PersistentData.set("abdata", {
		active = AutoBuilder.picking and true or false,
		id = AutoBuilder.loadedId,
		target = AutoBuilder.target,
		slot = AutoBuilder.currentSlot(),
	})
end

-- Fire the UI state-changed hook (refreshes the Start/Stop button), if one is registered.
local function notifyState()
	if AutoBuilder.onStateChanged then
		pcall(AutoBuilder.onStateChanged)
	end
end

---Resolve the executor's HTTP request function across the common names.
---@return function?
local function getRequester()
	if type(request) == "function" then
		return request
	elseif type(http_request) == "function" then
		return http_request
	elseif syn and type(syn.request) == "function" then
		return syn.request
	elseif http and type(http.request) == "function" then
		return http.request
	end

	return nil
end

---Pull the build id out of anything the user might paste: a full builder URL, a share URL, or the
---raw id on its own.
---@param input string
---@return string?
function AutoBuilder.extractId(input)
	if type(input) ~= "string" then
		return nil
	end

	input = input:gsub("%s+", "")

	if input == "" then
		return nil
	end

	local id = input:match("[?&]id=([%w%-_]+)")
		or input:match("/builds/([%w%-_]+)")
		or input:match("/builder/([%w%-_]+)")

	-- A bare id (no URL wrapping) -- accept it as-is.
	if not id and input:match("^[%w%-_]+$") then
		id = input
	end

	return id
end

---Fetch a URL and return the decoded JSON table (or nil, err). Sends browser-ish headers because the
---deepwoken.co proxy 403s without them.
---@param url string
---@param referer string
---@return table?, string?
local function httpGetJson(url, referer)
	local requester = getRequester()
	if not requester then
		return nil, "This executor has no HTTP request function."
	end

	local ok, response = pcall(requester, {
		Url = url,
		Method = "GET",
		Headers = {
			["Accept"] = "application/json",
			["Origin"] = "https://deepwoken.co",
			["Referer"] = referer,
			["User-Agent"] = USER_AGENT,
		},
	})

	if not ok or not response then
		return nil, "Request failed (network/executor error)."
	end

	local status = response.StatusCode or response.Status or 0
	if status == 404 then
		return nil, "Not found (404)."
	elseif status ~= 200 then
		return nil, string.format("Server returned %s.", tostring(status))
	end

	local body = response.Body
	if not body or body == "" then
		return nil, "Empty response body."
	end

	local decodeOk, decoded = pcall(JSON.decode, body)
	if not decodeOk or type(decoded) ~= "table" then
		return nil, "Couldn't parse JSON."
	end

	return decoded, nil
end

-- The raw master game-data dump (talents/mantras/etc.), fetched once and cached.
AutoBuilder.gameDump = nil

---Fetch (once, cached) deepwoken's master data dump (the `get?type=all` payload).
---@return table?, string?
function AutoBuilder.fetchGameData()
	if AutoBuilder.gameDump then
		return AutoBuilder.gameDump, nil
	end

	local dump, err = httpGetJson(GAMEDATA_URL, "https://deepwoken.co/builder")
	if not dump then
		return nil, err
	end

	AutoBuilder.gameDump = dump

	return dump, nil
end

---Fetch the raw build JSON by id. Returns (buildJson, nil) or (nil, errorMessage).
---@param id string
---@return table?, string?
function AutoBuilder.fetch(id)
	if not id or id == "" then
		return nil, "No build id."
	end

	local build, err = httpGetJson(PROXY_BASE .. id, "https://deepwoken.co/builder?id=" .. id)
	if not build then
		if err == "Not found (404)." then
			return nil, "Invalid link"
		end
		return nil, err
	end

	return build, nil
end

---Load a build from anything the user pasted (URL or id), store it as the current build, and return
---(build, nil) or (nil, errorMessage). Notifies the user either way.
---@param input string
---@return table?, string?
function AutoBuilder.load(input)
	-- Nothing pasted (e.g. an empty box restored by a config load) -> do nothing, silently. We only
	-- complain when there's actual text that failed to parse.
	if type(input) ~= "string" or input:gsub("%s+", "") == "" then
		return nil, "Empty."
	end

	local id = AutoBuilder.extractId(input)
	if not id then
		Logger.notify("Auto Builder: couldn't find a build id in that link.")
		return nil, "No id."
	end

	local buildJson, err = AutoBuilder.fetch(id)
	if not buildJson then
		Logger.notify("Auto Builder: " .. tostring(err))
		return nil, err
	end

	local dump, dumpErr = AutoBuilder.fetchGameData()
	if not dump then
		Logger.notify("Auto Builder: " .. tostring(dumpErr))
		return nil, dumpErr
	end

	-- Reuse Build Assistance's data objects: BuilderData wraps DeepwokenData (ddata) plus the pre/post
	-- AttributeData spreads, cleaned talent/mantra lists, and traits.
	local bdata = BuilderData.new(buildJson, dump)

	local stats = buildJson.stats or {}
	local meta = stats.meta or {}

	local build = {
		id = id,
		name = stats.buildName or "Unnamed Build",
		meta = {
			Origin = meta.Origin or "None",
			Oath = meta.Oath or "None",
		},
		pre = bdata.pre, -- AttributeData: .base / .weapon / .attunement
		post = bdata.post,
		talents = bdata.talents,
		mantras = bdata.mantras,
		traits = bdata.traits,
		bdata = bdata,
	}

	AutoBuilder.current = build
	AutoBuilder.loadedId = id

	local classified = AutoBuilder.classify(build)
	AutoBuilder.classified = classified

	-- If we're mid-run (e.g. a reload while active), keep the persisted state current.
	if AutoBuilder.picking then
		persist()
	end

	Logger.notify(
		string.format(
			"Auto Builder: loaded \"%s\" (%d talents, %d mantras).",
			build.name,
			#build.talents,
			#build.mantras
		)
	)

	return build, nil
end

---@param target "pre"|"post"
function AutoBuilder.setTarget(target)
	if target ~= "pre" and target ~= "post" then
		return
	end

	AutoBuilder.target = target

	if AutoBuilder.picking then
		persist()
	end

	notifyState()
end

---Start the automation (picking + investment) and persist it for server-hop resume.
function AutoBuilder.startPicking()
	AutoBuilder.picking = true
	persist()
	notifyState()
end

---Stop the automation and clear the persisted resume state.
function AutoBuilder.stopPicking()
	AutoBuilder.picking = false
	PersistentData.set("abdata", nil)
	notifyState()
end

---Resume a persisted Auto Builder session after a server hop. Only resumes if it was active and on
---the same data slot. Returns the reloaded build, or nil.
---@return table?
function AutoBuilder.resume()
	local data = PersistentData.get("abdata")
	if type(data) ~= "table" or not data.active or not data.id then
		return nil
	end

	-- Only resume on the same slot it was started on (a different character shouldn't inherit it).
	local slot = AutoBuilder.currentSlot()
	if data.slot ~= nil and slot ~= nil and slot ~= data.slot then
		return nil
	end

	AutoBuilder.setTarget(data.target == "post" and "post" or "pre")

	local build = AutoBuilder.load(data.id)
	if not build then
		return nil
	end

	AutoBuilder.startPicking()

	return build
end

-- ============================================================================================
-- Preshrine / postshrine split.
--
-- Uses Build Assistance's obtainability check (bdata.ddata:possible(name, spread)), which resolves
-- each talent/mantra's requirements against an AttributeData spread. A talent belongs to the
-- preshrine column when it's obtainable at the pre-shrine spread but NOT at the post-shrine spread
-- (you'd lose access after the Shrine of Order, so grab it first). Everything else is postshrine.
-- ============================================================================================

---Split the loaded build's talents + mantras into preshrine/postshrine columns.
---@param build table
---@return table? { pre = { talents = {}, mantras = {} }, post = { talents = {}, mantras = {} } }
function AutoBuilder.classify(build)
	local bdata = build and build.bdata
	if not bdata then
		return nil
	end

	local result = {
		pre = { talents = {}, mantras = {} },
		post = { talents = {}, mantras = {} },
	}

	local function place(bucket, name)
		local canPre = bdata.ddata:possible(name, bdata.pre)
		local canPost = bdata.ddata:possible(name, bdata.post)

		if canPre and not canPost then
			table.insert(result.pre[bucket], name)
		else
			table.insert(result.post[bucket], name)
		end
	end

	for _, talent in ipairs(bdata.talents) do
		place("talents", talent)
	end

	for _, mantra in ipairs(bdata.mantras) do
		place("mantras", mantra)
	end

	return result
end

-- ============================================================================================
-- Talent / mantra card picking.
--
-- The game shows talent/mantra cards as TextButtons under PlayerGui.TalentGui.ChoiceFrame, each named
-- after the talent/mantra it grants. Rather than clicking, we fire the exact remote the game's own
-- TalentClient fires on a card click:
--     ReplicatedStorage.Requests.TalentChoice:InvokeServer(cardName, isSpecial)
-- where isSpecial is true only for a "special" hand (resonance/foretold gift). We still read the card
-- buttons to know what's on offer and apply the build's priority: preshrine, then postshrine, then a
-- random real card.
-- ============================================================================================

local replicatedStorage = game:GetService("ReplicatedStorage")

local renderStepped = Signal.new(runService.RenderStepped)
local pickMaid = Maid.new()
local pickCooldownUntil = 0
local pickBusy = false
local investCooldownUntil = 0
local investBusy = false

---@return Instance? PlayerGui.TalentGui
local function getTalentGui()
	local playerGui = players.LocalPlayer and players.LocalPlayer:FindFirstChild("PlayerGui")
	return playerGui and playerGui:FindFirstChild("TalentGui")
end

---@return Instance? ReplicatedStorage.Requests.TalentChoice
local function getTalentChoiceRemote()
	local requests = replicatedStorage:FindFirstChild("Requests")
	return requests and requests:FindFirstChild("TalentChoice")
end

-- ============================================================================================
-- Point investment.
--
-- Each +1 to an attribute fires ReplicatedStorage.Requests.IncreaseAttribute:InvokeServer(<arg>),
-- where <arg> is the character attribute name minus the "Stat_" prefix (e.g. "Agility",
-- "WeaponMedium", "ElementWind"). We read the current value from the character's Stat_* attribute and
-- fire one increment at a time toward the target spread, so the server validates each point and we
-- never overshoot.
-- ============================================================================================

-- Ordered invest targets: { stat = <character attribute>, arg = <remote argument>, group, key }.
local INVEST_TARGETS = {
	-- Weapons + attunements first (they gate the most talents), then base attributes.
	{ group = "weapon", key = "Light Wep.", stat = "Stat_WeaponLight", arg = "WeaponLight" },
	{ group = "weapon", key = "Medium Wep.", stat = "Stat_WeaponMedium", arg = "WeaponMedium" },
	{ group = "weapon", key = "Heavy Wep.", stat = "Stat_WeaponHeavy", arg = "WeaponHeavy" },

	{ group = "attunement", key = "Flamecharm", stat = "Stat_ElementFire", arg = "ElementFire" },
	{ group = "attunement", key = "Frostdraw", stat = "Stat_ElementIce", arg = "ElementIce" },
	{ group = "attunement", key = "Thundercall", stat = "Stat_ElementLightning", arg = "ElementLightning" },
	{ group = "attunement", key = "Galebreathe", stat = "Stat_ElementWind", arg = "ElementWind" },
	{ group = "attunement", key = "Shadowcast", stat = "Stat_ElementShadow", arg = "ElementShadow" },
	{ group = "attunement", key = "Ironsing", stat = "Stat_ElementMetal", arg = "ElementMetal" },
	{ group = "attunement", key = "Bloodrend", stat = "Stat_ElementBlood", arg = "ElementBlood" },

	{ group = "base", key = "Strength", stat = "Stat_Strength", arg = "Strength" },
	{ group = "base", key = "Fortitude", stat = "Stat_Fortitude", arg = "Fortitude" },
	{ group = "base", key = "Agility", stat = "Stat_Agility", arg = "Agility" },
	{ group = "base", key = "Intelligence", stat = "Stat_Intelligence", arg = "Intelligence" },
	{ group = "base", key = "Willpower", stat = "Stat_Willpower", arg = "Willpower" },
	{ group = "base", key = "Charisma", stat = "Stat_Charisma", arg = "Charisma" },
}

---@return Instance? ReplicatedStorage.Requests.IncreaseAttribute
local function getIncreaseAttributeRemote()
	local requests = replicatedStorage:FindFirstChild("Requests")
	return requests and requests:FindFirstChild("IncreaseAttribute")
end

---The spread the script is currently working toward (pre or post), or nil if no build is loaded.
---@return table?
local function targetSpread()
	local build = AutoBuilder.current
	if not build then
		return nil
	end
	return AutoBuilder.target == "post" and build.post or build.pre
end

-- Attributes we recently couldn't raise (not enough XP): [stat] = os.clock() deadline to retry.
local investBlockedUntil = {}

---Invest a single point toward the target spread. Walks the attributes in order; for the first one
---below its goal that isn't temporarily blocked, it fires one increment and waits for the stat to
---actually rise (confirming the point landed before we ever fire again -- no overshoot). If the stat
---doesn't rise in time (can't afford it), that attribute is blocked for a while and we move on to the
---next one. Returns true if a point was successfully invested.
---@return boolean
function AutoBuilder.investOnce()
	local spread = targetSpread()
	if not spread then
		return false
	end

	local character = players.LocalPlayer and players.LocalPlayer.Character
	if not character then
		return false
	end

	local remote = getIncreaseAttributeRemote()
	if not remote then
		return false
	end

	for _, entry in ipairs(INVEST_TARGETS) do
		local goal = tonumber(spread[entry.group] and spread[entry.group][entry.key]) or 0
		if goal <= 0 then
			continue
		end

		local before = tonumber(character:GetAttribute(entry.stat)) or 0
		if before >= goal then
			continue
		end

		-- Skip attributes we recently failed to raise (retry only after the block expires).
		local blockedUntil = investBlockedUntil[entry.stat]
		if blockedUntil and os.clock() < blockedUntil then
			continue
		end

		-- Fire one increment and confirm it landed before returning.
		pcall(function()
			remote:InvokeServer(entry.arg)
		end)

		local deadline = os.clock() + 0.75
		while os.clock() < deadline do
			local now = tonumber(character:GetAttribute(entry.stat)) or 0
			if now > before then
				investBlockedUntil[entry.stat] = nil
				return true
			end
			task.wait(0.05)
		end

		-- Didn't rise -> can't afford it right now. Block it briefly and move on to the next attribute.
		investBlockedUntil[entry.stat] = os.clock() + 8
	end

	return false
end

-- ============================================================================================
-- Shrine progress (used by Auto Prog). Mirrors PR's ab_builder:shrine_ready / finished -- purely
-- attribute-based: compare the character's current Stat_* values to a target spread.
-- ============================================================================================

---Sum of still-unmet attribute points for a spread, plus the total wanted. Returns (missing, wanted).
---@param spread table?
---@return number, number
function AutoBuilder.missingForSpread(spread)
	local character = players.LocalPlayer and players.LocalPlayer.Character
	if not spread or not character then
		return 0, 0
	end

	local missing, wanted = 0, 0
	for _, entry in ipairs(INVEST_TARGETS) do
		local goal = tonumber(spread[entry.group] and spread[entry.group][entry.key]) or 0
		if goal > 0 then
			wanted = wanted + goal
			local current = tonumber(character:GetAttribute(entry.stat)) or 0
			missing = missing + math.max(0, goal - current)
		end
	end

	return missing, wanted
end

---Whether the preshrine attribute spread is fully invested (ready for the Shrine of Order).
---@return boolean
function AutoBuilder.shrineReady()
	local build = AutoBuilder.current
	if not build then
		return false
	end

	local missing, wanted = AutoBuilder.missingForSpread(build.pre)
	return wanted > 0 and missing == 0
end

---Whether the postshrine (final) attribute spread is fully invested -- the build is complete.
---@return boolean
function AutoBuilder.finished()
	local build = AutoBuilder.current
	if not build then
		return false
	end

	local missing, wanted = AutoBuilder.missingForSpread(build.post)
	return wanted > 0 and missing == 0
end

---Whether the current pending choice is a "special" hand (matches the game's u1 == "special"): the
---replicated SpecialChoice has a Type and a non-empty Selection.
---@return boolean
local function isSpecialChoice()
	local info = replicatedStorage:FindFirstChild("Info")
	local dataReplication = info and info:FindFirstChild("DataReplication")
	if not dataReplication then
		return false
	end

	local ok, module = pcall(require, dataReplication)
	if not ok or type(module) ~= "table" or type(module.GetData) ~= "function" then
		return false
	end

	local gotData, data = pcall(module.GetData)
	if not gotData or type(data) ~= "table" then
		return false
	end

	local special = data.SpecialChoice
	return type(special) == "table"
		and special.Type ~= nil
		and type(special.Selection) == "table"
		and #special.Selection > 0
end

---Fire the game's TalentChoice remote for a card by name. Returns true if the remote was invoked.
---@param name string
---@return boolean
local function chooseCard(name)
	local remote = getTalentChoiceRemote()
	if not remote then
		return false
	end

	local special = isSpecialChoice()

	return pcall(function()
		remote:InvokeServer(name, special)
	end)
end

---All TextButton descendants of PlayerGui.TalentGui.ChoiceFrame (the talent/mantra cards).
---@return table
local function getChoiceButtons()
	local talentGui = getTalentGui()
	local choiceFrame = talentGui and talentGui:FindFirstChild("ChoiceFrame")

	local buttons = {}
	if not choiceFrame then
		return buttons
	end

	for _, descendant in next, choiceFrame:GetDescendants() do
		if descendant:IsA("TextButton") then
			buttons[#buttons + 1] = descendant
		end
	end

	return buttons
end

---Pick a single card according to the priority (preshrine wanted > postshrine wanted > random real
---card) by firing the TalentChoice remote. Returns true if a card was chosen.
---@return boolean
function AutoBuilder.pickOnce()
	local buttons = getChoiceButtons()
	if #buttons == 0 then
		return false
	end

	-- "Only Roll 2 For Mantras": the special "Roll 2" card only appears in a Mantra Hand (it grants 2
	-- Talents instead of a mantra). When the toggle is on, always take it before considering any card --
	-- so mantra hands never pick a mantra. Talent hands have no Roll 2 card, so they fall through.
	if Toggles and Toggles.AutoBuilderOnlyRoll2 and Toggles.AutoBuilderOnlyRoll2.Value then
		for _, button in ipairs(buttons) do
			if button.Name == "Roll 2" then
				return chooseCard("Roll 2")
			end
		end
	end

	local classified = AutoBuilder.classified
	local bdata = AutoBuilder.current and AutoBuilder.current.bdata

	-- Build the wanted-name sets (talents + mantras) for each column.
	local wantedPre, wantedPost = {}, {}
	if classified then
		for _, name in ipairs(classified.pre.talents) do
			wantedPre[name] = true
		end
		for _, name in ipairs(classified.pre.mantras) do
			wantedPre[name] = true
		end
		for _, name in ipairs(classified.post.talents) do
			wantedPost[name] = true
		end
		for _, name in ipairs(classified.post.mantras) do
			wantedPost[name] = true
		end
	end

	-- 1) A preshrine pick that's on offer.
	for _, button in ipairs(buttons) do
		if wantedPre[button.Name] then
			return chooseCard(button.Name)
		end
	end

	-- 2) A postshrine pick that's on offer.
	for _, button in ipairs(buttons) do
		if wantedPost[button.Name] then
			return chooseCard(button.Name)
		end
	end

	-- 3) None of the wanted cards are here -- pick a random real talent/mantra card. We use ddata to
	-- confirm each button's name is an actual talent/mantra (avoids firing on non-card buttons). If
	-- ddata isn't available, fall back to any button.
	local cards = {}
	for _, button in ipairs(buttons) do
		if bdata and bdata.ddata:get(button.Name) ~= nil then
			cards[#cards + 1] = button
		end
	end
	if #cards == 0 then
		cards = buttons
	end

	return chooseCard(cards[math.random(1, #cards)].Name)
end

---Register the picking loop (guarded by AutoBuilder.picking). Call once at startup.
function AutoBuilder.init()
	pickMaid:add(renderStepped:connect("AutoBuilderPick_OnRender", function()
		if not AutoBuilder.picking then
			return
		end

		-- Point investment toward the selected spread. The remote yields, so run it off-thread with a
		-- busy guard to avoid overlapping invokes.
		if not investBusy and os.clock() >= investCooldownUntil then
			investBusy = true
			task.spawn(function()
				-- investOnce self-paces (it confirms each point), so retry immediately after a success
				-- and back off when there's nothing to do / everything's blocked.
				local fired = AutoBuilder.investOnce()
				investCooldownUntil = os.clock() + (fired and 0 or 1.0)
				investBusy = false
			end)
		end

		-- Card picking: paced by a cooldown, off-thread (the TalentChoice remote yields too). Picks as
		-- soon as cards are present (no wait for the list to "settle").
		if not pickBusy and os.clock() >= pickCooldownUntil then
			pickBusy = true
			task.spawn(function()
				local fired = AutoBuilder.pickOnce()
				pickCooldownUntil = os.clock() + (fired and 1.0 or 0.25)
				pickBusy = false
			end)
		end
	end))

	if not LPH_OBFUSCATED or (shared.Luma and shared.Luma.verbose) then
		Logger.warn("Auto Builder initialized.")
	end
end

function AutoBuilder.detach()
	pickMaid:clean()
end

return AutoBuilder
end)
__bundle_register("Features/Visuals/Objects/BuilderData", function(require, _LOADED, __bundle_register, __bundle_modules)
---@module Features.Visuals.Objects.DeepwokenData
local DeepwokenData = require("Features/Visuals/Objects/DeepwokenData")

---@module Features.Visuals.Objects.AttributeData
local AttributeData = require("Features/Visuals/Objects/AttributeData")

---@module Utility.Configuration
local Configuration = require("Utility/Configuration")

---@class BuilderData
---@field ddata DeepwokenData
---@field talents string[]
---@field mantras string[]
---@field pre AttributeData
---@field post AttributeData
---@field traits table<string, number>
local BuilderData = {}
BuilderData.__index = BuilderData

---@param drinfo table
---@return AttributeData
function BuilderData:attributes(drinfo)
	return (self:ipre(drinfo) == 0) and self.pre or self.post
end

---@param drinfo table
---@return number
function BuilderData:ipre(drinfo)
	local currentInvestedPoints = 0

	for idx, value in next, drinfo do
		if typeof(idx) ~= "string" or not idx:match("Stat") then
			continue
		end

		if typeof(value) ~= "number" then
			continue
		end

		currentInvestedPoints = currentInvestedPoints + value
	end

	local pointsToGetToShrine = self.pre:points()
	local shrineOverrideState = Configuration.expectOptionValue("ShrineOverrideState")

	if not Configuration.expectToggleValue("ShrineDetectionOverride") then
		shrineOverrideState = nil
	end

	if shrineOverrideState == "Pre-Shrine" then
		return 0
	end

	if shrineOverrideState == "Post-Shrine" then
		return 1
	end

	if shrineOverrideState == "Must Shrine" then
		return 2
	end

	if not self:dshrine() then
		return 1
	end

	if currentInvestedPoints < pointsToGetToShrine then
		return 0
	end

	if currentInvestedPoints == pointsToGetToShrine then
		return 2
	end

	return 1
end

---@note: No specific tag, but if they did not shrine of order, then post and pre are the same.
---@return boolean
function BuilderData:dshrine()
	return not self.pre:compare(self.post)
end

---@param values table
function BuilderData:load(values)
	if typeof(values.talents) == "table" then
		self.talents = values.talents
	end

	if typeof(values.mantras) == "table" then

		end

		self.post:load(values.attributes)

	end

	self.traits["Erudition"] = traits["Erudition"] or 0

	self.traits["Songchant"] = traits["Songchant"] or 0

__bundle_register("Features/Visuals/Objects/DeepwokenData", function(require, _LOADED, __bundle_register, __bundle_modules)
---@class DeepwokenData
---@field talents table
---@field mantras table
local DeepwokenData = {}
DeepwokenData.__index = DeepwokenData

---@param name string
function DeepwokenData:get(name)
	return self.talents[string.lower(name)] or self.mantras[string.lower(name)]
end

---@param name string
---@param adata AttributeData
---@return boolean
function DeepwokenData:possible(name, adata)
	local data = self:get(name)
	if not data then
		return false
	end

	local formatted = {}

	if data.requirements and data.requirements.stats then
		for idx, value in pairs(data.requirements.stats) do
			formatted[idx] = value
		end
	elseif data.reqs then
		if data.reqs.base then
			for idx, value in pairs(data.reqs.base) do
				formatted[idx] = value
			end
		end

		if data.reqs.weapon then
			for idx, value in pairs(data.reqs.weapon) do
				formatted[idx] = value
			end
		end

		if data.reqs.attunement then
			for idx, value in pairs(data.reqs.attunement) do
				formatted[idx] = value
			end
		end
	else
		return false
	end

	return adata:possible(formatted)
end

---Whether an entry carries stat requirements (either schema).
---@param entry table
---@return boolean
local function hasReqs(entry)
	return (entry.requirements ~= nil and entry.requirements.stats ~= nil) or entry.reqs ~= nil
end

---@param values table
function DeepwokenData:load(values)
	-- Many talents have a duplicate "(VOI)" Voidwalker variant sharing the same name: the base entry
	-- carries the stat requirements, the variant has none (shop/memento acquisition). Storing whichever
	-- comes last would mask the real one, so a requirement-bearing variant always wins.
	local function ingest(source, target)
		for key, entry in pairs(source) do
			if typeof(entry) == "table" then
				local name = entry.name or (typeof(key) == "string" and key or nil)
				if name then
					local lname = string.lower(name)
					local existing = target[lname]
					if not existing or (hasReqs(entry) and not hasReqs(existing)) then
						target[lname] = entry
					end
				end
			end
		end
	end

	if typeof(values.talents) == "table" then
		self.talents = {}
		ingest(values.talents, self.talents)
	end

	if typeof(values.mantras) == "table" then
		self.mantras = {}
		ingest(values.mantras, self.mantras)
	end
end

---@param values table?
---@return DeepwokenData
function DeepwokenData.new(values)
	local self = setmetatable({}, DeepwokenData)

	self.talents = {}
	self.mantras = {}

	if values then
		self:load(values)
	end

	return self
end

return DeepwokenData
end)

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

local function getNearest(list, filter)
	local closest, dist = nil, 9e9
	local r = root()
	if not r then
		return nil
	end
	for _, v in next, list do
		if filter(v) then
			local pos = getPosition(v)
			if pos then
				local d = (pos - r.Position).Magnitude
				if d < dist then
					dist = d
					closest = v
				end
			end
		end
	end
	return closest
end

---Nearest free altar across Floor1/2/3 (only advances a floor once the previous is fully filled).
local function getNearestAltar()
	local room = workspace:FindFirstChild("TrueAvatarBossRoom")
	if not room then
		return nil
	end
	local r = root()
	if not r then
		return nil
	end
	local closest = nil
	local function scan(floor)
		local allFilled = true
		if not floor then
			return allFilled
		end
		for _, altar in next, floor:GetChildren() do
			if not altar.Name:match("Altar") or altar:FindFirstChild("BoneSpear") then
				continue
			end
			allFilled = false
			local pivot = altar:GetPivot().Position
			if not closest or (r.Position - pivot).Magnitude < (r.Position - closest:GetPivot().Position).Magnitude then
				closest = altar
			end
		end
		return allFilled
	end
	if scan(room:FindFirstChild("Floor1Stuff")) then
		if scan(room:FindFirstChild("Floor2Stuff")) then
			scan(room:FindFirstChild("Floor3Stuff"))
		end
	end
	return closest

---Grab a boss/loot chest: tween to the lid, fire the prompt until the choice UI appears, then let it close.

		return
	end
	local gui = playerGui()
	local t = 0
	repeat
		pcall(function()
			tempTween(chest.Lid.CFrame, 170)
		end)
		pcall(function()
			fireProx(chest:FindFirstChildWhichIsA("ProximityPrompt"))
		end)
		task.wait(0.1)
		t = t + 0.1
		gui = playerGui()
	until (gui and gui:FindFirstChild("ChoicePrompt")) or not running or t >= 3
	local w = 0
	repeat
		task.wait(0.1)
		w = w + 0.1
		gui = playerGui()
	until not (gui and gui:FindFirstChild("ChoicePrompt")) or not running or w >= 3
end

local function handleBonekeeper()
	setStage("floor 1: bonekeeper")
	tempTween(L2_BONEKEEPER, 250)
	setVoidMobs(true)

	local keeper, realKeeper = nil, nil
	repeat
		if not keeper then
			pcall(function()
				tempTween(L2_BONEKEEPER_APPROACH, 150)
			end)
		end
		if realKeeper and realKeeper.Parent == nil and not keeper then
			break
		end
		for _, v in next, liveChildren() do
			if v ~= character() and v.Name:find("%.boneboy") then
				realKeeper = v
				keeper = v
			end
		end
		if not keeper then
			pcall(function()
				tempTween(L2_BONEKEEPER, 150)
			end)
		end
		task.wait()
	until keeper or not running

	tempTween(L2_BONEKEEPER_WATCH, 250)
	repeat
		task.wait()
		local triggers = workspace:FindFirstChild("EncounterTriggers")
		local bridge = triggers and (triggers:FindFirstChild("BonekeeperBridge") or triggers:FindFirstChild("BoneKeeperBridge"))
		if not bridge or not bridge:FindFirstChild("Barrier") then
			break
		end
	until not running
end
