	return instance

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

local function runElapsed()
	while running do
		local data = PersistentData.get("authoritydata")
		if elapsedLabel and data then
			elapsedLabel.Text = "Elapsed: " .. formatDuration(math.max(0, os.time() - (data.startedAt or os.time())))
		end
		task.wait(0.5)
	end
end

local function setSpeed(value)
	value = (Options and Options.AuthorityTweenSpeed and Options.AuthorityTweenSpeed.Value) or 100
	if Options and Options.TweenStudsPerSecond and Options.TweenStudsPerSecond.Value ~= value then
		Options.TweenStudsPerSecond:SetValue(value)
	end
end

---Blocking tween to a CFrame (PR's Tween.new(cf, true, speed).wait()).
local function tweenTo(cframe, speed)
	setSpeed(speed or TWEEN_SPEED)
	Tweening.goal(MOVE_ID, cframe, false)
	local goalPos = cframe.Position
	while running do
		local r = root()
		if r and (r.Position - goalPos).Magnitude <= 4 then
			break
		end
		task.wait()
	end
	Tweening.stop(MOVE_ID)
end

	end)

local sendDialogue = nil
local dialogueOpen = false -- PR's `got`
local dialogueFires = 0 -- PR's `fc`

local function connectDialogueListener()
	if sendDialogue then
		return
	end
	pcall(function()
		sendDialogue = replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue")
		maid:mark(sendDialogue.OnClientEvent:Connect(function(v)
			dialogueOpen = not (v and v.exit)
			dialogueFires = dialogueFires + 1
		end))
	end)
end

local function fireDialogue(payload)
	if sendDialogue then
		pcall(function()
			sendDialogue:FireServer(payload)
		end)
	end
end

---Tween to Captain Trist and fire his prompt until the dialogue opens.
local function reachTrist()
	setStage("talking to Captain Trist")
	repeat
		tweenTo(TRIST_HIGH, TWEEN_SPEED)
		tweenTo(TRIST_LOW, TWEEN_SPEED)
		local npcs = workspace:FindFirstChild("NPCs")
		local trist = npcs and npcs:FindFirstChild("Captain Trist")
		local prompt = trist and trist:FindFirstChild("InteractPrompt")
		if prompt then
			Prompt.interact(prompt)
		end
		task.wait()
	until dialogueOpen or not running
end

---Close any open dialogue (fire exit until the DialogueGui closes).
local function exitDialogue()
	local start = tick()
	repeat
		fireDialogue({ exit = true })
		task.wait(0.5)
	until not (playerGui() and playerGui():FindFirstChild("DialogueGui") and playerGui().DialogueGui.Enabled)
		or not running
		or tick() - start > 8
end

---Identify the accepted job by scanning Terrain for the JobTrackerGui attachment. Returns jobType, cframe.
local function identifyJob()
	local terrain = workspace:FindFirstChild("Terrain")
	if not terrain then
		return nil, nil
	end
	local deadline = os.clock() + 8
	repeat
		for _, entity in next, terrain:GetDescendants() do
			if entity:IsA("Attachment") then
				local jobTracker = entity:FindFirstChild("JobTrackerGui")
				local textLabel = jobTracker and jobTracker:FindFirstChild("TextLabel")
				if textLabel then
					local text = textLabel.Text
					local jobType = (text:find("Hostage") and "hostage")
						or (text:lower():find("secure") and "secure")
						or (text:find("Armanent Sabotage") and "barrels")
					if jobType then
						return jobType, entity.CFrame
					end
				end
			end
		end
		task.wait()
	until os.clock() >= deadline or not running
	return nil, nil
end

local function wantsJob(jobType)
	local map = {
		hostage = Toggles and Toggles.AuthorityHostage and Toggles.AuthorityHostage.Value,
		barrels = Toggles and Toggles.AuthorityBarrels and Toggles.AuthorityBarrels.Value,
		secure = Toggles and Toggles.AuthoritySecure and Toggles.AuthoritySecure.Value,
	}
	return map[jobType] == true
end

local function m1Hold(state)
	local feat = struct and struct.active_features and struct.active_features["m1_hold"]
	if feat then
		feat.held = state
	end
end

local function splashSaysReportBack()
	local gui = playerGui() and playerGui():FindFirstChild("SplashGui")
	local frame = gui and gui:FindFirstChild("TimeFrame")
	local label = frame and frame:FindFirstChild("TimeText")
	return label ~= nil and label.Text:lower():find("report back") ~= nil
end

local function missionHostage(object)
	setStage("mission: hostage")
	exitDialogue()

	tweenTo(object * CFrame.new(0, 2000, 0), TWEEN_SPEED)
	tweenTo(object, TWEEN_SPEED)

	local live = workspace:FindFirstChild("Live")
	local hostage
	repeat
		for _, entity in next, (live and live:GetChildren() or {}) do
			if entity.Name:find("%.hostage") and entity:FindFirstChild("HumanoidRootPart") then
				if (entity.HumanoidRootPart.Position - (root() and root().Position or Vector3.zero)).Magnitude <= 1000 then
					tweenTo(entity.HumanoidRootPart.CFrame, TWEEN_SPEED)
					hostage = entity
					break
				end
			end
		end
		task.wait()
	until hostage or not running
	if not hostage then
		return
	end

	tweenTo(hostage.HumanoidRootPart.CFrame, TWEEN_SPEED)

	local er = getEffectReplicator()
	local lastCarry = 0
	local carryingSince = nil
	repeat
		if er and er:FindEffect("Carrying") then
			carryingSince = carryingSince or tick()
		else
			carryingSince = nil
			local handler = character() and character():FindFirstChild("CharacterHandler")
			local requests = handler and handler:FindFirstChild("Requests")
			local carry = requests and requests:FindFirstChild("Carry")
			if carry and tick() - lastCarry > 0.75 then
				lastCarry = tick()
				pcall(function()
					carry:FireServer()
				end)
			end
		end
		if hostage and hostage:FindFirstChild("HumanoidRootPart") then
			tweenTo(hostage.HumanoidRootPart.CFrame, TWEEN_SPEED)
		end
		task.wait()
	until (carryingSince and tick() - carryingSince >= 2) or not running

	if er and er:FindEffect("Carrying") and root() then
		tweenTo(CFrame.new(root().Position.X, 2000, root().Position.Z), TWEEN_SPEED)
	end

	reachTrist()
end

local function missionBarrels(object)
	setStage("mission: barrels")
	exitDialogue()

	pcall(function()
		character().CharacterHandler.Requests.DrawWeapon:FireServer(true)
	end)
	tweenTo(object * CFrame.new(0, 1500, 0), TWEEN_SPEED)
	tweenTo(object * CFrame.new(0, 200, 0), TWEEN_SPEED)

	local destructibles = workspace:FindFirstChild("Destructibles")
	while running do
		if splashSaysReportBack() then
			break
		end
		local crates = {}
		if destructibles then
			for _, crate in next, destructibles:GetChildren() do
				if crate.Name:find("KhanCrate") and crate:IsA("Model") then
					local pivot = crate:GetPivot().Position
					local rp = root() and root().Position or Vector3.zero
					if (Vector3.new(pivot.X, 0, pivot.Z) - Vector3.new(rp.X, 0, rp.Z)).Magnitude < 1200 then
						table.insert(crates, crate)
					end
				end
			end
		end
		table.sort(crates, function(a, b)
			local rp = root() and root().Position or Vector3.zero
			return (a:GetPivot().Position - rp).Magnitude < (b:GetPivot().Position - rp).Magnitude
		end)
		for _, crate in ipairs(crates) do
			while running and crate:GetAttribute("StructureHealth") and crate:GetAttribute("StructureHealth") > 0 do
				m1Hold(true)
				tweenTo(CFrame.new((crate:GetPivot() * CFrame.new(0, -5, 0)).Position) * CFrame.Angles(math.rad(90), 0, 0), TWEEN_SPEED)
				task.wait()
			end
			m1Hold(false)
		end
		task.wait()
	end
	m1Hold(false)

	reachTrist()
end

local function missionSecure(object)
	setStage("mission: secure")

	tweenTo(object * CFrame.new(0, 1500, 0), TWEEN_SPEED)
	tweenTo(object, TWEEN_SPEED)

	local banner
	repeat
		for _, model in next, workspace:GetChildren() do
			if model.Name == "Authority Checkpoint" and model:IsA("Model") then
				local pivot = model:GetPivot().Position
				local rp = root() and root().Position or Vector3.zero
				if (Vector3.new(pivot.X, 0, pivot.Z) - Vector3.new(rp.X, 0, rp.Z)).Magnitude < 1200 then
					banner = model
				end
			end
		end
		task.wait()
	until banner or not running
	if not banner then
		return
	end

	-- PR's 3-phase capture loop: sit on the banner (0-35s), fly high above it (35-125s), sit again
	-- (125-155s), each phase bailing out the moment the objective completes ("report back").
	while running do
		tweenTo(banner:GetPivot() * CFrame.new(0, 8, 0), TWEEN_SPEED)
		local start = tick()
		local stop = false

		repeat
			task.wait()
			if splashSaysReportBack() then
				stop = true
				break
			end
		until tick() - start > 35 or not running
		if stop then
			break
		end

		repeat
			task.wait()
			if splashSaysReportBack() then
				stop = true
				break
			end
			tweenTo(banner:GetPivot() * CFrame.new(0, 3000, 0), TWEEN_SPEED)
		until tick() - start > 90 + 35 or not running
		if stop then
			break
		end

		repeat
			task.wait()
			if splashSaysReportBack() then
				stop = true
				break
			end
			tweenTo(banner:GetPivot() * CFrame.new(0, 8, 0), TWEEN_SPEED)
		until tick() - start > 120 + 35 or not running
		if stop then
			break
		end

		if splashSaysReportBack() then
			break
		end
	end

	reachTrist()
end

-- =========================================================================================================
-- State machine.
-- =========================================================================================================
local function runPlayerSafeWatch()
	-- PR's PostSimulation nearby-player check: if someone's within range and we're not in Danger, hop; also
	-- keep pulled up to altitude so nothing can hit us.
	maid:mark(task.spawn(function()
		local er = getEffectReplicator()
		while running do
			if Finder.pnear(root() and root().Position or Vector3.new(9e9, 9e9, 9e9), PLAYER_SAFE_RADIUS) then
				if not (er and er:HasEffect("Danger")) then
					serverHop()
					return
				end
				local r = root()
				if r then
					r.CFrame = CFrame.new(r.CFrame.X, 3000, r.CFrame.Z)
				end
			end
			task.wait()
		end
	end))
end

local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "continuation", from = "_check_area", to = "_check_area" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			if game.PlaceId ~= AUTHORITY_PLACE_ID then
				return
			end

			pcall(function()
				replicatedStorage:WaitForChild("Requests"):WaitForChild("UpdateUXSettings"):FireServer("FriendlyFire", true)
			end)

			connectDialogueListener()

			-- Talk to Captain Trist + accept the job.
			reachTrist()
			if not running then
				return
			end

			-- Advance past the intro with "Yes, Sir." until the dialogue state moves on.
			setStage("accepting job")
			local startFires = dialogueFires
			repeat
				fireDialogue({ choice = "Yes, Sir." })
				task.wait()
			until startFires ~= dialogueFires or not running

			-- Background: keep the conversation closed once accepted.
			maid:mark(task.spawn(function()
				while running and dialogueOpen do
					fireDialogue({ exit = true })
					task.wait()
				end
			end))

			-- Figure out which job we accepted.
			setStage("identifying job")
			local jobType, object = identifyJob()

			if not jobType or not object or not wantsJob(jobType) then
				-- Rejoining is faster than doing an unwanted job.
				serverHop()
				return
			end

			if jobType == "hostage" then
				missionHostage(object)
			elseif jobType == "barrels" then
				missionBarrels(object)
			elseif jobType == "secure" then
				missionSecure(object)
			end

			exitDialogue()
			self:continuation()
		end,
	},
})

-- =========================================================================================================
-- Struct.
-- =========================================================================================================
local function buildStruct()
	return AutomationStruct.construct({
		id = "auto_authority",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"fly",
			"noclip",
			"no_fall",
			"no_kill_bricks",
			"m1_hold",
			"no_fire",
			"auto_equip_weapon",
			"auto_log",
		},
	})
end

-- =========================================================================================================
-- Public interface.
-- =========================================================================================================
function AutoAuthority.start()
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end
	if not AutoAuthority.inValidPlace() then
		return Logger.notify("Auto Authority: not in the Eastern Luminant.")
	end

	local slot = currentSlot()
	if not slot then
		return Logger.notify("Auto Authority: no data slot found.")
	end

	local data = PersistentData.get("authoritydata") or {}
	data.slot = slot
	data.startedAt = data.startedAt or os.time()
	PersistentData.set("authoritydata", data)

	running = true
	dialogueOpen = false
	dialogueFires = 0
	buildOverlay()
	AntiAFK.start("AutoAuthority")

	maid:mark(task.spawn(runElapsed))
	runPlayerSafeWatch()

	stateMachine.current = "idle"
	stateMachine.asyncState = StateMachine.NONE
	stateMachine.currentTransitioningEvent = nil

	struct = buildStruct()
	struct:run()
end

function AutoAuthority.stop()
	if not running then
		return
	end
	running = false

	if struct then
		pcall(function()
			struct:stop()
		end)
		struct = nil
	end

	Tweening.stop(MOVE_ID)
	pcall(function()
		AntiAFK.stop("AutoAuthority")
	end)

	maid:clean()
	sendDialogue = nil
	destroyOverlay()

	PersistentData.set("authoritydata", nil)
end

return AutoAuthority
end)

-- Stage text was removed, but setStage now re-arms the per-action watchdog on every stage change (the void
-- phase, "fighting titus", gets the shorter VOID_TIMEOUT cap). Same mechanism as the echo farm.
local function setStage(name)
	actionStartedAt = os.clock()
	currentActionTimeout = (name == "fighting titus") and VOID_TIMEOUT or ACTION_TIMEOUT
end

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

local function runElapsed()
	while running do
		local data = PersistentData.get("titusdata")
		if elapsedLabel and data then
			elapsedLabel.Text = "Elapsed: " .. formatDuration(math.max(0, os.time() - (data.startedAt or os.time())))
		end
		task.wait(0.5)
	end
end

-- =========================================================================================================
-- Movement primitives (ported from PR's autotitus table).
-- =========================================================================================================
local function setSpeed(value, force)
	-- Normally the per-automation Tween Speed slider wins; `force` lets a specific move (the dungeon-door
	-- travel) pin an exact speed regardless of the slider.
	if not force then
		value = (Options and Options.TitusTweenSpeed and Options.TitusTweenSpeed.Value) or 100
	end
	if Options and Options.TweenStudsPerSecond and Options.TweenStudsPerSecond.Value ~= value then
		Options.TweenStudsPerSecond:SetValue(value)
	end
end

---When `force` is set (the dungeon-door travel), step the root toward the goal at an EXACT studs/second
---ourselves instead of going through the shared Tween Speed slider. That slider is clamped (Min) and shared
---across every automation, so it can't reliably pin a speed at runtime -- this keeps the door approach at
---the speed we actually ask for (e.g. 50 studs/s) no matter what the slider says.
local function tempTween(cframe, speed, force)
	if force then
		Tweening.stop(MOVE_ID) -- make sure the engine isn't also driving us toward a stale goal
		local targetPos = cframe.Position
		local studsPerSecond = speed or 150
		local dt = task.wait()
		while running do
			local r = root()
			if not r then
				return
			end
			r.AssemblyLinearVelocity = Vector3.zero
			local pos = r.Position
			local toTarget = targetPos - pos
			local dist = toTarget.Magnitude
			if dist <= 1 then
				break
			end
			local stepDist = studsPerSecond * dt
			if stepDist >= dist then
				r.CFrame = CFrame.new(targetPos)
				break
			end
			r.CFrame = CFrame.new(pos + (toTarget / dist) * stepDist)
			dt = task.wait()
		end
		return
	end

	setSpeed(speed or 150, force)
	Tweening.goal(MOVE_ID, cframe, false)
	local goalPos = cframe.Position
	while running do
		local r = root()
		if r and (r.Position - goalPos).Magnitude <= 10 then
			break
		end
		task.wait()
	end
	Tweening.stop(MOVE_ID)
end

local function playerCheck(mag, position)
	local origin = position or (root() and root().Position)
	if not origin then
		return false
	end
	for _, other in next, players:GetPlayers() do
		if other == localPlayer() or not other.Character then
			continue
		end
		local hrp = other.Character:FindFirstChild("HumanoidRootPart")
		if hrp and (origin - hrp.Position).Magnitude <= mag then
			return true
		end
	end
	return false
end

local function serverHop()
	-- currentSlot() reads the live DataSlot attribute, which is often absent inside the Titus dungeon;
	-- fall back to the farm's persisted slot so the rejoin resolves a real save slot instead of no-opping
	-- on nil.
	local data = PersistentData.get("titusdata")
	local slot = currentSlot() or (data and data.slot)
	pcall(function()
		ServerHop.hop(slot, true)
	end)
end

---Per-action watchdog (same mechanism as the echo farm): if a stage stalls past its cap -- the Titus void
---("fighting titus") gets VOID_TIMEOUT, everything else ACTION_TIMEOUT -- server-hop to rejoin a fresh server
---and re-enter (titusdata resume brings the farm back). Unlike the echo farm this REJOINS rather than wiping:
---a relic-farm character must survive, so it never obliterates.
local function runActionWatchdog()
	while running do
		if actionStartedAt > 0 and os.clock() - actionStartedAt > currentActionTimeout then
			actionStartedAt = 0 -- disarm so we only fire once
			serverHop()
			return
		end
		task.wait(0.25)
	end
end

---Player-safe tween: hop if anyone is near, else fly there at altitude and snap down (PR relic version).
---@param groundLevel boolean? when true, travel at ground level (no altitude lift) instead of high up.
local function playerSafeTween(cf, speed, groundLevel, force)
	if playerCheck(PLAYER_SAFE_RADIUS) then
		serverHop()
		while task.wait() do
		end
		return
	end
	local r = root()
	if not r then
		return
	end
	if groundLevel then
		-- Match the dungeon door's Y: snap to the target's height, then tween horizontally across to it at
		-- that same Y level (no altitude lift).
		r.CFrame = CFrame.new(r.Position.X, cf.Position.Y, r.Position.Z)
		tempTween(cf, speed, force)
	else
		local y = math.random(8000, 10000)
		r.CFrame = r.CFrame + Vector3.new(0, y, 0)
		tempTween(cf + Vector3.new(0, y, 0), speed, force)
	end
	if playerCheck(PLAYER_SAFE_RADIUS, cf.Position) then
		serverHop()
		while task.wait() do
		end
		return
	end
	r = root()
	if r then
		r.CFrame = cf
	end
end

local function needFood()
	local c = character()
	if not c then
		return false
	end
	local water = c:FindFirstChild("Water")
	local stomach = c:FindFirstChild("Stomach")
	if not water or not stomach then
		return false
	end
	return (water.Value <= (water.MaxValue * 0.3)) and (stomach.Value <= (stomach.MaxValue * 0.3))
end

local function refillFood()
	setStage("refill food")
	local c = character()
	local water = c and c:FindFirstChild("Water")
	if not water then
		return
	end

	local function nearestWell()
		local closest
		local terrain = workspace:FindFirstChild("Terrain")
		local waterFolder = terrain and terrain:FindFirstChild("Water")
		if not waterFolder then
			return nil
		end
		for _, well in next, waterFolder:GetChildren() do
			if well.Name:match("DrinkingWater") then
				if not closest then
					closest = well
				else
					local cd = (root().Position - closest:GetPivot().Position).Magnitude
					local d = (root().Position - well:GetPivot().Position).Magnitude
					if d < cd then
						closest = well
					end
				end
			end
		end
		return closest
	end

	if not (water.Value >= (water.MaxValue * 0.75)) then
		local well = nearestWell()
		if not well then
			return
		end
		playerSafeTween(well:GetPivot(), 250)
		repeat
			well = nearestWell()
			task.wait()
		until well or not running
		repeat
			pcall(function()
				Prompt.interact(well.InteractPrompt)
			end)
			task.wait(0.2)
		until (water.Value >= (water.MaxValue * 0.75)) or not running
	end
end

local function enterDungeon()
	setStage("enter dungeon")
	-- Force noclip on for the trip to the dungeon door so geometry can't snag us on the way in.
	pcall(function()
		if Toggles and Toggles.NoClip then
			Toggles.NoClip:SetValue(true)
		end
	end)
	pcall(function()
		replicatedStorage.Requests.UpdateUXSettings:FireServer("FriendlyFire", true)
	end)

	-- Keep noclip enforced for the WHOLE trip to the door (not just once we're there): re-assert the NoClip
	-- toggle and brute-force every character part CanCollide=false each frame while we travel, so geometry
	-- can't snag us on the way in.
	local travelNoclip = runService.Heartbeat:Connect(function()
		if Toggles and Toggles.NoClip and not Toggles.NoClip.Value then
			Toggles.NoClip:SetValue(true)
		end
		local c = character()
		if c then
			for _, part in next, c:GetDescendants() do
				if part:IsA("BasePart") and part.CanCollide then
					part.CanCollide = false
				end
			end
		end
	end)

	playerSafeTween(ENTRANCE, ENTRANCE_SPEED, true, true) -- ground level (y=0) to the door; force 50 studs/s regardless of the Tween Speed slider
	pcall(function()
		travelNoclip:Disconnect()
	end)
	pcall(function()
		replicatedStorage.Requests.UpdateUXSettings:FireServer("FriendlyFire", true)
	end)

	local entry = workspace:WaitForChild("Map", 10)
	entry = entry and entry:WaitForChild("MeritEntry", 10)

	local holdConn = runService.Heartbeat:Connect(function()
		local r = root()
		if r then
			r.CFrame = ENTRANCE
		end
		-- Brute-force noclip for the door approach: force every character part CanCollide=false each frame so
		-- geometry can't snag us (doesn't rely on the NoClip toggle / Movement loop).
		local c = character()
		if c then
			for _, part in next, c:GetDescendants() do
				if part:IsA("BasePart") and part.CanCollide then
					part.CanCollide = false
				end
			end
		end
	end)

	local start = tick()
	repeat
		-- Keep noclip ON for the whole door approach (re-assert each pass -- a one-shot set can get cleared by
		-- the fresh-character spawn / feature management, same reason the no-fall loop re-asserts).
		if Toggles and Toggles.NoClip and not Toggles.NoClip.Value then
			Toggles.NoClip:SetValue(true)
		end
		if playerCheck(PLAYER_SAFE_RADIUS, ENTRANCE.Position) then
			if holdConn then
				holdConn:Disconnect()
			end
			serverHop()
			while task.wait() do
			end
			return
		end
		if entry and entry:FindFirstChild("InteractPrompt") then
			Prompt.interact(entry.InteractPrompt)
		end
		task.wait(0.1)
	until tick() - start > 30 or game.PlaceId ~= EASTERN_PLACE_ID or not running

	if holdConn then
		holdConn:Disconnect()
	end
end

-- Old pathfind (mob_ai) breaker, baked straight into the Titus fight (no toggle). Every Heartbeat we
-- snapshot the root's real velocity, then slam it straight down (0, -9e11, 0); we restore the snapshot
-- BEFORE the next physics step so we never actually fall -- the slam lives for one tick, just long enough
-- to replicate the fling. The restore runs on PreRender AND PreSimulation: PreRender covers the focused
-- case, PreSimulation also fires while the window is UNFOCUSED (PreRender doesn't), so an AFK/tabbed-out
-- session can't skip the restore and let the slam rocket us into the void / null zone.
local titusBreakerHB = nil
local titusBreakerPR = nil
local titusBreakerPS = nil
local titusBreakerLastVel = nil

local function stopTitusBreaker()
	if titusBreakerHB then
		pcall(function()
			titusBreakerHB:Disconnect()
		end)
		titusBreakerHB = nil
	end
	if titusBreakerPR then
		pcall(function()
			titusBreakerPR:Disconnect()
		end)
		titusBreakerPR = nil
	end
	if titusBreakerPS then
		pcall(function()
			titusBreakerPS:Disconnect()
		end)
		titusBreakerPS = nil
	end
	if titusBreakerLastVel then
		local r = root()
		if r then
			pcall(function()
				r.AssemblyLinearVelocity = titusBreakerLastVel
			end)
		end
		titusBreakerLastVel = nil
	end
end

local function startTitusBreaker()
	stopTitusBreaker()

	-- Restore the pre-slam velocity. Runs on every pre-physics tick so the slam is always undone before
	-- physics integrates it (otherwise -9e11 would carry us straight into the void).
	local function restoreBreakerVelocity()
		if not titusBreakerLastVel then
			return
		end
		local r = root()
		if r then
			r.AssemblyLinearVelocity = titusBreakerLastVel
		end
		titusBreakerLastVel = nil
	end

	titusBreakerHB = runService.Heartbeat:Connect(function()
		local r = root()
		if not r then
			return
		end
		titusBreakerLastVel = r.AssemblyLinearVelocity
		r.AssemblyLinearVelocity = Vector3.new(0, -9e11, 0)
	end)

	-- PreRender restores while the window is focused; PreSimulation (falling back to Stepped) also fires
	-- when it's unfocused, so the restore can never be skipped and let the slam fling us to the null zone.
	titusBreakerPR = runService.PreRender:Connect(restoreBreakerVelocity)
	local preSim = runService.PreSimulation or runService.Stepped
	if preSim then
		titusBreakerPS = preSim:Connect(restoreBreakerVelocity)
	end
end

local function attachToTitus()
	setStage("fighting titus")

	local titus
	repeat
		for _, ent in next, workspace.Live:GetChildren() do
			if ent.Name:lower():find("titus") and ent.Name:find("%.") then
				titus = ent
				break
			end
		end
		task.wait()
	until titus or not running
	if not titus then
		return
	end

	local torso = titus:WaitForChild("Torso")
	if not torso then
		return
	end

	local playerRoot = root()
	if not playerRoot then
		return
	end

	maid:mark(task.spawn(function()
		while titus and titus.Parent and running do
			pcall(function()
				InputClient.dodge({ direct = true })
			end)
			task.wait(0.35)
		end
	end))

	local targetCF = CFrame.new(torso.CFrame.X, APPROACH_Y, torso.CFrame.Z)
	playerRoot.AssemblyLinearVelocity = Vector3.zero
	tempTween(targetCF, 250)
	task.wait(0.5)
	tempTween(targetCF, 250)

	-- Kill by ramming up and down THROUGH Titus's torso: oscillate between his torso and a point VOID_SWING
	-- studs above it (position-only target -- a straight vertical lookAt is degenerate and NaNs the CFrame).
	-- The pathfind breaker does the damage; this just drives the motion. Done when he dies, or hop if he
	-- somehow gets flung up off the map.
	startTitusBreaker()
	-- Ram hard and FAST through Titus's torso at a fixed, slider-independent speed (the shared Tween Speed
	-- slider can't be trusted to hold a value at runtime, and the old setSpeed(200) was non-force so it
	-- silently ran at the ~100 slider default). Step the root straight between his torso and VOID_SWING studs
	-- above it, flipping the instant we reach an end, slamming the whole time via the breaker.
	local VOID_SWING = 100
	local RAM_SPEED = 900
	-- If Titus gets flung farther than this from us, stop chasing him. He's on his way out (flung/voiding),
	-- and gliding across the map after his torso is exactly the bug we're fixing. Hold position instead and
	-- keep swinging; he dies or comes back, and the per-action watchdog server-hops if the fight truly stalls.
	local MAX_RAM_DISTANCE = 10000
	local voidGoingIn = true
	local dt = task.wait()
	while running do
		if not titus or not titus.Parent or not torso.Parent then
			break
		end
		if torso.Position.Y > 2000 then
			stopTitusBreaker()
			serverHop()
			return
		end

		playerRoot = root()
		if playerRoot then
			playerRoot.AssemblyLinearVelocity = Vector3.zero

			local ramPos = torso.Position + Vector3.new(0, voidGoingIn and VOID_SWING or 0, 0)
			local toTarget = ramPos - playerRoot.Position
			local dist = toTarget.Magnitude
			-- Only ram/move while he's close. Flung too far -> don't chase; lift up to Y=1000 and wait him out.
			if dist <= MAX_RAM_DISTANCE then
				local stepDist = RAM_SPEED * dt
				if dist <= stepDist or dist < 3 then
					playerRoot.CFrame = CFrame.new(ramPos)
					voidGoingIn = not voidGoingIn
				else
					playerRoot.CFrame = CFrame.new(playerRoot.Position + (toTarget / dist) * stepDist)
				end
			else
				-- Titus flung past MAX_RAM_DISTANCE: sit at Y=1000 (keep our X/Z) instead of gliding after him.
				local pos = playerRoot.Position
				playerRoot.CFrame = CFrame.new(pos.X, 1000, pos.Z)
			end
			pcall(function()
				character().CharacterHandler.Requests.DrawWeapon:FireServer(true)
			end)
		end

		dt = task.wait()
	end
	stopTitusBreaker()
	Tweening.stop(MOVE_ID)
end

---Loot the relic chest (relic: open + force LOOT_ALL, then EXIT).
local function grabChest()
	setStage("looting chest")

	-- Find the chest (wait for it to spawn after the kill).
	local chest
	local findStart = tick()
	repeat
		local thrown = workspace:FindFirstChild("Thrown")
		if thrown then
			for _, object in next, thrown:GetChildren() do
				if object:FindFirstChild("Lid") then
					chest = object
					break
				end
			end
		end
		if not chest then
			task.wait()
		end
	until chest or not running or (tick() - findStart > 15)
	if not chest then
		return
	end

	-- Chest appeared: snap straight onto it and hold there. After the post-kill teleport we're ~1300 studs
	-- up, so don't rely on a slow descend tween -- lock onto the chest position immediately, then the
	-- re-tween loop below keeps us glued to it while we open + loot.
	do
		local rr = root()
		if rr then
			rr.AssemblyLinearVelocity = Vector3.zero
			rr.CFrame = CFrame.new(chest.Lid.CFrame.Position)
		end
	end

	-- Open: same logic as AutoSaramed -- tween onto the lid, then keep firing the prompt (re-tweening each
	-- pass so we stay glued to it) until the ChoicePrompt appears.
	tempTween(chest.Lid.CFrame, 170)
	local promptStart = tick()
	repeat
		task.wait(0.1)
		pcall(function()
			Prompt.interact(chest:FindFirstChildWhichIsA("ProximityPrompt"))
		end)
		pcall(function()
			tempTween(chest.Lid.CFrame, 170)
		end)
	until (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) or not running or (tick() - promptStart > 15)

	if not (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) then
		return
	end

	-- Force Loot All: spam LOOT_ALL while the ChoicePrompt is up (up to 3s), then EXIT to close -- same as
	-- the echo farm. (Was a plain 3s wait + EXIT that only relied on the AutoLoot feature to pull.)
	local time = tick()
	repeat
		task.wait()
		local gui = playerGui()
		local cp = gui and gui:FindFirstChild("ChoicePrompt")
		if cp then
			pcall(function()
				cp:FindFirstChild("Choice"):FireServer("LOOT_ALL")
			end)
		end
	until not (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) or not running or (tick() - time >= 3)

	local gui = playerGui()
	local cp = gui and gui:FindFirstChild("ChoicePrompt")
	if cp then
		pcall(function()
			cp:FindFirstChild("Choice"):FireServer("EXIT")
		end)
	end
	-- Intentionally NOT tweening back down to Y=0 -- stay at altitude after looting.
end

---Deposit relic stacks (>= BANK_THRESHOLD) at the Banker.
local function bankRelics()
	setStage("banking relics")

	local npcs = workspace:FindFirstChild("NPCs")
	local banker = npcs and npcs:WaitForChild("Banker", 10)
	if not banker then
		return
	end

	pcall(function()
		playerSafeTween(banker.HumanoidRootPart.CFrame * CFrame.new(0, -6, 0), 150)
	end)
	pcall(function()
		local r = root()
		if r then
			r.CFrame = banker.HumanoidRootPart.CFrame * CFrame.new(0, -6, 0)
		end
	end)

	local sendDialogue = replicatedStorage.Requests.SendDialogue
	local dl = tick()
	repeat
		local dg = playerGui() and playerGui():FindFirstChild("DialogueGui")
		if dg and dg.Enabled then
			pcall(function()
				sendDialogue:FireServer({ choice = "Can I access my account?" })
			end)
		else
			pcall(function()
				Prompt.interact(banker.InteractPrompt)
			end)
		end
		task.wait(0.5)
	until (playerGui() and playerGui():FindFirstChild("BankGui")) or tick() - dl > 15 or not running

	local bankGui = playerGui() and playerGui():FindFirstChild("BankGui")
	if not bankGui then
		return
	end

	local function bankFull()
		local cp = playerGui() and playerGui():FindFirstChild("ChoicePrompt")
		if not cp then
			return false
		end
		local ds = cp:FindFirstChild("DescSheet", true)
		if not ds or not ds:FindFirstChild("Ok") then
			return false
		end
		local dc = ds:FindFirstChild("Desc", true)
		return dc ~= nil and dc.Text == "You don't have any bank slots free."
	end

	local function confirmMax()
		local cp = playerGui() and playerGui():FindFirstChild("ChoicePrompt")
		if not cp then
			return false
		end
		local mx = cp:GetAttribute("Max")
		local ch = cp:FindFirstChild("Choice")
		if mx and ch then
			task.spawn(function()
				pcall(function()
					ch:InvokeServer(mx)
				end)
			end)
			return true
		end
		return false
	end

	for name, selected in pairs(bankItems()) do
		if selected and countItem(name) >= BANK_THRESHOLD then
			local n = 0
			while countItem(name) > 0 and n < 6 and running do
				n = n + 1
				local before = countItem(name)
				local deposit = {}
				for _, item in next, backpack():GetChildren() do
					if item.Name == name then
						deposit[#deposit + 1] = item
					end
				end
				if #deposit == 0 then
					break
				end
				pcall(function()
					bankGui:WaitForChild("Choice"):FireServer("deposit", deposit)
				end)
				local bt = tick()
				repeat
					if bankFull() then
						Logger.notify("Bank full -- Auto Titus stopped.")
						AutoTitus.stop()
						return
					end
					confirmMax()
					task.wait(0.3)
				until countItem(name) < before or tick() - bt > 8 or not running
			end
		end
	end
end

local function isCharacterCreation()
	local cc = workspace:FindFirstChild("CharacterCreator")
	local c = character()
	return cc ~= nil and c ~= nil and c.Parent == cc

---Tap a key once via VirtualInputManager (input-only; no remote).
local function pressKey(keyCode)
	pcall(function()
		local vim = Instance.new("VirtualInputManager")
		vim:SendKeyEvent(true, keyCode, false, game)
		vim:SendKeyEvent(false, keyCode, false, game)
	end)
end

---Blow through the loading screen and wait for our character to spawn into workspace.Live. While the
---LoadingGui is up, spams A to click through it (input-only -- the old Requests.StartMenu.Start remote
---is an anticheat honeypot, firing it = instant ban). Skipped in character creation.
local function requestStart()
	local deadline = os.clock() + 25
	repeat
		if isCharacterCreation() then
			return
		end
		local c = character()
		if c and c.Parent and c.Parent.Name == "Live" then
			return
		end
		local pg = players.LocalPlayer and players.LocalPlayer:FindFirstChildOfClass("PlayerGui")
		if pg and pg:FindFirstChild("LoadingGui") then
			pressKey(Enum.KeyCode.A)
		end
		task.wait(0.2)
	until not running or os.clock() >= deadline
end

-- Kept the name (callers unchanged); tears down the inline pathfind breaker (the Titus kill mechanism).
local function disableBreaker()
	stopTitusBreaker()
end

-- Ferryman-style auto log: throughout the entire farm, hop the moment another player comes within the
-- per-automation auto-log distance (adjustable 0-2000 in the menu; 0 = off). Saved per automation.
local function watchForPlayers()
	while running do
		task.wait(1)
		if not running then
			break
		end
		-- Don't hop mid-creation (lobby players would softlock the character creator).
		if isCharacterCreation() then
			continue
		end
		local dist = AutoLog.get("auto_titus")
		if dist > 0 and playerCheck(dist) then
			serverHop()
			return
		end
	end
end

local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "continuation", from = "_create_character", to = "_check_area" },
		{ name = "continuation", from = "_bank_relics", to = "_check_area" },
		{ name = "bank", from = "_check_area", to = "_bank_relics" },
		{ name = "cc", from = "_check_area", to = "_create_character" },
		{ name = "enter", from = "_check_area", to = "_enter_dungeon" },
		{ name = "run", from = "_check_area", to = "_run_dungeon" },
		{ name = "exit", from = "_run_dungeon", to = "_exit" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")

			-- Merit / dungeon: existing character here, so it's safe to blow through the loading screen.
			if game.PlaceId == EASTERN_PLACE_ID then
				requestStart()
				-- Bank first if we're holding a full stack of any selected relic (cooldown-gated).
				for name, selected in pairs(bankItems()) do
					if selected and countItem(name) >= BANK_THRESHOLD and tick() - lastBank > BANK_COOLDOWN then
						lastBank = tick()
						return self:bank()
					end
				end
				-- No wipe: turn the breaker off in the hub, refill food if low, then enter.
				disableBreaker()
				if needFood() then
					refillFood()
				end
				return self:enter()
			end

			if game.PlaceId == DUNGEON_PLACE_ID then
				requestStart()
				return self:run()
			end

			-- Anywhere else (lobby / character creator): DO NOT fire StartMenu.Start -- that would rush the
			-- character creator and roll the wrong character. Wait patiently for the creator to come up.
			local deadline = os.clock() + 20
			repeat
				if isCharacterCreation() then
					return self:cc()
				end
				task.wait()
			until os.clock() >= deadline or not running
		end,

		onenter_create_character = function(self)
			if not running then
				return
			end
			setStage("create character")
			struct:create_character()
			self:continuation()
		end,

		onenter_enter_dungeon = function(self)
			if not running then
				return
			end
			enterDungeon()
		end,

		onenter_run_dungeon = function(self)
			if not running then
				return
			end
			setStage("run dungeon")

			local startTime = tick()
			repeat
				task.wait(0.2)
			until (currentArea() and tostring(currentArea()):match("Detainment")) or not running

			task.wait(1)
			attachToTitus()

			-- Re-arm the watchdog off the shorter void cap now the fight's over, so the exit-wait + loot get a
			-- fresh ACTION_TIMEOUT instead of inheriting "fighting titus"'s 45s.
			setStage("boss defeated")

			-- Titus is down: reset the 30s DungeonExitFinal cap to measure from the KILL, not the dungeon-run
			-- start, so a slow fight can't eat the window the run gets to wait for the exit to spawn.
			startTime = tick()

			local hum = humanoid()
			if hum then
				hum.Died:Connect(function()
					serverHop()
				end)
			end

			local finished = false
			local core = workspace:WaitForChild("DetainmentCore", 10)
			local conn
			if core then
				conn = core.ChildAdded:Connect(function(child)
					if child.Name == "DungeonExitFinal" then
						finished = true
					end
				end)
			end

			repeat
				task.wait(0.2)
				if tick() - startTime > 30 then
					if conn then
						conn:Disconnect()
					end
					serverHop()
					while task.wait() do
					end
					return
				end
			until finished or not running

			if conn then
				conn:Disconnect()
			end

			task.wait(0.5)

			disableBreaker()

			local r = root()
			if r then
				r.CFrame = r.CFrame * CFrame.new(0, 1300, 0)
			end

			xpcall(function()
				grabChest()
			end, function(err)
				Logger.notify("AutoTitus loot error: %s", tostring(err))
			end)

			task.wait(2)

			r = root()
			if r then
				r.CFrame = r.CFrame * CFrame.new(0, 1300, 0)
			end

			local hum2 = humanoid()
			if hum2 and hum2:GetAttribute("DangerExpiration") and hum2:GetAttribute("DangerExpiration") > 0 then
				repeat
					task.wait()
				until not (hum2:GetAttribute("DangerExpiration") and hum2:GetAttribute("DangerExpiration") > 0)
					or not running
			end

			self:exit()
		end,

		onenter_bank_relics = function(self)
			if not running then
				return
			end
			bankRelics()
			self:continuation()
		end,

		onenter_exit = function(self)
			if not running then
				return
			end
			setStage("respawn")
			-- Fast way out: replicate the Kill signal, then void ourselves if we somehow survived.
			if replicatesignal then
				pcall(function()
					replicatesignal(localPlayer().Kill)
				end)
				task.delay(1, function()
					local hum = humanoid()
					if hum and hum.Health <= 1 then
						return
					end
					local r = root()
					if r then
						r.CFrame = r.CFrame * CFrame.new(0, 9e9, 0)
					end
				end)
			else
				local r = root()
				if r then
					r.CFrame = r.CFrame * CFrame.new(0, 9e9, 0)
				end
			end
		end,
	},
})

local function buildStruct()
	return AutomationStruct.construct({
		id = "auto_titus",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"no_fall",
			"noclip",
			"fly",
			"auto_loot",
			"auto_equip_weapon",
		},
		character_creator_handler_used = true,
		character_creator_handler_opts = {
			origin = "Merit",
		},
	})
end

-- =========================================================================================================
-- Public interface.
-- =========================================================================================================
function AutoTitus.start()
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end
	if not AutoTitus.inValidPlace() then
		return Logger.notify("Auto Titus: Not in Eastern Luminant")
	end

	local slot = currentSlot()
	if not slot then
		return Logger.notify("Auto Titus: no data slot found.")
	end

	local data = PersistentData.get("titusdata") or {}
	data.slot = slot
	data.startedAt = data.startedAt or os.time()
	PersistentData.set("titusdata", data)

	running = true
	buildOverlay()
	AntiAFK.start("AutoTitus")

	maid:mark(task.spawn(runElapsed))

	-- Keep No Fall Damage forced on for the entire farm (respawns / new characters can drop it).
	maid:mark(task.spawn(function()
		while running do
			if Toggles and Toggles.NoFallDamage and not Toggles.NoFallDamage.Value then
				Toggles.NoFallDamage:SetValue(true)
			end
			task.wait(0.25)
		end
	end))

	-- Ferryman-style auto log: hop if any player comes within 1000 studs.
	maid:mark(task.spawn(watchForPlayers))

	-- Per-action watchdog: rejoin if a stage (esp. the Titus void) stalls past its cap.
	maid:mark(task.spawn(runActionWatchdog))

	stateMachine.current = "idle"
	stateMachine.asyncState = StateMachine.NONE
	stateMachine.currentTransitioningEvent = nil

	struct = buildStruct()
	struct:run()
end

function AutoTitus.stop()
	if not running then
		return
	end
	running = false

	stopTitusBreaker()

	if struct then
		pcall(function()
			struct:stop()
		end)
		struct = nil
	end

	Tweening.stop(MOVE_ID)

	pcall(function()
		AntiAFK.stop("AutoTitus")
	end)

	maid:clean()
	destroyOverlay()

	PersistentData.set("titusdata", nil)
end

return AutoTitus
end)

---Exploit-safe GUI parent: gethui -> CoreGui -> PlayerGui.
---@return Instance?
local function guiParent()
	local ok, hui = pcall(function()
		return gethui and gethui()
	end)
	if ok and hui then
		return hui
	end

	local ok2, coreGui = pcall(function()
		return game:GetService("CoreGui")
	end)
	if ok2 and coreGui then
		return coreGui
	end

	return localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
end

---@param class string
---@param props table
---@param parent Instance?
---@return Instance
local function mk(class, props, parent)
	local instance = Instance.new(class)
	for key, value in pairs(props) do
		instance[key] = value
	end
	if parent then
		instance.Parent = parent
	end
	return instance
end

---@param seconds number
---@return string
local function formatElapsed(seconds)
	local minutes = math.floor(seconds / 60)
	local secs = math.floor(seconds % 60)
	return string.format(TEXT.elapsed, minutes, secs)
end

---Total elapsed time this run, including time carried over from before a server hop.
---@return number
local function currentElapsed()
	return baseElapsed + (os.clock() - startClock)
end

---The player's current data slot, or nil.
---@return any
local function currentSlot()
	return localPlayer and localPlayer:GetAttribute("DataSlot")
end

---Persist the resume flag together with cycles + elapsed + start slot so they survive a server hop.
local function saveState()
	PersistentData.set("afdata", { active = true, cycles = cycles, elapsed = currentElapsed(), slot = startSlot })
end

---Pin the global tween speed to a fixed value for the run, remembering the old one.

	Options.TweenStudsPerSecond:SetValue((Options and Options.FerrymanTweenSpeed and Options.FerrymanTweenSpeed.Value) or 100)

end

---Turn off the "Disable When Window Not Active" Auto Defense filter for the run, remembering whether
---it was on so we can put it back.
local function applyDefenseFilter()
	local option = Options and Options.AutoDefenseFilters
	if not option then
		return
	end

	local current = option.Value or {}
	removedWindowFilter = current[WINDOW_FILTER] == true
	if not removedWindowFilter then
		return
	end

	local newValue = {}
	for key, value in pairs(current) do
		newValue[key] = value
	end
	newValue[WINDOW_FILTER] = nil
	option:SetValue(newValue)
end

local function restoreDefenseFilter()
	local option = Options and Options.AutoDefenseFilters
	if option and removedWindowFilter then
		local newValue = {}
		for key, value in pairs(option.Value or {}) do
			newValue[key] = value
		end
		newValue[WINDOW_FILTER] = true
		option:SetValue(newValue)
	end
	removedWindowFilter = false
end

---A stronger noclip than the toggle feature, plus physics stabilization. Every frame (before the
---physics step) we drop CanCollide on ALL character parts, force the AirController so nothing (the
---ground included) blocks our tweens, and zero every part's velocity so a ragdoll can't build
---momentum that fights the CFrame tween (which is what made the tween bug out when knocked).
local function applyNoClip()
	if noclipConnection then
		return
	end

	noclipConnection = runService.Stepped:Connect(function()
		if not AutoFerryman.running then
			return
		end

		local character = localPlayer and localPlayer.Character
		if not character then
			return
		end

		for _, part in next, character:GetDescendants() do
			if part:IsA("BasePart") then
				pcall(function()
					if part.CanCollide then
						part.CanCollide = false
					end
					-- Kill momentum: a ragdoll gives limbs (their own assemblies once the joints
					-- break) physics velocity that fights the tween. Zero it every frame.
					part.AssemblyLinearVelocity = Vector3.zero
					part.AssemblyAngularVelocity = Vector3.zero
				end)
			end
		end

		-- Use the air controller so the ground controller can't keep us grounded (lets us sink
		-- through the floor while tweening).
		local controllerManager = character:FindFirstChild("ControllerManager")
		local airController = controllerManager and controllerManager:FindFirstChild("AirController")
		if controllerManager and airController then
			pcall(function()
				controllerManager.ActiveController = airController
			end)
		end
	end)
end

local function restoreNoClip()
	if noclipConnection then
		pcall(function()
			noclipConnection:Disconnect()
		end)
		noclipConnection = nil
	end
end

---Set only the "Current Action:" label text, WITHOUT touching the timeout. Used by the tween loop,
---which updates the distance every frame but keeps its own long timeout budget.
---@param text string?
local function setActionText(text)
	if not actionLabel then
		return
	end

	if text and text ~= "" then
		actionLabel.Text = TEXT.actionPrefix .. text
	else
		actionLabel.Text = TEXT.actionPrefixEmpty
	end
end

---Arm the timeout to `seconds` from now (nil clears it).
---@param seconds number?
local function armTimeout(seconds)
	timeoutDeadline = seconds and (os.clock() + seconds) or nil
end

---Tween to a world position, updating the "Current Action" line with the live distance, until we
---arrive or the automation stops.
---@param goalPosition Vector3
---@param destName string? Name shown in the action line (defaults to "Ferryman Island").
local function tweenTo(goalPosition, destName)
	destName = destName or TEXT.destIsland
	Tweening.goal(TWEEN_IDENTIFIER, CFrame.new(goalPosition), false)

	-- Long timeout for the whole fly-up; the label updates every frame but must not re-arm it.
	armTimeout(TWEEN_TIMEOUT)

	while AutoFerryman.running do
		local character = localPlayer and localPlayer.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if root then
			-- Horizontal distance only -- the vertical leg snaps, so the Y gap isn't meaningful.
			local dx = goalPosition.X - root.Position.X
			local dz = goalPosition.Z - root.Position.Z
			local distance = math.sqrt(dx * dx + dz * dz)
			setActionText(string.format(TEXT.tweening, destName, distance))
			if distance <= 1.0 then
				break
			end
		end
		runService.Heartbeat:Wait()
	end
end

---Watchdog: Auto Log (always on) hops when a player comes within range, and the timeout hops if an
---action has been stuck past its budget.
local function watchForNearbyPlayers()
	while AutoFerryman.running do
		task.wait(AUTO_LOG_WATCH_INTERVAL)

		if not AutoFerryman.running or hopping then
			break
		end

		-- Ended up in the Depths mid-run -- stop the automation and leave.
		if game.PlaceId == DEPTHS_PLACE_ID then
			AutoFerryman.stop()
			pcall(function()
				localPlayer:Kick(TEXT.kickDepths)
			end)
			return
		end

		-- On a different slot than the one we started on -- stop.
		local slot = currentSlot()
		if startSlot ~= nil and slot ~= nil and slot ~= startSlot then
			AutoFerryman.setAction(TEXT.wrongSlot)
			Logger.notify(TEXT.wrongSlotNotify)
			AutoFerryman.stop()
			return
		end

		-- Timeout: assume a stuck edge case and hop.
		if timeoutDeadline and os.clock() >= timeoutDeadline then
			AutoFerryman.hop(TEXT.reasonTimeout)
			return
		end

		-- A moderator is in the server (mod notifications logic) -- hop away.
		local hasMods = false
		pcall(function()
			hasMods = PlayerScanning.hasModerators()
		end)
		if hasMods then
			AutoFerryman.hop(TEXT.reasonModerator)
			return
		end

		-- Hold off auto-log briefly after a descent -- players loading in around the freshly-streamed
		-- island shouldn't trigger an instant log.
		if lastDescentClock and (os.clock() - lastDescentClock) < DESCENT_LOG_GRACE then
			continue
		end

		local character = localPlayer and localPlayer.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not root then
			continue
		end

		local autoLogDist = AutoLog.get("auto_ferryman")
		if autoLogDist > 0 and Finder.pnear(root.Position, autoLogDist) then
			AutoFerryman.hop(TEXT.reasonPlayerNearby)
			return
		end
	end
end

---Sum a tool's "Quantity" IntValue across the backpack AND the character (a stack that's currently
---equipped lives under the Character, not the Backpack -- missing that was why the count sometimes
---read low). Returns 0 if the tool (or the value) isn't found.
---@param toolName string
---@return number
local function getToolQuantity(toolName)
	local total = 0

	local containers = {
		localPlayer and localPlayer:FindFirstChild("Backpack"),
		localPlayer and localPlayer.Character,
	}

	for _, container in next, containers do
		if container then
			for _, tool in next, container:GetChildren() do
				if tool:IsA("Tool") and tool.Name == toolName then
					local quantity = tool:FindFirstChild("Quantity")
					if quantity and quantity:IsA("IntValue") then
						total = total + quantity.Value
					end
				end
			end
		end
	end

	return total
end

---Count an item across the backpack AND character. Uses the "Quantity" IntValue when present,
---otherwise counts one per matching tool (some items -- e.g. Flint, Coal -- may not carry a stack
---value). Returns 0 if none are found.
---@param name string
---@return number
local function getItemCount(name)
	local total = 0

	local containers = {
		localPlayer and localPlayer:FindFirstChild("Backpack"),
		localPlayer and localPlayer.Character,
	}

	for _, container in next, containers do
		if container then
			for _, tool in next, container:GetChildren() do
				if tool:IsA("Tool") and tool.Name == name then
					local quantity = tool:FindFirstChild("Quantity")
					if quantity and quantity:IsA("IntValue") then
						total = total + quantity.Value
					else
						total = total + 1
					end
				end
			end
		end
	end

	return total
end

---Total remaining flint "Uses" across the backpack AND character, counting both Flint and Umbral
---Flint (each carries a "Uses" IntValue). One use lights one campfire. Returns 0 if there's none.
---@return number
local function flintUses()
	local total = 0

	local containers = {
		localPlayer and localPlayer:FindFirstChild("Backpack"),
		localPlayer and localPlayer.Character,
	}

	for _, container in next, containers do
		if container then
			for _, tool in next, container:GetChildren() do
				if tool:IsA("Tool") and (tool.Name == "Flint" or tool.Name == "Umbral Flint") then
					local uses = tool:FindFirstChild("Uses")
					if uses and uses:IsA("IntValue") then
						total = total + uses.Value
					end
				end
			end
		end
	end

	return total
end

---Find an instance named "Choice" of the given class. The craft menu's per-session remotes are
---nil-parented (their DebugIds change every session, so we match by name + class), but fall back to
---a full game scan in case this executor parents them somewhere real.
---@param className string
---@param timeout number?
---@return Instance?
local function findChoiceRemote(className, timeout)
	local deadline = os.clock() + (timeout or 5)

	local function scan()
		-- Nil-parented instances (the documented location).
		if getnilinstances then
			local ok, instances = pcall(getnilinstances)
			if ok and type(instances) == "table" then
				for _, object in next, instances do
					local matches = false
					pcall(function()
						matches = object.Name == "Choice" and object:IsA(className)
					end)
					if matches then
						return object
					end
				end
			end
		end

		-- Fallback: anywhere under game.
		local ok, found = pcall(function()
			for _, object in next, game:GetDescendants() do
				if object.Name == "Choice" and object:IsA(className) then
					return object
				end
			end
		end)
		if ok and found then
			return found
		end

		return nil
	end

	repeat
		local found = scan()
		if found then
			return found
		end
		task.wait()
	until os.clock() >= deadline

	return nil
end

---Craft `count` of a recipe, mimicking the game's crafting exactly (CraftScript + ChoiceScript):
--- 1) `Requests.Craft:InvokeServer(recipeTable, product)` -- recipeTable is the ingredient set
---    ({ [ingredient] = true, ... }) and product is the recipe name. For stackable recipes this opens the
---    quantity-slider ChoicePrompt (and yields until it's answered, so we fire it on its own thread).
--- 2) submit the amount like the slider's Submit button: `Choice:InvokeServer(count)` (a RemoteFunction).
---    Recipes that craft directly (no slider) never spawn a Choice, so that step is skipped after a short
---    wait.
---@param materials table Ingredient set, e.g. { Wood = true } or { Coal = true, Rock = true }.
---@param recipeName string The product to craft, e.g. "Campfire Pit" or "Flint".
---@param count number
---@return boolean
local function craftBatch(materials, recipeName, count)
	if count <= 0 then
		return true
	end

	local requests = replicatedStorage:FindFirstChild("Requests")
	local craftRemote = requests and requests:FindFirstChild("Craft")
	if not craftRemote then
		Logger.warn(TEXT.craftRemoteMissing)
		return false
	end

	-- 1) Fire the craft (opens the quantity slider for stackable recipes). Spawned because it yields
	-- until the quantity Choice is answered.
	task.spawn(function()
		pcall(function()
			craftRemote:InvokeServer(materials, recipeName)
		end)
	end)

	-- 2) Submit the quantity via the ChoicePrompt's Choice RemoteFunction (the slider's Submit), asking
	-- for the full count in one go. If no slider appears (direct craft), skip.
	local quantityChoice = findChoiceRemote("RemoteFunction", 5)
	if quantityChoice then
		task.spawn(function()
			pcall(function()
				quantityChoice:InvokeServer(count)
			end)
		end)
	end

	return true
end

---Craft `amount` Campfire Pits from Wood (1:1). Sticks are not used.
---@param amount number
---@return boolean crafted Whether crafting ran.
local function craftCampfires(amount)
	return craftBatch({ Wood = true }, "Campfire Pit", amount)
end

---How many campfires the current Wood stock can make (1 per Wood). Sticks are not used.
---@return number
local function craftableCampfires()
	return math.floor(getToolQuantity("Wood") / WOOD_PER_CAMPFIRE)
end

---Nearest live tree model in workspace.Destructibles (a Model with "Tree" in its name), restricted
---to trees within TREE_ISLAND_RANGE of the island so we never wander off for a far-off tree.
---@return Model?
local function findTree()
	local destructibles = workspace:FindFirstChild("Destructibles")
	if not destructibles then
		return nil
	end

	local character = localPlayer and localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local origin = (root and root.Position) or FERRYMAN_TARGET

	local best, bestDistance = nil, math.huge

	for _, model in next, destructibles:GetChildren() do
		if model:IsA("Model") and model.Parent and model.Name:find("Tree") then
			local ok, pivot = pcall(function()
				return model:GetPivot().Position
			end)
			if ok then
				-- Must be within range of the island, else skip it entirely.
				if (pivot - FERRYMAN_TARGET).Magnitude <= TREE_ISLAND_RANGE then
					local distance = (pivot - origin).Magnitude
					if distance < bestDistance then
						best = model
						bestDistance = distance
					end
				end
			end
		end
	end

	return best
end

---Equip the Lumber Axe. If one is already on the character it's reused; otherwise it's pulled from
---the backpack. If the player has no Lumber Axe at all, leave the game (kick).
---@return Tool?
local function equipLumberAxe()
	local character = localPlayer and localPlayer.Character
	local humanoid = character and character:FindFirstChildOfClass("Humanoid")
	if not character or not humanoid then
		return nil
	end

	local equipped = character:FindFirstChild("Lumber Axe")
	if equipped and equipped:IsA("Tool") then
		return equipped
	end

	local backpack = localPlayer:FindFirstChild("Backpack")
	local tool = backpack and backpack:FindFirstChild("Lumber Axe")
	if not (tool and tool:IsA("Tool")) then
		Logger.notify(TEXT.noLumberAxeNotify)
		AutoFerryman.stop()
		pcall(function()
			localPlayer:Kick(TEXT.kickNoLumberAxe)
		end)
		return nil
	end

	pcall(function()
		humanoid:EquipTool(tool)
	end)

	return character:FindFirstChild("Lumber Axe")
end

---Nearest live "Campfire" in Destructibles within CAMPFIRE_BREAK_RANGE of the island spot we tweened
---down to. Campfires can be a Model or a bare MeshPart, so match either.
---@return Instance?
local function findNearbyCampfire()
	local destructibles = workspace:FindFirstChild("Destructibles")
	if not destructibles then
		return nil
	end

	local reference = Vector3.new(FERRYMAN_TARGET.X, FERRYMAN_DESCENT_Y, FERRYMAN_TARGET.Z)
	local best, bestDistance = nil, math.huge

	for _, object in next, destructibles:GetChildren() do
		if (object:IsA("Model") or object:IsA("BasePart")) and object.Parent and object.Name:find("Campfire") then
			local ok, pivot = pcall(function()
				return object:GetPivot().Position
			end)
			if ok then
				local distance = (pivot - reference).Magnitude
				if distance <= CAMPFIRE_BREAK_RANGE and distance < bestDistance then
					best = object
					bestDistance = distance
				end
			end
		end
	end

	return best
end

---Equip the axe, tween to a destructible (Model or MeshPart), and keep swinging until it is deleted.
---@param model Instance
---@return boolean destroyed True once it is gone; false if kicked (no axe) or stopped.
local function chopModel(model)
	if not equipLumberAxe() then
		return false -- kicked (no axe) or no character
	end

	local position = model:GetPivot().Position

	while AutoFerryman.running and model.Parent do
		-- Hold position at the model.
		Tweening.goal(CHOP_TWEEN_IDENTIFIER, CFrame.new(position), false)

		-- Keep the axe equipped (a respawn/unequip would otherwise silently stop the chopping).
		local axe = equipLumberAxe()
		if not axe then
			Tweening.stop(CHOP_TWEEN_IDENTIFIER)
			return false
		end

		-- Swing.
		pcall(function()
			axe:Activate()
		end)

		task.wait(CHOP_SWING_INTERVAL)
	end

	Tweening.stop(CHOP_TWEEN_IDENTIFIER)

	return model.Parent == nil
end

---Chop the nearest tree down for wood.
---@return boolean destroyed True once a tree was chopped down; false if none/kicked/stopped.
local function chopTree()
	-- Wait for a tree within range to appear (they respawn). If none shows within 60s this server is
	-- barren -- hop to a fresh one.
	local tree = findTree()
	local deadline = os.clock() + 60
	while not tree and AutoFerryman.running do
		if os.clock() >= deadline then
			AutoFerryman.hop(TEXT.reasonNoTrees)
			return false
		end
		AutoFerryman.setAction(TEXT.waitingForTree)
		task.wait(1)
		tree = findTree()
	end

	if not AutoFerryman.running or not tree then
		return false
	end

	AutoFerryman.setAction(TEXT.choppingTree)
	return chopModel(tree)
end

---Equip a flint to light campfires -- prefer an Umbral Flint if we have one, else a normal Flint.

	local character = localPlayer and localPlayer.Character

---Tween to a CFrame (position + facing) and hold there until we arrive or stop.
---@param cframe CFrame
local function moveTo(cframe)
	Tweening.goal(TWEEN_IDENTIFIER, cframe, false)

	while AutoFerryman.running do
		local character = localPlayer and localPlayer.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if root then
			local dx = cframe.Position.X - root.Position.X
			local dz = cframe.Position.Z - root.Position.Z
			if math.sqrt(dx * dx + dz * dz) <= 2 then
				break
			end
		end
		runService.Heartbeat:Wait()
	end

	Tweening.stop(TWEEN_IDENTIFIER)
end

---Tween to each placement spot and place a campfire there.
local function placeCampfires()
	for index, cframe in ipairs(PLACE_LOCATIONS) do
		if not AutoFerryman.running then
			return
		end

		AutoFerryman.setAction(string.format(TEXT.placingCampfire, index, #PLACE_LOCATIONS))

		moveTo(cframe)
		if not AutoFerryman.running then
			return
		end

		-- Nail the exact position + facing before placing.
		local character = localPlayer and localPlayer.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if root then
			root.CFrame = cframe
		end

		local tool = equipCampfirePit()
		if tool then
			task.wait(0.3)
			pcall(function()
				tool:Activate()
			end)
		end

		task.wait(1)

		-- Light the campfire: equip Flint, then click it. Kicks if the player has none.
		local flint = equipFlint()
		if not flint then
			return -- kicked (no flint) or stopped
		end

		task.wait(0.3)
		pcall(function()
			flint:Activate()
		end)

		-- Extra second after using the flint before moving to the next campfire.
		task.wait(1)
	end
end

---Get to REQUIRED_CAMPFIRES: chop trees for wood as needed, then craft once. Crafting is a single
---attempt -- if it doesn't produce campfires, we stop retrying and let the action timeout hop us.
---@return boolean ok
local function acquireCampfires()
	while AutoFerryman.running do
		AutoFerryman.setAction(TEXT.checkingCampfires)

		local campfires = getToolQuantity("Campfire Pit")
		if campfires >= REQUIRED_CAMPFIRES then
			return true
		end

		local needed = REQUIRED_CAMPFIRES - campfires

		if craftableCampfires() >= needed then
			-- Enough materials on hand: craft the shortfall once, then wait for the count to rise.
			AutoFerryman.setAction(string.format(TEXT.craftingCampfires, needed))
			craftCampfires(needed)

			local before = campfires
			local deadline = os.clock() + 12
			repeat
				task.wait(0.5)
			until (not AutoFerryman.running)
				or getToolQuantity("Campfire Pit") > before
				or os.clock() >= deadline

			if not AutoFerryman.running then
				return false
			end

			if getToolQuantity("Campfire Pit") <= before then
				-- One craft attempt produced nothing -- hop to a fresh server right away.
				AutoFerryman.setAction(TEXT.campfireCraftFailed)
				AutoFerryman.hop(TEXT.reasonCraftFailed)
				return false
			end
			-- Progress -- loop and re-check.
		else
			-- Short on materials: chop a tree for wood, then re-check. Give the drop a moment to
			-- replicate so the next material count includes it.
			if not chopTree() then
				return false -- no trees left, no axe (kicked), or stopped
			end
			task.wait(1)
		end
	end

	return false
end

---Nearest mesh named `meshName` in workspace.Ingredients (searched deep), by distance from the HRP,
---restricted to meshes within MINER_RANGE of Miner's Landing.
---@param meshName string
---@return BasePart?
local function findNearestIngredient(meshName)
	local ingredients = workspace:FindFirstChild(INGREDIENTS_FOLDER)
	if not ingredients then
		return nil
	end

	local character = localPlayer and localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	local origin = (root and root.Position) or MINER_TARGET

	local best, bestDistance = nil, math.huge

	for _, object in next, ingredients:GetDescendants() do
		if object:IsA("BasePart") and object.Name == meshName and object.Parent then
			-- Must be within range of Miner's Landing, else skip it entirely.
			if (object.Position - MINER_TARGET).Magnitude <= MINER_RANGE then
				local distance = (object.Position - origin).Magnitude
				if distance < bestDistance then
					best = object
					bestDistance = distance
				end
			end
		end
	end

	return best
end

---Mine one `itemName` off the nearest `meshName` in Ingredients: tween to it, keep the Pickaxe
---equipped, and fire its InteractPrompt until our item count goes up.
---@param meshName string
---@param itemName string
---@return boolean ok True once we've gained one; false if none found / stuck / stopped.
local function mineIngredient(meshName, itemName)
	local before = getItemCount(itemName)

	while AutoFerryman.running do
		if getItemCount(itemName) > before then
			return true
		end

		-- Wait for an in-range mesh to appear (they respawn). If none shows within 60s this server is
		-- barren -- hop to a fresh one.
		local mesh = findNearestIngredient(meshName)
		local deadline = os.clock() + 60
		while not mesh and AutoFerryman.running do
			if os.clock() >= deadline then
				AutoFerryman.hop(string.format(TEXT.reasonNoMesh, meshName))
				return false
			end
			AutoFerryman.setAction(string.format(TEXT.waitingForMesh, meshName))
			task.wait(1)
			mesh = findNearestIngredient(meshName)
		end
		if not AutoFerryman.running or not mesh then
			return false
		end

		AutoFerryman.setAction(string.format(TEXT.miningMesh, meshName))
		moveTo(CFrame.new(mesh.Position))
		if not AutoFerryman.running then
			return false
		end

		-- Keep the pickaxe equipped and fire the mesh's proximity prompt.
		equipPickaxe()
		Prompt.interact(mesh:FindFirstChild(INTERACT_PROMPT_NAME))

		task.wait(1)
	end

	return false
end

---Craft one flint (a single attempt). If we have Umbral Obsidian, craft an Umbral Flint (1 Umbral
---Obsidian + 1 Coal); otherwise a normal Flint (Coal + Rock). If it produces nothing we stop retrying
---and let the action timeout hop us.
---@return boolean ok
local function craftFlint()
	if not AutoFerryman.running then
		return false
	end

	local umbral = getItemCount("Umbral Obsidian") >= 1

	AutoFerryman.setAction(umbral and TEXT.craftingUmbralFlint or TEXT.craftingFlint)

	local before = flintUses()
	if umbral then
		craftBatch({ ["Umbral Obsidian"] = true, Coal = true }, "Umbral Flint", 1)
	else
		craftBatch({ Coal = true, Rock = true }, "Flint", 1)
	end

	local deadline = os.clock() + 12
	repeat
		task.wait(0.5)
	until (not AutoFerryman.running) or flintUses() > before or os.clock() >= deadline

	if not AutoFerryman.running then
		return false
	end

	if flintUses() > before then
		return true
	end

	-- One craft attempt produced nothing -- hop to a fresh server right away.
	AutoFerryman.setAction(TEXT.flintCraftFailed)
	AutoFerryman.hop(TEXT.reasonFlintCraftFailed)
	return false
end

---Out of flint: make sure we have 1 Coal + 1 Rock (mining them at Miner's Landing only if we don't
---already have them), then craft a Flint. Kicks only if we actually need to mine but have no Pickaxe.
---@return boolean ok True once a flint has been crafted; false if kicked / hopped / stopped.
local function mineForFlint()
	-- With Umbral Obsidian we craft Umbral Flint (Obsidian + Coal), so Rock isn't needed.
	local umbral = getItemCount("Umbral Obsidian") >= 1
	local needCoal = getItemCount("Coal") < 1
	local needRock = (not umbral) and getItemCount("Rock") < 1

	-- Only make the mining trip if we're missing a material -- otherwise craft with what we have.
	if needCoal or needRock then
		-- Pickaxe gate (only needed to mine).
		local character = localPlayer and localPlayer.Character
		local backpack = localPlayer and localPlayer:FindFirstChild("Backpack")
		local hasPickaxe = (character and character:FindFirstChild("Pickaxe"))
			or (backpack and backpack:FindFirstChild("Pickaxe"))
		if not hasPickaxe then
			Logger.notify(TEXT.noPickaxeNotify)
			AutoFerryman.stop()
			pcall(function()
				localPlayer:Kick(TEXT.kickNoPickaxe)
			end)
			return false
		end

		-- Fly to Miner's Landing (skip the fly-up if we're already within range horizontally).
		local startRoot = localPlayer and localPlayer.Character
			and localPlayer.Character:FindFirstChild("HumanoidRootPart")
		local alreadyNear = false
		if startRoot then
			local dx = MINER_TARGET.X - startRoot.Position.X
			local dz = MINER_TARGET.Z - startRoot.Position.Z
			alreadyNear = math.sqrt(dx * dx + dz * dz) <= ISLAND_NEAR_RANGE
		end

		if not alreadyNear then
			flying = true
			tweenTo(Vector3.new(MINER_TARGET.X, FERRYMAN_TWEEN_Y, MINER_TARGET.Z), "Miners Landing")
		end
		if not AutoFerryman.running then
			return false
		end

		-- Instant descend to the ground.
		Tweening.stop(TWEEN_IDENTIFIER)
		local root = localPlayer and localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
		if root then
			root.CFrame = CFrame.new(MINER_TARGET)
		end
		flying = false
		lastDescentClock = os.clock()

		if not AutoFerryman.running then
			return false
		end

		equipPickaxe()

		-- Mine only what we're missing.
		if needCoal and not mineIngredient("Coal", "Coal") then
			return false
		end
		if needRock and not mineIngredient("Rock", "Rock") then
			return false
		end
	end

	-- Craft the flint.
	return craftFlint()
end

---The (early) ferryman routine: fly up, descend onto the island, then make sure we have campfires.
local function runRoutine()
	-- Flint check FIRST (before flying to the island): one Use per campfire. If we're short, go mine
	-- coal + rock and craft flint at Miner's Landing, then rerun the whole loop.
	if flintUses() < REQUIRED_CAMPFIRES then
		if mineForFlint() then
			return runRoutine() -- got flint; restart the whole loop (tail call, no stack growth)
		end
		return -- kicked (no pickaxe), hopped (craft/mining failed), or stopped
	end

	-- Skip the fly-up entirely if we're already within range of the island (horizontally).
	local startCharacter = localPlayer and localPlayer.Character
	local startRoot = startCharacter and startCharacter:FindFirstChild("HumanoidRootPart")
	local alreadyNear = false
	if startRoot then
		local dx = FERRYMAN_TARGET.X - startRoot.Position.X
		local dz = FERRYMAN_TARGET.Z - startRoot.Position.Z
		alreadyNear = math.sqrt(dx * dx + dz * dz) <= ISLAND_NEAR_RANGE
	end

	if not alreadyNear then
		flying = true
		tweenTo(Vector3.new(FERRYMAN_TARGET.X, FERRYMAN_TWEEN_Y, FERRYMAN_TARGET.Z))
	end

	if not AutoFerryman.running then
		return
	end

	-- Instant descend onto the island.
	Tweening.stop(TWEEN_IDENTIFIER)

	local character = localPlayer and localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if root then
		root.CFrame = CFrame.new(FERRYMAN_TARGET.X, FERRYMAN_DESCENT_Y, FERRYMAN_TARGET.Z)
	end
	flying = false
	lastDescentClock = os.clock()

	if not AutoFerryman.running then
		return
	end

	-- Phase: campfire pits -- chop trees for wood as needed, craft, repeat until enough.
	if not acquireCampfires() then
		-- If a hop is in flight (craft failure), leave the resume flag alone; otherwise this is a
		-- terminal failure (e.g. no trees left) -- stop for good.
		if not hopping then
			AutoFerryman.stop()
		end
		return
	end

	if not AutoFerryman.running then
		return
	end

	AutoFerryman.setAction(TEXT.campfiresReady)

	-- Phase: break existing campfires near the island with the Lumber Axe.
	while AutoFerryman.running do
		local campfire = findNearbyCampfire()
		if not campfire then
			break
		end

		AutoFerryman.setAction(TEXT.breakingCampfire)
		if not chopModel(campfire) then
			return -- kicked (no axe) or stopped
		end
	end

	if not AutoFerryman.running then
		return
	end

	-- Phase: place the fresh campfires at the 4 spots (flint was ensured at the top of the loop).
	placeCampfires()

	if not AutoFerryman.running then
		return
	end

	AutoFerryman.setAction(TEXT.campfiresPlaced)
end

---Which attack the boss is currently playing: "triple" (pause our M1 + hold), "javelin" (drop to the
---ground to dodge), or nil. Mirrors PR ferryman's AnimationPlayed handling.
---@param boss Instance
---@return string?
local function bossAnimState(boss)
	local animator = boss:FindFirstChildWhichIsA("Animator", true)
	if not animator then
		return nil
	end

	local ok, tracks = pcall(function()
		return animator:GetPlayingAnimationTracks()
	end)
	if not ok or not tracks then
		return nil
	end

	for _, track in next, tracks do
		local anim = track.Animation
		local id = anim and tostring(anim.AnimationId)
		if id then
			if id:find(BOSS_JAVELIN_ANIM_ID, 1, true) then
				return "javelin"
			end

			for _, tripleId in next, BOSS_TRIPLE_ANIM_IDS do
				if id:find(tripleId, 1, true) then
					return "triple"
				end
			end
		end
	end

	return nil
end

---Whether the weapon is currently drawn (the "Equipped" effect is present).
---@return boolean
local function isWeaponDrawn()
	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local module = effectReplicator and require(effectReplicator)
	if not module then
		return false
	end
	local ok, has = pcall(function()
		return module:HasEffect("Equipped")
	end)
	return ok and has == true
end

---Draw the "Weapon" tool via the DrawWeapon remote (instead of EquipTool). Returns the weapon tool if
---one exists (character or backpack), nil otherwise.
---@return Tool?
local function equipWeapon()
	local character = localPlayer and localPlayer.Character
	if not character then
		return nil
	end

	-- The Weapon tool, wherever it lives.
	local tool = character:FindFirstChild(WEAPON_NAME)
	if not (tool and tool:IsA("Tool")) then
		local backpack = localPlayer:FindFirstChild("Backpack")
		tool = backpack and backpack:FindFirstChild(WEAPON_NAME)
	end
	if not (tool and tool:IsA("Tool")) then
		return nil
	end

	-- Draw it via the remote, only if it isn't already out.
	if not isWeaponDrawn() then
		local handler = character:FindFirstChild("CharacterHandler")
		local requests = handler and handler:FindFirstChild("Requests")
		local drawEvent = requests and requests:FindFirstChild("DrawWeapon")
		if drawEvent then
			pcall(function()
				drawEvent:FireServer(true)
			end)
		end
	end

	return tool
end

---Turn on the combat features we want for the boss fight: Auto Defense, Block Input (PVE, blocking

		end
		if Options.AutoFeintOwnTags then
			Options.AutoFeintOwnTags:SetValue({ M1s = true, Mantras = true })
		end
		if Options.AutoFeintAgainstTags then
			Options.AutoFeintAgainstTags:SetValue({ M1s = true, Mantras = true, Criticals = true })
		end
	end

	if Toggles then
		if Toggles.EnableAutoDefense then
			Toggles.EnableAutoDefense:SetValue(true)
		end
		if Toggles.BlockInput then
			Toggles.BlockInput:SetValue(true)
		end
		if Toggles.AutoFeint then
			Toggles.AutoFeint:SetValue(true)
		end

			Toggles.Fling:SetValue(false)
		end
	end

---Name-matched loot: fire the ChoicePrompt's Choice remote for every option button whose title matches
---one of `names` (case-insensitive substring). Mirrors the automation folder's loot_from_prompt.
---@param choicePrompt Instance
---@param names table
---@return boolean selected
local function lootMatchingItems(choicePrompt, names)
	local frame = choicePrompt:FindFirstChild("ChoiceFrame")
	local options = frame and frame:FindFirstChild("Options")
	local choice = choicePrompt:FindFirstChild("Choice")
	if not options or not choice then
		return false
	end

	local selected = false
	for _, button in next, options:GetChildren() do
		if not button:IsA("TextButton") then
			continue
		end
		local title = button:FindFirstChild("Title")
		local text = title and title.Text
		if not text then
			continue
		end
		local lower = text:lower()
		for _, name in next, names do
			if type(name) == "string" and lower:find(name:lower(), 1, true) then
				pcall(function()
					choice:FireServer(button.Name)
				end)
				selected = true
				task.wait(0.1)
				break
			end
		end
	end
	return selected
end

---All chests near us: workspace.Thrown descendants named "Lid" (BaseParts) -> their parent model, within
---`maxDistance`. Direct port of PR ferryman's find_chests.
---@param maxDistance number?
---@return table
local function findChests(maxDistance)
	local chests = {}
	local thrown = workspace:FindFirstChild(THROWN_FOLDER)
	if not thrown then
		return chests
	end
	local character = localPlayer and localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	for _, object in next, thrown:GetDescendants() do
		if object.Name == "Lid" and object:IsA("BasePart") then
			if not maxDistance or not root or (root.Position - object.Position).Magnitude <= maxDistance then
				table.insert(chests, object.Parent)
			end
		end
	end
	return chests
end

---After the boss dies, loot the chests -- direct port of PR ferryman's grab_chests: tween onto each lid,
---spam its proximity prompt until the ChoicePrompt opens, name-match loot the wanted items, wait for the
---prompt to close, handle a Craft prompt (-> Campfire Pit), and tag the chest looted.
local function lootChests()
	AutoFerryman.setAction(TEXT.lootingChests)

	-- PR ferryman takes Umbral Obsidian + Coal; merge in the user's configured items too.
	local wanted = { ALWAYS_LOOT, "Coal" }
	for _, name in next, (Configuration.expectOptionValues("ItemNameList") or {}) do
		wanted[#wanted + 1] = name
	end

	local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
	local function choicePrompt()
		return playerGui and playerGui:FindFirstChild("ChoicePrompt")
	end

	-- Wait for the chests to spawn (up to 30s).
	local spawnDeadline = os.clock() + 30
	repeat
		AutoFerryman.setAction(TEXT.waitingForChests)
		task.wait(0.5)
	until #findChests(2000) > 0 or os.clock() >= spawnDeadline or not AutoFerryman.running

	for _, chest in next, findChests(2000) do
		if not AutoFerryman.running then
			break
		end
		if not collectionService:HasTag(chest, LOOTED_TAG) and chest:FindFirstChild("Lid") then
			AutoFerryman.setAction(TEXT.lootingChest)

			-- Tween onto the lid and spam the chest's proximity prompt until the ChoicePrompt opens.
			pcall(function()
				moveTo(chest.Lid.CFrame)
			end)
			local openDeadline = os.clock() + 3
			repeat
				pcall(function()
					Prompt.interact(chest:FindFirstChildWhichIsA("ProximityPrompt", true))
				end)
				pcall(function()
					moveTo(chest.Lid.CFrame)
				end)
				task.wait(1)
			until choicePrompt() or os.clock() >= openDeadline or not AutoFerryman.running

			-- Name-match the wanted items out of the prompt.
			if choicePrompt() then
				pcall(function()
					lootMatchingItems(choicePrompt(), wanted)
				end)
			end

			-- Wait for the prompt to close (up to 3s).
			local closeDeadline = os.clock() + 3
			repeat
				task.wait(1)
			until not choicePrompt() or os.clock() >= closeDeadline or not AutoFerryman.running

			-- A Craft prompt (campfire pit) can pop -- pick Campfire Pit to clear it (PR).
			local cp = choicePrompt()
			if cp then
				local title = cp:FindFirstChild("Title", true)
				local choice = cp:FindFirstChild("Choice")
				if title and choice and tostring(title.Text):find("Craft") then
					pcall(function()
						choice:FireServer("Campfire Pit")
					end)
				end
			end

			pcall(function()
				collectionService:AddTag(chest, LOOTED_TAG)
			end)
		end
	end
end

---The boss-dungeon routine: tween to the Ferryman, start the interaction, and spam "1" to blow
---through the dialogue until the Ferryman leaves the NPCs folder (the fight is next -- built later).
local function bossRoutine()
	local ferryman = findFerrymanNpc()
	if not ferryman then
		AutoFerryman.setAction(TEXT.ferrymanNotFound)
		AutoFerryman.stop()
		return
	end

	-- Tween to the Ferryman.
	AutoFerryman.setAction(TEXT.tweeningToFerryman)
	local ok, pivot = pcall(function()
		return ferryman:GetPivot().Position
	end)
	if ok then
		moveTo(CFrame.new(pivot))
	end
	if not AutoFerryman.running then
		return
	end

	-- Interact with the Ferryman: fire its InteractPrompt until the dialogue opens, then walk its wager
	-- dialogue tree by firing Requests.SendDialogue with the exact choice strings (deterministic -- picks
	-- name -> wager -> deal, instead of blindly spamming "1"). setAction arms the timeout once; the loop
	-- uses setActionText so the timeout actually counts down, and we hop if it runs out.
	AutoFerryman.setAction(TEXT.talkingToFerryman)

	local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")

	-- Fire the Ferryman's InteractPrompt until the dialogue GUI opens (matches PR: raw fireproximityprompt
	-- against workspace.NPCs['The Ferryman'].InteractPrompt). Timeout/running guards keep it from hanging.
	while AutoFerryman.running do
		if timeoutDeadline and os.clock() >= timeoutDeadline then
			AutoFerryman.hop(TEXT.reasonTimeout)
			return
		end

		local dialogueGui = playerGui and playerGui:FindFirstChild("DialogueGui")
		if dialogueGui and dialogueGui.Enabled then
			break -- dialogue is open, stop prompting
		end

		pcall(function()
			Prompt.interact(workspace.NPCs["The Ferryman"].InteractPrompt)
		end)

		setActionText(TEXT.talkingToFerryman)
		task.wait(0.5)
	end

	if not AutoFerryman.running then
		return
	end

	-- Walk the wager dialogue by firing the choices in order.
	setActionText(TEXT.advancingDialogue)
	local requests = replicatedStorage:FindFirstChild("Requests")
	local sendDialogue = requests and requests:FindFirstChild("SendDialogue")
	for index, choice in ipairs(FERRYMAN_DIALOGUE_CHOICES) do
		if not AutoFerryman.running then
			return
		end
		if sendDialogue then
			pcall(function()
				sendDialogue:FireServer({ choice = choice })
			end)
		end
		if index < #FERRYMAN_DIALOGUE_CHOICES then
			task.wait(0.5)
		end
	end

	AutoFerryman.setAction(TEXT.dialogueComplete)

	-- Wait for the boss to spawn as ".ferryman1" in workspace.Live.
	AutoFerryman.setAction(TEXT.waitingForBoss)
	while AutoFerryman.running and not findBossModel() do
		task.wait(0.5)
	end
	if not AutoFerryman.running then
		return
	end

	-- Turn on the combat features for the fight.
	enableBossCombat()

	-- Equip the weapon.
	local weapon = equipWeapon()
	if not weapon then
		AutoFerryman.setAction(TEXT.noWeapon)
		Logger.notify(TEXT.noWeaponNotify)
		AutoFerryman.stop()
		return
	end

	-- Continuously tween under the boss (facing up at it) and swing until it's gone. No fight timeout -- clear
	-- any armed deadline so the fight runs as long as it needs (it only ends when the boss is gone).
	armTimeout(nil)
	local healing = false
	while AutoFerryman.running do
		local boss = findBossModel()
		if not boss then
			break -- boss defeated / despawned (next phase later)
		end

		local root = localPlayer and localPlayer.Character and localPlayer.Character:FindFirstChild("HumanoidRootPart")
		local humanoid = localPlayer and localPlayer.Character and localPlayer.Character:FindFirstChildOfClass("Humanoid")
		local bossHRP = boss:FindFirstChild("HumanoidRootPart")

		if root and humanoid and bossHRP then
			-- Heal hysteresis: once below 35% HP, go heal until back above 85% (PR's heal_cf loop).
			if healing then
				if humanoid.Health >= humanoid.MaxHealth * BOSS_HEAL_HIGH then
					healing = false
				end
			elseif humanoid.Health <= humanoid.MaxHealth * BOSS_HEAL_LOW then
				healing = true
			end

			local animState = bossAnimState(boss)

			if healing then
				-- Low HP: stop swinging and sit on the safe spot until we're topped up.
				setActionText(TEXT.fightingFerryman)
				Tweening.goal(CHOP_TWEEN_IDENTIFIER, BOSS_HEAL_CFRAME, false)
			elseif animState == "javelin" then
				-- Javelin throw: drop straight down to the ground to dodge it (PR drops to Y 0).
				Tweening.goal(CHOP_TWEEN_IDENTIFIER, CFrame.new(root.Position.X, 0, root.Position.Z), false)
			elseif animState == "triple" then
				-- Triple slash: hold still and stop attacking, let Auto Defense handle it.
				Tweening.stop(CHOP_TWEEN_IDENTIFIER)
			else
				-- Sit below + behind the boss in its own frame, facing it, then swing (PR: CFrame.new(0, -7, 4)).
				local targetCFrame = bossHRP.CFrame * CFrame.new(0, -BOSS_UNDER_OFFSET, BOSS_BEHIND_OFFSET)
				Tweening.goal(CHOP_TWEEN_IDENTIFIER, CFrame.lookAt(targetCFrame.Position, bossHRP.Position), false)

				equipWeapon()
				setActionText(TEXT.fightingFerryman)
				local aim = CFrame.new(root.Position, bossHRP.Position)
				pcall(function()
					InputClient.left(aim, true)
				end)
			end
		end

		task.wait(CHOP_SWING_INTERVAL)
	end

	Tweening.stop(CHOP_TWEEN_IDENTIFIER)

	if not AutoFerryman.running then
		return
	end

	-- Ferryman killed -- count the cycle and persist it.
	cycles = cycles + 1
	saveState()
	AutoFerryman.setAction(TEXT.bossDefeated)

	-- Loot the chests that spawn after the kill.
	lootChests()

	if not AutoFerryman.running then
		return
	end

	AutoFerryman.setAction(TEXT.chestsLooted)

	-- Make sure the chest's ChoicePrompt is fully closed before we hop (fire EXIT if it's still open,
	-- then wait for it to actually close, up to 3s).
	do
		local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
		local cp = playerGui and playerGui:FindFirstChild("ChoicePrompt")
		if cp then
			local choice = cp:FindFirstChild("Choice")
			if choice then
				pcall(function()
					choice:FireServer("EXIT")
				end)
			end
			local closeDeadline = os.clock() + 3
			repeat
				task.wait(0.1)
			until not (playerGui and playerGui:FindFirstChild("ChoicePrompt")) or os.clock() >= closeDeadline
		end
	end

	-- Restart the whole process on a fresh server (carries cycles + elapsed over).
	AutoFerryman.hop(TEXT.reasonCycleComplete)
end

---Update the "Current Action:" line. Each call counts as progress, so it re-arms the default
---timeout.
---@param text string?
function AutoFerryman.setAction(text)
	setActionText(text)
	armTimeout(DEFAULT_TIMEOUT)
end

---Is a tool present on the character or in the backpack?
---@param name string
---@return boolean
local function toolPresent(name)
	local character = localPlayer and localPlayer.Character
	local backpack = localPlayer and localPlayer:FindFirstChild("Backpack")
	if character and character:FindFirstChild(name) then
		return true
	end
	if backpack and backpack:FindFirstChild(name) then
		return true
	end
	return false
end

---Are we loaded in enough to start: character exists and the backpack contents have replicated.
---@return boolean
local function requirementsMet()
	local character = localPlayer and localPlayer.Character
	if not character then
		return false
	end
	local backpack = localPlayer and localPlayer:FindFirstChild("Backpack")
	return backpack ~= nil and #backpack:GetChildren() > 0
end

---Wait for the backpack to exist and have its contents replicated (up to `timeout`s).
---@param timeout number?
---@return boolean loaded
local function waitForBackpack(timeout)
	local deadline = os.clock() + (timeout or 10)
	while os.clock() < deadline do
		local backpack = localPlayer and localPlayer:FindFirstChild("Backpack")
		if backpack and #backpack:GetChildren() > 0 then
			return true
		end
		task.wait(0.1)
	end
	return false
end

---Start the automation (overlay + timer + highlight). The routine itself is not wired up yet.
function AutoFerryman.start()
	if AutoFerryman.running then
		return
	end

	-- Ended up in the Depths -- something went wrong; end the automation and leave immediately.
	if game.PlaceId == DEPTHS_PLACE_ID then
		AutoFerryman.stop()
		pcall(function()
			localPlayer:Kick(TEXT.kickDepths)
		end)
		return
	end

	-- Pick the mode from where we are: the Eastern Luminant runs the gather-to-dungeon routine, the
	-- boss dungeon (Ferryman present) runs the fight routine.
	local mode
	if AutoFerryman.inEasternLuminant() then
		mode = "gather"
	elseif AutoFerryman.inBossDungeon() then
		mode = "boss"
	else
		return Logger.notify(TEXT.notInPlace)
	end

	hopping = false
	flying = false
	timeoutDeadline = nil
	lastDescentClock = nil
	startClock = os.clock()

	-- Carry cycles + elapsed over from before a server hop (nil/fresh start -> 0). Loaded before the
	-- LoadingGui check so a loading-timeout hop preserves the running totals.
	local resumeData = PersistentData.get("afdata") or {}
	cycles = resumeData.cycles or 0
	baseElapsed = resumeData.elapsed or 0

	-- A manual first start (no resume flag yet) vs. an auto-resume after a hop.
	local firstStart = not resumeData.active

	-- Remember the slot we started on (a manual start locks in the current slot; a resume keeps it).
	if firstStart then
		startSlot = currentSlot()
	else
		startSlot = resumeData.slot
	end

	-- If the loading screen (LoadingGui) is up, fire the StartMenu.Start remote to blow through it until
	-- the character + backpack are loaded (10s cap). Handles resuming right after a hop. If we never load
	-- in, hop to a fresh server like the other timeouts.
	local loadingGuiParent = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
	if loadingGuiParent and loadingGuiParent:FindFirstChild("LoadingGui") then
		local deadline = os.clock() + 10
		while not requirementsMet() and os.clock() < deadline do
			General.fireStartMenu()
			task.wait(0.25)
		end

		if not requirementsMet() then
			-- Mark running so hop()'s guard/teardown behave, then hop.
			AutoFerryman.running = true
			return AutoFerryman.hop(TEXT.reasonLoadingTimeout)
		end
	end

	-- Wait for the backpack contents to replicate before we start (and before the tool check).
	waitForBackpack(10)

	-- First (manual) start: require a Pickaxe and Lumber Axe up front, else don't start.
	if firstStart and not (toolPresent("Pickaxe") and toolPresent("Lumber Axe")) then
		return Logger.notify(TEXT.needTools)
	end

	AutoFerryman.running = true

	-- Resume flag: survives server hops (MemStorage), so the farm restarts on the next server.
	saveState()

	buildOverlay()
	AutoFerryman.setAction(nil)
	if timerLabel then
		timerLabel.Text = formatElapsed(baseElapsed)
	end
	if timeoutLabel then
		timeoutLabel.Text = TEXT.timeoutNone
	end
	if cyclesLabel then
		cyclesLabel.Text = string.format(TEXT.cycles, cycles)
	end
	if screenGui then
		screenGui.Enabled = true
	end

	applyHighlight()
	applyTweenSpeed()
	applyNoFallDamage()
	applyFly()
	applyDefenseFilter()
	applyNoClip()

	-- Fast swing: strip attack cooldowns so m1s aren't rate-limited during the Ferryman fight.
	if not fastSwingConn then
		fastSwingConn = General.connectFastSwing()
	end

	-- Re-apply the highlight and re-pin the replication focus if we respawn mid-run.
	characterAddedConnection = localPlayer.CharacterAdded:Connect(function()
		if not AutoFerryman.running then
			return
		end
		task.wait(0.2)
		applyHighlight()
	end)

	-- Tick the elapsed timer + the timeout countdown.
	timerConnection = runService.Heartbeat:Connect(function()
		if not AutoFerryman.running then
			return
		end
		if timerLabel then
			timerLabel.Text = formatElapsed(currentElapsed())
		end
		if cyclesLabel then
			cyclesLabel.Text = string.format(TEXT.cycles, cycles)
		end
		if timeoutLabel then
			if timeoutDeadline then
				local remaining = math.max(0, timeoutDeadline - os.clock())
				timeoutLabel.Text = string.format(TEXT.timeout, math.floor(remaining / 60), math.floor(remaining % 60))
			else
				timeoutLabel.Text = TEXT.timeoutNone
			end
		end
	end)

	-- Keep the weapon drawn during the boss fight (only while the boss is present, so it doesn't fight
	-- the pickaxe/lumber-axe phases): redraw it whenever it's sheathed, like the other farms.
	weaponConnection = runService.Heartbeat:Connect(function()
		if not AutoFerryman.running then
			return
		end
		if not findBossModel() then
			return
		end
		if not isWeaponDrawn() then
			equipWeapon()
		end
	end)

	-- Wait 2s before the first action so the auto-log watcher gets a chance to catch a nearby player
	-- (and hop) before we start flying.
	if mode == "boss" then
		-- Instanced dungeon -- no auto-log / server-hopping, just run the fight routine.
		task.spawn(function()
			task.wait(2)
			if AutoFerryman.running then
				bossRoutine()
			end
		end)
	else
		task.spawn(watchForNearbyPlayers)
		task.spawn(function()
			task.wait(2)
			if AutoFerryman.running then
				runRoutine()
			end
		end)
	end

	Logger.notify(TEXT.started)
end

---Tear down this run (threads, overlay, highlight, tweens, restored toggles) WITHOUT clearing the
---resume flag. Shared by stop() and the server-hop path.
local function teardown()
	AutoFerryman.running = false
	timeoutDeadline = nil

	if fastSwingConn then
		pcall(function()
			fastSwingConn:Disconnect()
		end)
		fastSwingConn = nil
	end

	-- If we were flying up when stopped, drop the character back to a normal Y (onto the island)
	-- instead of leaving them stranded in the sky.
	if flying then
		Tweening.stop(TWEEN_IDENTIFIER)
		local character = localPlayer and localPlayer.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if root then
			pcall(function()
				root.CFrame = CFrame.new(FERRYMAN_TARGET.X, FERRYMAN_DESCENT_Y, FERRYMAN_TARGET.Z)
			end)
		end
	end
	flying = false

	if timerConnection then
		pcall(function()
			timerConnection:Disconnect()
		end)
		timerConnection = nil
	end

	if characterAddedConnection then
		pcall(function()
			characterAddedConnection:Disconnect()
		end)
		characterAddedConnection = nil
	end

	if weaponConnection then
		pcall(function()
			weaponConnection:Disconnect()
		end)
		weaponConnection = nil
	end

	removeHighlight()

	Tweening.stop(TWEEN_IDENTIFIER)
	Tweening.stop(CHOP_TWEEN_IDENTIFIER)
	restoreTweenSpeed()
	restoreNoFallDamage()
	restoreFly()
	restoreDefenseFilter()
	restoreNoClip()

	if screenGui then
		screenGui.Enabled = false
	end
end

---Blacklist the current server and hop to a fresh one, keeping the resume flag set so Auto Ferryman
---restarts automatically after the teleport. Used by auto-log and craft-failure.
---@param reason string
function AutoFerryman.hop(reason)
	if hopping then
		return
	end
	hopping = true

	Logger.notify(TEXT.hopNotify, reason)

	-- Blacklist the current server (by JobId) so the hopper won't rejoin it, matching ServerHop's
	-- own blacklist shape (id -> tick()).
	local blacklist = PersistentData.get("sblacklist") or {}
	blacklist[game.JobId] = tick()
	PersistentData.set("sblacklist", blacklist)

	-- Keep the resume flag set (with cycles + elapsed) so the fresh server restarts and carries the
	-- running totals over.
	saveState()

	-- Stop this run but keep afdata, then hop (ServerHop returns to the menu, which rejoins a new
	-- server on the same slot).
	teardown()

	-- Use the slot we captured at start (persisted in afdata) rather than re-reading DataSlot now --
	-- a nil read here would set shslot = nil and the menu resume would never rejoin a server (looks
	-- like the automation just stopped).
	local slot = startSlot or (localPlayer and localPlayer:GetAttribute("DataSlot"))
	ServerHop.hop(slot, false)
end

---Stop the automation for good (manual): tears down and clears the resume flag.
function AutoFerryman.stop()
	if not AutoFerryman.running and not PersistentData.get("afdata") then
		return
	end

	teardown()

	-- Manual stop: clear the resume flag so a re-execute doesn't restart the farm.
	PersistentData.set("afdata", nil)

	Logger.notify(TEXT.stopped)
end

return AutoFerryman
end)

local function fetchName(player)
	local spoofName = Configuration.expectToggleValue("InfoSpoofing")
		and Configuration.expectToggleValue("SpoofOtherPlayers")

	return spoofName and "[REDACTED]"
		or string.format("(%s) %s", player:GetAttribute("CharacterName") or "Unknown Character Name", player.Name)

---@param list table<string, any>
---@param value string
---@return string?
local partialStringFind = LPH_NO_VIRTUALIZE(function(list, value)
	for _, str in next, list do
		if not value:match(str) then
			continue
		end

		return str
	end

	return nil
end)

local function checkInventoriesForTools()
	for _, player in next, players:GetPlayers() do
		if not Configuration.expectToggleValue("NotifyItems") then
			continue
		end

		if player == players.LocalPlayer then
			continue
		end

		local backpack = player:FindFirstChild("Backpack")
		if not backpack then
			continue
		end

		local notifyItemsList = Options.NotifyItemsList and Options.NotifyItemsList.Values
		if not notifyItemsList then
			continue
		end

		for _, tool in next, backpack:GetChildren() do
			if seenTools[tool] then
				continue
			end

			local itemName = tool:GetAttribute("ItemName")
			if typeof(itemName) ~= "string" or itemName == "" then
				continue
			end

			local matchedString = partialStringFind(notifyItemsList, itemName)
			if not matchedString then
				continue
			end

			seenTools[tool] = matchedString

			Logger.longNotify("%s has item '%s' in their inventory.", fetchName(player), itemName)
		end

		for tool, matched in next, seenTools do
			if table.find(notifyItemsList, matched) then
				continue
			end

			seenTools[tool] = nil
		end
	end

	lastCheckedTimestamp = os.clock()
end

local runPlayerScans = LPH_NO_VIRTUALIZE(function()
	local localPlayer = players.LocalPlayer
	if not localPlayer then
		return
	end

	for player, _ in next, PlayerScanning.scanQueue do
		if shared.Luma.dpscanning then
			continue
		end

		if not PlayerScanning.scanDataCache[player] then
			local handledSuccess, handledResult = nil, nil

			local unhandledSuccess, unhandledResult = pcall(function()
				handledSuccess, handledResult = PlayerScanning.getStaffRank(player)
			end)

			if not unhandledSuccess then
				Logger.warn(
					"Scan player %s ran into error '%s' while getting staff rank.",
					player.Name,
					unhandledResult
				)

				Logger.mnnotify("Failed to scan player %s for moderator status.", fetchName(player), unhandledResult)

				PlayerScanning.scanQueue[player] = nil

				continue
			end

			if not handledSuccess then
				continue
			end

			if Configuration.expectToggleValue("NotifyMod") and handledResult then
				Logger.mnnotify("%s is a staff member with the rank '%s' in group.", fetchName(player), handledResult)

				if Configuration.expectToggleValue("NotifyModSound") then
					moderatorSound.SoundId = "rbxassetid://6045346303"
					moderatorSound.PlaybackSpeed = 1
					moderatorSound.Volume = Configuration.expectToggleValue("NotifyModSoundVolume") or 10
					moderatorSound:Play()
				end
			end

			PlayerScanning.scanDataCache[player] = { staffRank = handledResult }
		end

		local backpack = player:FindFirstChild("Backpack")
		if not backpack then
			return
		end

		if not collectionService:HasTag(backpack, "Loaded") or #backpack:GetChildren() < 1 then
			if not PlayerScanning.waitingForLoad[player] then
				Logger.warn("Player scanning is waiting for %s to load in the game.", fetchName(player))
			end

			PlayerScanning.waitingForLoad[player] = true

			continue
		end

		if
			Configuration.expectToggleValue("NotifyVoidWalker")
			and backpack:FindFirstChild("Talent:Voidwalker Contract")
		then
			Logger.longNotify("%s has the Voidwalker Contract talent.", fetchName(player))
		end

		PlayerScanning.scanQueue[player] = nil

		PlayerScanning.friendCache[player] = localPlayer:GetFriendStatus(player) == Enum.FriendStatus.Friend

		Logger.warn("Player scanning finished scanning %s in queue.", fetchName(player))
	end
end)

---@return table

	for _, scanData in next, PlayerScanning.scanDataCache do
		if not scanData.staffRank then

	end

	return false
end

---@param player Player
---@return boolean
function PlayerScanning.isAlly(player)
	---@note: bruh we can call ReputationSystem
	local localPlayerGuild = players.LocalPlayer:GetAttribute("Guild")
	local usernameList = Options["UsernameList"]

	if usernameList then
		local displayNameFound = player and table.find(usernameList.Values, player.DisplayName)
		local usernameFound = player and table.find(usernameList.Values, player.Name)

		if displayNameFound or usernameFound then
			return true
		end
	end

	return PlayerScanning.friendCache[player]
		or ((localPlayerGuild and #localPlayerGuild >= 1) and player:GetAttribute("Guild") == localPlayerGuild)
end

---@param player Player
---@return boolean, string?
function PlayerScanning.getStaffRank(player)
	local mapped = ModeratorMap[player.UserId]
	if mapped then
		return true, mapped
	end

	local rankSuccess, rank = pcall(player.GetRankInGroup, player, STAFF_GROUP_ID)
	if not rankSuccess then
		return false, rank
	end

	if typeof(rank) ~= "number" or rank <= 0 then
		return true, nil
	end

	local roleSuccess, role = pcall(player.GetRoleInGroup, player, STAFF_GROUP_ID)
	if not roleSuccess or typeof(role) ~= "string" then
		return true, "Staff"
	end

	return true, role
end

---@note: Request will yield - so we need a debounce to prevent multiple scan loops.
---@note: We must defer the error back to the caller and reset the scanning debounce so errors will not break the scanning loop.
function PlayerScanning.update()
	if os.clock() - lastCheckedTimestamp >= 5.0 then
		checkInventoriesForTools()
	end

	if os.clock() - lastPlayerScanTimestamp <= 1.0 then
		return
	end

	lastPlayerScanTimestamp = os.clock()

	if PlayerScanning.scanning then
		return
	end

	PlayerScanning.scanning = true

	local success, result = pcall(runPlayerScans)

	PlayerScanning.scanning = false

	if success then
		return
	end

	return error(result)
end

---@param player Player
---@param status Enum.FriendStatus
function PlayerScanning.friend(player, status)
	PlayerScanning.friendCache[player] = status == Enum.FriendStatus.Friend
end

---@param player Player
function PlayerScanning.onPlayerAdded(player)
	if player == players.LocalPlayer then
		return
	end

	PlayerScanning.scanQueue[player] = true
end

---@param player Player
function PlayerScanning.onPlayerRemoving(player)
	PlayerScanning.scanQueue[player] = nil
	PlayerScanning.scanDataCache[player] = nil
	PlayerScanning.friendCache[player] = nil
	PlayerScanning.waitingForLoad[player] = nil
end

function PlayerScanning.init()
	local playerAddedSignal = Signal.new(players.PlayerAdded)
	local playerRemovingSignal = Signal.new(players.PlayerRemoving)
	local renderSteppedSignal = Signal.new(runService.RenderStepped)
	local friendStatusChanged = Signal.new(players.LocalPlayer.FriendStatusChanged)

	playerScanningMaid:add(friendStatusChanged:connect("PlayerScanning_OnFriendStatusChanged", PlayerScanning.friend))
	playerScanningMaid:add(renderSteppedSignal:connect("PlayerScanning_Update", PlayerScanning.update))
	playerScanningMaid:add(playerAddedSignal:connect("PlayerScanning_OnPlayerAdded", PlayerScanning.onPlayerAdded))
	playerScanningMaid:add(
		playerRemovingSignal:connect("PlayerScanning_OnPlayerRemoving", PlayerScanning.onPlayerRemoving)
	)

	for _, player in next, players:GetPlayers() do
		PlayerScanning.onPlayerAdded(player)
	end
end

function PlayerScanning.detach()
	playerScanningMaid:clean()
end

return PlayerScanning
end)

	if lootQueue[1] ~= currentChoice then
		error("We assumed that the current choice was the first in the queue, was the queue modified?")
	end

	table.remove(lootQueue, 1)

	currentChoice = nil
end

local function resetAutoLoot()
	choiceRemote = nil
	lootQueue = {}
	currentChoice = nil
	currentPrompt = nil
	lootAll = false
end

---@param descendant Instance
local function onPlayerGuiDescendantAdded(descendant)
	if descendant.Name ~= "ChoicePrompt" then
		return
	end

	if AutoLoot.ignore then
		return
	end

	if not Configuration.expectToggleValue("AutoLoot") then
		return
	end

	descendant:WaitForChild("ChoiceFrame"):WaitForChild("Options")

	AutoLoot.process(
		descendant,
		AutoLootOptions.new(
			Configuration.expectOptionValue("AutoLootStarsMin") or 0,
			Configuration.expectOptionValue("AutoLootStarsMax") or 0,
			Configuration.expectOptionValues("ItemNameList") or {},
			Configuration.expectToggleValue("AutoLootAll") or false
		)
	)
end

---@param prompt ScreenGui
---@param remote RemoteEvent
---@param param string
local function waitForPromptDeletion(prompt, remote, param)
	repeat
		remote:FireServer(param)

		task.wait()
	until not prompt.Parent

	resetAutoLoot()
end

---@return boolean
function AutoLoot.active()
	return choiceRemote ~= nil and currentPrompt ~= nil and #lootQueue > 0
end

function AutoLoot.update()
	if os.clock() - lastTimestamp <= 0.15 then
		return
	end

	lastTimestamp = os.clock()

	if not AutoLoot.ignore and not Configuration.expectToggleValue("AutoLoot") then
		return resetAutoLoot()
	end

	if not choiceRemote or not choiceRemote.Parent then
		return resetAutoLoot()
	end

	if not currentPrompt or not currentPrompt.Parent then
		return resetAutoLoot()
	end

	local choiceFrame = currentPrompt:FindFirstChild("ChoiceFrame")
	local options = choiceFrame and choiceFrame:FindFirstChild("Options")
	if not options then
		return resetAutoLoot()
	end

	if lootAll then
		return waitForPromptDeletion(currentPrompt, choiceRemote, "LOOT_ALL")
	end

	if currentChoice then
		return waitForChoiceDeletion(options, choiceRemote)
	end

	local itemToLoot = lootQueue[1]
	if not itemToLoot then
		return waitForPromptDeletion(currentPrompt, choiceRemote, "EXIT")
	end

	currentChoice = itemToLoot

	choiceRemote:FireServer(itemToLoot)
end

---@param prompt ScreenGui The choice prompt to process.
---@param options AutoLootOptions The options to use.
function AutoLoot.process(prompt, options)
	local remote = prompt:FindFirstChild("Choice")
	if not remote or not remote:IsA("RemoteEvent") then
		return
	end

	local jsonLootData = prompt:GetAttribute("LootData")
	if typeof(jsonLootData) ~= "string" then
		return
	end

	local lootData = JSON.decode(jsonLootData)
	if typeof(lootData) ~= "table" then
		return
	end

	resetAutoLoot()

	currentPrompt = prompt
	choiceRemote = remote
	lootAll = options.lall

	for _, item in next, lootData do
		local rich = item.rich:gsub("%$%S*", "")
		local text = item.text

		if not text then
			continue
		end

		-- Always-loot list: matched items are queued regardless of the star filter.
		local alwaysWanted, _ = Table.find(options.always or {}, function(value, _)
			return rich:match(value)
		end)
		if alwaysWanted then
			lootQueue[#lootQueue + 1] = text
			continue
		end

		local matched = text:match("★")
		local stars = matched and #matched or 0

		if stars < options.smin or stars > options.smax then
			continue
		end

		local wanted, _ = Table.find(options.wanted, function(value, _)
			return rich:match(value)
		end)

		if not wanted then
			continue
		end

		lootQueue[#lootQueue + 1] = text
	end
end

function AutoLoot.init()
	local localPlayer = players.LocalPlayer
	local playerGui = localPlayer:WaitForChild("PlayerGui")

	local preRender = Signal.new(runService.PreRender)
	local playerGuiDescendantAdded = Signal.new(playerGui.DescendantAdded)

	autoLootMaid:mark(playerGuiDescendantAdded:connect("AutoLoot_PlayerGuiDescendantAdded", onPlayerGuiDescendantAdded))
	autoLootMaid:mark(preRender:connect("AutoLoot_PreRender", AutoLoot.update))
end

function AutoLoot.detach()
	autoLootMaid:clean()
end

return AutoLoot
end)

---@param start number
---@param stop number
---@return any[]

	local out = {}

	if start == nil then
		start = 1
	elseif start < 0 then
		start = #input + start + 1
	end
	if stop == nil then
		stop = #input
	elseif stop < 0 then
		stop = #input + stop + 1
	end

	for idx = start, stop do
		table.insert(out, input[idx])
	end

	return out
end)

---@param input any[]
---@param predicate fun(value: any, index: number): boolean
---@return number, any

	for idx, value in next, input do
		if not predicate(value, idx) then
			continue
		end

		return idx, value
	end
end)

---Auto Ferryman's custom noclip: every frame drop CanCollide on all character parts, force the
---AirController so the ground can't hold us, and zero velocity so physics can't fight teleports.

		return
	end

	noclipConnection = runService.Stepped:Connect(function()
		if not AutoVoI.running then
			return
		end

		local character = localPlayer and localPlayer.Character
		if not character then
			return
		end

		for _, part in next, character:GetDescendants() do
			if part:IsA("BasePart") then
				pcall(function()
					if part.CanCollide then
						part.CanCollide = false
					end
					part.AssemblyLinearVelocity = Vector3.zero
					part.AssemblyAngularVelocity = Vector3.zero
				end)
			end
		end

		local controllerManager = character:FindFirstChild("ControllerManager")
		local airController = controllerManager and controllerManager:FindFirstChild("AirController")
		if controllerManager and airController then
			pcall(function()
				controllerManager.ActiveController = airController
			end)
		end
	end)
end

		return animator:GetPlayingAnimationTracks()

---The first live entity in workspace.Live whose name contains `match`, or nil.
---@param match string
---@return Instance?
local function findLiveEnemyMatching(match)
	local live = workspace:FindFirstChild("Live")
	if not live then
		return nil
	end

	local myCharacter = localPlayer and localPlayer.Character

	for _, entity in next, live:GetChildren() do
		if entity == myCharacter then
			continue
		end
		if not string.find(string.lower(entity.Name), match, 1, true) then
			continue
		end

		local humanoid = entity:FindFirstChildWhichIsA("Humanoid")
		local root = entity:FindFirstChild("HumanoidRootPart")
		if humanoid and root and humanoid.Health > 0 then
			return entity
		end
	end

	return nil
end

---Find the "Choice" RemoteEvent. It lives in PlayerGui (a sibling of the ChoiceFrame prompt), so we
---search there; getnilinstances is a fallback.
---@return Instance?
local function getChoiceRemote()
	-- PlayerGui (where the choice prompt UI + its Choice remote live).
	local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
	if playerGui then
		for _, object in ipairs(playerGui:GetDescendants()) do
			if object.Name == CHOICE_REMOTE_NAME and object:IsA("RemoteEvent") then
				return object
			end
		end
	end

	-- Fallback: parentless instances.
	if getnilinstances then
		local ok, nils = pcall(getnilinstances)
		if ok and nils then
			for _, object in next, nils do
				local nameOk = pcall(function()
					return object.Name
				end)
				if nameOk and object.Name == CHOICE_REMOTE_NAME and object:IsA("RemoteEvent") then
					return object
				end
			end
		end
	end

	return nil
end

---Teleport to a CFrame and wiggle around it a bit so the server registers the touch/arrival, then
---settle back on it.
---@param cframe CFrame
local function jitterTeleport(cframe)
	local character = localPlayer and localPlayer.Character
	if not character then
		return
	end

	pcall(function()
		character:PivotTo(cframe)
	end)

	for _ = 1, 6 do
		if not AutoVoI.running then
			break
		end
		local offset = Vector3.new(math.random(-2, 2), math.random(-1, 1), math.random(-2, 2))
		pcall(function()
			character:PivotTo(cframe + offset)
		end)
		task.wait(0.1)
	end

	pcall(function()
		character:PivotTo(cframe)
	end)
end

---Teleport the character onto workspace.NexusExit (1 stud above), wiggling so the touch registers.
---@return boolean
local function teleportToNexusExit()
	local nexusExit = workspace:FindFirstChild("NexusExit")
	if not nexusExit then
		return false
	end

	local ok, pivot = pcall(function()
		return nexusExit:GetPivot()
	end)
	if not ok then
		return false
	end

	jitterTeleport(pivot + Vector3.new(0, 1, 0))
	return true
end

---Tween (fly) to a world position, Auto-Ferryman style (Y snaps to the goal Y, moves horizontally).
---Blocks until we arrive horizontally or the automation stops.
---@param goalPos Vector3
local function tweenTo(goalPos)
	-- Force a fast tween speed for the crossing, restoring the user's afterwards.
	local prevSpeed
	if Options and Options.TweenStudsPerSecond then
		prevSpeed = Options.TweenStudsPerSecond.Value
		setTweenSpeed(TWEEN_SPEED)
	end

	Tweening.goal(TWEEN_ID, CFrame.new(goalPos), false)

	while AutoVoI.running do
		local _, _, hrp = characterParts()
		if hrp then
			local dx = goalPos.X - hrp.Position.X
			local dz = goalPos.Z - hrp.Position.Z
			setAction(TEXT.tweening) -- re-arms the timeout while we travel
			if math.sqrt(dx * dx + dz * dz) <= ARRIVE_DISTANCE then
				break
			end
		end
		task.wait()
	end

	Tweening.stop(TWEEN_ID)

	if prevSpeed ~= nil and Options and Options.TweenStudsPerSecond then
		Options.TweenStudsPerSecond:SetValue(prevSpeed)
	end
end

---A part/model's CFrame.
---@param part Instance
---@return CFrame
local function partCFrame(part)
	return part:IsA("BasePart") and part.CFrame or part:GetPivot()
end

---Whether we've arrived: within `radius` of `pos` (nil pos = never "arrived").
---@param pos Vector3?
---@param radius number?
---@return boolean
local function arrivedNear(pos, radius)
	if not pos then
		return false
	end
	local _, _, root = characterParts()
	return root ~= nil and (root.Position - pos).Magnitude <= (radius or WITHIN_DISTANCE)
end

---Continuously teleport onto a part (by name). When the part is present we jitter-teleport onto it;
---when it isn't we hold at `holdPos` (to stream it in). Exits when we arrive near `arrivePos`, the
---automation stops, or -- for non-persistent teleports -- the part despawns (we moved on).
---
---`persistent` = true means it NEVER stops on the part despawning; use it for boundaries that change
---the place id (the place change / VM reset ends the loop).
---@param name string
---@param waitText string?
---@param teleportText string?
---@param holdPos Vector3?
---@param arrivePos Vector3?
---@param arriveRadius number?
---@param persistent boolean?
local function teleportToPartContinuously(name, waitText, teleportText, holdPos, arrivePos, arriveRadius, persistent)
	local seen = false

	while AutoVoI.running do
		if arrivedNear(arrivePos, arriveRadius) then
			break
		end

		local part = workspace:FindFirstChild(name, true)
		if part then
			seen = true
			if teleportText then
				setAction(teleportText)
			end
			jitterTeleport(partCFrame(part))
		else
			-- Part not present. If we'd already used it and this isn't a persistent teleport, we've
			-- moved on. Otherwise keep waiting (holding at holdPos to stream it in).
			if seen and not persistent then
				break
			end
			if waitText then
				setAction(waitText)
			end
			if holdPos then
				local character = localPlayer and localPlayer.Character
				if character then
					pcall(function()
						character:PivotTo(CFrame.new(holdPos))
					end)
				end
			end
		end

		task.wait(0.3)
	end
end

---The player's current Notes, read from the currency UI (PlayerGui.CurrencyGui.CurrencyFrame.Notes),
---or nil if it can't be read.
---@return number?
local function getNotes()
	local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
	local currencyGui = playerGui and playerGui:FindFirstChild("CurrencyGui")
	local currencyFrame = currencyGui and currencyGui:FindFirstChild("CurrencyFrame")
	local notesLabel = currencyFrame and currencyFrame:FindFirstChild("Notes")
	local amount = notesLabel and notesLabel:FindFirstChild("Amount")
	if not amount then
		return nil
	end

	-- Strip commas / non-digits from the displayed amount.
	local digits = tostring(amount.Text or ""):gsub("[^%d]", "")
	if digits == "" then
		return nil
	end
	return tonumber(digits)
end

---Generic boss fight (same logic as Auto Ferryman): Pathfind Breaker + Aggressive on, keep the weapon
---drawn, stand in front of any Live entity matching `match` and M1 it. Returns true once the boss was
---present and is gone (defeated), false if the automation stopped first.
---In-front melee boss fight (same logic as Auto Ferryman): Pathfind Breaker + Aggressive on, keep the
---weapon drawn, stand in front of any Live entity matching `match` and M1 it. Returns true once the
---boss was present and is gone (defeated), false if the automation stopped first.
---@param match string
---@param waitText string
---@param fightText string
---@return boolean
local function bossFight(match, waitText, fightText)
	-- Remember Pathfind Breaker + Aggressive Mode prior states so we can restore them after the fight.
	if Toggles and Toggles.AggressiveMode and previousAggressive == nil then
		previousAggressive = Toggles.AggressiveMode.Value
	end
	if Toggles and Toggles.NoFallDamage and previousNoFall == nil then
		previousNoFall = Toggles.NoFallDamage.Value
	end

	applyFly()
	applyNoClip()

	local bossWasPresent = false
	local dodgingAbove = false

	while AutoVoI.running do
		if Toggles and Toggles.AggressiveMode then
			Toggles.AggressiveMode:SetValue(true)
		end
		if Toggles and Toggles.NoFallDamage then
			Toggles.NoFallDamage:SetValue(true)
		end
		if Toggles and Toggles.Fly then
			Toggles.Fly:SetValue(true)
		end

		if not isWeaponDrawn() then
			drawWeapon()
		end

		local enemy = findLiveEnemyMatching(match)
		if not enemy then
			if bossWasPresent then
				break
			end
			setAction(waitText)
			Tweening.stop(FIGHT_TWEEN_ID)
			task.wait(0.3)
		else
			bossWasPresent = true

			-- Pin the boss's velocity so our fast tween into it can't fling it around.
			for _, part in next, enemy:GetDescendants() do
				if part:IsA("BasePart") then
					pcall(function()
						part.AssemblyLinearVelocity = Vector3.zero
						part.AssemblyAngularVelocity = Vector3.zero
					end)
				end
			end

			local ok, enemyCFrame = pcall(function()
				return enemy:GetPivot()
			end)
			if ok then
				local enemyPos = enemyCFrame.Position
				local character = localPlayer and localPlayer.Character
				local _, _, hrp = characterParts()

				if isPlayingActionAnim(enemy) then
					-- Shogun is doing an Action-priority move: snap 1000 studs directly above it (keeping our
					-- X/Z) and don't m1 until it's done.
					dodgingAbove = true
					Tweening.stop(FIGHT_TWEEN_ID)
					if character and hrp then
						pcall(function()
							character:PivotTo(
								CFrame.new(hrp.Position.X, enemyPos.Y + SHOGUN_DODGE_ABOVE, hrp.Position.Z)
							)
						end)
					end
				else
					-- Animation ended: snap back down to the fight spot, then resume normal following.
					if dodgingAbove then
						dodgingAbove = false
						local snapBack = (enemyCFrame * CFrame.new(0, 0, -FIGHT_FRONT_OFFSET)).Position
						if character then
							pcall(function()
								character:PivotTo(CFrame.lookAt(snapBack, enemyPos))
							end)
						end
					end

					setAction(fightText)
					setTweenSpeed(FIGHT_TWEEN_SPEED)

					local frontPos = (enemyCFrame * CFrame.new(0, 0, -FIGHT_FRONT_OFFSET)).Position
					Tweening.goal(FIGHT_TWEEN_ID, CFrame.lookAt(frontPos, enemyPos), false)

					local aim = hrp and CFrame.new(hrp.Position, enemyPos) or CFrame.new(enemyPos)
					pcall(function()
						InputClient.left(aim, true)
					end)
				end
			end
			task.wait(0.3)
		end
	end

	Tweening.stop(FIGHT_TWEEN_ID)
	return AutoVoI.running and bossWasPresent
end

---Rat King fight: normal Pathfind Breaker on, stay 5 studs behind + 3 above it holding m1 (Auto Defense
---+ Block Input + Auto Feint). Follows at 150 speed; if it blinks >500 studs away, crawl at 10 instead
---of flinging across the map. Returns true once it's present-then-gone (killed).
---@param match string
---@param waitText string
---@param fightText string
---@return boolean
local function ratKingFight(match, waitText, fightText)
	-- Remember toggles we force so we can restore them.
	if Toggles and Toggles.NoFallDamage and previousNoFall == nil then
		previousNoFall = Toggles.NoFallDamage.Value
	end
	if Toggles and Toggles.EnableAutoDefense and previousAutoDefense == nil then
		previousAutoDefense = Toggles.EnableAutoDefense.Value
	end
	if Toggles and Toggles.BlockInput and previousBlockInput == nil then
		previousBlockInput = Toggles.BlockInput.Value
	end
	if Toggles and Toggles.AutoFeint and previousAutoFeint == nil then
		previousAutoFeint = Toggles.AutoFeint.Value
	end
	if Toggles and Toggles.Fly and previousFly == nil then
		previousFly = Toggles.Fly.Value
	end

	-- Custom noclip on (CanCollide off + AirController + zero our velocity). Fly off -- movement is via
	-- tweening.
	applyNoClip()

	local bossWasPresent = false
	local dodgingAbove = false

	while AutoVoI.running do
		-- Just normal Pathfind Breaker + the combat toggles.
		if Toggles and Toggles.NoFallDamage then
			Toggles.NoFallDamage:SetValue(true)
		end
		if Toggles and Toggles.Fly then
			Toggles.Fly:SetValue(false)
		end
		if Toggles and Toggles.EnableAutoDefense then
			Toggles.EnableAutoDefense:SetValue(true)
		end
		if Toggles and Toggles.BlockInput then
			Toggles.BlockInput:SetValue(true)
		end
		if Toggles and Toggles.AutoFeint then
			Toggles.AutoFeint:SetValue(true)
		end

		if not isWeaponDrawn() then
			drawWeapon()
		end

		local _, _, localRoot = characterParts()
		if not localRoot then
			task.wait()
			continue
		end

		local enemy = findLiveEnemyMatching(match)
		if not enemy then
			if bossWasPresent then
				break
			end
			setAction(waitText)
			Tweening.stop(FIGHT_TWEEN_ID)
			task.wait(0.3)
		else
			bossWasPresent = true

			-- Anti body-part velocity: zero the boss's velocity so our tween near it can't fling it.
			for _, part in next, enemy:GetDescendants() do
				if part:IsA("BasePart") then
					pcall(function()
						part.AssemblyLinearVelocity = Vector3.zero
						part.AssemblyAngularVelocity = Vector3.zero
					end)
				end
			end

			local targetRoot = enemy:FindFirstChild("HumanoidRootPart") or enemy.PrimaryPart
			if targetRoot then
				local character = localPlayer and localPlayer.Character

				if isPlayingActionAnim(enemy) then
					-- Rat king is doing an Action-priority move: snap 500 studs directly above it (keeping
					-- our X/Z) and don't m1 until it's done.
					dodgingAbove = true
					Tweening.stop(FIGHT_TWEEN_ID)
					if character then
						pcall(function()
							character:PivotTo(
								CFrame.new(localRoot.Position.X, targetRoot.Position.Y + RATKING_DODGE_ABOVE, localRoot.Position.Z)
							)
						end)
					end
				else
					-- Animation ended: snap back down to the fight spot, then resume normal following.
					if dodgingAbove then
						dodgingAbove = false
						local snapBack = (targetRoot.CFrame * CFrame.new(0, RATKING_ABOVE, RATKING_BEHIND)).Position
						if character then
							pcall(function()
								character:PivotTo(CFrame.lookAt(snapBack, targetRoot.Position))
							end)
						end
					end

					-- Normal 150 speed; crawl at 10 if it's blinked far away (so we don't fling ourselves).
					local dist = (localRoot.Position - targetRoot.Position).Magnitude
					setTweenSpeed(dist > RATKING_FAR_DISTANCE and RATKING_FAR_SPEED or RATKING_TWEEN_SPEED)

					-- Stay 5 studs behind it and 3 studs above, facing it.
					local behindPos = (targetRoot.CFrame * CFrame.new(0, RATKING_ABOVE, RATKING_BEHIND)).Position
					Tweening.goal(FIGHT_TWEEN_ID, CFrame.lookAt(behindPos, targetRoot.Position), false)

					setAction(fightText)

					-- Spam m1 like Auto Ferryman (respect block input so blocks/punishes still land).
					if not Defense.shouldBlockInput("M1s") then
						local aim = CFrame.new(localRoot.Position, targetRoot.Position)
						pcall(function()
							InputClient.left(aim, true)
						end)
					end
				end
			end
			task.wait()
		end
	end

	Tweening.stop(FIGHT_TWEEN_ID)
	return AutoVoI.running and bossWasPresent
end

---Pick Enmity's fight targets: the boss (Live entity whose name contains ENMITY_ENEMY_MATCH) and the
---nearest "enforcer" (any OTHER live, alive, non-self entity in workspace.Live). Enforcers are killed
---first, so the caller fights the enforcer when one exists and only falls back to the boss once they're
---all cleared.
---@return Instance? boss, Instance? nearestEnforcer
local function findEnmityTargets()
	local live = workspace:FindFirstChild("Live")
	if not live then
		return nil, nil
	end

	local myCharacter = localPlayer and localPlayer.Character
	local _, _, localRoot = characterParts()

	local boss = nil
	local nearestEnforcer, nearestDist = nil, math.huge

	for _, entity in next, live:GetChildren() do
		if entity == myCharacter then
			continue
		end

		local humanoid = entity:FindFirstChildWhichIsA("Humanoid")
		local root = entity:FindFirstChild("HumanoidRootPart") or entity.PrimaryPart
		if not (humanoid and root and humanoid.Health > 0) then
			continue
		end

		if string.find(string.lower(entity.Name), ENMITY_ENEMY_MATCH, 1, true) then
			boss = entity
		else
			-- An enforcer / other enemy: track the nearest one so we clear it first.
			local dist = localRoot and (localRoot.Position - root.Position).Magnitude or 0
			if dist < nearestDist then
				nearestDist = dist
				nearestEnforcer = entity
			end
		end
	end

	return boss, nearestEnforcer
end

---Enmity fight (arena-only). Normal Pathfind Breaker on, custom noclip, always stay ENMITY_BEHIND studs
---behind + ENMITY_ABOVE up the target holding m1 (no dodge). ALWAYS spams m1 (no block-input gate), and
---prioritizes killing Enmity's enforcers (any other Live enemy) before the boss. Completion is tracked on
---the boss specifically, so enforcers spawning/dying never ends the run. Returns true once the boss was
---present and is gone (killed).
---@param waitText string
---@param fightText string
---@return boolean
local function enmityFight(waitText, fightText)
	-- Remember toggles we force so we can restore them.
	if Toggles and Toggles.NoFallDamage and previousNoFall == nil then
		previousNoFall = Toggles.NoFallDamage.Value
	end
	if Toggles and Toggles.EnableAutoDefense and previousAutoDefense == nil then
		previousAutoDefense = Toggles.EnableAutoDefense.Value
	end
	if Toggles and Toggles.BlockInput and previousBlockInput == nil then
		previousBlockInput = Toggles.BlockInput.Value
	end
	if Toggles and Toggles.AutoFeint and previousAutoFeint == nil then
		previousAutoFeint = Toggles.AutoFeint.Value
	end
	if Toggles and Toggles.Fly and previousFly == nil then
		previousFly = Toggles.Fly.Value
	end

	applyNoClip()

	local bossWasPresent = false

	while AutoVoI.running do
		if Toggles and Toggles.NoFallDamage then
			Toggles.NoFallDamage:SetValue(true)
		end
		if Toggles and Toggles.Fly then
			Toggles.Fly:SetValue(false)
		end
		if Toggles and Toggles.EnableAutoDefense then
			Toggles.EnableAutoDefense:SetValue(true)
		end
		if Toggles and Toggles.BlockInput then
			Toggles.BlockInput:SetValue(true)
		end
		if Toggles and Toggles.AutoFeint then
			Toggles.AutoFeint:SetValue(true)
		end

		if not isWeaponDrawn() then
			drawWeapon()
		end

		local _, _, localRoot = characterParts()
		if not localRoot then
			task.wait()
			continue
		end

		local boss, enforcer = findEnmityTargets()
		if boss then
			bossWasPresent = true
		end

		-- Enforcers first; only fight the boss once they're all cleared.
		local enemy = enforcer or boss
		if not enemy then
			-- No boss and no enforcers left: if the boss had been here, it's dead -> done.
			if bossWasPresent then
				break
			end
			setAction(waitText)
			Tweening.stop(FIGHT_TWEEN_ID)
			task.wait(0.3)
		else
			-- Anti body-part velocity: zero the target's velocity so our tween near it can't fling it.
			for _, part in next, enemy:GetDescendants() do
				if part:IsA("BasePart") then
					pcall(function()
						part.AssemblyLinearVelocity = Vector3.zero
						part.AssemblyAngularVelocity = Vector3.zero
					end)
				end
			end

			local targetRoot = enemy:FindFirstChild("HumanoidRootPart") or enemy.PrimaryPart
			if targetRoot then
				-- Follow speed; crawl if it's blinked far away so we don't fling ourselves.
				local dist = (localRoot.Position - targetRoot.Position).Magnitude
				setTweenSpeed(dist > RATKING_FAR_DISTANCE and RATKING_FAR_SPEED or RATKING_TWEEN_SPEED)

				-- Stay behind + above the target, facing it: enforcers 3/3, the boss 15/10.
				local behindOff = enforcer and ENMITY_ENFORCER_BEHIND or ENMITY_BEHIND
				local aboveOff = enforcer and ENMITY_ENFORCER_ABOVE or ENMITY_ABOVE
				local behindPos = (targetRoot.CFrame * CFrame.new(0, aboveOff, behindOff)).Position
				Tweening.goal(FIGHT_TWEEN_ID, CFrame.lookAt(behindPos, targetRoot.Position), false)

				setAction(enforcer and TEXT.fightingEnforcers or fightText)

				-- ALWAYS spam m1 (no block-input gate).
				local aim = CFrame.new(localRoot.Position, targetRoot.Position)
				pcall(function()
					InputClient.left(aim, true)
				end)
			end
			task.wait()
		end
	end

	Tweening.stop(FIGHT_TWEEN_ID)
	return AutoVoI.running and bossWasPresent
end

local function routine()
	-- In the boss arena: fight the run's target, then end the automation.
	if AutoVoI.inArena() then
		if currentTarget == "enmity" then
			enmityFight(TEXT.waitingForEnmity, TEXT.fightingEnmity)
		elseif currentTarget == "ratking" then
			ratKingFight(RATKING_ENEMY_MATCH, TEXT.waitingForRatKing, TEXT.fightingRatKing)
		else
			bossFight(SHOGUN_ENEMY_MATCH, TEXT.waitingForShogun, TEXT.fightingShogun)
		end

		-- Boss defeated (or stopped) -> end the automation.
		setAction(TEXT.done)
		AutoVoI.stop()
		return
	end

	-- Enmity is arena-only; there's no VoI-side path to it, so idle out here (start guards this).
	if currentTarget == "enmity" then
		setAction(TEXT.waitingForEnmity)
		return
	end

	-- In the VoI: teleport to the target boss's boundary (place change into the arena ends this).
	if currentTarget == "ratking" then
		teleportToPartContinuously(
			GOLDENRAT_BOUNDARY_NAME,
			TEXT.waitingGoldenRat,
			TEXT.teleportingGoldenRat,
			GOLDENRAT_WAIT_POS,
			nil,
			nil,
			true -- persistent: never stops (place change ends it)
		)
	else
		setAction(TEXT.teleportingCrossing)
		teleportToPartContinuously(
			SHOGUN_PART_NAME,
			TEXT.waitingShogun,
			TEXT.teleportingShogun,
			Vector3.new(CROSSING_X, DROP_Y, CROSSING_Z),
			nil,
			nil,
			true -- persistent: never stops (place change ends it)
		)
	end
end

local function teardown()
	AutoVoI.running = false

	if timerConnection then
		pcall(function()
			timerConnection:Disconnect()
		end)
		timerConnection = nil
	end
	if characterAddedConnection then
		pcall(function()
			characterAddedConnection:Disconnect()
		end)
		characterAddedConnection = nil
	end

	restoreNoClip()
	removeHighlight()
	holdM1(false)

	timeoutDeadline = nil

	pcall(function()
		Tweening.stop(TWEEN_ID)
		Tweening.stop(FIGHT_TWEEN_ID)
	end)

	-- Restore anything the run forced.
	if previousTweenSpeed ~= nil and Options and Options.TweenStudsPerSecond then
		Options.TweenStudsPerSecond:SetValue(previousTweenSpeed)
	end
	if previousAggressive ~= nil and Toggles and Toggles.AggressiveMode then
		Toggles.AggressiveMode:SetValue(previousAggressive)
	end
	if previousFly ~= nil and Toggles and Toggles.Fly then
		Toggles.Fly:SetValue(previousFly)
	end
	if previousVoidMobs ~= nil and Toggles and Toggles.VoidMobs then
		Toggles.VoidMobs:SetValue(previousVoidMobs)
	end
	if previousNoFall ~= nil and Toggles and Toggles.NoFallDamage then
		Toggles.NoFallDamage:SetValue(previousNoFall)
	end
	if previousAutoDefense ~= nil and Toggles and Toggles.EnableAutoDefense then
		Toggles.EnableAutoDefense:SetValue(previousAutoDefense)
	end
	if previousBlockInput ~= nil and Toggles and Toggles.BlockInput then
		Toggles.BlockInput:SetValue(previousBlockInput)
	end
	if previousAutoFeint ~= nil and Toggles and Toggles.AutoFeint then
		Toggles.AutoFeint:SetValue(previousAutoFeint)
	end
	if previousM1Hold ~= nil and Toggles and Toggles.M1Hold then
		Toggles.M1Hold:SetValue(previousM1Hold)
	end
	previousTweenSpeed = nil
	previousPathfind = nil
	previousAggressive = nil
	previousFly = nil
	previousNoFall = nil
	previousVoidMobs = nil
	previousAutoDefense = nil
	previousBlockInput = nil
	previousAutoFeint = nil
	previousM1Hold = nil

	if screenGui then
		screenGui.Enabled = false
	end
end

---@param target string? "shogun" or "ratking" for a manual start; nil = server-hop resume (read the
---persisted target).
function AutoVoI.start(target)
	if AutoVoI.running then
		return
	end

	if not AutoVoI.inValidPlace() then
		return Logger.notify(TEXT.notInPlace)
	end

	-- Resolve the target: a manual Start passes it; a resume reads the persisted one.
	if target then
		currentTarget = target
	else
		local existing = PersistentData.get("voidata")
		currentTarget = existing and existing.target
	end
	if not currentTarget then
		return
	end

	-- Enmity is an arena-only boss -- it can only be started from inside the boss arena.
	if currentTarget == "enmity" and not AutoVoI.inArena() then
		return Logger.notify(TEXT.notInArena)
	end

	-- If the loading screen (LoadingGui) is up (e.g. we just resumed after a place teleport), spam A to
	-- blow through it until the character + backpack have replicated (10s cap) -- same as Auto Ferryman.
	local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
	if playerGui and playerGui:FindFirstChild("LoadingGui") then
		local deadline = os.clock() + 10
		while not requirementsMet() and os.clock() < deadline do
			pressKey(Enum.KeyCode.A)
			task.wait(0.25)
		end
	end

	startClock = os.clock()
	AutoVoI.running = true

	-- Capture the user's tween speed so we can restore it after the run.
	if Options and Options.TweenStudsPerSecond then
		previousTweenSpeed = Options.TweenStudsPerSecond.Value
	end

	-- Resume flag: survives server hops (MemStorage).
	saveState()

	buildOverlay()
	setAction(nil)
	if timerLabel then
		timerLabel.Text = formatClock(TEXT.elapsed, 0)
	end
	if timeoutLabel then
		timeoutLabel.Text = TEXT.timeoutNone
	end
	if screenGui then
		screenGui.Enabled = true
	end

	-- Fly + custom noclip for the run.
	applyFly()
	applyNoClip()

	-- Highlight the character, and re-apply it (+ noclip via its connection) if we respawn mid-run.
	applyHighlight()
	characterAddedConnection = localPlayer.CharacterAdded:Connect(function()
		if not AutoVoI.running then
			return
		end
		task.wait(0.2)
		applyHighlight()
		applyFly()
	end)

	-- Tick the elapsed timer + the timeout countdown (and enforce the timeout).
	timerConnection = runService.Heartbeat:Connect(function()
		if not AutoVoI.running then
			return
		end

		if timerLabel then
			timerLabel.Text = formatClock(TEXT.elapsed, os.clock() - startClock)
		end

		if timeoutLabel then
			if timeoutDeadline then
				timeoutLabel.Text = formatClock(TEXT.timeout, math.max(0, timeoutDeadline - os.clock()))
			else
				timeoutLabel.Text = TEXT.timeoutNone
			end
		end

		if timeoutDeadline and os.clock() >= timeoutDeadline then
			Logger.notify(TEXT.timedOut)
			AutoVoI.stop()
		end
	end)

	-- Run the routine off-thread so pathfinding can block.
	task.spawn(routine)
end

function AutoVoI.stop()
	if not AutoVoI.running and not PersistentData.get("voidata") then
		return
	end

	teardown()

	-- Clear the target + resume flag so a re-execute doesn't restart it.
	currentTarget = nil
	PersistentData.set("voidata", nil)
end

function AutoVoI.init()
	if not LPH_OBFUSCATED or (shared.Luma and shared.Luma.verbose) then
		Logger.warn("Auto VoI initialized.")
	end
end

return AutoVoI
end)

---Turn off the "Disable When Window Not Active" Auto Defense filter for the run (so defense/block
---input still work when the Roblox window isn't focused), remembering whether it was on.
local function applyDefenseFilter()
	local option = Options and Options.AutoDefenseFilters
	if not option then
		return
	end

	local current = option.Value or {}
	removedWindowFilter = current[WINDOW_FILTER] == true
	if not removedWindowFilter then
		return
	end

	local newValue = {}
	for key, value in pairs(current) do
		newValue[key] = value
	end
	newValue[WINDOW_FILTER] = nil
	option:SetValue(newValue)
end

local function restoreDefenseFilter()
	local option = Options and Options.AutoDefenseFilters
	if option and removedWindowFilter then
		local newValue = {}
		for key, value in pairs(option.Value or {}) do
			newValue[key] = value
		end
		newValue[WINDOW_FILTER] = true
		option:SetValue(newValue)
	end
	removedWindowFilter = false
end

---The first live entity in workspace.Live that isn't our own character (an enemy), or nil.
---@return Instance?
local function findEnemyInLive()
	local live = workspace:FindFirstChild(LIVE_FOLDER)
	if not live then
		return nil
	end

	local myCharacter = localPlayer and localPlayer.Character

	for _, entity in next, live:GetChildren() do
		if entity == myCharacter then
			continue
		end

		-- Skip anything with "hostage" in the name so we never target the trial hostages.
		if string.find(string.lower(entity.Name), "hostage", 1, true) then
			continue
		end

		local humanoid = entity:FindFirstChildWhichIsA("Humanoid")
		local root = entity:FindFirstChild("HumanoidRootPart")
		if humanoid and root and humanoid.Health > 0 then
			return entity
		end
	end

	return nil
end

---Whether any live entity in workspace.Live has "alpha" in its name (the trial's final boss).
---@return boolean
local function alphaEnemyPresent()
	local live = workspace:FindFirstChild(LIVE_FOLDER)
	if not live then
		return false
	end

	local myCharacter = localPlayer and localPlayer.Character

	for _, entity in next, live:GetChildren() do
		if entity == myCharacter then
			continue
		end
		if not string.find(string.lower(entity.Name), "alpha", 1, true) then
			continue
		end

		local humanoid = entity:FindFirstChildWhichIsA("Humanoid")
		if humanoid and humanoid.Health > 0 then
			return true
		end
	end

	return false
end

-- ============================================================================================
-- Step machine
-- ============================================================================================

local function runStep()
	-- ---- Etrean Luminant: teleport to the Trial of One (toggles already applied on start) ----
	if AutoTo1.inEtreanLuminant() then
		if step == 0 then
			setAction(TEXT.teleportingToTrial)
			Teleport.start("TrialOfOne")
			step = 1
		end
		return
	end

	-- ---- Trial arena (8668476218): go to workspace.One.OneTrigger, then fight enemies ----
	if AutoTo1.inArena() then
		local character = localPlayer.Character
		local root = character and character:FindFirstChild("HumanoidRootPart")
		if not root then
			return
		end

		-- Trial completion: once an "alpha" enemy that was present is gone, wait 5s then kick out.
		if alphaEnemyPresent() then
			alphaWasPresent = true
		elseif alphaWasPresent and completeAt == nil then
			completeAt = os.clock() + 5
		end

		if completeAt then
			setAction(TEXT.completing)
			if os.clock() >= completeAt then
				-- Clear the resume flag so it doesn't restart, then kick.
				PersistentData.set("to1data", nil)
				pcall(function()
					localPlayer:Kick(TEXT.kickMessage)
				end)
			end
			return
		end

		-- Resolve workspace.One.OneTrigger's position (part or model).
		local one = workspace:FindFirstChild("One")
		local trigger = one and one:FindFirstChild("OneTrigger")
		if not trigger then
			setAction(TEXT.waitingForOneTrigger)
			return
		end

		local triggerPos = trigger:IsA("BasePart") and trigger.Position or trigger:GetPivot().Position

		if step == 0 then
			-- Slow tween speed + Pathfind Breaker off while travelling to OneTrigger.
			setTweenSpeed(ONETRIGGER_TWEEN_SPEED)
			setToggle("Fling", false)
			setAction(TEXT.tweeningToOneTrigger)
			Tweening.goal(ARENA_TWEEN_ID, CFrame.new(triggerPos), false)

			if (root.Position - triggerPos).Magnitude <= 3 then
				-- Reached OneTrigger: draw the weapon and keep it drawn from here on, then wait for an
				-- enemy (no wait timer).
				weaponWanted = true
				drawWeapon()
				step = 2
			end
			return
		end

		if step == 2 then
			-- Normal tween speed + Pathfind Breaker off while holding + waiting for an enemy.
			setTweenSpeed(normalTweenSpeed())
			setToggle("Fling", false)
			Tweening.goal(ARENA_TWEEN_ID, CFrame.new(triggerPos), false)
			setAction(TEXT.waitingForEnemy)

			if findEnemyInLive() then
				step = 3
			end
			return
		end

		-- Step 3: fight -- fast-tween onto the enemy and m1 until it's gone, then wait for the next.
		local enemy = findEnemyInLive()
		if not enemy then
			-- Enemy dead / despawned: back to normal speed, Pathfind Breaker off, wait for more enemies.
			setTweenSpeed(normalTweenSpeed())
			setToggle("Fling", false)
			step = 2
			return
		end

		local ok, enemyCFrame = pcall(function()
			return enemy:GetPivot()
		end)
		if ok then
			local enemyPos = enemyCFrame.Position
			local targetCFrame

			local lowerName = string.lower(enemy.Name)
			local isEnforcer = string.find(lowerName, "enforcer", 1, true) ~= nil

			-- Fling stays OFF for this automation (never flings, even on enforcers).
			setToggle("Fling", false)

			setAction(TEXT.fightingEnemy)

			if string.find(lowerName, "stoneknight", 1, true) or isEnforcer then
				-- Stoneknight / Enforcer: stay 5 studs behind it and 8 studs above it, facing it.
				local behind = enemyPos - (enemyCFrame.LookVector * 5) + Vector3.new(0, 8, 0)
				targetCFrame = CFrame.lookAt(behind, enemyPos)
			else
				-- Track the enemy horizontally but hold the Y under OneTrigger, facing up at it (small
				-- nudge keeps CFrame.lookAt stable when directly overhead).
				local underPos = Vector3.new(enemyPos.X + 0.1, triggerPos.Y - ENEMY_UNDER_OFFSET, enemyPos.Z)
				targetCFrame = CFrame.lookAt(underPos, enemyPos)
			end

			-- Fast tween speed ONLY while fighting, so positioning onto the enemy is effectively instant.
			setTweenSpeed(TWEEN_SPEED)
			Tweening.goal(ARENA_TWEEN_ID, targetCFrame, false)

			-- Respect Block Input: don't swing while it wants to hold our M1 (block/punish window).
			if not Defense.shouldBlockInput("M1s") then
				local aim = CFrame.new(root.Position, enemyPos)
				pcall(function()
					InputClient.left(aim, true)
				end)
			end
		end
		return
	end

	-- Neither place yet -- wait.
	setAction(TEXT.waitingForTrial)
end

-- Start / stop
-- ============================================================================================

local function teardown()
	AutoTo1.running = false

	if timerConnection then
		pcall(function()
			timerConnection:Disconnect()
		end)
		timerConnection = nil
	end
	if stepConnection then
		pcall(function()
			stepConnection:Disconnect()
		end)
		stepConnection = nil
	end
	if characterAddedConnection then
		pcall(function()
			characterAddedConnection:Disconnect()
		end)
		characterAddedConnection = nil
	end
	if weaponConnection then
		pcall(function()
			weaponConnection:Disconnect()
		end)
		weaponConnection = nil
	end

	weaponWanted = false

	pcall(function()
		Tweening.stop(ARENA_TWEEN_ID)
	end)
	pcall(Teleport.stop)
	restoreToggles()

	if screenGui then
		screenGui.Enabled = false
	end
end

function AutoTo1.start()
	if AutoTo1.running then
		return
	end

	if not AutoTo1.inValidPlace() then
		return Logger.notify(TEXT.notInPlace)
	end

	-- If the loading screen (LoadingGui) is up (e.g. we just resumed after a place teleport), fire the
	-- StartMenu.Start remote to blow through it until the character + backpack have replicated (10s cap).
	local playerGui = localPlayer and localPlayer:FindFirstChildOfClass("PlayerGui")
	if playerGui and playerGui:FindFirstChild("LoadingGui") then
		local deadline = os.clock() + 10
		while not requirementsMet() and os.clock() < deadline do
			General.fireStartMenu()
			task.wait(0.25)
		end
	end

	step = 0
	weaponWanted = false
	alphaWasPresent = false
	completeAt = nil
	startClock = os.clock()
	AutoTo1.running = true

	-- Resume flag: survives server hops (MemStorage), so it restarts on the next server.
	saveState()

	-- No Fall Damage + Fly + Auto Defense/Block Input/Auto Feint + Void Mobs on for the run.
	applyToggles()

	buildOverlay()
	setAction(nil)
	if timerLabel then
		timerLabel.Text = formatElapsed(0)
	end
	if screenGui then
		screenGui.Enabled = true
	end

	-- Re-run current step if we respawn mid-automation.
	characterAddedConnection = localPlayer.CharacterAdded:Connect(function()
		if not AutoTo1.running then
			return
		end
		task.wait(0.2)
	end)

	-- Elapsed timer.
	timerConnection = runService.Heartbeat:Connect(function()
		if not AutoTo1.running then
			return
		end
		if timerLabel then
			timerLabel.Text = formatElapsed(os.clock() - startClock)
		end
	end)

	-- Keep the weapon drawn: once we've reached OneTrigger, continuously check if it's not drawn and
	-- fire the DrawWeapon remote if so.
	weaponConnection = runService.Heartbeat:Connect(function()
		if not AutoTo1.running or not weaponWanted then
			return
		end
		if not isWeaponDrawn() then
			drawWeapon()
		end
	end)

	-- Step machine (throttled).
	local lastStepAt = 0
	stepConnection = runService.Heartbeat:Connect(function()
		if not AutoTo1.running then
			return
		end
		if os.clock() - lastStepAt < 0.25 then
			return
		end
		lastStepAt = os.clock()
		runStep()
	end)
end

function AutoTo1.stop()
	if not AutoTo1.running and not PersistentData.get("to1data") then
		return
	end

	teardown()

	-- Manual stop: clear the resume flag so a re-execute doesn't restart it.
	PersistentData.set("to1data", nil)
end

function AutoTo1.init()
	if not LPH_OBFUSCATED or (shared.Luma and shared.Luma.verbose) then
		Logger.warn("Auto To1 initialized.")
	end
end

return AutoTo1

-- PR-style per-entity animator tracking. `trackedEntities` maps a live entity -> its Animator (or `true`
-- while the Animator is still being resolved, which claims the slot synchronously so a racing ChildAdded +
-- the initial scan can't double-track the same entity). Only models under workspace.Live get an
-- AnimatorDefender, so the number of parry trackers is bounded by the live combat entities (tens) instead of
-- every Animator in the DataModel (thousands) -- that unbounded count was the fiber-exhaustion crash.
local trackedEntities = {}

---@param entity Instance
local untrackLiveEntity = LPH_NO_VIRTUALIZE(function(entity)
	local animator = trackedEntities[entity]
	if not animator then
		return
	end

	trackedEntities[entity] = nil

	-- `true` is only the "resolving" placeholder -- just a real Animator has a defender to tear down.
	if typeof(animator) == "Instance" then
		onGameDescendantRemoved(animator)
	end
end)

---@param entity Instance
local trackLiveEntity = LPH_NO_VIRTUALIZE(function(entity)
	if not entity:IsA("Model") or trackedEntities[entity] then
		return
	end

	-- Claim the slot synchronously (before the yield below) so a racing ChildAdded / initial-scan pass can't
	-- spin up a second AnimatorDefender for the same entity.
	trackedEntities[entity] = true

	task.spawn(function()
		-- The Animator is usually parented a frame or two after the entity enters Live, so wait briefly.
		local animator = entity:FindFirstChildWhichIsA("Animator", true)
		if not animator then
			local deadline = os.clock() + 5
			repeat
				task.wait()
				animator = entity:FindFirstChildWhichIsA("Animator", true)
			until animator or os.clock() >= deadline or not entity.Parent
		end

		-- Bail if it never showed, the entity already left, or it was untracked while we waited.
		if not animator or not entity.Parent or trackedEntities[entity] ~= true then
			if trackedEntities[entity] == true then
				trackedEntities[entity] = nil
			end
			return
		end

		trackedEntities[entity] = animator
		addAnimatorDefender(animator)
	end)
end)

local function isMantraAllowed(mantra, optionKey)
	local selectedMantras = Configuration.expectOptionValue(optionKey)
	if type(selectedMantras) ~= "table" then
		return true
	end

	local isEmpty = true
	for _, v in pairs(selectedMantras) do
		if v then
			isEmpty = false
			break
		end
	end

	if isEmpty then
		return false
	end

	local defaultName = mantra and mantra.GetAttribute and mantra:GetAttribute("DefaultName")
	if not defaultName then
		return false
	end

	return selectedMantras[defaultName] == true
end

---@param optionName string
---@param randomize boolean? If true, pick a random off-cooldown mantra instead of the first in order.
local function getSelectedMantraTool(optionName, randomize)
	local backpack = localPlayer and localPlayer:FindFirstChild("Backpack")
	if not backpack then
		return nil
	end

	local selectedMantras = Configuration.expectOptionValue(optionName)
	if type(selectedMantras) ~= "table" then
		return nil
	end

	-- Selected mantra names (the multi-dropdown stores a { [name] = true } set); ignore any that
	-- were toggled off, and sort so "the next mantra" is a stable, predictable fallback order.
	local names = {}
	for name, selected in next, selectedMantras do
		if selected then
			names[#names + 1] = name
		end
	end

	if #names == 0 then
		return nil
	end

	table.sort(names)

	-- DefaultNames of every mantra currently on cooldown (a ToolLockCD effect whose Value names the tool).
	local onCooldown = {}
	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local effectReplicatorModule = effectReplicator and require(effectReplicator)

	if effectReplicatorModule and effectReplicatorModule.GetEffects then
		for _, effect in pairs(effectReplicatorModule:GetEffects()) do
			if type(effect) ~= "table" then
				continue
			end

			local class = effect.Class or (effect.index and effect.index.Class)
			local value = effect.Value or (effect.index and effect.index.Value)

			if class == "ToolLockCD" and value then
				local tool = backpack:FindFirstChild(tostring(value))
				local defaultName = tool and tool:GetAttribute("DefaultName")
				if defaultName then
					onCooldown[defaultName] = true
				end
			end
		end
	end

	---Find the backpack Tool for a mantra DefaultName.
	local function toolFor(name)
		for _, tool in pairs(backpack:GetChildren()) do
			if tool:IsA("Tool") and tool:GetAttribute("DefaultName") == name then
				return tool
			end
		end
		return nil
	end

	if randomize then
		-- Collect every selected mantra that exists and is off cooldown, then pick one at random.
		local available = {}
		for _, name in ipairs(names) do
			if not onCooldown[name] then
				local tool = toolFor(name)
				if tool then
					available[#available + 1] = tool
				end
			end
		end

		if #available == 0 then
			return nil
		end

		return available[math.random(1, #available)]
	end

	-- Walk the selection in order and return the first mantra that exists and is off cooldown,
	-- skipping straight past any that are still cooling down.
	for _, name in ipairs(names) do
		if not onCooldown[name] then
			local tool = toolFor(name)
			if tool then
				return tool
			end
		end
	end

	return nil
end

local function tryActivateBackstabMovestack()
	if not Configuration.expectToggleValue("BackstabMovestack") then
		return false
	end

	local tool = getSelectedMantraTool("BackstabMovestackMantras")
	if not tool then
		return false
	end

	local localPlayer = game:GetService("Players").LocalPlayer
	local character = localPlayer and localPlayer.Character
	local characterHandler = character and character:FindFirstChild("CharacterHandler")
	local requests = characterHandler and characterHandler:FindFirstChild("Requests")
	local activateMantra = requests and requests:FindFirstChild("ActivateMantra")

	if activateMantra then
		activateMantra:FireServer(tool)
		return true
	end

	return false
end

-- ---- Auto Mixup: Roll Cancel Behind Back ----
-- Nearest living, non-ally enemy's root part within range (self-contained, no aimbot config needed).
local function getNearestEnemyRoot()
	local character = localPlayer and localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if not root then
		return nil
	end

	local live = workspace:FindFirstChild("Live")
	if not live then
		return nil
	end

	local best, bestDistance = nil, math.huge
	for _, entity in next, live:GetChildren() do
		if entity == character then
			continue
		end

		local humanoid = entity:FindFirstChildWhichIsA("Humanoid")
		local entityRoot = entity:FindFirstChild("HumanoidRootPart")
		if not humanoid or not entityRoot or humanoid.Health <= 0 then
			continue
		end

		local player = players:GetPlayerFromCharacter(entity)
		if player and PlayerScanning.isAlly(player) then
			continue
		end

		-- Optionally restrict roll-behind to real players (skip NPCs/mobs).
		if Configuration.expectToggleValue("AutoMixupRollBehindPlayersOnly") and not player then
			continue
		end

		local distance = (entityRoot.Position - root.Position).Magnitude
		if distance <= 60 and distance < bestDistance then
			best = entityRoot
			bestDistance = distance
		end
	end

	return best
end

-- Roll -> lock onto the opponent through the roll -> roll cancel -> m1 again (the Easy Roll Cancel path).
local performRollCancelBehindBack = LPH_NO_VIRTUALIZE(function(targetRoot)
	-- Dodge gate over the whole sequence: if we can't dodge right now, do nothing -- no roll, no
	-- roll cancel, no m1 -- rather than feinting and m1ing on a roll that never happened.
	if not StateListener.cdodge() then
		return
	end

	-- Re-entry guard: this routine holds a ~0.5s camera lock (Heartbeat:Wait loops below), and it has two
	-- drivers -- the m1 remote-hook mixup path and the Roll Parry alias (Defense.rollCancelBehindBack). If a
	-- second call enters while the first is still mid-lock, both BindToRenderStep the SAME name with a fresh
	-- closure, so they stack (2-3 deep) and Roblox warns "removed different functions with same reference name
	-- AutoMixup_RollBehindLockOn N times" every unbind -- the churn that saturated the crash log. One at a time.
	if rollBehindActive then
		return
	end

	-- Suppress Easy Roll Cancel / Easy Mantra Feint for the duration so they don't stack their own
	-- feint + m1 on top of this sequence's roll cancel.
	rollBehindActive = true

	local intensity = math.clamp((Configuration.expectOptionValue("AutoMixupRollBehindLockOnIntensity") or 50) / 100, 0.0, 1.0)

	-- How long before the roll cancel, and how long the camera stays locked on overall.
	local ROLL_CANCEL_DELAY = 0.18
	local LOCK_ON_DURATION = 0.5

	-- Roll -- same rolling logic as the auto-defense roll (honors the Blatant Roll setting). We do
	-- the cancel ourselves below (Easy Roll Cancel path), so leave the dodge's own rollCancel off.
	local rollOptions = DodgeOptions.new()
	rollOptions.direct = Configuration.expectToggleValue("BlatantRoll")
	InputClient.dodge(rollOptions)

	-- Lock onto the opponent with the CAMERA. Bound just above the camera render step so our aim is
	-- the last write each frame. Lock On Intensity = lerp strength.
	local bindName = "AutoMixup_RollBehindLockOn"
	-- Always clear any prior bind under this name before re-binding, so a run that errored out of its wait
	-- loop before reaching the unbind below can never leave a stale closure stacked under the same name.
	pcall(runService.UnbindFromRenderStep, runService, bindName)
	if intensity > 0 then
		pcall(function()
			runService:BindToRenderStep(bindName, Enum.RenderPriority.Camera.Value + 1, function()
				local camera = workspace.CurrentCamera
				if camera and targetRoot and targetRoot.Parent then
					camera.CFrame = camera.CFrame:Lerp(CFrame.lookAt(camera.CFrame.Position, targetRoot.Position), intensity)
				end
			end)
		end)
	end

	local lockStart = os.clock()

	-- Short window before the roll cancel.
	while os.clock() - lockStart < ROLL_CANCEL_DELAY do
		runService.Heartbeat:Wait()
	end

	-- Roll cancel + m1 again (Easy Roll Cancel path), aimed via the camera (mouse hit).
	local character = localPlayer and localPlayer.Character
	local root = character and character:FindFirstChild("HumanoidRootPart")
	if root then
		-- Only feint if we're actually still mid-roll (Dodge / ClientDodge effect); otherwise a stray
		-- feint would get buffered onto the m1 below and feint it.
		local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
		local effectReplicatorModule = effectReplicator and require(effectReplicator)
		if effectReplicatorModule and (effectReplicatorModule:HasEffect("Dodge") or effectReplicatorModule:HasEffect("ClientDodge")) then
			InputClient.feint()
			task.wait()
		end

		local mouse = localPlayer and localPlayer:GetMouse()
		local aim = (mouse and mouse.Hit) or root.CFrame
		InputClient.left(aim, true)
	end

	-- Hold the camera lock for the rest of the set lock-on duration, then release it.
	while os.clock() - lockStart < LOCK_ON_DURATION do
		runService.Heartbeat:Wait()
	end

	pcall(function()
		runService:UnbindFromRenderStep(bindName)
	end)

	rollBehindActive = false
end)

-- Public alias so other modules (e.g. the Roll Parry defense option) can run the roll-behind ->
-- roll cancel -> m1 sequence against a given target root.
Defense.rollCancelBehindBack = performRollCancelBehindBack

-- Called from the m1 remote hook. Rolls each enabled Auto Mixup action's probability, then picks
-- ONE of the actions that hit at random -- no fixed priority order. Returns true to BLOCK the
-- manual m1 (roll-cancel-behind-back takes over), false to let it fire (a feint / feint-into-mantra
-- rides on the m1 that goes out).
Defense.tryAutoMixup = LPH_NO_VIRTUALIZE(function()
	if not Configuration.expectToggleValue("AutoMixup") then
		return false
	end

	if os.clock() - lastAutoMixupAt < 0.2 then
		return false
	end

	if not (localPlayer and localPlayer.Character) then
		return false
	end

	-- Only mix up while the weapon is actually drawn. "Equipped" is the drawn-weapon effect (same one
	-- weapon-draw detection keys off). Sheathed / no weapon out -> leave the manual m1 alone.
	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local effectReplicatorModule = effectReplicator and require(effectReplicator)
	if not (effectReplicatorModule and effectReplicatorModule:HasEffect("Equipped")) then
		return false
	end

	-- Build the pool of actions that rolled this m1. Each is gated by its own feasibility check so
	-- we never queue something the player can't actually do.
	local candidates = {}
	local mantraTool = nil
	local enemyRoot = nil

	local feintChance = Configuration.expectOptionValue("AutoMixupFeintProbability") or 0.0
	if feintChance > 0 and Random.new():NextNumber(1.0, 100.0) <= feintChance and StateListener.cfeint() then
		candidates[#candidates + 1] = "feint"
	end

	local mantraChance = Configuration.expectOptionValue("AutoMixupFeintIntoMantraProbability") or 0.0
	if mantraChance > 0 and Random.new():NextNumber(1.0, 100.0) <= mantraChance then
		mantraTool = getSelectedMantraTool("AutoMixupMantras", true)
		if mantraTool then
			candidates[#candidates + 1] = "mantra"
		end
	end

	local rollChance = Configuration.expectOptionValue("AutoMixupRollBehindProbability") or 0.0
	-- Never roll-cancel-behind if the player can't dodge right now (NoRoll cooldown / stun / blocked).
	if rollChance > 0 and Random.new():NextNumber(1.0, 100.0) <= rollChance and StateListener.cdodge() then
		enemyRoot = getNearestEnemyRoot()
		if enemyRoot then
			candidates[#candidates + 1] = "rollbehind"
		end
	end

	if #candidates == 0 then
		return false
	end

	local choice = candidates[Random.new():NextInteger(1, #candidates)]
	lastAutoMixupAt = os.clock()

	if choice == "rollbehind" then
		task.spawn(function()
			performRollCancelBehindBack(enemyRoot)
		end)
		return true
	end

	if choice == "mantra" then
		local mantraDelay = Configuration.expectOptionValue("AutoMixupFeintIntoMantraDelay") or 0.0
		task.spawn(function()
			if mantraDelay > 0 then
				task.wait(mantraDelay)
			end
			InputClient.amantra(mantraTool)
		end)
		return false
	end

	if choice == "feint" then
		task.spawn(function()
			task.wait() -- let the manual m1 go out first, then feint it
			InputClient.feint()
		end)
		return false
	end

	return false
end)

local function isBackstabActionLocked()
	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local effectReplicatorModule = effectReplicator and require(effectReplicator)
	if not effectReplicatorModule then
		return false
	end

	return effectReplicatorModule:HasAny("Action", "MobileAction", "UsingAbility", "LightAttack")
end

local function getBackstabMovestackDelay()
	local delay = Configuration.expectOptionValue("BackstabMovestackDelay") or 0.05
	return math.max(delay, 0.0)
end

local function scheduleBackstabMovestackActivation(label)
	TaskSpawner.delay(label, function()
		return getBackstabMovestackDelay()
	end, function()
		backstabMovestackReady = false
		tryActivateBackstabMovestack()
	end)
end

local function isBackstabTriggerAnimationTrack(track)
	if not track or not track.Animation then
		return false
	end

	local aid = tostring(track.Animation.AnimationId)
	if aid == BACKSTAB_TRIGGER_ANIMATION_ID then
		return true
	end

	local numericId = aid:match("%d+")
	return numericId == BACKSTAB_TRIGGER_ANIMATION_NUMERIC_ID
end

local onBackstabLocalAnimationPlayed = LPH_NO_VIRTUALIZE(function(track)
	if not Configuration.expectToggleValue("BackstabMovestack") then
		return
	end

	if not isBackstabTriggerAnimationTrack(track) then
		return
	end

	if os.clock() - backstabAnimationTriggerTimestamp <= 0.15 then
		return
	end

	backstabMovestackReady = true
	backstabAnimationTriggerTimestamp = os.clock()

	if not isBackstabActionLocked() and tryActivateBackstabMovestack() then
		backstabMovestackReady = false
		return
	end

	TaskSpawner.spawn("Defense_BackstabMovestackImmediateRetry", function()
		local startedAt = os.clock()

		while os.clock() - startedAt <= 0.35 do
			if not backstabMovestackReady then
				return
			end

			if not Configuration.expectToggleValue("BackstabMovestack") then
				return
			end

			if not isBackstabActionLocked() and tryActivateBackstabMovestack() then
				backstabMovestackReady = false
				return
			end

			runService.RenderStepped:Wait()
		end

		scheduleBackstabMovestackActivation("Defense_BackstabMovestackFromAnimation")
	end)
end)

local function bindBackstabAnimationListener(character)
	if not character then
		return
	end

	local animator = character:FindFirstChildWhichIsA("Animator", true)
	if not animator then
		return
	end

	local animationPlayed = Signal.new(animator.AnimationPlayed)
	defenseMaid["Defense_BackstabAnimationPlayed"] =
		animationPlayed:connect("Defense_BackstabAnimationPlayed", onBackstabLocalAnimationPlayed)
end

local function onBackstabCharacterAdded(character)
	if not character then
		return
	end

	bindBackstabAnimationListener(character)

	local descendantAdded = Signal.new(character.DescendantAdded)
	defenseMaid["Defense_BackstabCharacterDescendantAdded"] = descendantAdded:connect(
		"Defense_BackstabCharacterDescendantAdded",
		function(descendant)
			if not descendant:IsA("Animator") then
				return
			end

			bindBackstabAnimationListener(character)
		end
	)
end

---@param effect table
local function onBackstabEffectAdded(effect)
	if not Configuration.expectToggleValue("BackstabMovestack") then
		return
	end

	if effect.Class == "backstabDebounce" then
		backstabMovestackReady = true
		task.delay(3, function()
			backstabMovestackReady = false
		end)
	end
end

---@param effect table
local function onBackstabEffectRemoved(effect)
	if not Configuration.expectToggleValue("BackstabMovestack") then
		return
	end

	if os.clock() - backstabAnimationTriggerTimestamp <= 1.0 then
		return
	end

	if effect.Class ~= "Action" or not backstabMovestackReady then
		return
	end

	scheduleBackstabMovestackActivation("Defense_BackstabMovestack")
end



local function attemptRollCast(mantra)
	if not isMantraAllowed(mantra, "RollCastMantrasList") then
		return
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local effectReplicatorModule = effectReplicator and require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	if effectReplicatorModule:HasAny(
		"AirCombo",
		"NoSlide",
		"Action",
		"MobileAction",
		"UsingAbility",
		"LightAttack",
		"Blocking",
		"Parried",
		"ClientSlide2",
		"ClientSwim"
	) then
		return
	end

	local keybindsModule = replicatedStorage:FindFirstChild("KeyBinds")
	if not keybindsModule then
		return
	end

	local Keybinds = require(keybindsModule)
	if not Keybinds or not Keybinds.ForceActionDown then
		return
	end

	Keybinds.ForceActionDown("Dodge")
	Keybinds.ForceActionUp("Dodge")

	task.delay(0.09, function()
		local clientFeint = effectReplicatorModule:CreateEffect("ClientFeint")
		if clientFeint then
			clientFeint:Debris(0.1)
		end
	end)
end

local function attemptSlideCast(mantra)
	if not isMantraAllowed(mantra, "SlideCastMantrasList") then
		return
	end

	local character = localPlayer and localPlayer.Character
	if not character then
		return
	end

	local ether = character:FindFirstChild("Ether")
	local spellCost = mantra and mantra:GetAttribute("SpellCost")
	if not ether or not spellCost or ether.Value <= spellCost then
		return
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	local effectReplicatorModule = effectReplicator and require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	if effectReplicatorModule:HasAny(
		"AirCombo",
		"NoSlide",
		"Action",
		"MobileAction",
		"UsingAbility",
		"LightAttack",
		"Blocking",
		"Parried",
		"ClientSlide2",
		"ClientSwim"
	) then
		return
	end

	local humanoid = character:FindFirstChildWhichIsA("Humanoid")
	if not humanoid then
		return
	end

	local moveDirection = humanoid.MoveDirection
	if moveDirection.Magnitude <= 0.1 then
		return
	end

	local rootPart = character:FindFirstChild("HumanoidRootPart")
	if not rootPart then
		return
	end

	local hitPart = workspace:FindPartOnRayWithIgnoreList(
		Ray.new(rootPart.Position, Vector3.new(0, -10, 0)),
		{ workspace:FindFirstChild("Live"), workspace:FindFirstChild("NPCs"), workspace:FindFirstChild("Thrown") }
	)

	if not hitPart then
		return
	end

	if not effectReplicatorModule:FindEffect("Sprinting") then
		return
	end

	local keybindsModule = replicatedStorage:FindFirstChild("KeyBinds")
	if not keybindsModule then
		return
	end

	local Keybinds = require(keybindsModule)
	if not Keybinds or not Keybinds.ForceActionDown then
		return
	end

	if not effectReplicatorModule:FindEffect("CastingSpell") then
		local castingSpell = effectReplicatorModule:CreateEffect("CastingSpell")
		if castingSpell then
			castingSpell:Debris(0.05)
		end
	end

	Keybinds.ForceActionDown("Crouch")
	Keybinds.ForceActionUp("Crouch")
end

local function updateMantraRequests()
	local mantra = StateListener.lMantraActivated
	if not mantra then
		return
	end

	local activatedAt = StateListener.lMantraActivatedAt
	if activatedAt == lastMantraActivatedAt then
		return
	end

	lastMantraActivatedTool = mantra
	lastMantraActivatedAt = activatedAt

	if Configuration.expectToggleValue("RollCastMantras") then
		attemptRollCast(mantra)
	end

	if Configuration.expectToggleValue("SlideCastMantras") then
		attemptSlideCast(mantra)
	end
end

local updateAutoAirCounter = LPH_NO_VIRTUALIZE(function()
	if not Configuration.expectToggleValue("AutoAirCounter") then
		return
	end

	if os.clock() - lastAutoAirCounterUpdate < 0.1 then
		return
	end
	lastAutoAirCounterUpdate = os.clock()

	local character = localPlayer and localPlayer.Character
	if not character then
		return
	end

	local characterHandler = character:FindFirstChild("CharacterHandler")
	local requests = characterHandler and characterHandler:FindFirstChild("Requests")
	local antiAir = requests and requests:FindFirstChild("AntiAir")
	if not antiAir then
		return
	end

	local rootPart = character:FindFirstChild("HumanoidRootPart")
	if not rootPart then
		return
	end

	if not Configuration.expectToggleValue("AutoAirCounterOnGround") then
		local isGrounded = false
		local controllerManager = character:FindFirstChild("ControllerManager")
		if controllerManager then
			local groundSensor = controllerManager:FindFirstChild("GroundSensor")
			if groundSensor and groundSensor.SensedPart then
				isGrounded = true
			end
		end
		
		if not isGrounded then
			local humanoid = character:FindFirstChildWhichIsA("Humanoid")
			if humanoid and humanoid.FloorMaterial ~= Enum.Material.Air then
				isGrounded = true
			end
		end

		if not isGrounded then
			local raycastParams = RaycastParams.new()
			raycastParams.FilterType = Enum.RaycastFilterType.Exclude
			raycastParams.FilterDescendantsInstances = { workspace:FindFirstChild("Live"), workspace:FindFirstChild("NPCs"), workspace:FindFirstChild("Thrown") }
			local hit = workspace:Raycast(rootPart.Position, Vector3.new(0, -4, 0), raycastParams)
			if hit and hit.Instance then
				isGrounded = true
			end
		end

		if isGrounded then
			return
		end
	end

	local live = workspace:FindFirstChild("Live")
	if not live then
		return
	end

	local range = Configuration.expectOptionValue("AutoAirCounterRange") or 60
	local allyCheck = Configuration.expectToggleValue("AutoAirCounterAllyCheck")

	for _, entity in next, live:GetChildren() do
		if entity == character then
			continue
		end

		if entity.Name:sub(1, 1) == "." then
			continue
		end

		local entityRoot = entity:FindFirstChild("HumanoidRootPart")
		if not entityRoot then
			continue
		end

		local distance = (entityRoot.Position - rootPart.Position).Magnitude
		if distance > range then
			continue
		end

		if allyCheck then
			local entityPlayer = players:GetPlayerFromCharacter(entity)
			if entityPlayer then
				local pvpAttribute = entity:GetAttribute("PVPEnabled")
				if pvpAttribute == false or PlayerScanning.isAlly(entityPlayer) then
					continue
				end
			end
		end

		antiAir:FireServer(entity)
	end
end)

local updateVisualizations = LPH_NO_VIRTUALIZE(function()
	if not Configuration.expectToggleValue("EnableVisualizations") then
		return
	end

	if os.clock() - lastVisualizationUpdate <= 5.0 then
		return
	end

	lastVisualizationUpdate = os.clock()

	for _, object in next, defenderObjects do
		for idx, hitbox in next, object.hmaid._tasks do
			if typeof(hitbox) ~= "Instance" then
				continue

		end
	end

---@param position number
---@param str string
local hssp = LPH_NO_VIRTUALIZE(function(position, str)
	if autoWispLocked then
		return Logger.warn("Auto Wisp is locked.")
	end

	if position <= 0 or position > #str then
		return Logger.warn("Invalid position (%i vs. %i) for Auto Wisp.", position, #str)
	end

	local delay = Configuration.expectOptionValue("AutoWispDelay") or 0

	if lastAutoWispShift and os.clock() - lastAutoWispShift <= 0.15 + delay then
		return Logger.warn(
			"(%.2f - %.2f = %.2f vs. %.2f) Auto Wisp shift is on forced cooldown.",
			os.clock(),
			lastAutoWispShift,
			os.clock() - lastAutoWispShift,
			0.15 + delay
		)
	end

	local character = str:sub(position, position)
	if character ~= "Z" and character ~= "X" and character ~= "C" and character ~= "V" then
		return Logger.warn("Invalid character (%s) for Auto Wisp.", tostring(character))
	end

	local localPlayer = players.LocalPlayer
	local localPlayerCharacter = localPlayer.Character
	if not localPlayerCharacter then
		return
	end

	local mantras = replicatedStorage:FindFirstChild("Requests")
	mantras = mantras and mantras:FindFirstChild("Mantras")
	local spellInput = mantras and mantras:FindFirstChild("RitualSpellInput")
	if not spellInput then
		return
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return
	end

	if not effectReplicatorModule:HasEffect("RitualCastingSpell") then
		return Logger.warn("RitualCastingSpell effect not found.")
	end

	if effectReplicatorModule:HasEffect("Knocked") then
		return Logger.warn("Knocked effect found.")
	end

	Logger.warn("Sending event for character (%s) with position (%i) and string (%s).", character, position, str)

	autoWispLocked = true
	lastAutoWispUpdate = os.clock()

	pcall(function()
		spellInput:FireServer(WISP_INPUT_MAP[character] or character)
	end)

	task.delay(0.5 + delay, function()
		autoWispLocked = false
	end)
end)

local updateDefenders = LPH_NO_VIRTUALIZE(function()
	Defense.updateBlockInputState()

	if Configuration.expectToggleValue("M1Hold") and leftClickState and not Defense.shouldBlockInput("M1s") then
		-- Held m1s are fired by our own code, so they bypass the m1 hook's Auto Mixup (which only runs
		-- on non-checkcaller input). Run the mixup here too: if it takes over (roll-cancel-behind), skip
		-- this held m1; otherwise fire it (a feint / feint-into-mantra rides on it).
		if not Defense.tryAutoMixup() then
			InputClient.left(players.LocalPlayer:GetMouse().Hit, true)
		end
	end

	if Configuration.expectToggleValue("AutoWisp") and cwp and cws then
		hssp(cwp, cws)
	end

	if not Configuration.expectToggleValue("EnableAutoDefense") then
		return
	end

	for _, object in next, defenderAnimationObjects do
		object:update()
	end

	for _, object in next, defenderPartObjects do
		object:update()
	end
end)

local function isThreatImminent(object, timing)
	local entity = object.entity
	if not entity then
		return true
	end

	if not entity:FindFirstChild("HumanoidRootPart") then
		return true
	end

	local root = entity:FindFirstChild("HumanoidRootPart")

	if Configuration.expectOptionValue("BlockInputRangeCheck") == "Inside Hitbox" then
		local hoptions = HitboxOptions.new(root, timing)
		hoptions.visualize = false
		-- Enlarge the detection hitbox by 1.2x so it triggers a touch more leniently than the exact hitbox
		-- (hc reads the size from options:hitbox(), which returns this cache).
		hoptions.cache = hoptions:hitbox() * 1.2

		if hoptions:hitbox().Magnitude > 0 then
			return object:hc(hoptions, nil) == true
		end
	end

	local distance = object:distance(entity)
	if not distance then
		return true
	end

	return distance <= (timing.imxd or math.huge)
end

---@note: Runs on a safe thread (postSimulation) so the input hooks only ever read the cached
Defense.updateBlockInputState = LPH_NO_VIRTUALIZE(function()
	if
		not Configuration.expectToggleValue("EnableAutoDefense")
		or not Configuration.expectToggleValue("BlockInput")
	then
		blockInputActive = {}
		return
	end

	local allowedTargets = Configuration.expectOptionValue("BlockInputAllowedTargets") or {
		PVE = true,
		PVP = true,
	}

	local useTargeting = Configuration.expectToggleValue("BlockInputTargeting")
	local criticalPunishTime = (Configuration.expectOptionValue("BlockInputCriticalPunishmentTime") or 0) / 1000
	local m1PunishTime = (Configuration.expectOptionValue("BlockInputM1PunishmentTime") or 0) / 1000
	local halfPing = Latency.rdelay()

	local active = {}

	for object in next, Defender.armed do
		local timing = object.timing
		if timing and timing.ibi then
			continue
		end

		local ownerModel = object.entity or object.owner
		if not ownerModel and object.part then
			ownerModel = object.part:FindFirstAncestorWhichIsA("Model")
		end

		local isPvp = ownerModel and players:GetPlayerFromCharacter(ownerModel) ~= nil
		if isPvp and not allowedTargets.PVP then
			continue
		end

		if not isPvp and not allowedTargets.PVE then
			continue
		end

		if useTargeting and not (ownerModel and Targeting.find(ownerModel, BLOCK_INPUT_TARGETING_KEYS)) then
			continue
		end

		if timing and not isThreatImminent(object, timing) then
			continue
		end

		if object:blocking() then
			active.Mantras = true
		end

		for _, tsk in next, object.tasks do
			if coroutine.status(tsk.thread) == "dead" then
				continue
			end

			local when = tsk:when()
			local tail = when + tsk.after

			if os.clock() >= when - criticalPunishTime - halfPing and os.clock() <= tail then
				active.Criticals = true
			end

			if os.clock() >= when - m1PunishTime - halfPing and os.clock() <= tail then
				active.M1s = true
			end
		end
	end

	blockInputActive = active
end)

---@note: Cheap read only (booleans + option table) so it is safe to call from input hooks.
---@param moveType string
---@return boolean
Defense.shouldBlockInput = LPH_NO_VIRTUALIZE(function(moveType)
	local blockedMoves = Configuration.expectOptionValue("BlockedSafeInputUserMoves") or {}
	return blockedMoves[moveType] == true and blockInputActive[moveType] == true
end)

---@param aid string
---@return PlaybackData?
Defense.agpd = LPH_NO_VIRTUALIZE(function(aid)
	---@note: Grabbing from 'rpbdata' means that we know that the data has been fully recorded.
	for _, object in next, defenderAnimationObjects do
		local pbdata = object.rpbdata[aid]
		if not pbdata then
			continue
		end

		return pbdata
	end

	---@note: Fallback to deleted playback data if that doesn't exist.
	for _, rpbdata in next, deletedPlaybackData do
		local pbdata = rpbdata[aid]
		if not pbdata then
			continue
		end

		return pbdata
	end
end)

---@param name string
---@param data any?
local onSpellEvent = LPH_NO_VIRTUALIZE(function(name, data)
	if name == "close" then
		cws = nil
		cwp = nil
		lastAutoWispUpdate = nil
		lastAutoWispShift = nil
		autoWispLocked = false
	end

	if name == "start" then
		cws = data
		cwp = 1
		lastAutoWispUpdate = nil
		lastAutoWispShift = nil
		autoWispLocked = false
	end

	if cws and cwp and name == "shift" then
		cwp = cwp + 1
		lastAutoWispShift = os.clock()
		autoWispLocked = false
	end
end)

function Defense.fflourish()
	if not Configuration.expectToggleValue("FeintFlourish") then
		return
	end

	local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
	if not effectReplicator then
		return Logger.warn("EffectReplicator not found for FeintFlourish.")
	end

	local effectReplicatorModule = require(effectReplicator)
	if not effectReplicatorModule then
		return Logger.warn("EffectReplicator module not found for FeintFlourish.")
	end

	if not effectReplicatorModule:FindEffect("MidAttack") then
		return Logger.warn("Not MidAttack effect for FeintFlourish.")
	end

	local mantra = Finder.ncdm()
	if not mantra then
		return Logger.warn("No mantra found for FeintFlourish.")
	end

	if effectReplicatorModule:FindEffect("FeintCool") then
		return Logger.warn("No avaliable feint for FeintFlourish.")
	end

	local lastAnimationTiming = StateListener.lAnimTiming
	if not lastAnimationTiming then
		return Logger.warn("No last animation timing found for FeintFlourish.")
	end

	if not lastAnimationTiming.smod:match("Flourish") then
		return Logger.warn("Last animation is not a flourish for FeintFlourish.")
	end

	Logger.warn("(%s) FeintFlourish activated mantra.", mantra.Name)

	InputClient.amantra(mantra)

	task.wait(0.1)

	Logger.warn("(%s) FeintFlourish feinted mantra.", mantra.Name)

	InputClient.feint()
end

function Defense.init()
	local WAIT = 30

	local assetFolder = replicatedStorage:WaitForChild("Assets", WAIT)
	local animationFolder = assetFolder and assetFolder:WaitForChild("Anims", WAIT)
	local mobsAnimationFolder = animationFolder and animationFolder:WaitForChild("Mobs", WAIT)

	for _, animation in next, mobsAnimationFolder and mobsAnimationFolder:QueryDescendants("Animation") or {} do
		if animation.Name == "RunningAttack" then
			continue
		end

		mobAnimations[animation.AnimationId] = animation
	end

	local requests = replicatedStorage:WaitForChild("Requests", WAIT)
	local clientEffect = requests and requests:WaitForChild("ClientEffect", WAIT)
	local clientEffectLarge = requests and requests:WaitForChild("ClientEffectLarge", WAIT)
	local clientEffectDirect = requests and requests:WaitForChild("ClientEffectDirect", WAIT)
	local mantras = requests and requests:WaitForChild("Mantras", WAIT)
	local spell = mantras and mantras:WaitForChild("RitualSpellInput", WAIT)

	local gameDescendantAdded = Signal.new(game.DescendantAdded)
	local gameDescendantRemoved = Signal.new(game.DescendantRemoving)
	local characterAdded = Signal.new(localPlayer.CharacterAdded)
	local inputBegan = Signal.new(userInputService.InputBegan)
	local inputEnded = Signal.new(userInputService.InputEnded)
	local renderStepped = Signal.new(runService.RenderStepped)
	local postSimulation = Signal.new(runService.PostSimulation)
	local clientEffectEvent = clientEffect and Signal.new(clientEffect.OnClientEvent)
	local clientEffectLargeEvent = clientEffectLarge and Signal.new(clientEffectLarge.OnClientEvent)
	local cientEffectDirect = clientEffectDirect
		and Signal.new(clientEffectDirect:IsA("BindableEvent") and clientEffectDirect.Event or clientEffectDirect.OnClientEvent)
	local spellEvent = spell and Signal.new(spell.OnClientEvent)

	local effectReplicator = replicatedStorage:WaitForChild("EffectReplicator", WAIT)
	if not effectReplicator then
		return error("EffectReplicator not found.")
	end

	local effectReplicatorModule = require(effectReplicator)
	local effectAddedSignal = Signal.new(effectReplicatorModule.EffectAdded)
	local effectRemovedSignal = Signal.new(effectReplicatorModule.EffectRemoved)

	---@note: Need this to detect UI presses / game processed inputs.
	defenseMaid:mark(inputBegan:connect("Defense_OnInputBegan", function(input, gameProcessed)
		if gameProcessed then
			return
		end

		if input.UserInputType == Enum.UserInputType.MouseButton2 then
			TaskSpawner.delay("StateListener_FeintFlourish", function()
				return 0.07
			end, Defense.fflourish)
		end

		if input.UserInputType ~= Enum.UserInputType.MouseButton1 then
			return
		end

		do
			local character = localPlayer.Character
			local root = character and character:FindFirstChild("HumanoidRootPart")
			-- Skip while Auto Mixup's roll-behind runs its own roll cancel, so we don't stack a second
			-- feint + m1 on it. Time-bounded so a stuck flag can't disable Easy Roll Cancel for good.
			local mixupBusy = rollBehindActive and (os.clock() - lastAutoMixupAt) < 1.0
			if root and not mixupBusy and (os.clock() - lastEasyFeint) > 0.3 and StateListener.cfeint() then
				if
					Configuration.expectToggleValue("EasyMantraFeint")
					and (effectReplicatorModule:HasEffect("UsingSpell") or effectReplicatorModule:HasEffect("CastingSpell"))
				then
					lastEasyFeint = os.clock()
					task.spawn(function()
						InputClient.feint()
						task.wait()
						InputClient.left(root.CFrame, true)
					end)
				elseif
					Configuration.expectToggleValue("EasyRollCancel")
					and (effectReplicatorModule:HasEffect("Dodge") or effectReplicatorModule:HasEffect("ClientDodge"))
				then
					lastEasyFeint = os.clock()
					task.spawn(function()
						InputClient.feint()
						task.wait()
						InputClient.left(root.CFrame, true)
					end)
				end
			end
		end

		leftClickState = true
	end))

	defenseMaid:mark(inputEnded:connect("Defense_OnInputEnded", function(input, _)
		if input.UserInputType ~= Enum.UserInputType.MouseButton1 then
			return
		end

		leftClickState = false
	end))

	defenseMaid:mark(gameDescendantAdded:connect("Defense_OnDescendantAdded", onGameDescendantAdded))
	defenseMaid:mark(gameDescendantRemoved:connect("Defense_OnDescendantRemoved", onGameDescendantRemoved))

	-- Per-entity animator tracking off workspace.Live (replaces the game-wide Animator branch of
	-- handleGameDescendant). One AnimatorDefender per live entity, torn down the instant it leaves Live --
	-- ChildRemoved fires on a reparent-out too, which game.DescendantRemoving misses. Spawned so a missing
	-- Live folder can't block init.
	defenseMaid:mark(task.spawn(function()
		local live = workspace:WaitForChild("Live", 30)
		if not live then
			return
		end

		defenseMaid:mark(Signal.new(live.ChildAdded):connect("Defense_LiveEntityAdded", trackLiveEntity))
		defenseMaid:mark(Signal.new(live.ChildRemoved):connect("Defense_LiveEntityRemoved", untrackLiveEntity))

		for _, entity in next, live:GetChildren() do
			trackLiveEntity(entity)
		end
	end))

	defenseMaid:mark(renderStepped:connect("Defense_UpdateVisualizations", updateVisualizations))
	defenseMaid:mark(renderStepped:connect("Defense_UpdateHistory", updateHistory))
	defenseMaid:mark(renderStepped:connect("Defense_UpdateGoldenTongue", updateGoldenTongue))
	defenseMaid:mark(renderStepped:connect("Defense_UpdateAutoArdour", updateAutoArdour))
	defenseMaid:mark(postSimulation:connect("Defense_UpdateDefenders", updateDefenders))
	defenseMaid:mark(postSimulation:connect("Defense_UpdateMantraRequests", updateMantraRequests))
	defenseMaid:mark(postSimulation:connect("Defense_UpdateAutoAirCounter", updateAutoAirCounter))

	if clientEffectEvent then
		defenseMaid:mark(clientEffectEvent:connect("Defense_ClientEffectEvent", onClientEffectEvent))
	end

	if clientEffectLargeEvent then
		defenseMaid:mark(clientEffectLargeEvent:connect("Defense_ClientEffectEventLarge", onClientEffectEvent))
	end

	if cientEffectDirect then
		defenseMaid:mark(cientEffectDirect:connect("Defense_ClientEffectEventDirect", onClientEffectEvent))
	end

	if spellEvent then
		defenseMaid:mark(spellEvent:connect("Defense_SpellEvent", onSpellEvent))
	end
	defenseMaid:mark(characterAdded:connect("Defense_BackstabCharacterAdded", onBackstabCharacterAdded))
	defenseMaid:mark(effectAddedSignal:connect("Defense_BackstabEffectAdded", onBackstabEffectAdded))
	defenseMaid:mark(effectRemovedSignal:connect("Defense_BackstabEffectRemoved", onBackstabEffectRemoved))

	onBackstabCharacterAdded(localPlayer.Character)

	-- Initial catch-up sweep for parts/sounds/animators that already exist. This walks the ENTIRE
	-- DataModel (tens of thousands of instances, most of them BaseParts that miss every timing) and
	-- was the single biggest chunk of startup (~425ms, fully blocking). Run it in the background with
	-- a ~4ms/frame budget instead: game.DescendantAdded (connected just above) already covers anything
	-- that appears while we sweep, so nothing is missed -- pre-existing static hazards just get their
	-- defender within a few frames instead of blocking the whole load.
	defenseMaid:mark(task.spawn(function()
		local descendants = game:GetDescendants()
		local budgetStart = os.clock()

		for index = 1, #descendants do
			onGameDescendantAdded(descendants[index])

			if os.clock() - budgetStart > 0.004 then
				runService.Heartbeat:Wait()
				budgetStart = os.clock()
			end
		end
	end))

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		Logger.warn("Defense initialized.")
	end
end

function Defense.detach()
	for _, object in next, defenderObjects do
		object:detach()
	end

	table.clear(defenderObjects)
	table.clear(defenderPartObjects)
	table.clear(defenderAnimationObjects)
	table.clear(deletedPlaybackData)

	defenseMaid:clean()

	Logger.warn("Defense detached.")
end

return Defense
end)
__bundle_register("Features/Combat/Objects/HitboxOptions", function(require, _LOADED, __bundle_register, __bundle_modules)
---@class HitboxOptions
---@note: Options for the hitbox check.
---@field part BasePart? If this is specified and it exists, it will be used for the position.
---@field cframe CFrame? Else, the part's CFrame will be used.
---@field timing Timing|AnimationTiming|SoundTiming
---@field action Action?
---@field cache Vector3 Cached hitbox size.
---@field filter Instance[]
---@field spredict boolean If true, a check will run for predicted positions.
---@field ptime number? The predicted time in seconds for extrapolation.
---@field entity Model? The entity for extrapolation.
---@field phcolor Color3 The color for predicted hitboxes.
---@field pmcolor Color3 The color for predicted missed hitboxes.
---@field hcolor Color3 The color for hitboxes.
---@field mcolor Color3 The color for missed hitboxes.
---@field hmid number? Hitbox visualization ID for normal hitbox check.
---@field visualize boolean Whether to visualize the hitbox.
local HitboxOptions = {}
HitboxOptions.__index = HitboxOptions

local players = game:GetService("Players")

---@return Color3
HitboxOptions.ghcolor = LPH_NO_VIRTUALIZE(function(self, result)
	return result and self.hcolor or self.mcolor
end)

---@return Color3
HitboxOptions.gphcolor = LPH_NO_VIRTUALIZE(function(self, result)
	return result and self.phcolor or self.pmcolor
end)

---@return HitboxOptions
HitboxOptions.clone = LPH_NO_VIRTUALIZE(function(self)
	local options = setmetatable({}, HitboxOptions)
	options.action = self.action
	options.spredict = self.spredict
	options.entity = self.entity
	options.ptime = self.ptime
	options.entity = self.entity
	options.phcolor = self.phcolor
	options.pmcolor = self.pmcolor
	options.hcolor = self.hcolor
	options.mcolor = self.mcolor
	options.hmid = self.hmid
	options.filter = self.filter
	options.part = self.part
	options.cframe = self.cframe
	options.timing = self.timing
	options.visualize = self.visualize
	return options
end)

HitboxOptions.ucache = LPH_NO_VIRTUALIZE(function(self)
	if not self.timing then
		return error("HitboxOptions:update - no timing specified")
	end

	local hitbox = self.action and self.action.hitbox or self.timing.hitbox

	if self.timing.duih then
		hitbox = self.timing.hitbox
	end

	if not hitbox then
		self.cache = Vector3.new(0, 0, 0)
	else
		self.cache = Vector3.new(hitbox.X, hitbox.Y, hitbox.Z)
	end
end)

---@return Vector3
HitboxOptions.hitbox = LPH_NO_VIRTUALIZE(function(self)
	if not self.cache then
		self:ucache()
	end

	return self.cache
end)

---@return CFrame
HitboxOptions.extrapolate = LPH_NO_VIRTUALIZE(function(self)
	if not self.part then
		return error("HitboxOptions.extrapolate - unimplemented for CFrame")
	end

	if not self.ptime then
		return error("HitboxOptions.extrapolate - no predicted time specified")
	end

	return self.part.CFrame + (self.part.AssemblyLinearVelocity * self.ptime)
end)

---@return CFrame
HitboxOptions.pos = LPH_NO_VIRTUALIZE(function(self)
	if self.cframe then
		return self.cframe
	end

	if self.part then
		return self.part.CFrame
	end

	return error("HitboxOptions.pos - impossible condition")
end)

---@param target Instance|CFrame
---@param timing Timing|AnimationTiming|SoundTiming
---@param filter Instance[]?
---@return HitboxOptions
HitboxOptions.new = LPH_NO_VIRTUALIZE(function(target, timing, filter)
	local self = setmetatable({}, HitboxOptions)
	self.part = typeof(target) == "Instance" and target:IsA("BasePart") and target
	self.cframe = typeof(target) == "CFrame" and target
	self.timing = timing
	self.action = nil
	self.filter = filter or {}
	self.cache = Vector3.zero
	self.spredict = false
	self.hmid = nil
	self.entity = nil
	self.phcolor = Color3.new(1, 0, 1)
	self.pmcolor = Color3.new(0.349019, 0.345098, 0.345098)
	self.hcolor = Color3.new(0, 1, 0)
	self.mcolor = Color3.new(1, 0, 0)
	self.ptime = nil
	self.visualize = true

	if not self.part and not self.cframe or not self.timing then
		return error("HitboxOptions: No part or CFrame or timing specified.")
	end

	self:ucache()

	if filter then
		return self
	end

	local character = players.LocalPlayer.Character
	if not character then
		return self
	end

	self.filter = { character }

	return self
end)

return HitboxOptions
end)
