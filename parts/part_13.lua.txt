local updateTalentSheet = LPH_NO_VIRTUALIZE(function(rframe)
	task.wait()

	local talentSheet = rframe:FindFirstChild("TalentSheet")
	local container = talentSheet and talentSheet:FindFirstChild("Container")
	local talentScroll = container and container:FindFirstChild("TalentScroll")
	if not talentScroll then
		return
	end

	local drinfo = Visuals.drinfo
	if not drinfo then
		return
	end

	---@type BuilderData
	local bdata = Visuals.bdata
	if not bdata then
		return
	end

	local talentFrameTemplate = nil
	local categoryGroupTemplate = nil
	for _, category in next, talentScroll:GetChildren() do
		if category:IsA("Frame") and category:FindFirstChild("Title") then
			if not categoryGroupTemplate then
				categoryGroupTemplate = category
			end

			for _, talent in next, category:GetChildren() do
				if talent:IsA("Frame") and talent:FindFirstChild("Title") then
					talentFrameTemplate = talent
					break
				end
			end

			if talentFrameTemplate then
				break
			end
		end
	end

	if not talentFrameTemplate or not categoryGroupTemplate then
		return
	end

	builderAssistanceMaid:clean()

	labelMap = {}

	local filteredTalents = table.clone(bdata.talents)

	local talentNameToIdx = {}
	for idx, talent in next, filteredTalents do
		local talentClean = stripTags(talent)
		talentNameToIdx[talentClean] = idx
		talentNameToIdx[talent] = idx
	end

	for _, instance in next, talentScroll:GetDescendants() do
		if not instance:IsA("TextLabel") then
			continue
		end

		if instance.Name ~= "Title" then
			continue
		end

		local idx = talentNameToIdx[instance.Text]

		if not idx then
			continue
		end

		buildAssistanceMap:add(instance, "TextColor3", Color3.fromRGB(9, 255, 0))

		local talent = filteredTalents[idx]
		talentNameToIdx[stripTags(talent)] = nil
		talentNameToIdx[talent] = nil
		filteredTalents[idx] = nil
	end

	local missingTalentGroup = InstanceWrapper.mark(builderAssistanceMaid, "missingTalentGroup", categoryGroupTemplate:Clone())
	for _, child in next, missingTalentGroup:GetChildren() do
		if child:IsA("Frame") then
			child:Destroy()
		end
	end

	missingTalentGroup.Name = "LMissingTalents"
	missingTalentGroup.LayoutOrder = 9990

	local missingTalentTitle = missingTalentGroup:FindFirstChild("Title")
	if missingTalentTitle then
		missingTalentTitle.Text = "MISSING"
		missingTalentTitle.TextColor3 = Color3.fromRGB(255, 0, 2)
	end

	local hasMissingTalents = false
	local talentOrder = 1
	for _, talent in next, filteredTalents do
		local data = bdata.ddata:get(talent)
		if not data then
			continue
		end

		hasMissingTalents = true

		local cleanTalent = stripTags(talent)

		local newFrame = InstanceWrapper.mark(builderAssistanceMaid, talent, talentFrameTemplate:Clone())
		local pshlocked = (bdata.ddata:possible(talent, bdata.pre) and not bdata.ddata:possible(talent, bdata.post))
		newFrame.Name = "M" .. cleanTalent
		newFrame.LayoutOrder = talentOrder
		talentOrder = talentOrder + 1

		local icon = newFrame:FindFirstChild("Icon")
		if icon then
			icon:Destroy()
		end

		local title = newFrame:FindFirstChild("Title")
		if title then
			title.Name = "M" .. cleanTalent
			title.Text = cleanTalent
			title.TextColor3 = pshlocked and Color3.fromRGB(255, 4, 255) or Color3.fromRGB(255, 0, 2)
			title.TextTransparency = 0.4
		end

		newFrame.Parent = missingTalentGroup

		labelMap["M" .. cleanTalent] = data
	end

	if hasMissingTalents then
		missingTalentGroup.Parent = talentScroll
	end

	end

	local hasMissingMantras = false
	local mantraOrder = 1
	for _, mantra in next, bdata.mantras do
		local data = bdata.ddata:get(mantra)
		if not data then
			continue
		end

		local cleanMantra = correctBuilderName(stripTags(mantra))

		local idx = Table.find(players.LocalPlayer.Backpack:GetChildren(), function(value, _)
			local displayName = value:GetAttribute("DisplayName")
			if not displayName then
				return false
			end
			local cleanDisplayName = stripTags(displayName)
			return cleanDisplayName == cleanMantra or displayName == cleanMantra
		end)

		hasMissingMantras = true

		local newFrame = InstanceWrapper.mark(builderAssistanceMaid, mantra, talentFrameTemplate:Clone())
		local pshlocked = (bdata.ddata:possible(mantra, bdata.pre) and not bdata.ddata:possible(mantra, bdata.post))
		newFrame.Name = "Z" .. cleanMantra
		newFrame.LayoutOrder = mantraOrder

		if icon then
			icon:Destroy()
		end

		local title = newFrame:FindFirstChild("Title")
		if title then
			title.Name = "Z" .. cleanMantra
			title.Text = cleanMantra
			title.TextColor3 = pshlocked and Color3.fromRGB(255, 4, 255) or Color3.fromRGB(255, 0, 2)
			title.TextTransparency = 0.4

			if idx then
				title.TextColor3 = Color3.fromRGB(9, 255, 0)
			end
		end

		newFrame.Parent = missingMantraGroup

		labelMap["Z" .. cleanMantra] = data
	end

	if hasMissingMantras then
		missingMantraGroup.Parent = talentScroll
	end

		if not train then
			continue
		end

		if not train.Visible then
			continue
		end

		local valueLabel = instance:FindFirstChild("Value")
		if not valueLabel then
			continue
		end

		local statInvested = tonumber(drinfo["Stat" .. instance.Name])
		if not statInvested then
			continue
		end

		local value = (bdata:ipre(drinfo) == 0 and statInvested <= 0) and statInvested or tonumber(valueLabel.Text)
		if not value then
			continue
		end

		local color = expectedValue ~= value and Color3.fromRGB(9, 136, 0) or Color3.fromRGB(127, 0, 2)

		if value > expectedValue then
			color = Color3.fromRGB(128, 128, 128)
		end

		buildAssistanceMap:add(train, "ImageColor3", color)
	end
end)

local function applyAutoFavour()
	local bdata = Visuals.bdata
	if not bdata then
		return
	end

	local drinfo = Visuals.drinfo
	if not drinfo then
		return
	end

	local requests = replicatedStorage:FindFirstChild("Requests")
	if not requests then
		return
	end

	local cards = requests:FindFirstChild("Cards")
	if not cards then
		return
	end

	local favourCard = cards:FindFirstChild("FavourCard")
	if not favourCard then
		return
	end

	local removeFavourRemote = cards:FindFirstChild("RemoveFavour")
	if not removeFavourRemote then
		return
	end

	local names = {}
	local buildSet = {}

	for _, talent in next, bdata.talents do
		local name = correctBuilderName(stripTags(talent))
		table.insert(names, name)
		buildSet[name] = true
	end

	for _, mantra in next, bdata.mantras do
		local displayName = correctBuilderName(stripTags(mantra))
		local internalName = mantraDisplayToInternal[displayName] or displayName
		table.insert(names, internalName)
		buildSet[internalName] = true
	end

	local cardsFavoured = drinfo.CardsFavoured
	local cardsForetold = drinfo.CardsForetold

	local favourNames = {}
	for _, name in next, names do
		if not (cardsFavoured and cardsFavoured[name]) then
			table.insert(favourNames, name)
		end
	end

	local unfavourNames = {}

	if cardsFavoured then
		for cardName, _ in next, cardsFavoured do
			if not buildSet[cardName] and not (cardsForetold and cardsForetold[cardName]) then
				table.insert(unfavourNames, cardName)
			end
		end
	end

	lastFavouredNames = names
	autoFavourApplied = true
	autoFavourBdata = bdata

	TaskSpawner.spawn("Visuals_AutoFavourCards", function()
		for _, name in next, favourNames do
			pcall(favourCard.InvokeServer, favourCard, name)
			task.wait(0.1)
		end

		for _, name in next, unfavourNames do
			pcall(removeFavourRemote.InvokeServer, removeFavourRemote, name)
			task.wait(0.1)
		end

		Logger.notify("Favoured %d cards, unfavoured %d non-build cards.", #favourNames, #unfavourNames)
	end)
end

local lastFilterLabels = {}

local updateFilterLabelOptions = LPH_NO_VIRTUALIZE(function()
	for identifier, group in next, groups do
		local option = Options and Options[Configuration.identify(identifier, "FilterLabelList")]
		if not option then
			continue
		end

		local seen = {}
		local labels = {}

		for label, _ in next, option.Value do
			seen[label] = true
			table.insert(labels, label)
		end

		for _, object in next, group:data() do
			local label = object.__type == "FilteredESP" and object.object.label
			if not label or seen[label] then
				continue
			end

			seen[label] = true
			table.insert(labels, label)
		end

		table.sort(labels)

		local key = table.concat(labels, "|")
		if lastFilterLabels[identifier] == key then
			continue
		end

		lastFilterLabels[identifier] = key

		option:SetValues(labels)
		option:SetValue(option.Value)
	end
end)

local onThrownChildAdded = LPH_NO_VIRTUALIZE(function(child)
	local name = child.Name

	if name == "MinistryCacheIndicator" then
		return emplaceObject(child, PartESP.new("MinistryCacheIndicator", child, "Ministry Cache Indicator"))
	end

	if name == "BigArtifact" and child:IsA("Model") then
		return emplaceObject(child, ModelESP.new("Artifact", child, "Artifact"))
	end

	if name == "BellMeteor" then
		return emplaceObject(child, ModelESP.new("BellMeteor", child, "Bell Meteor"))
	end

	if name == "ExplodeCrate" then
		return emplaceObject(child, PartESP.new("ExplosiveBarrel", child, "Explosive Barrel"))
	end

	if name == "BagDrop" then
		return emplaceObject(child, PartESP.new("BagDrop", child, "Bag"))
	end

	if name == "EventFeatherRef" then
		return emplaceObject(child, PartESP.new("OwlFeathers", child, "Owl Feathers"))
	end

	if name == "BoneSpear" then
		return emplaceObject(child, PartESP.new("BoneSpear", child, "Bone Spear"))
	end

	if name == "Chest" and child:GetAttribute("LootName") ~= nil then
		return emplaceObject(child, ChestESP.new("Chest", child, "Chest"))
	end

	if not child:IsA("Model") and not child:IsA("Part") then
		return
	end

	if child:FindFirstChild("LootUpdated") then
		return emplaceObject(child, ChestESP.new("Chest", child, "Chest"))
	end

	-- Project Rain's chest watcher logic (its InstanceWatcher on workspace.Thrown, src/features/visuals/
	-- base_esp.lua): a child is a chest if it is named "Chest" OR a "Lid" child appears within 2.5s, re-checked
	-- once after 0.1s. GATE on top (the one thing PR doesn't do): only engage for children named "Chest". PR
	-- runs that inline WaitForChild("Lid", 2.5) for EVERY non-chest Thrown child, and on this low-FPS Saramed
	-- farm the Thrown folder gets spammed (that was the x1233 Visuals_ChestCheck pile-up), so without the gate
	-- PR's version re-leaks here. Named chests still match instantly; PR's structure is kept intact behind it.
	if child.Name ~= "Chest" then
		return
	end
	local function isChest()
		return child.Name == "Chest" or child:WaitForChild("Lid", 2.5)
	end
	if isChest() then
		return emplaceObject(child, ChestESP.new("Chest", child, "Chest"))
	end
	task.delay(0.1, function()
		if child.Parent and isChest() then
			emplaceObject(child, ChestESP.new("Chest", child, "Chest"))
		end
	end)
end)

---@param instance Instance
---@param identifier string
---@param addedCallback function
---@param removingCallback function
---@param childFlag boolean
local createListener = LPH_NO_VIRTUALIZE(function(instance, identifier, addedCallback, removingCallback, childFlag)
	local type = childFlag and "Child" or "Descendant"
	local added = Signal.new(childFlag and instance.ChildAdded or instance.DescendantAdded)
	local removed = Signal.new(childFlag and instance.ChildRemoved or instance.DescendantRemoving)

	visualsMaid:add(added:connect(string.format("Visuals_%sOn%sAdded", identifier, type), addedCallback))
	visualsMaid:add(removed:connect(string.format("Visuals_%sOn%sRemoved", identifier, type), removingCallback))

	Profiler.run(string.format("Visuals_%sAddInitial", identifier), function()
		for _, child in next, (childFlag and instance:GetChildren() or instance:GetDescendants()) do
			addedCallback(child)
		end
	end)
end)

---@param player Player
local onPlayerAdded = LPH_NO_VIRTUALIZE(function(player)
	if player == players.LocalPlayer then
		return
	end

	local characterAdded = Signal.new(player.CharacterAdded)
	local characterRemoving = Signal.new(player.CharacterRemoving)
	local playerDestroying = Signal.new(player.Destroying)

	local characterAddedId = nil
	local characterRemovingId = nil
	local playerDestroyingId = nil

	characterAddedId = visualsMaid:add(characterAdded:connect("Visuals_OnCharacterAdded", function(character)
		emplaceObject(player, PlayerESP.new("Player", player, character))
	end))

	characterRemovingId = visualsMaid:add(characterRemoving:connect("Visuals_OnCharacterRemoving", function()
		onInstanceRemoving(player)
	end))

	playerDestroyingId = visualsMaid:add(playerDestroying:connect("Visuals_OnPlayerDestroying", function()
		visualsMaid[characterAddedId] = nil
		visualsMaid[characterRemovingId] = nil
		visualsMaid[playerDestroyingId] = nil
	end))

	local character = player.Character
	if not character then
		return
	end

	emplaceObject(player, PlayerESP.new("Player", player, character))
end)

function Visuals.init()
	local WAIT = 30

	local live = workspace:WaitForChild("Live", WAIT)
	local npcs = workspace:WaitForChild("NPCs", WAIT)
	local ingredients = workspace:WaitForChild("Ingredients", WAIT)
	local thrown = workspace:WaitForChild("Thrown", WAIT)
	local terrain = workspace:WaitForChild("Terrain", WAIT)
	local shops = workspace:WaitForChild("Shops", WAIT)

	if terrain then
		createListener(terrain, "Terrain", onTerrainChildAdded, onInstanceRemoving, true)
	end

	createListener(workspace, "Workspace", onWorkspaceChildAdded, onInstanceRemoving, true)

	if thrown then
		createListener(thrown, "Thrown", onThrownChildAdded, onInstanceRemoving, true)
	end

	if live then
		createListener(live, "Live", onLiveChildrenAdded, onInstanceRemoving, true)
	end

	if npcs then
		createListener(npcs, "NPCs", onNPCsChildAdded, onInstanceRemoving, true)
	end

	if ingredients then
		createListener(ingredients, "Ingredients", onIngredientsChildAdded, onInstanceRemoving, true)
	end

	createListener(players, "Players", onPlayerAdded, onInstanceRemoving, true)

	if shops then
		createListener(shops, "Shops", onShopChildAdded, onInstanceRemoving, true)
	end

	---@note: We only need to get this once.
	local markerWorkspace = replicatedStorage:WaitForChild("MarkerWorkspace", WAIT)

	for _, descendant in next, markerWorkspace and markerWorkspace:GetDescendants() or {} do
		if descendant.Name ~= "AreaMarker" then
			continue
		end

		local areaMarkerName = descendant.Parent.Name or "Unidentified Area Marker"
		emplaceObject(descendant, FilteredESP.new(PartESP.new("AreaMarker", descendant, areaMarkerName)))
	end

	local localPlayer = players.LocalPlayer
	local playerGui = localPlayer:WaitForChild("PlayerGui", WAIT)
	if not playerGui then
		return
	end

	local playerGuiDescendantAdded = Signal.new(playerGui.DescendantAdded)
	local playerGuiDescendantRemoving = Signal.new(playerGui.DescendantRemoving)

	visualsMaid:mark(TaskSpawner.spawn("Visuals_UIVanityWait", function()
		local uiVanity = playerGui:WaitForChild("UIVanity", 10)

		if uiVanity then
			uiVanity.Enabled = false
			task.wait()
			uiVanity.Enabled = true
		end

		visualsMaid:add(playerGuiDescendantAdded:connect("Visuals_OnPlayerGuiDescendantAdded", onPlayerGuiDescendantAdded))
		visualsMaid:add(
			playerGuiDescendantRemoving:connect("Visuals_OnPlayerGuiDescendantRemoving", onPlayerGuiDescendantRemoving)
		)
		visualsMaid:add(renderStepped:connect("Visuals_RenderStepped", updateVisuals))

		for _, descendant in next, playerGui:GetDescendants() do
			onPlayerGuiDescendantAdded(descendant)
		end
	end))

	local info = replicatedStorage:WaitForChild("Info", WAIT)
	local dataReplication = info and info:WaitForChild("DataReplication", WAIT)
	local dataReplicationModule = dataReplication and require(dataReplication)

	local success, drinfo = pcall(function()
		return dataReplicationModule.GetData()
	end)

	if success and type(drinfo) == "table" then
		Visuals.drinfo = drinfo
	else
		Logger.warn("GetData() failed, reading stats directly from character...")

		local character = localPlayer and localPlayer.Character
		local humanoid = character and character:FindFirstChildOfClass("Humanoid")

		if humanoid then
			local fallbackData = {}
			local statNames = {
				"Agility",
				"Strength",
				"Fortitude",
				"Intelligence",
				"Willpower",
				"Charisma",
				"ElementBlood",
				"ElementFire",
				"ElementIce",
				"ElementLightning",
				"ElementWind",
				"ElementShadow",
				"ElementMetal",
				"WeaponHeavy",
				"WeaponMedium",
				"WeaponLight",
			}

			for _, statName in ipairs(statNames) do
				local key = "Stat" .. statName
				fallbackData[key] = humanoid:GetAttribute(key) or character:GetAttribute(key) or 0
			end

			fallbackData["TraitHealth"] = humanoid:GetAttribute("TraitHealth") or character:GetAttribute("TraitHealth") or 0
			fallbackData["TraitEther"] = humanoid:GetAttribute("TraitEther") or character:GetAttribute("TraitEther") or 0
			fallbackData["TraitWeaponDamage"] = humanoid:GetAttribute("TraitWeaponDamage")
				or character:GetAttribute("TraitWeaponDamage")
				or 0
			fallbackData["TraitMantraDamage"] = humanoid:GetAttribute("TraitMantraDamage")
				or character:GetAttribute("TraitMantraDamage")
				or 0

			Visuals.drinfo = fallbackData
			Logger.warn("Successfully read stats from character attributes (fallback mode).")
		else
			Visuals.drinfo = nil
			Logger.warn("Could not read character attributes - BuildAssistance disabled.")
		end
	end

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		Logger.warn("Visuals initialized.")
	end
end

function Visuals.detach()
	for _, group in next, groups do
		group:detach()
	end

	mysteryRevealMap:restore()
	autoFavourApplied = false
	autoFavourBdata = nil
	lastFavouredNames = {}
	dismissRoll2()
	roll2Notified = false
	cachedRoll2Selection = nil

	visualsMaid:clean()
	builderAssistanceMaid:clean()

	Logger.warn("Visuals detached.")
end

return Visuals

InstanceESP.build = LPH_NO_VIRTUALIZE(function(self, label, tags)
	if #tags <= 0 then
		return label
	end

	local start = true

	for _, tag in next, tags do
		local line = lines[#lines] or label

		if not start and #line > Configuration.expectOptionValue("ESPSplitLineLength") then
			lines[#lines + 1] = tag
			continue
		end

		line = line .. " " .. tag

		lines[start and 1 or #lines] = line

	end

---@module Utility.ReferencedMap
local ReferencedMap = require("Utility/ReferencedMap")

---@module Utility.Profiler
local Profiler = require("Utility/Profiler")

---@module Utility.Logger
local Logger = require("Utility/Logger")

---@module Utility.Configuration
local Configuration = require("Utility/Configuration")

---@class Group: ReferencedMap
---@field part number
---@field icount number
---@field updated boolean
---@field identifier string
local Group = setmetatable({}, ReferencedMap)
Group.__index = Group

---@param self Group
---@param object ModelESP|PartESP|FilteredESP
Group.object = LPH_NO_VIRTUALIZE(function(self, object)
	self.count = self.count + 1

	if not self.warned and self.count >= 500 then
		Logger.longNotify("(%s) Too many objects will cause your elements to stop updating.", object.identifier)

		self.warned = true
	end

	---@note: If we're updating too many objects, it will cause Roblox to hide UI elements and kick us from the game.
	if self.count >= 500 then
		return
	end

	Profiler.run(self.updateLabel, object.update, object)
end)

---@param self Group
Group.update = LPH_NO_VIRTUALIZE(function(self)
	local map = self:data()

	if not Configuration.idToggleValue(self.identifier, "Enable") then
		return self:hide()
	end

	if Configuration.expectToggleValue("ESPSplitUpdates") then
		local totalElements = #map
		local totalFrames = Configuration.expectOptionValue("ESPSplitFrames")

		local objectsPerPart = math.ceil(totalElements / totalFrames)
		local currentPart = self.part

		local startIdx = (currentPart - 1) * objectsPerPart + 1
		local endIdx = math.min(currentPart * objectsPerPart, totalElements)

		for idx = startIdx, endIdx do
			self:object(map[idx])
		end

		self.part = self.part + 1

		if self.part > totalFrames then
			self.count = 0
			self.part = 1
		end
	else
		for _, object in next, map do
			self:object(object)
		end

		self.part = 1
		self.count = 0
	end

	self.updated = true
end)

---@param self Group
Group.hide = LPH_NO_VIRTUALIZE(function(self)
	if not self.updated then
		return
	end

	for _, object in next, self:data() do
		object:visible(false)
	end

	self.updated = false
end)

---@param self Group
Group.detach = LPH_NO_VIRTUALIZE(function(self)
	for _, object in next, self:data() do
		object:detach()
	end
end)

---@param identifier string
---@return Group
function Group.new(identifier)
	local self = setmetatable(ReferencedMap.new(), Group)
	self.part = 1
	self.count = 0
	self.warned = false
	self.updated = true
	self.identifier = identifier
	self.updateLabel = string.format("ESP_Update_%s", identifier)
	return self

__bundle_register("Utility/ReferencedMap", function(require, _LOADED, __bundle_register, __bundle_modules)
return LPH_NO_VIRTUALIZE(function()
	---@class ReferencedMap
	---@field _map table
	---@field _references table
	local ReferencedMap = {}
	ReferencedMap.__index = ReferencedMap

	---@param ref any
	---@param element any
	function ReferencedMap:insert(ref, element)
		local key = #self._map + 1
		self._map[key] = element
		self._references[ref] = element
	end

	---@param ref any
	---@return any?
	function ReferencedMap:remove(ref)
		local element = self._references[ref]
		if not element then
			return nil
		end

		self._references[ref] = nil

		local position = table.find(self._map, element)
		if not position then
			return nil
		end

		table.remove(self._map, position)

		return element
	end

	---@return number
	function ReferencedMap:size()
		return #self._map
	end

	---@return table
	function ReferencedMap:data()
		return self._map
	end

	---@return ReferencedMap
	function ReferencedMap.new()
		local self = setmetatable({}, ReferencedMap)
		self._map = {}
		self._references = {}
		return self
	end

	return ReferencedMap
end)()

EntityESP.btext = LPH_NO_VIRTUALIZE(function(self, label, tags)
	if #tags <= 0 then
		return label
	end

	local lines = {}
	local start = true

	for _, tag in next, tags do
		local line = lines[#lines] or label

		if not start and #line > Configuration.expectOptionValue("ESPSplitLineLength") then
			lines[#lines + 1] = tag
			continue
		end

		line = line .. " " .. tag

		lines[start and 1 or #lines] = line

		start = false
	end

	return table.concat(lines, "\n"), #lines
end)

---@param self PlayerESP
EntityESP.hchanged = LPH_NO_VIRTUALIZE(function(self)
	if not self.lhealth then
		return
	end

	if self.billboard.Enabled == false then
		return
	end

	if not Configuration.idToggleValue(self.identifier, "ShowHealthChanges") then
		return
	end

	local character = self.entity
	if not character or not character:IsA("Model") then
		return
	end

	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then
		return
	end

	local root = character:FindFirstChild("HumanoidRootPart")
	if not root then
		return
	end

	local debris = replicatedStorage:FindFirstChild("Debris")
	if not debris then
		return
	end

	local newHealth = humanoid.Health
	local differenceColor = Color3.fromRGB(255, 0, 0)
	local differenceAmount = newHealth - self.lhealth

	if differenceAmount >= 0 then
		differenceColor = Color3.fromRGB(0, 255, 0)
	end

	self.lhealth = newHealth

	if differenceAmount >= -0.5 and differenceAmount <= 0.5 then
		return
	end

	local clientEffectModules = replicatedStorage:FindFirstChild("ClientEffectModules")
	local replication = clientEffectModules and clientEffectModules:FindFirstChild("Replication")
	local replicationModule = replication and replication:FindFirstChild("Replication")
	if not replicationModule then
		return
	end

	local damageSplash = replicationModule:FindFirstChild("DamageSplash")
	if not damageSplash then
		return
	end

	local thrown = workspace:FindFirstChild("Thrown")
	if not thrown then
		return
	end

	local initialOffset = Vector3.new(math.random() * 2 - 1, math.random() * 2 - 1, math.random() * 2 - 1) * 2
	local clonedSplash = damageSplash:Clone()
	clonedSplash.Adornee = root
	clonedSplash.Size = UDim2.new(2.5, 0, 0.5, 0)
	clonedSplash.Parent = thrown
	clonedSplash.TextLabel.Text = string.format("%.02f", differenceAmount)
	clonedSplash.StudsOffsetWorldSpace = initialOffset

	if differenceColor then
		clonedSplash.TextLabel.TextColor3 = differenceColor
	end

	local textLabel = clonedSplash and clonedSplash:FindFirstChild("TextLabel")
	if not textLabel then
		return
	end

	local uiStroke = textLabel:FindFirstChild("UIStroke")
	if not uiStroke then
		return
	end

	uiStroke.Transparency = 1.0

	tweenService
		:Create(clonedSplash, TweenInfo.new(0.5, Enum.EasingStyle.Back), {
			Size = UDim2.new(5, 0, 1),
		})
		:Play()

	tweenService
		:Create(clonedSplash, TweenInfo.new(5, Enum.EasingStyle.Quad, Enum.EasingDirection.In), {
			StudsOffsetWorldSpace = initialOffset + Vector3.new(0, 10, 0),
		})
		:Play()

	self.maid:mark(TaskSpawner.delay("PlayerESP_RemoveDamageSplash", function()
		return 3.0
	end, function()
		tweenService
			:Create(textLabel, TweenInfo.new(2), {
				TextTransparency = 1,
			})
			:Play()
	end))

	debris:Fire(clonedSplash, 5)
end)

---@param self InstanceESP
---@param tags string[]
EntityESP.update = LPH_NO_VIRTUALIZE(function(self, tags)
	local identifier = self.identifier

	local localPlayer = playersService.LocalPlayer
	local localCharacter = localPlayer and localPlayer.Character
	local localRoot = localCharacter and localCharacter:FindFirstChild("HumanoidRootPart")

	local entity = self.entity
	local entityHumanoid = self._cachedHumanoid
	local entityRoot = self._cachedRoot

	if not entityHumanoid or not entityHumanoid.Parent then
		entityHumanoid = entity and entity:FindFirstChildOfClass("Humanoid")
		self._cachedHumanoid = entityHumanoid
	end

	if not entityRoot or not entityRoot.Parent then
		entityRoot = entity and entity:FindFirstChild("HumanoidRootPart")
		self._cachedRoot = entityRoot
	end

	local position = entityRoot and entityRoot.Position

	if not Configuration.idToggleValue(identifier, "Enable") then
		return self:visible(false)
	end

	if not entityHumanoid then
		return self:visible(false)
	end

	if not localRoot then
		return self:visible(false)
	end

	if not position then
		return self:visible(false)
	end

	local distance = (localRoot.Position - position).Magnitude

	if distance > Configuration.idOptionValue(identifier, "MaxDistance") then
		return self:visible(false)
	end

	---@todo: This was a lazy way to connect to the signal.
	if not self.maid["PlayerESP_OnHealthChanged"] and not self.lhealth then
		local onHealthChanged = Signal.new(entityHumanoid.HealthChanged)

		self.lhealth = entityHumanoid.Health

		self.maid["PlayerESP_OnHealthChanged"] = onHealthChanged:connect("PlayerESP_OnHealthChanged", function()
			self:hchanged()
		end)
	end

	self.billboard.Adornee = entityRoot

	local bbstroke = self.bbstroke
	local wbstroke = self.wbstroke
	local dcontainer = self.dcontainer
	local ncontainer = self.ncontainer
	local hbar = self.hbar

	bbstroke.Visible = Configuration.idToggleValue(identifier, "BoundingBox")
	wbstroke.Visible = Configuration.idToggleValue(identifier, "BoundingBox")
	dcontainer.Visible = Configuration.idToggleValue(identifier, "ShowDistance")
	hbar.Visible = Configuration.idToggleValue(identifier, "HealthBar")

	local fontSize = Configuration.expectOptionValue("FontSize") or 13
	local text, lines = self:btext(self.label, tags)
	self:utext(ncontainer, text)

	local name = self._cachedNameElement
	if not name then
		name = self:find("Name")
		self._cachedNameElement = name
	end
	if name then
		name.space = (lines * fontSize) + (ELEMENT_PADDING * 2)
	end

	local delement = self._cachedDistElement
	if not delement then
		delement = self:find("Distance")
		self._cachedDistElement = delement
	end
	if delement then
		delement.space = fontSize + (ELEMENT_PADDING * 2)
	end

	if dcontainer then
		self:utext(dcontainer, string.format("%im", math.floor(distance)))
	end

	local bar = self.gb(hbar)

	if hbar and bar then
		local percent = entityHumanoid.Health / entityHumanoid.MaxHealth
		local fullColor = Configuration.idOptionValue(identifier, "FullColor")
		local emptyColor = Configuration.idOptionValue(identifier, "EmptyColor")

		self.mbs(hbar, true, percent)

		bar.BackgroundColor3 = emptyColor:Lerp(fullColor, math.clamp(percent, 0.0, 1.0))

		self:useperators(distance)
	end

	self:build()

	self:visible(true)
end)

EntityESP.build = LPH_NO_VIRTUALIZE(function(self)
	local sideOffsets = { top = 0, bottom = 0, left = 0, right = 0 }

	for side, elementList in next, self.elements do
		local offsetElementCount = 0

		for _, item in next, elementList do
			if not item.container.Visible then
				continue
			end

			sideOffsets[side] = sideOffsets[side] + item.space + ELEMENT_PADDING
			offsetElementCount = offsetElementCount + 1
		end

		if offsetElementCount > 0 then
			sideOffsets[side] = sideOffsets[side] - ELEMENT_PADDING
		end
	end

	local maxHorizontalPadding = math.max(sideOffsets.left, sideOffsets.right)
	local maxVerticalPadding = math.max(sideOffsets.top, sideOffsets.bottom)

	local now = os.clock()
	if not self._lastExtentsTime or (now - self._lastExtentsTime) >= 1.0 then
		local extentsSize = self.entity:GetExtentsSize()
		self._lastExtentsTime = now

		if self.lextents and math.abs(self.lextents.Magnitude - extentsSize.Magnitude) >= 10.0 then
			self.sextents = nil
			self._cachedBoneParts = nil
		end

		self.lextents = extentsSize
	end

	if not self.sextents then
		local parts = self._cachedBoneParts
		if not parts then
			parts = {}
			for _, inst in next, self.entity:GetDescendants() do
				if inst:IsA("BasePart") then
					if not inst.Parent:IsA("BasePart") and (inst:FindFirstChildWhichIsA("Weld") or inst:FindFirstChildWhichIsA("Motor6D") or inst.Name == "HumanoidRootPart") then
						parts[#parts + 1] = inst
					end
				end
			end
			self._cachedBoneParts = parts
		end

		local minX, minY, minZ = math.huge, math.huge, math.huge
		local maxX, maxY, maxZ = -math.huge, -math.huge, -math.huge
		local found = false

		for _, inst in next, parts do
			if inst.Parent then
				local size = inst.Size
				local pos = inst.CFrame.Position
				minX = math.min(minX, pos.X - size.X / 2)
				minY = math.min(minY, pos.Y - size.Y / 2)
				minZ = math.min(minZ, pos.Z - size.Z / 2)
				maxX = math.max(maxX, pos.X + size.X / 2)
				maxY = math.max(maxY, pos.Y + size.Y / 2)
				maxZ = math.max(maxZ, pos.Z + size.Z / 2)
				found = true
			end
		end

		if found then
			self.sextents = Vector3.new(maxX - minX, maxY - minY, maxZ - minZ)
		else
			self.sextents = self.lextents or self.entity:GetExtentsSize()
		end
	end

	self.billboard.Size = UDim2.new(
		self.sextents.X + 1.5,
		maxHorizontalPadding * 2 + BILLBOARD_MIN_WIDTH,
		self.sextents.Y + 1.5,
		maxVerticalPadding * 2 + BILLBOARD_MIN_HEIGHT
	)

	self.bbox.Position = UDim2.new(0, maxHorizontalPadding, 0, maxVerticalPadding)
	self.bbox.Size = UDim2.new(1, -(maxHorizontalPadding * 2), 1, -(maxVerticalPadding * 2))

	local currentTopOffset = 0

	for _, element in next, self.elements.top do
		if not element.container.Visible then
			continue
		end

		if element.container.Parent ~= self.canvas then
			element.container.Parent = self.canvas
		end
		element.container.AnchorPoint = Vector2.new(0, 1)
		element.container.Position = UDim2.new(0, maxHorizontalPadding, 0, maxVerticalPadding - currentTopOffset)
		element.container.Size = UDim2.new(1, -(maxHorizontalPadding * 2), 0, element.space)
		currentTopOffset = currentTopOffset + element.space + ELEMENT_PADDING

		if element.created then
			continue
		end

		if not element.create then
			continue
		end

		element.create(element.container)
		element.created = true
	end

	local currentBottomOffset = 0

	for _, element in next, self.elements.bottom do
		if not element.container.Visible then
			continue
		end

		if element.container.Parent ~= self.bbox then
			element.container.Parent = self.bbox
		end
		element.container.AnchorPoint = Vector2.new(0, 0)
		element.container.Position = UDim2.new(0, 0, 1, currentBottomOffset)
		element.container.Size = UDim2.new(1, 0, 0, element.space)
		currentBottomOffset = currentBottomOffset + element.space + ELEMENT_PADDING

		if element.created then
			continue
		end

		if not element.create then
			continue
		end

		element.create(element.container)
		element.created = true
	end

	local currentLeftOffset = 0

	for _, element in next, self.elements.left do
		if not element.container.Visible then
			continue
		end

		if element.container.Parent ~= self.canvas then
			element.container.Parent = self.canvas
		end
		element.container.AnchorPoint = Vector2.new(1, 0)
		element.container.Position = UDim2.new(0, maxHorizontalPadding - currentLeftOffset, 0, maxVerticalPadding)
		element.container.Size = UDim2.new(0, element.space, 1, -(maxVerticalPadding * 2))
		currentLeftOffset = currentLeftOffset + element.space + ELEMENT_PADDING

		if element.created then
			continue
		end

		if not element.create then
			continue
		end

		element.create(element.container)
		element.created = true
	end

	local currentRightOffset = 0

	for _, element in next, self.elements.right do
		if not element.container.Visible then
			continue
		end

		if element.container.Parent ~= self.bbox then
			element.container.Parent = self.bbox
		end
		element.container.AnchorPoint = Vector2.new(0, 0)
		element.container.Position = UDim2.new(1, currentRightOffset, 0, 0)
		element.container.Size = UDim2.new(0, element.space, 1, 0)
		currentRightOffset = currentRightOffset + element.space + ELEMENT_PADDING

		if element.created then
			continue
		end

		if not element.create then
			continue
		end

		element.create(element.container)
		element.created = true
	end
end)

				groundSensor.UpdateType = Enum.SensorUpdateType.OnRead
			else
				groundSensor.SensorMode = Enum.SensorMode.Floor
				groundSensor.UpdateType = Enum.SensorUpdateType.Manual
			end
		end

		-- Uncap speed while flying.
		if effectReplicatorModule and not effectReplicatorModule:FindEffect("OverrideSpeedCap") then
			pcall(function()
				effectReplicatorModule:CreateEffect("OverrideSpeedCap")
			end)
		end
	end

	local lastMomentumUpdate = 0
	local function updateMaxMomentumSpoofer()
		if tick() - lastMomentumUpdate < 0.1 then
			return
		end
		lastMomentumUpdate = tick()

		local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
		local effectReplicatorModule = effectReplicator and require(effectReplicator)
		if not effectReplicatorModule then
			return
		end

		local forcedMomentumEffect = effectReplicatorModule:FindEffect("ForceMomentum")

		if not Configuration.expectToggleValue("MaxMomentumSpoof") then
			if forcedMomentumEffect then
				forcedMomentumEffect.Disabled = true
				forcedMomentumEffect.Value = 0
			end
			return
		end

		if not forcedMomentumEffect then
			return effectReplicatorModule:CreateEffect("ForceMomentum", { Value = 10 })
		end

		if forcedMomentumEffect.Value ~= 10 then
			forcedMomentumEffect.Value = 10
		end

		if forcedMomentumEffect.Disabled ~= false then
			forcedMomentumEffect.Disabled = false
		end
	end

	---@note: The client recomputes WalkSpeed from effects every frame, so a
	local function updateForceSpeedBuff()
		local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
		local effectReplicatorModule = effectReplicator and require(effectReplicator)
		if not effectReplicatorModule then
			return
		end

		local forceSpeedEffect = effectReplicatorModule:FindEffect("ForceSpeedBuff")

		if not Configuration.expectToggleValue("ForceSpeedBuff") then
			if forceSpeedEffect then
				forceSpeedEffect.Value = 0
				forceSpeedEffect.Disabled = true
			end
			return
		end

		if not forceSpeedEffect then
			return effectReplicatorModule:CreateEffect("ForceSpeedBuff")
		end

		local amount = Configuration.expectOptionValue("ForceSpeedBuffAmount") or 0.0
		if forceSpeedEffect.Value ~= amount then
			forceSpeedEffect.Value = amount
		end

		forceSpeedEffect.Disabled = false
	end

	---@param character Model
	local function updateAgilitySpoofer(character)
		local agility = character:FindFirstChild("PassiveAgility")
		if not agility then
			return
		end

		---@note: For every 10 investment points, there are two real agility points.
		local agilitySpoofValue = 8 + (Options.AgilitySpoof.Value / 10) * 2

		if Toggles.BoostAgilityDirectly.Value then
			agilitySpoofValue = Options.AgilitySpoof.Value
		end

		agilitySpoofer:set(agility, "Value", agilitySpoofValue)
	end

	local function updateTweenToBack()
		local validTargets = Finder.geir(300, Configuration.expectToggleValue("AttachIgnorePlayers"))
		if not validTargets then
			return true
		end

		local attachTarget = Configuration.expectToggleValue("StickyAttach") and cachedTarget or validTargets[1]
		if not attachTarget then
			return true
		end

		if not attachTarget.Parent then
			return false
		end

		local attachTargetHrp = attachTarget:FindFirstChild("HumanoidRootPart")
		if not attachTargetHrp then
			return false
		end

		local attachTargetHumanoid = attachTarget:FindFirstChild("Humanoid")
		if not attachTargetHumanoid then
			return false
		end

		if attachTargetHumanoid.Health <= 0 then
			return false
		end

		cachedTarget = cachedTarget or attachTarget

		local offsetCFrame = CFrame.new(
			0.0,
			Configuration.expectOptionValue("HeightOffset"),
			Configuration.expectOptionValue("BackOffset")
		)

		local targetPosition = (attachTargetHrp.CFrame * offsetCFrame).Position
		local goalCFrame = CFrame.lookAt(targetPosition, attachTargetHrp.Position)

		Tweening.goal("TweenToBack", goalCFrame, false)

		return true
	end

	-- Infinite zoom out: crank the local player's max camera zoom while enabled, restore it when off.
	local function updateInfiniteZoom()
		local localPlayer = players.LocalPlayer
		if not localPlayer then
			return
		end

		if Configuration.expectToggleValue("InfiniteZoom") then
			if originalMaxZoom == nil then
				originalMaxZoom = localPlayer.CameraMaxZoomDistance
			end
			-- Re-assert every frame; the game resets this on some state changes.
			if localPlayer.CameraMaxZoomDistance ~= 1000000 then
				localPlayer.CameraMaxZoomDistance = 1000000
			end
		elseif originalMaxZoom ~= nil then
			localPlayer.CameraMaxZoomDistance = originalMaxZoom
			originalMaxZoom = nil
		end
	end

	-- Freecam: Project Rain's freecam module. Detaches the camera (WASD move, Q/E down/up, mouse
	-- look, Up/Down adjust speed, LeftShift slow). The character stays put. The module runs its own
	-- BindToRenderStep, so this just enters/exits on the toggle and feeds the speed slider in.
	local function updateFreecam()
		if Configuration.expectToggleValue("Freecam") then
			if not Freecam.running() then
				Freecam.enter()
			end

			Freecam.set_speed((Configuration.expectOptionValue("FreecamSpeed") or 60) / 60)
		elseif Freecam.running() then
			Freecam.exit()
		end
	end

	---@param dt number
	local function updateMovement(dt)
		updateInfiniteZoom()
		updateFreecam(dt)

		local localPlayer = players.LocalPlayer
		local character = localPlayer.Character
		if not character then
			return
		end

		local rootPart = character:FindFirstChild("HumanoidRootPart")
		if not rootPart then
			return
		end

		local humanoid = character:FindFirstChild("Humanoid")
		if not humanoid then
			return
		end

		if not Configuration.expectToggleValue("TweenToBack") or not updateTweenToBack() then
			cachedTarget = nil
		end

		if Configuration.expectToggleValue("AgilitySpoof") then
			updateAgilitySpoofer(character)
		else
			agilitySpoofer:restore()
		end

		updateMaxMomentumSpoofer()

		updateForceSpeedBuff()

		if Tweening.active or Configuration.expectToggleValue("NoClip") then
			updateNoClip(character, rootPart)
			updateNoClipHasAnyHook(true)
		else
			noClipMap:restore()
			updateNoClipHasAnyHook(false)
		end

		if Tweening.active then
			return
		end

		if Configuration.expectToggleValue("Fly") then
			flyActive = true
			updateFlyHack(character, rootPart, humanoid)
		else
			movementMaid["flyBodyVelocity"] = nil
			if flyActive then
				flyActive = false
				-- Restore normal movement: hand control back to the ground controller, reset the ground
				-- sensor, and drop the speed-cap override the fly loop created.
				local controllerManager = character and character:FindFirstChild("ControllerManager")
				if controllerManager then
					local groundController = controllerManager:FindFirstChild("GroundController")
					if groundController then
						controllerManager.ActiveController = groundController
					end
				end
				local groundSensor = rootPart and rootPart:FindFirstChild("GroundSensor")
				if groundSensor then
					groundSensor.SensorMode = Enum.SensorMode.Floor
					groundSensor.UpdateType = Enum.SensorUpdateType.Manual
				end
				local effectReplicator = replicatedStorage:FindFirstChild("EffectReplicator")
				local effectReplicatorModule = effectReplicator and require(effectReplicator)
				local capEffect = effectReplicatorModule and effectReplicatorModule:FindEffect("OverrideSpeedCap")
				if capEffect then
					pcall(function()
						capEffect:Debris(1)
					end)
				end
			end
		end
		
		if Configuration.expectToggleValue("BoatSpeed") then
			updateBoatfly(rootPart, humanoid)
		end

		if Configuration.expectToggleValue("Speedhack") then
			updateSpeedHack(rootPart, humanoid)
		end

		-- CFrame Speed is standalone and can be used independently of Speedhack.
		if Configuration.expectToggleValue("CFrameSpeed") then
			updateCFrameSpeed(rootPart, humanoid, dt)
		end

		if Configuration.expectToggleValue("InfiniteJump") then
			updateInfiniteJump(rootPart)
		end
	end

	function Movement.init()
		movementMaid:add(preSimulation:connect("Movement_PreSimulation", updateMovement))
		movementMaid:add(heartbeat:connect("Movement_Heartbeat", Tweening.update))

		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("Movement initialized.")
		end
	end

	function Movement.detach()
		movementMaid:clean()

		if Freecam.running() then
			Freecam.exit()
		end

		Logger.warn("Movement detached.")
	end

	return Movement
end)()
end)
__bundle_register("Utility/Freecam", function(require, _LOADED, __bundle_register, __bundle_modules)
---@class Freecam
--- Project Rain's freecam, adapted for luma. Detaches the camera from the character:
---   WASD  - move (relative to look)      Q / E        - down / up
---   mouse - look                         Up / Down    - adjust speed live
---   LeftShift - move slowly
--- The character stays put. Speed is scaled by the FreecamSpeed slider via set_speed().
local Freecam = {}

local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local ContextActionService = game:GetService("ContextActionService")

local Camera = workspace.CurrentCamera
workspace:GetPropertyChangedSignal("CurrentCamera"):Connect(function()
	local newCamera = workspace.CurrentCamera
	if newCamera then
		Camera = newCamera
	end
end)

local INPUT_PRIORITY = Enum.ContextActionPriority.High.Value

local running = false
local cameraPos = Vector3.new()
local cameraRot = Vector2.new()

local NAV_KEYBOARD_SPEED = Vector3.new(1, 1, 1)
local PAN_MOUSE_SPEED = Vector2.new(1, 1) * (math.pi / 64)
local NAV_ADJ_SPEED = 0.75
local NAV_SHIFT_MUL = 0.25

local navSpeed = 2

local keyboard = { W = 0, A = 0, S = 0, D = 0, E = 0, Q = 0, Up = 0, Down = 0 }
local mouse = { Delta = Vector2.new() }

local function inputVel(dt)
	navSpeed = math.clamp(navSpeed + dt * (keyboard.Up - keyboard.Down) * NAV_ADJ_SPEED, 0.01, 4)

	local kKeyboard = Vector3.new(
		keyboard.D - keyboard.A,
		keyboard.E - keyboard.Q,
		keyboard.S - keyboard.W
	) * NAV_KEYBOARD_SPEED

	local shift = UserInputService:IsKeyDown(Enum.KeyCode.LeftShift)

	return kKeyboard * (navSpeed * (shift and NAV_SHIFT_MUL or 1))
end

local function inputPan()
	local kMouse = mouse.Delta * PAN_MOUSE_SPEED
	mouse.Delta = Vector2.new()
	return kMouse
end

local function keypress(_, state, input)
	keyboard[input.KeyCode.Name] = state == Enum.UserInputState.Begin and 1 or 0
	return Enum.ContextActionResult.Sink
end

local function mousePan(_, _, input)
	local delta = input.Delta
	mouse.Delta = Vector2.new(-delta.Y, -delta.X)
	return Enum.ContextActionResult.Sink
end

local function zero(t)
	for k in next, t do
		t[k] = 0
	end
end

local function startCapture()
	ContextActionService:BindActionAtPriority(
		"FreecamKeyboard",
		keypress,
		false,
		INPUT_PRIORITY,
		Enum.KeyCode.W,
		Enum.KeyCode.A,
		Enum.KeyCode.S,
		Enum.KeyCode.D,
		Enum.KeyCode.E,
		Enum.KeyCode.Q,
		Enum.KeyCode.Up,
		Enum.KeyCode.Down
	)
	ContextActionService:BindActionAtPriority(
		"FreecamMousePan",
		mousePan,
		false,
		INPUT_PRIORITY,
		Enum.UserInputType.MouseMovement
	)
end

local function stopCapture()
	navSpeed = 1
	zero(keyboard)
	mouse.Delta = Vector2.new()
	ContextActionService:UnbindAction("FreecamKeyboard")
	ContextActionService:UnbindAction("FreecamMousePan")
end

local function getFocusDistance(cameraFrame)
	local znear = 0.1
	local viewport = Camera.ViewportSize
	local projy = 2 * math.tan(70 / 2)
	local projx = viewport.X / viewport.Y * projy
	local fx = cameraFrame.RightVector
	local fy = cameraFrame.UpVector
	local fz = cameraFrame.LookVector

	local minVect = Vector3.new()
	local minDist = 512

	for x = 0, 1, 0.5 do
		for y = 0, 1, 0.5 do
			local cx = (x - 0.5) * projx
			local cy = (y - 0.5) * projy
			local offset = fx * cx - fy * cy + fz
			local origin = cameraFrame.Position + offset * znear

			local result = workspace:Raycast(origin, offset.Unit * minDist)
			local hitPos = result and result.Position or (origin + offset.Unit * minDist)
			local dist = (hitPos - origin).Magnitude
			if minDist > dist then
				minDist = dist
				minVect = offset.Unit
			end
		end
	end

	return fz:Dot(minVect) * minDist
end

local function stepFreecam(dt)
	local vel = inputVel(dt)
	local pan = inputPan()

	cameraRot = cameraRot + pan * Vector2.new(0.75, 1) * 8 * dt
	cameraRot = Vector2.new(math.clamp(cameraRot.X, -math.rad(90), math.rad(90)), cameraRot.Y % (2 * math.pi))

	local cameraCFrame = CFrame.new(cameraPos)
		* CFrame.fromOrientation(cameraRot.X, cameraRot.Y, 0)
		* CFrame.new(vel * 64 * dt)
	cameraPos = cameraCFrame.Position

	Camera.CFrame = cameraCFrame
	Camera.Focus = cameraCFrame * CFrame.new(0, 0, -getFocusDistance(cameraCFrame))
end

-- Saved player camera + mouse state (restored on exit).
local savedCameraType, savedCFrame, savedFocus, savedMouseIcon, savedMouseBehavior

local function push()
	savedCameraType = Camera.CameraType
	Camera.CameraType = Enum.CameraType.Custom

	savedCFrame = Camera.CFrame
	savedFocus = Camera.Focus

	savedMouseIcon = UserInputService.MouseIconEnabled
	UserInputService.MouseIconEnabled = true

	savedMouseBehavior = UserInputService.MouseBehavior
	UserInputService.MouseBehavior = Enum.MouseBehavior.Default
end

local function pop()
	Camera.CameraType = savedCameraType or Enum.CameraType.Custom
	Camera.CFrame = savedCFrame or Camera.CFrame
	Camera.Focus = savedFocus or Camera.Focus
	UserInputService.MouseIconEnabled = savedMouseIcon
	if savedMouseBehavior then
		UserInputService.MouseBehavior = savedMouseBehavior
	end

	savedCameraType, savedCFrame, savedFocus, savedMouseIcon, savedMouseBehavior = nil, nil, nil, nil, nil
end

---@param pos CFrame? Optional starting camera CFrame.
function Freecam.enter(pos)
	if running then
		Freecam.exit()
	end

	local cameraCFrame = pos or Camera.CFrame
	cameraRot = Vector2.new()
	cameraPos = cameraCFrame.Position

	push()
	RunService:BindToRenderStep("Freecam", Enum.RenderPriority.Camera.Value, stepFreecam)
	startCapture()
	running = true
end

function Freecam.exit()
	if not running then
		return
	end

	stopCapture()
	RunService:UnbindFromRenderStep("Freecam")
	pop()
	running = false
end

---@param speed number Multiplier on the base nav speed (1 = default feel).
function Freecam.set_speed(speed)
	NAV_KEYBOARD_SPEED = Vector3.new(speed, speed, speed)
end

---@return boolean
function Freecam.running()
	return running
end

return Freecam
end)

		return
	end

		if not instance:IsA("TextButton") then
			continue
		end

		local number = tonumber(instance.Name)
		if not number then
			continue
		end

		table.insert(answers, number)
	end

	local descriptionText = tostring(description.Text)
	local parsedDescription = descriptionText:gsub("What is", ""):gsub(" ", ""):gsub("?", ""):gsub("by", "")

	local operation = nil

	if parsedDescription:match("divided") then
		operation = "divided"
	end

	if parsedDescription:match("times") then
		operation = "times"
	end

	if parsedDescription:match("plus") then
		operation = "plus"
	end

	if parsedDescription:match("minus") then
		operation = "minus"
	end

	if not operation then
		return
	end

	local parsedNumbers = parsedDescription:split(operation)
	local firstNumber = tonumber(parsedNumbers[1] or "")
	local secondNumber = tonumber(parsedNumbers[2] or "")
	if not firstNumber or not secondNumber then
		return

	if operation == "divided" then

		answer = firstNumber + secondNumber
	end

	if operation == "minus" then
		answer = firstNumber - secondNumber
	end

	local closestDifference = nil

	for _, possible in next, answers do
		local delta = math.abs(answer - possible)

		if closestDifference and delta > closestDifference then
			continue
		end

		closestAnswer = possible
		closestDifference = delta
	end

	if not closestAnswer then
		return
	end

	choiceRemote:FireServer(tostring(closestAnswer))
end)

local players = game:GetService("Players")
local userInputService = game:GetService("UserInputService")

local renderStepped = Signal.new(runService.RenderStepped)

local menuMaid = Maid.new()

local MENU_TITLE = "Book of the Depths | CM1"

if LRM_UserNote then
	MENU_TITLE = string.format(
		"Book of the Depths | CM1",
		string.sub("6c10af8d79e3dc253ba6db7f343613361c37e378", 1, 6)
	)
end

function Menu.init()
	local window = Library:CreateWindow({
		Title = MENU_TITLE,
		Center = true,
		AutoShow = not shared.Luma.silent,
		TabPadding = 8,
		MenuFadeTime = 0.0,
		Size = UDim2.fromOffset(625, 650),
	})

	ThemeManager:SetLibrary(Library)
	ThemeManager:SetFolder("CM1-Deepwoken-Themes")

	SaveManager:SetLibrary(Library)
	SaveManager:IgnoreThemeSettings()
	SaveManager:SetFolder("CM1-Deepwoken-Configs")
	SaveManager:SetIgnoreIndexes({
		"Fly",
		"NoClip",
		"Speedhack",
		"InfiniteJump",
		"TweenToObjective",
		"TweenToBack",
	})

	-- Eager: only the first tab, so the menu has content the instant it auto-shows. The remaining
	-- tabs are built on background frames (see the deferred builder below) so the menu appears
	-- immediately instead of blocking ~1s on control creation.
	HomeTab.init(window)
	CombatTab.init(window)
	CombatTab.initAntiExploit(window)

	local menuUIHidden = false

	local lastUpdate = os.clock()

	menuMaid:add(renderStepped:connect(
		"CM1_WatermarkUpdate",
		LPH_NO_VIRTUALIZE(function()
			if os.clock() - lastUpdate <= 0.5 then
				return
			end

			lastUpdate = os.clock()

			local networkStats = stats:FindFirstChild("Network")
			local workspaceStats = stats:FindFirstChild("Workspace")
			local performanceStats = stats:FindFirstChild("PerformanceStats")
			local serverStats = networkStats and networkStats:FindFirstChild("ServerStatsItem") or nil

			local pingData = serverStats and serverStats:FindFirstChild("Data Ping") or nil
			local heartbeatData = workspaceStats and workspaceStats:FindFirstChild("Heartbeat") or nil
			local cpuData = performanceStats and performanceStats:FindFirstChild("CPU") or nil
			local gpuData = performanceStats and performanceStats:FindFirstChild("GPU") or nil

			local ping = pingData and pingData:GetValue() or 0.0
			local fps = heartbeatData and heartbeatData:GetValue() or 0.0
			local cpu = cpuData and cpuData:GetValue() or 0.0
			local gpu = gpuData and gpuData:GetValue() or 0.0

			local mouse = players.LocalPlayer and players.LocalPlayer:GetMouse()
			local character = players.LocalPlayer and players.LocalPlayer.Character
			local humanoidRootPart = character and character:FindFirstChild("HumanoidRootPart")
			local position = humanoidRootPart and humanoidRootPart.Position
			local positionFormat = position and string.format("(%.2f, %.2f, %.2f)", position.X, position.Y, position.Z)
				or "N/A"

			local str = string.format("%s | %.2fms | %.1f/s | %.1fms | %.1fms", MENU_TITLE, ping, fps, cpu, gpu)

			if Configuration.expectToggleValue("ShowDebugInformation") then
				str = str .. string.format(" | %s", positionFormat)
				str = str .. string.format(" | %s", mouse and mouse.Target and mouse.Target:GetFullName() or "N/A")
			end

			-- YSL's SetWatermark(Text, RText, Sections) measures the frame width from RText, not Text, so
			-- it must be passed too (Obsidian sized off the single arg). Without it the frame collapses and
			-- the watermark renders blank.
			if not menuUIHidden then
				Library:SetWatermark(str)
			end
		end)
	))

	menuMaid:add(renderStepped:connect(
		"Menu_HideUIInMenus",
		LPH_NO_VIRTUALIZE(function()
			local playerGui = players.LocalPlayer and players.LocalPlayer:FindFirstChild("PlayerGui")
			local backpackGui = playerGui and playerGui:FindFirstChild("BackpackGui")
			local leftFrame = backpackGui and backpackGui:FindFirstChild("LeftFrame")
			local inMenu = leftFrame and leftFrame:IsA("GuiObject") and leftFrame.Visible

			if Configuration.expectToggleValue("HideUIInMenus") and inMenu then
				Library.KeybindFrame.Visible = false
				Library:SetWatermarkVisibility(false)
				menuUIHidden = true
				return
			end

			if not menuUIHidden then
				return
			end

			menuUIHidden = false
			Library.KeybindFrame.Visible = Options.KeybindList and Options.KeybindList:GetState() or false
			Library:SetWatermarkVisibility(Options.Watermark and Options.Watermark:GetState() or false)
		end)
	))

	local inputBegan = Signal.new(userInputService.InputBegan)

	inputBegan:connect("Menu_InputBegan", function(input, gameProcessed)
		if gameProcessed then
			return
		end

		if not Configuration.expectToggleValue("ShowDebugInformation") then
			return
		end

		if input.KeyCode == Enum.KeyCode.F8 then
			local mouse = players.LocalPlayer and players.LocalPlayer:GetMouse()
			local target = mouse and mouse.Target
			if not target then
				return
			end

			target.Name = tostring(math.random(100000, 1000000))

			return Logger.mnnotify("The target has been renamed to '%s' as a marker.", target.Name)
		end

		if input.KeyCode == Enum.KeyCode.Equals and Library.Recording then
			local Character = players.LocalPlayer and players.LocalPlayer.Character
			local HumanoidRootPart = Character and Character:FindFirstChild("HumanoidRootPart")
			local Position = HumanoidRootPart and HumanoidRootPart.Position
			local PositionFormat = Position
					and string.format("CFrame.new(%.2f, %.2f, %.2f)", Position.X, Position.Y, Position.Z)
				or "N/A"

			Library.PositionList[#Library.PositionList + 1] = PositionFormat
		end
	end)

	-- Deferred tab build: every tab except the first is created here, one per frame, so the menu is
	-- visible and interactive immediately and the rest pop into the sidebar as they finish. The menu
	-- keybind binding and autoload run only AFTER every tab exists -- their controls (Toggles/Options/
	-- KeyPickers, incl. Options.MenuKeybind) must be built or saved configs, keybinds and default-on
	-- features would read nil and silently not apply.
	task.spawn(function()
		local function diagTab(n, f)
			local ok, e = pcall(f, window)
			if not ok then warn("[Luma/diag] " .. n .. ".init FAILED: " .. tostring(e)) end
			task.wait()
		end
		diagTab("TimingBuilderTab", TimingBuilderTab.init)
		diagTab("VisualsTab", VisualsTab.init)
		diagTab("UtilityTab", UtilityTab.init)
		diagTab("AutomationTab", AutomationTab.init)
		diagTab("LumaTab", LumaTab.init)
		-- Custom tab is unfinished -- not registered yet (CustomTab.init is intentionally not called).
		-- diagTab("CustomTab", CustomTab.init)
		diagTab("ConfigTab", ConfigTab.init)

		Library.ToggleKeybind = Options.MenuKeybind

		SaveManager:LoadAutoloadConfig()

		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("Menu initialized.")
		end
	end)
end

function Menu.detach()
	menuMaid:clean()

	-- Tear down any running custom features (their render/step/input connections outlive the ScreenGui).
	pcall(Custom.stopAll)

	TimingBuilderTab.detach()

	Library:Unload()

	Logger.warn("Menu detached.")
end

return Menu
end)

-- Manager for user-built custom features: storage (workspace files), sandboxed execution via CustomAPI,
-- and optional AI code generation (OpenAI / Anthropic / Google) against a described Luma feature API.

---@module Utility.Filesystem
local Filesystem = require("Utility/Filesystem")

---@module Utility.JSON
local JSON = require("Utility/JSON")

---@module Features.Custom.CustomAPI
local CustomAPI = require("Features/Custom/CustomAPI")

---@module Utility.Logger
local Logger = require("Utility/Logger")

local Custom = {}

local storage = Filesystem.new("Luma/Custom")

local host = nil -- the panel widgets render into (set by CustomTab)
local running = {} -- name -> CustomAPI instance
local settingsCache = nil

local DEFAULT_SETTINGS = {
	provider = "OpenAI",
	model = "gpt-4o-mini",
	apiKey = "",
	autorun = {},
}

Custom.PROVIDERS = { "OpenAI", "Anthropic", "Google" }
Custom.DEFAULT_MODELS = {
	OpenAI = "gpt-4o-mini",
	Anthropic = "claude-3-5-sonnet-latest",
	Google = "gemini-1.5-flash",
}

Custom.TEMPLATE = table.concat({
	"-- Custom feature. The Luma API is available as globals (and under `Luma`).",
	"-- Widgets appear in a section on the right; everything is cleaned up on reload.",
	"",
	'local enabled = Toggle("Enabled", function(on)',
	'    Notify(on and "Enabled" or "Disabled")',
	"end)",
	"",
	'Slider("Speed", { min = 16, max = 200, default = 16, callback = function(v)',
	"    local hum = Humanoid()",
	"    if hum then hum.WalkSpeed = v end",

	"    if not enabled.get() then return end",

-- ------------------------------------------------------------ settings
function Custom.settings()
	if settingsCache then
		return settingsCache
	end
	local data = nil
	pcall(function()
		if storage:file("settings.json") then
			local ok, decoded = pcall(JSON.decode, storage:read("settings.json"))
			if ok and type(decoded) == "table" then
				data = decoded
			end
		end
	end)
	data = data or {}
	for k, v in pairs(DEFAULT_SETTINGS) do
		if data[k] == nil then
			data[k] = v
		end
	end
	if type(data.autorun) ~= "table" then
		data.autorun = {}
	end
	settingsCache = data
	return data
end

function Custom.saveSettings()
	local s = Custom.settings()
	pcall(function()
		storage:write("settings.json", JSON.encode(s))
	end)
end

function Custom.setSetting(key, value)
	local s = Custom.settings()
	s[key] = value
	Custom.saveSettings()
end

-- ------------------------------------------------------------ storage
---@return string[]
function Custom.list()
	local out = {}
	pcall(function()
		for _, filename in ipairs(storage:list()) do
			if type(filename) == "string" and filename:sub(-4) == ".lua" then
				out[#out + 1] = filename:sub(1, #filename - 4)
			end
		end
	end)
	table.sort(out)
	return out
end

---@param name string
---@return string?
function Custom.load(name)
	local source
	pcall(function()
		if storage:file(name .. ".lua") then
			source = storage:read(name .. ".lua")
		end
	end)
	return source
end

---@param name string
---@param source string
---@return boolean ok, string? err
function Custom.save(name, source)
	if type(name) ~= "string" or name:gsub("%s", "") == "" then
		return false, "invalid name"
	end
	-- Keep filenames flat/safe.
	name = name:gsub("[\\/:*?\"<>|]", "_")
	local ok, err = pcall(function()
		storage:write(name .. ".lua", source or "")
	end)
	if not ok then
		return false, tostring(err)
	end
	return true
end

---@param name string
function Custom.delete(name)
	Custom.stop(name)
	pcall(function()
		if storage:file(name .. ".lua") then
			storage:delete(name .. ".lua")
		end
	end)
	local s = Custom.settings()
	if s.autorun[name] then
		s.autorun[name] = nil
		Custom.saveSettings()
	end
end

-- ------------------------------------------------------------ execution
function Custom.setHost(panel)
	host = panel
end

function Custom.isRunning(name)
	return running[name] ~= nil
end

function Custom.runningNames()
	local out = {}
	for name in pairs(running) do
		out[#out + 1] = name
	end
	table.sort(out)
	return out
end

function Custom.stop(name)
	local inst = running[name]
	if inst then
		pcall(inst.destroy)
		running[name] = nil
	end
end

function Custom.stopAll()
	for name in pairs(running) do
		Custom.stop(name)
	end
end

---Run (or re-run) a feature's source. Cleans any previous instance of the same name first.
---@param name string
---@param source string
---@return boolean ok, string? err
function Custom.run(name, source)
	if not host then
		return false, "custom panel not ready"
	end

	Custom.stop(name)

	local chunk, compileErr = loadstring(source)
	if not chunk then
		local clean = tostring(compileErr):gsub('^%[string "[^"]*"%]:', "line ")
		return false, "syntax: " .. clean
	end

	local inst = CustomAPI.new(host, name)
	pcall(setfenv, chunk, inst.env)

	local ranOk, runErr = pcall(chunk)
	if not ranOk then
		pcall(inst.destroy)
		local clean = tostring(runErr):gsub('^%[string "[^"]*"%]:', "line ")
		return false, "runtime: " .. clean
	end

	running[name] = inst
	return true
end

function Custom.setAutorun(name, on)
	local s = Custom.settings()
	s.autorun[name] = on and true or nil
	Custom.saveSettings()
end

function Custom.isAutorun(name)
	return Custom.settings().autorun[name] == true
end

-- Run every feature flagged autorun (called by CustomTab once the host panel exists).
function Custom.autorunAll()
	local s = Custom.settings()
	for name in pairs(s.autorun) do
		local source = Custom.load(name)
		if source then
			local ok, err = Custom.run(name, source)
			if not ok then
				Logger.notify("[Custom] autorun '%s' failed: %s", tostring(name), tostring(err))
			end
		end
	end
end

-- ------------------------------------------------------------ AI code generation
local function httpRequest()
	return (syn and syn.request)
		or (http and http.request)
		or http_request
		or request
		or (getgenv and getgenv().request)
end

local API_REFERENCE = table.concat({
	"You write Luau code for the Roblox game Deepwoken, run inside the 'luma' exploit menu's custom-feature sandbox.",
	"The following API is available as GLOBAL functions (also under the `Luma` table). Use ONLY this API for UI + lifecycle:",
	"",
	"UI (each returns a control object):",
	"  Toggle(text, function(value) end)  -- or Toggle(text, { default=bool, callback=fn }); control.get()/control.set(v)",
	"  Button(text, function() end)",
	"  Slider(text, { min=number, max=number, default=number, decimals=number, callback=function(value) end }); control.get()/set(v)",
	"  Dropdown(text, { values={...}, default=any, callback=function(value) end }); control.get()/set(v)/setValues(t)",
	"  Input(text, { default=string, callback=function(text) end }); control.get()/set(v)",
	"  Keybind(text, { default='E', callback=function(key) end })  -- fires callback when the bound key is pressed",
	"  Label(text); control.set(text)",
	"  Section(title) -> a sub-surface with the same UI methods",
	"  Window(title) -> a floating draggable window, a sub-surface with the same UI methods",
	"",
	"Lifecycle (auto-cleaned on reload):",
	"  OnRender(function(dt) end)   -- every RenderStepped frame",
	"  OnStep(function(dt) end)     -- every Heartbeat",
	"  OnInput(function(input, gameProcessed) end)",
	"  Interval(seconds, function() end)",
	"  OnCleanup(function() end)",
	"  Notify(text, seconds)",
	"",
	"Game helpers:",
	"  LocalPlayer, Character(), HRP(), Humanoid(), Mouse(), Camera()",
	"  GetTarget() -> { character, humanoid, root, du } (nearest aim target) ; GetTargets() -> list",
	"  Value(flag) -> read an existing luma toggle/option by flag; Toggles, Options, Services, Library",
	"  Standard Roblox globals (game, workspace, task, math, Vector3, CFrame, etc.) are available.",
	"",
	"Rules: output ONLY runnable Luau code, no markdown fences, no explanation. Do not call loadstring/HttpGet/setfenv.",
	"Guard every frame loop with a toggle. Keep it concise and correct.",
}, "\n")

local function extractCode(text)
	if type(text) ~= "string" then
		return nil
	end
	-- Prefer a fenced code block if the model added one.
	local fenced = text:match("```[%w]*\n(.-)```")
	if fenced and fenced:gsub("%s", "") ~= "" then
		return fenced
	end
	return text
end

---Generate feature code from a natural-language prompt. Runs async; cb(ok, codeOrErr).
---@param prompt string
---@param cb fun(ok: boolean, result: string)
function Custom.generate(prompt, cb)
	task.spawn(function()
		local s = Custom.settings()
		local req = httpRequest()
		if not req then
			return cb(false, "no HTTP request function on this executor")
		end
		if not s.apiKey or s.apiKey:gsub("%s", "") == "" then
			return cb(false, "set your API key first")
		end

		local provider = s.provider or "OpenAI"
		local model = (s.model and s.model:gsub("%s", "") ~= "" and s.model) or Custom.DEFAULT_MODELS[provider] or "gpt-4o-mini"

		local url, headers, body
		if provider == "Anthropic" then
			url = "https://api.anthropic.com/v1/messages"
			headers = {
				["x-api-key"] = s.apiKey,
				["anthropic-version"] = "2023-06-01",
				["Content-Type"] = "application/json",
			}
			body = JSON.encode({
				model = model,
				max_tokens = 2048,
				system = API_REFERENCE,
				messages = { { role = "user", content = prompt } },
			})
		elseif provider == "Google" then
			url = "https://generativelanguage.googleapis.com/v1beta/models/"
				.. model
				.. ":generateContent?key="
				.. s.apiKey
			headers = { ["Content-Type"] = "application/json" }
			body = JSON.encode({
				system_instruction = { parts = { { text = API_REFERENCE } } },
				contents = { { role = "user", parts = { { text = prompt } } } },
			})
		else
			url = "https://api.openai.com/v1/chat/completions"
			headers = {
				["Authorization"] = "Bearer " .. s.apiKey,
				["Content-Type"] = "application/json",
			}
			body = JSON.encode({
				model = model,
				temperature = 0.4,
				messages = {
					{ role = "system", content = API_REFERENCE },
					{ role = "user", content = prompt },
				},
			})
		end

		local ok, res = pcall(req, {
			Url = url,
			Method = "POST",
			Headers = headers,
			Body = body,
		})
		if not ok or type(res) ~= "table" then
			return cb(false, "request failed: " .. tostring(res))
		end
		local status = res.StatusCode or res.Status or 0
		if type(res.Body) ~= "string" or res.Body == "" then
			return cb(false, "empty response (status " .. tostring(status) .. ")")
		end

		local decodeOk, decoded = pcall(JSON.decode, res.Body)
		if not decodeOk or type(decoded) ~= "table" then
			return cb(false, "bad JSON (status " .. tostring(status) .. ")")
		end

		-- API-level error?
		if decoded.error then
			local msg = type(decoded.error) == "table" and decoded.error.message or tostring(decoded.error)
			return cb(false, "api error: " .. tostring(msg))
		end

		local content
		if provider == "Anthropic" then
			content = decoded.content and decoded.content[1] and decoded.content[1].text
		elseif provider == "Google" then
			content = decoded.candidates
				and decoded.candidates[1]
				and decoded.candidates[1].content
				and decoded.candidates[1].content.parts
				and decoded.candidates[1].content.parts[1]
				and decoded.candidates[1].content.parts[1].text
		else
			content = decoded.choices and decoded.choices[1] and decoded.choices[1].message and decoded.choices[1].message.content
		end

		if type(content) ~= "string" or content:gsub("%s", "") == "" then
			return cb(false, "no code in response (status " .. tostring(status) .. ")")
		end

		return cb(true, extractCode(content))
	end)
end

return Custom
end)
__bundle_register("Features/Custom/CustomAPI", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Builds the sandboxed API a user's custom-feature code runs against. new(hostBody, name) returns
-- { env, destroy }: `env` is the environment to setfenv the user chunk with (API helpers are available
-- as globals AND under the `Luma`/`api` table); `destroy()` tears down every widget, window, connection
-- and loop the feature created. Everything the feature makes is tracked so a reload leaves nothing behind.

---@module GUI.Library
local Library = require("GUI/Library")

---@module Features.Custom.CustomWidgets
local Widgets = require("Features/Custom/CustomWidgets")

---@module Utility.Configuration
local Configuration = require("Utility/Configuration")

---@module Features.Combat.Targeting
local Targeting = require("Features/Combat/Targeting")

---@module Utility.Logger
local Logger = require("Utility/Logger")

local Players = game:GetService("Players")
local RunService = game:GetService("RunService")
local UserInputService = game:GetService("UserInputService")
local Workspace = game:GetService("Workspace")

local CustomAPI = {}

local function normalizeOpts(arg)
	if type(arg) == "function" then
		return { callback = arg }
	end
	if type(arg) == "table" then
		return arg
	end
	return {}
end

---@param hostBody Instance where this feature's widgets are parented
---@param name string
---@return table
function CustomAPI.new(hostBody, name)
	local conns = {}
	local roots = {}
	local cleanupFns = {}
	local intervals = {}
	local destroyed = false

	local function track(control)
		if control and control.conns then
			for _, c in ipairs(control.conns) do
				conns[#conns + 1] = c
			end
		end
		return control
	end

	local function trackConn(c)
		conns[#conns + 1] = c
		return c
	end

	-- Section for this feature inside the host panel; all its root widgets live here.
	local rootSection = Widgets.section(hostBody, name or "Custom Feature")
	roots[#roots + 1] = rootSection.instance

	-- Builds a widget surface bound to a particular parent body (root section, a sub-section, or a window).
	local function buildSurface(parent)
		local surface = {}

		function surface.Label(text)
			return track(Widgets.label(parent, text))
		end

		function surface.Button(text, callback)
			return track(Widgets.button(parent, text, callback))
		end

		function surface.Toggle(text, arg)
			local o = normalizeOpts(arg)
			return track(Widgets.toggle(parent, text, o.default, o.callback))
		end

		function surface.Slider(text, arg)
			local o = normalizeOpts(arg)
			return track(Widgets.slider(parent, text, o.min, o.max, o.default, o.callback, o.decimals))
		end

		function surface.Dropdown(text, arg)
			local o = normalizeOpts(arg)
			return track(Widgets.dropdown(parent, text, o.values, o.default, o.callback))
		end

		function surface.Input(text, arg)
			local o = normalizeOpts(arg)
			return track(Widgets.input(parent, text, o.default, o.callback))
		end

		function surface.Keybind(text, arg)
			local o = normalizeOpts(arg)
			return track(Widgets.keybind(parent, text, o.default, o.callback))
		end

		function surface.Section(title)
			local sec = track(Widgets.section(parent, title))
			return buildSurface(sec.body)
		end

		return surface
	end

	local root = buildSurface(rootSection.body)

	-- Floating, draggable window parented to the menu ScreenGui. Returns a widget surface for its body.
	local function makeWindow(title)
		local gui = Library.ScreenGui
		local outer = Library:Create("Frame", {
			Name = "CustomWindow",
			AnchorPoint = Vector2.new(0.5, 0.5),
			Position = UDim2.fromScale(0.5, 0.5),
			Size = UDim2.fromOffset(240, 60),
			AutomaticSize = Enum.AutomaticSize.Y,
			BackgroundColor3 = Library.MainColor,
			BorderSizePixel = 0,
			ZIndex = 70,
			Parent = gui,
		})
		Library:Create("UICorner", { CornerRadius = UDim.new(0, 6), Parent = outer })
		local os = Library:Create("UIStroke", { Color = Library.AccentColor, Thickness = 1, Transparency = 0.2, Parent = outer })
		Library:AddToRegistry(outer, { BackgroundColor3 = "MainColor" })
		Library:AddToRegistry(os, { Color = "AccentColor" })
		pcall(function() Library:AddUIScale(outer) end)

		local bar = Library:Create("TextLabel", {

		Library:AddToRegistry(bar, { BackgroundColor3 = "BackgroundColor", TextColor3 = "FontColor" })
		pcall(function() Library:MakeDraggable(outer, 24) end)

			Size = UDim2.new(1, -12, 0, 0),

			ZIndex = 71,
			Parent = outer,
		})
		Library:Create("UIListLayout", { SortOrder = Enum.SortOrder.LayoutOrder, Padding = UDim.new(0, 4), Parent = body })
		Library:Create("UIPadding", { PaddingBottom = UDim.new(0, 6), Parent = body })

		roots[#roots + 1] = outer
		return buildSurface(body)
	end

	-- ------------------------------------------------------------ lifecycle + helpers
	local api = {}

	-- widget methods (root surface)
	for k, v in pairs(root) do
		api[k] = v
	end

	function api.Window(title)
		return makeWindow(title)
	end

	function api.OnRender(fn)
		trackConn(RunService.RenderStepped:Connect(function(dt)
			if not destroyed then
				local ok, err = pcall(fn, dt)
				if not ok then
					Logger.notify("[Custom] render error: %s", tostring(err))
				end
			end
		end))
	end

	function api.OnStep(fn)
		trackConn(RunService.Heartbeat:Connect(function(dt)
			if not destroyed then
				pcall(fn, dt)
			end
		end))
	end

	function api.OnInput(fn)
		trackConn(UserInputService.InputBegan:Connect(function(input, gameProcessed)
			if not destroyed then
				pcall(fn, input, gameProcessed)
			end
		end))
	end

	function api.Interval(seconds, fn)
		seconds = tonumber(seconds) or 1
		local token = { running = true }
		intervals[#intervals + 1] = token
		task.spawn(function()
			while token.running and not destroyed do
				task.wait(seconds)
				if token.running and not destroyed then
					pcall(fn)
				end
			end
		end)
	end

	function api.OnCleanup(fn)
		cleanupFns[#cleanupFns + 1] = fn
	end

	function api.Notify(text, time)
		pcall(function()
			Library:Notify(tostring(text), time)
		end)
	end

	-- game helpers
	api.LocalPlayer = Players.LocalPlayer
	api.Services = {
		Players = Players,
		RunService = RunService,
		UserInputService = UserInputService,
		Workspace = Workspace,
		ReplicatedStorage = game:GetService("ReplicatedStorage"),
		TweenService = game:GetService("TweenService"),
		VirtualInputManager = nil,
	}

	function api.Character()
		local lp = Players.LocalPlayer
		return lp and lp.Character
	end

	function api.HRP()
		local char = api.Character()
		return char and char:FindFirstChild("HumanoidRootPart")
	end

	function api.Humanoid()
		local char = api.Character()
		return char and char:FindFirstChildOfClass("Humanoid")
	end

	function api.Mouse()
		local lp = Players.LocalPlayer
		return lp and lp:GetMouse()
	end

	function api.Camera()
		return Workspace.CurrentCamera
	end

	-- Current best aim target (Targeting respects the user's targeting config), and the full list.
	function api.GetTarget()
		local ok, list = pcall(Targeting.best)
		if ok and list then
			return list[1]
		end
		return nil
	end

	function api.GetTargets()
		local ok, list = pcall(Targeting.best)
		if ok and list then
			return list
		end
		return {}
	end

	-- Read any existing luma control value by flag.
	function api.Value(flag)
		local v = Configuration.expectToggleValue(flag)
		if v ~= nil then
			return v
		end
		return Configuration.expectOptionValue(flag)
	end

	api.Toggles = getgenv().Toggles
	api.Options = getgenv().Options
	api.Library = Library

	-- ------------------------------------------------------------ sandbox env
	local base = getfenv(1)
	local gg = getgenv and getgenv() or nil
	local backing = {}
	local env = setmetatable(backing, {
		__index = function(_, key)
			local a = api[key]
			if a ~= nil then
				return a
			end
			local b = base[key]
			if b ~= nil then
				return b
			end
			if gg then
				return gg[key]
			end
			return nil
		end,
		__newindex = function(t, key, value)
			rawset(t, key, value)
		end,
	})
	-- Expose the whole surface as a table too, for users who prefer `Luma.Toggle(...)`.
	backing.Luma = api
	backing.api = api

	local function destroy()
		if destroyed then
			return
		end
		destroyed = true
		for _, token in ipairs(intervals) do
			token.running = false
		end
		for _, c in ipairs(conns) do
			pcall(function()
				c:Disconnect()
			end)
		end
		for _, fn in ipairs(cleanupFns) do
			pcall(fn)
		end
		for _, inst in ipairs(roots) do
			pcall(function()
				inst:Destroy()
			end)
		end
	end

	return { env = env, destroy = destroy, api = api }
end

return CustomAPI
end)

function CustomWidgets.slider(parent, text, min, max, default, callback, decimals)
	min = tonumber(min) or 0
	max = tonumber(max) or 100
	if max <= min then
		max = min + 1
	end
	decimals = tonumber(decimals) or 0
	local value = math.clamp(tonumber(default) or min, min, max)

	local f = Library:Create("Frame", {
		BackgroundColor3 = Library.BackgroundColor,
		BackgroundTransparency = 0.3,
		BorderSizePixel = 0,
		Size = UDim2.new(1, 0, 0, 34),
		Parent = parent,
	})
	corner(f, 4)
	Library:AddToRegistry(f, { BackgroundColor3 = "BackgroundColor" })

	local nameLabel = label(f, tostring(text or "Slider"), 12, Enum.TextXAlignment.Left, UDim2.new(1, -60, 0, 14))
	nameLabel.Position = UDim2.fromOffset(8, 2)

	local valueLabel = Library:Create("TextLabel", {
		AnchorPoint = Vector2.new(1, 0),
		BackgroundTransparency = 1,
		Position = UDim2.new(1, -8, 0, 2),
		Size = UDim2.fromOffset(54, 14),
		Font = Enum.Font.GothamSemibold,
		TextSize = 12,
		TextColor3 = Library.AccentColor,
		TextXAlignment = Enum.TextXAlignment.Right,
		Text = "",
		Parent = f,
	})
	Library:AddToRegistry(valueLabel, { TextColor3 = "AccentColor" }, true)

	local track = Library:Create("Frame", {
		BackgroundColor3 = Library.MainColor,
		BorderSizePixel = 0,
		Position = UDim2.new(0, 8, 0, 22),
		Size = UDim2.new(1, -16, 0, 6),
		Parent = f,
	})
	corner(track, 3)
	Library:AddToRegistry(track, { BackgroundColor3 = "MainColor" })

	local fill = Library:Create("Frame", {
		BackgroundColor3 = Library.AccentColor,
		BorderSizePixel = 0,
		Size = UDim2.fromScale(0, 1),
		Parent = track,
	})
	corner(fill, 3)
	Library:AddToRegistry(fill, { BackgroundColor3 = "AccentColor" })

	local function round(v)
		local m = 10 ^ decimals
		return math.floor(v * m + 0.5) / m
	end

	local control
	local function apply(fireCallback)
		local ratio = (value - min) / (max - min)
		fill.Size = UDim2.fromScale(math.clamp(ratio, 0, 1), 1)
		valueLabel.Text = tostring(round(value))
		if fireCallback and callback then
			pcall(callback, value)
		end
	end

	local dragging = false
	local function setFromX(x)
		local rel = math.clamp((x - track.AbsolutePosition.X) / math.max(track.AbsoluteSize.X, 1), 0, 1)
		value = round(min + rel * (max - min))
		apply(true)
	end

	track.InputBegan:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = true
			setFromX(input.Position.X)
		end
	end)
	local c1 = UserInputService.InputEnded:Connect(function(input)
		if input.UserInputType == Enum.UserInputType.MouseButton1 then
			dragging = false
		end
	end)
	local c2 = UserInputService.InputChanged:Connect(function(input)
		if dragging and input.UserInputType == Enum.UserInputType.MouseMovement then
			setFromX(input.Position.X)
		end
	end)

	apply(false)

	control = {
		instance = f,
		conns = { c1, c2 },
		get = function()
			return value
		end,
		set = function(v, fire)
			value = math.clamp(round(tonumber(v) or min), min, max)
			apply(fire ~= false)
		end,
	}
	return control
end

	Library:AddToRegistry(keyBtn, { BackgroundColor3 = "MainColor", TextColor3 = "AccentColor" })

		listening = true
		keyBtn.Text = "..."
	end)

	local conn = UserInputService.InputBegan:Connect(function(input, gameProcessed)
		if listening and input.UserInputType == Enum.UserInputType.Keyboard then
			listening = false
			if input.KeyCode == Enum.KeyCode.Backspace or input.KeyCode == Enum.KeyCode.Escape then
				key = nil
				keyBtn.Text = "None"
			else
				key = input.KeyCode.Name
				keyBtn.Text = key
			end
			return
		end
		if gameProcessed then
			return
		end
		if key and input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode.Name == key then
			if callback then
				pcall(callback, key)
			end
		end
	end)

	return {
		instance = f,
		conns = { conn },
		get = function()
			return key
		end,
		set = function(k)
			key = k
			keyBtn.Text = k and tostring(k) or "None"
		end,
	}

__bundle_register("Features/Custom/AIChat", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Conversational AI agent for the Custom tab. Keeps a message transcript, calls the configured provider
-- (OpenAI / Anthropic / Google) as a multi-turn chat, and runs a ReAct-style tool loop: the model asks
-- for a tool by replying with a JSON object, we run it against the live game (GameTools) and feed the
-- result back, repeating until the model gives a final answer. Provider-agnostic (plain chat, no native
-- function-calling), so it works the same across all three.

---@module Utility.JSON
local JSON = require("Utility/JSON")

---@module Features.Custom.GameTools
local GameTools = require("Features/Custom/GameTools")

---@module Features.Custom.Custom
local Custom = require("Features/Custom/Custom")

local AIChat = {}

local MAX_STEPS = 10

AIChat.busy = false
AIChat.lastCode = nil
AIChat.messages = nil
AIChat.current = nil -- active run token (for interrupt)

-- Stop the in-flight run. Takes effect between steps / mid-reveal (a blocking request() can't be aborted).
function AIChat.cancel()
	if AIChat.current then
		AIChat.current.cancelled = true
	end
end

local function httpRequest()
	return (syn and syn.request)
		or (http and http.request)
		or http_request
		or request
		or (getgenv and getgenv().request)
end

local function systemPrompt()
	return table.concat({
		"You are an expert assistant embedded in 'luma', a menu for the Roblox game Deepwoken. You help the",
		"user understand the game and build custom features. You can inspect the LIVE game and decompile",
		"scripts via tools before answering, so prefer checking real data over guessing.",
		"",
		GameTools.API_REFERENCE,
		"",
		GameTools.TOOLS_DOC,
		"",
		"Be concise. When the user asks for a feature, inspect what you need, then give the final code in a",
		"single ```lua block using only the API above.",
	}, "\n")
end

function AIChat.reset()
	AIChat.messages = { { role = "system", content = systemPrompt() } }
	AIChat.lastCode = nil
end

local function ensureInit()
	if not AIChat.messages then
		AIChat.reset()
	end
end

-- ------------------------------------------------------------ provider chat completion (synchronous)
local function chatComplete(messages)
	local s = Custom.settings()
	local req = httpRequest()
	if not req then
		return false, "no HTTP request function on this executor"
	end
	if not s.apiKey or s.apiKey:gsub("%s", "") == "" then
		return false, "set your API key in the Custom tab first"
	end

	local provider = s.provider or "OpenAI"
	local model = (s.model and s.model:gsub("%s", "") ~= "" and s.model) or Custom.DEFAULT_MODELS[provider] or "gpt-4o-mini"

	local url, headers, body
	if provider == "Anthropic" then
		local system = {}
		local msgs = {}
		for _, m in ipairs(messages) do
			if m.role == "system" then
				system[#system + 1] = m.content
			else
				msgs[#msgs + 1] = { role = m.role, content = m.content }
			end
		end
		url = "https://api.anthropic.com/v1/messages"
		headers = {
			["x-api-key"] = s.apiKey,
			["anthropic-version"] = "2023-06-01",
			["Content-Type"] = "application/json",
		}
		body = JSON.encode({
			model = model,
			max_tokens = 4096,
			system = table.concat(system, "\n"),
			messages = msgs,
		})
	elseif provider == "Google" then
		local system = {}
		local contents = {}
		for _, m in ipairs(messages) do
			if m.role == "system" then
				system[#system + 1] = m.content
			else
				local role = (m.role == "assistant") and "model" or "user"
				contents[#contents + 1] = { role = role, parts = { { text = m.content } } }
			end
		end
		url = "https://generativelanguage.googleapis.com/v1beta/models/" .. model .. ":generateContent?key=" .. s.apiKey
		headers = { ["Content-Type"] = "application/json" }
		body = JSON.encode({
			system_instruction = { parts = { { text = table.concat(system, "\n") } } },
			contents = contents,
		})
	else
		url = "https://api.openai.com/v1/chat/completions"
		headers = {
			["Authorization"] = "Bearer " .. s.apiKey,
			["Content-Type"] = "application/json",
		}
		body = JSON.encode({
			model = model,
			temperature = 0.3,
			messages = messages,
		})
	end

	local ok, res = pcall(req, { Url = url, Method = "POST", Headers = headers, Body = body })
	if not ok or type(res) ~= "table" then
		return false, "request failed: " .. tostring(res)
	end
	local status = res.StatusCode or res.Status or 0
	if type(res.Body) ~= "string" or res.Body == "" then
		return false, "empty response (status " .. tostring(status) .. ")"
	end
	local decodeOk, decoded = pcall(JSON.decode, res.Body)
	if not decodeOk or type(decoded) ~= "table" then
		return false, "bad JSON (status " .. tostring(status) .. ")"
	end
	if decoded.error then
		local msg = type(decoded.error) == "table" and decoded.error.message or tostring(decoded.error)
		return false, "api error: " .. tostring(msg)
	end

	local content
	if provider == "Anthropic" then
		if decoded.content then
			local parts = {}
			for _, p in ipairs(decoded.content) do
				if p.type == "text" and p.text then
					parts[#parts + 1] = p.text
				end
			end
			content = table.concat(parts)
		end
	elseif provider == "Google" then
		local cand = decoded.candidates and decoded.candidates[1]
		if cand and cand.content and cand.content.parts then
			local parts = {}
			for _, p in ipairs(cand.content.parts) do
				if p.text then
					parts[#parts + 1] = p.text
				end
			end
			content = table.concat(parts)
		end
	else
		content = decoded.choices and decoded.choices[1] and decoded.choices[1].message and decoded.choices[1].message.content
	end

	if type(content) ~= "string" or content:gsub("%s", "") == "" then
		return false, "no content in response (status " .. tostring(status) .. ")"
	end
	return true, content
end

-- ------------------------------------------------------------ parsing
local function stripFence(text)
	local inner = text:match("^%s*```[%w]*%s*\n(.-)\n?%s*```%s*$")
	return inner or text
end

local function parseToolCall(text)
	local t = stripFence(text)
	t = t:gsub("^%s+", ""):gsub("%s+$", "")
	if t:sub(1, 1) ~= "{" then
		return nil
	end
	local ok, decoded = pcall(JSON.decode, t)
	if ok and type(decoded) == "table" and type(decoded.tool) == "string" then
		return { tool = decoded.tool, args = type(decoded.args) == "table" and decoded.args or {} }
	end
	return nil
end

local function extractCode(text)
	local c = text:match("```lua%s*\n(.-)```") or text:match("```%s*\n(.-)```")
	if c and c:gsub("%s", "") ~= "" then
		return c
	end
	return nil
end
AIChat.extractCode = extractCode

-- Reveal the final answer progressively (typewriter). request() returns the whole body at once, so this
-- is a cosmetic reveal, interruptible via the run token. Chunk size scales with length (~160 ticks total).
local function streamOut(text, onEvent, token)
	AIChat.lastCode = extractCode(text) or AIChat.lastCode
	onEvent("assistant_begin")
	local n = #text
	local chunkSize = math.max(2, math.floor(n / 160))
	local i = 1
	while i <= n do
		if token.cancelled then
			onEvent("assistant_delta", text:sub(i)) -- reveal the remainder instantly on stop
			break
		end
		onEvent("assistant_delta", text:sub(i, i + chunkSize - 1))
		i = i + chunkSize
		task.wait(0.012)
	end
	onEvent("assistant_end")
end

-- ------------------------------------------------------------ agent loop
---@param userText string
---@param onEvent fun(kind: string, text: string?)
-- kinds: user, assistant_begin, assistant_delta, assistant_end, tool, toolresult, error, status, done
function AIChat.send(userText, onEvent)
	ensureInit()
	if AIChat.busy then
		onEvent("error", "still working on the previous message...")
		return
	end
	if type(userText) ~= "string" or userText:gsub("%s", "") == "" then
		return
	end

	local token = { cancelled = false }
	AIChat.current = token
	AIChat.busy = true
	table.insert(AIChat.messages, { role = "user", content = userText })
	onEvent("user", userText)

	task.spawn(function()
		local steps = 0
		local finished = false
		while steps < MAX_STEPS do
			if token.cancelled then
				onEvent("status", "stopped.")
				finished = true
				break
			end
			steps = steps + 1
			onEvent("status", steps > 1 and ("thinking... (step " .. steps .. ")") or "thinking...")

			local ok, text = chatComplete(AIChat.messages)
			if token.cancelled then
				onEvent("status", "stopped.")
				finished = true
				break
			end
			if not ok then
				onEvent("error", text)
				finished = true
				break
			end

			table.insert(AIChat.messages, { role = "assistant", content = text })

			local call = parseToolCall(text)
			if call then
				local argsStr = ""
				pcall(function()
					argsStr = JSON.encode(call.args)
				end)
				onEvent("tool", call.tool .. " " .. argsStr)
				local result = GameTools.run(call.tool, call.args)
				onEvent("toolresult", result)
				table.insert(AIChat.messages, {
					role = "user",
					content = "TOOL RESULT (" .. call.tool .. "):\n" .. result,
				})
			else
				streamOut(text, onEvent, token)
				finished = true
				break
			end
		end

		if not finished then
			onEvent("error", "reached the tool-step limit; ask again or be more specific")
		end

		AIChat.busy = false
		AIChat.current = nil
		onEvent("done")
	end)
end

return AIChat
end)

-- ------------------------------------------------------------ path resolution
-- Resolve a dot path from the DataModel, e.g. "game.Workspace.Live", "Players.LocalPlayer",
-- "Players.LocalPlayer.Character". Handles services + the LocalPlayer/Character property shortcuts.
local function resolvePath(path)
	if type(path) ~= "string" or path:gsub("%s", "") == "" then
		return game
	end
	path = path:gsub("/", ".")
	local cur = game
	local first = true
	for token in path:gmatch("[^%.]+") do
		if first and (token == "game" or token == "Game" or token == "DataModel") then
			cur = game
		elseif token == "LocalPlayer" then
			cur = Players.LocalPlayer
			if not cur then
				return nil, "LocalPlayer is nil"
			end
		elseif token == "Character" and typeof(cur) == "Instance" and cur:IsA("Player") then
			cur = cur.Character
			if not cur then
				return nil, "Character is nil"
			end
		else
			local nextInst = nil
			if cur == game then
				pcall(function()
					nextInst = game:GetService(token)
				end)
			end
			if not nextInst then
				nextInst = cur:FindFirstChild(token)
			end
			if not nextInst then
				return nil, "not found: '" .. token .. "' under " .. cur:GetFullName()
			end
			cur = nextInst
		end
		first = false
	end
	return cur
end

-- ------------------------------------------------------------ tools
local function tExplore(path, depth)
	local inst, err = resolvePath(path)
	if not inst then
		return "Error: " .. tostring(err)
	end
	depth = math.clamp(tonumber(depth) or 1, 1, 3)
	local lines = { inst:GetFullName() .. "  [" .. inst.ClassName .. "]" }
	local count = 0
	local function walk(node, d, indent)
		if count >= 200 then
			return
		end
		local ok, children = pcall(function()
			return node:GetChildren()
		end)
		if not ok then
			return
		end
		for _, child in ipairs(children) do
			count = count + 1
			if count >= 200 then
				lines[#lines + 1] = indent .. "... (truncated)"
				return
			end
			lines[#lines + 1] = indent .. child.Name .. "  [" .. child.ClassName .. "]"
			if d > 1 then
				walk(child, d - 1, indent .. "  ")
			end
		end
	end
	walk(inst, depth, "  ")
	if count == 0 then
		lines[#lines + 1] = "  (no children)"
	end
	return table.concat(lines, "\n")
end

local function tProperties(path)
	local inst, err = resolvePath(path)
	if not inst then
		return "Error: " .. tostring(err)
	end
	local out = { inst:GetFullName() .. "  [" .. inst.ClassName .. "]" }
	for _, p in ipairs(PROP_CANDIDATES) do

			return inst[p]
		end)
		if ok and v ~= nil then

		end
	end
	pcall(function()
		if inst.Parent then

	end)
	local okA, attrs = pcall(function()
		return inst:GetAttributes()

		for k, v in pairs(attrs) do
			out[#out + 1] = "@" .. tostring(k) .. " = " .. tostring(v)
		end
	end
	return table.concat(out, "\n")
end

local function matchClass(className, filter)
	if filter == nil or filter == "" then
		return true
	end
	return className == filter
end

local function tSearch(query, class, rootPath)
	local root, err = resolvePath(rootPath or "game")
	if not root then
		return "Error: " .. tostring(err)
	end
	query = tostring(query or ""):lower()
	local results = {}
	local scanned = 0
	local ok, descendants = pcall(function()
		return root:GetDescendants()
	end)
	if not ok then
		return "Error: could not enumerate descendants."
	end
	for _, d in ipairs(descendants) do
		scanned = scanned + 1
		if scanned > 60000 then
			break
		end
		if matchClass(d.ClassName, class) and (query == "" or d.Name:lower():find(query, 1, true)) then
			results[#results + 1] = d:GetFullName() .. "  [" .. d.ClassName .. "]"
			if #results >= 60 then
				results[#results + 1] = "... (more results omitted)"
				break
			end
		end
	end
	if #results == 0 then
		return "No matches (scanned " .. scanned .. " descendants)."
	end
	return table.concat(results, "\n")
end

local SCRIPT_CLASSES = { LocalScript = true, ModuleScript = true, Script = true }

local function tScripts(rootPath)
	local root, err = resolvePath(rootPath or "game")
	if not root then
		return "Error: " .. tostring(err)
	end
	local results = {}
	local scanned = 0
	local ok, descendants = pcall(function()
		return root:GetDescendants()
	end)
	if not ok then
		return "Error: could not enumerate descendants."
	end
	for _, d in ipairs(descendants) do
		scanned = scanned + 1
		if scanned > 60000 then
			break
		end

			if #results >= 80 then
				results[#results + 1] = "... (more omitted)"
				break
			end
		end
	end

	return table.concat(results, "\n")
end

local function tDecompile(path)
	local inst, err = resolvePath(path)
	if not inst then
		return "Error: " .. tostring(err)
	end
	if not (typeof(inst) == "Instance" and inst:IsA("LuaSourceContainer")) then
		return "Not a script: " .. inst.ClassName
	end
	local dec = decompile or (getgenv and getgenv().decompile)
	if not dec then
		return "decompile() is not available on this executor."
	end
	local ok, src = pcall(dec, inst)
	if not ok or type(src) ~= "string" then
		return "Decompile failed: " .. tostring(src)
	end
	if #src > 9000 then
		src = src:sub(1, 9000) .. "\n-- [truncated; ask to decompile a nested module for more]"
	end
	return src
end

-- ------------------------------------------------------------ dispatch
---@param name string
---@param args table?
---@return string
function GameTools.run(name, args)
	args = args or {}
	local ok, result = pcall(function()
		if name == "explore" then
			return tExplore(args.path, args.depth)
		elseif name == "properties" then
			return tProperties(args.path)
		elseif name == "search" then
			return tSearch(args.query, args.class, args.root)
		elseif name == "scripts" then
			return tScripts(args.root)
		elseif name == "decompile" then
			return tDecompile(args.path)
		elseif name == "api_reference" then
			return GameTools.API_REFERENCE
		elseif name == "run_feature" then
			local fname = (type(args.name) == "string" and args.name:gsub("%s", "") ~= "") and args.name or "AI Test"
			if type(args.code) ~= "string" or args.code:gsub("%s", "") == "" then
				return "Error: no 'code' provided."
			end
			Custom.save(fname, args.code)
			local okRun, err = Custom.run(fname, args.code)
			if okRun then
				return "Ran '" .. fname .. "' successfully; its controls are now in the Custom Controls panel. "
					.. "Toggle them to verify behaviour, then give the user the final code in a ```lua block."
			end
			return "Run FAILED: " .. tostring(err) .. "\nFix the code and call run_feature again."
		elseif name == "stop_feature" then
			local fname = type(args.name) == "string" and args.name or ""
			Custom.stop(fname)
			return "Stopped '" .. fname .. "'."
		elseif name == "list_features" then
			local names = Custom.list()
			if #names == 0 then
				return "No saved features."
			end
			return table.concat(names, "\n")
		end
		return "Unknown tool: " .. tostring(name)
	end)
	if not ok then
		return "Tool error: " .. tostring(result)
	end
	return tostring(result)
end

-- Reusable Lua code editor widget (line gutter, custom caret, selection highlight, syntax colouring).
-- Lifted from the timing builder's module editor so the Custom tab can offer the exact same editor.
-- create(host, height, placeholder) -> { get, set, setEditable } ; host is a groupbox/tab (has .Container + :Resize()).

---@module GUI.Library
local Library = require("GUI/Library")

local CodeEditor = {}

local LUA_KEYWORDS = {}
for _, kw in ipairs({
	"and", "break", "do", "else", "elseif", "end", "false", "for", "function", "goto",
	"if", "in", "local", "nil", "not", "or", "repeat", "return", "then", "true", "until",
	"while", "continue", "self",
}) do
	LUA_KEYWORDS[kw] = true
end

local HL_KEYWORD = "#f3a8d6"
local HL_STRING = "#a8e6b0"
local HL_NUMBER = "#f5c07a"
local HL_COMMENT = "#8b7fa0"

local function escapeRich(s)
	return (s:gsub("&", "&amp;"):gsub("<", "&lt;"):gsub(">", "&gt;"))
end

local function span(color, text)
	return '<font color="' .. color .. '">' .. escapeRich(text) .. "</font>"
end

local function highlightLua(code)
	local out = {}
	local i, n = 1, #code

	while i <= n do
		local c = code:sub(i, i)
		local two = code:sub(i, i + 1)

		if two == "--" then
			local eqs = code:sub(i + 2):match("^%[(=*)%[")
			if eqs then
				local close = "]" .. eqs .. "]"
				local endpos = code:find(close, i + 2, true)
				local chunk = endpos and code:sub(i, endpos + #close - 1) or code:sub(i)
				out[#out + 1] = span(HL_COMMENT, chunk)
				i = i + #chunk
			else
				local nl = code:find("\n", i, true)
				local chunk = nl and code:sub(i, nl - 1) or code:sub(i)
				out[#out + 1] = span(HL_COMMENT, chunk)
				i = i + #chunk
			end
		elseif c == '"' or c == "'" then
			local j = i + 1
			while j <= n do
				local cj = code:sub(j, j)
				if cj == "\\" then
					j = j + 2
				elseif cj == c or cj == "\n" then
					j = j + 1
					break
				else
					j = j + 1
				end
			end
			local chunk = code:sub(i, j - 1)
			out[#out + 1] = span(HL_STRING, chunk)
			i = i + #chunk
		elseif c == "[" and code:sub(i):match("^%[=*%[") then
			local eqs = code:sub(i):match("^%[(=*)%[")
			local close = "]" .. eqs .. "]"
			local endpos = code:find(close, i + #eqs + 2, true)
			local chunk = endpos and code:sub(i, endpos + #close - 1) or code:sub(i)
			out[#out + 1] = span(HL_STRING, chunk)
			i = i + #chunk
		elseif c:match("%d") or (c == "." and code:sub(i + 1, i + 1):match("%d")) then
			local num = code:sub(i):match("^0[xX]%x+")
				or code:sub(i):match("^%d*%.?%d+[eE][%+%-]?%d+")
				or code:sub(i):match("^%d*%.?%d+")
				or c
			out[#out + 1] = span(HL_NUMBER, num)
			i = i + #num
		elseif c:match("[%a_]") then
			local word = code:sub(i):match("^[%w_]+")
			if LUA_KEYWORDS[word] then
				out[#out + 1] = span(HL_KEYWORD, word)
			else
				out[#out + 1] = escapeRich(word)
			end
			i = i + #word
		else
			out[#out + 1] = escapeRich(c)
			i = i + 1
		end
	end

	return table.concat(out)
end

CodeEditor.highlight = highlightLua
CodeEditor.escape = escapeRich

---@param host table|Instance a groupbox/tab (needs .Container and :Resize()) OR a raw parent frame
---@param height number
---@param placeholder string?
---@return table
function CodeEditor.create(host, height, placeholder, zbase)
	local GUTTER_W = 34
	local PAD = 6
	local PLACEHOLDER = placeholder or "-- write your feature here"
	-- The ScreenGui uses Global ZIndexBehavior, so when the editor lives in a high-ZIndex floating window
	-- its elements must be lifted above the window's background (pass zbase ~205), else it renders hidden.
	local Z = zbase or 5

	-- Accept either a luma groupbox/tab (parent into .Container, call :Resize afterwards) or a plain frame.
	local isContainer = typeof(host) ~= "Instance"
	local parentInst = isContainer and host.Container or host

	local holder = Library:Create("Frame", {
		BackgroundColor3 = Library.MainColor,
		BorderSizePixel = 0,
		Size = UDim2.new(1, -4, 0, height),
		LayoutOrder = 9999,
		ZIndex = Z,
		Parent = parentInst,
	})

	Library:Create("UICorner", { CornerRadius = UDim.new(0, 4), Parent = holder })
	Library:AddToRegistry(holder, { BackgroundColor3 = "MainColor" })

	local stroke = Library:Create("UIStroke", {
		Color = Library.OutlineColor,
		Thickness = 1,
		Transparency = 0.5,
		Parent = holder,
	})
	Library:AddToRegistry(stroke, { Color = "OutlineColor" })

	local gutterHolder = Library:Create("Frame", {
		BackgroundColor3 = Library.BackgroundColor,
		BackgroundTransparency = 0.4,
		BorderSizePixel = 0,
		ClipsDescendants = true,
		Position = UDim2.new(0, 0, 0, PAD),
		Size = UDim2.new(0, GUTTER_W, 1, -PAD * 2),
		ZIndex = Z + 1,
		Parent = holder,
	})

	local gutter = Library:Create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 0, 0, 0),
		Size = UDim2.new(1, -4, 0, 0),
		AutomaticSize = Enum.AutomaticSize.Y,
		Font = Enum.Font.Code,
		TextSize = 13,
		TextColor3 = Library.OutlineColor,
		TextXAlignment = Enum.TextXAlignment.Right,
		TextYAlignment = Enum.TextYAlignment.Top,
		Text = "1",
		ZIndex = Z + 1,
		Parent = gutterHolder,
	})

	local scroll = Library:Create("ScrollingFrame", {
		BackgroundTransparency = 1,
		BorderSizePixel = 0,
		Position = UDim2.new(0, GUTTER_W + 2, 0, PAD),
		Size = UDim2.new(1, -(GUTTER_W + 2 + PAD), 1, -PAD * 2),
		CanvasSize = UDim2.new(0, 0, 0, 0),
		AutomaticCanvasSize = Enum.AutomaticSize.XY,
		ScrollingDirection = Enum.ScrollingDirection.XY,
		ScrollBarThickness = 4,
		ZIndex = Z + 1,
		Parent = holder,
	})

	local selectionFrames = {}

	local highlight = Library:Create("TextLabel", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 2, 0, 0),
		Size = UDim2.new(0, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.XY,
		RichText = true,
		Font = Enum.Font.Code,
		TextSize = 13,
		TextColor3 = Color3.fromRGB(150, 150, 162),
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Top,
		Text = escapeRich(PLACEHOLDER),
		ZIndex = Z + 1,
		Parent = scroll,
	})

	local box = Library:Create("TextBox", {
		BackgroundTransparency = 1,
		Position = UDim2.new(0, 2, 0, 0),
		Size = UDim2.new(0, 0, 0, 0),
		AutomaticSize = Enum.AutomaticSize.XY,
		MultiLine = true,
		ClearTextOnFocus = false,
		TextEditable = true,
		TextWrapped = false,
		Font = Enum.Font.Code,
		TextSize = 13,
		TextColor3 = Library.FontColor,
		TextTransparency = 1,
		TextXAlignment = Enum.TextXAlignment.Left,
		TextYAlignment = Enum.TextYAlignment.Top,
		Text = "",
		PlaceholderText = "",
		ZIndex = Z + 2,
		Parent = scroll,
	})

	local TextService = game:GetService("TextService")
	local HUGE = Vector2.new(1e6, 1e6)
	local LINE_H = 15
	pcall(function()
		LINE_H = TextService:GetTextSize("Ag", 13, Enum.Font.Code, HUGE).Y
	end)

	local caret = Library:Create("Frame", {
		BackgroundColor3 = Library.FontColor,
		BorderSizePixel = 0,
		Size = UDim2.new(0, 1, 0, LINE_H),
		Position = UDim2.new(0, 2, 0, 0),
		Visible = false,
		ZIndex = Z + 3,
		Parent = scroll,
	})
	Library:AddToRegistry(caret, { BackgroundColor3 = "FontColor" })

	local function updateCaret()
		if not box:IsFocused() then
			caret.Visible = false
			return
		end
		local pos = box.CursorPosition
		if pos < 1 then
			caret.Visible = false
			return
		end
		local before = box.Text:sub(1, pos - 1)
		local line = 0
		for _ in before:gmatch("\n") do
			line = line + 1
		end
		local seg = before:match("[^\n]*$") or ""
		local width = 0
		pcall(function()
			width = TextService:GetTextSize(seg, 13, Enum.Font.Code, HUGE).X
		end)
		caret.Position = UDim2.new(0, 2 + width, 0, line * LINE_H)
		caret.Visible = true
		caret.BackgroundTransparency = 0
	end

	task.spawn(function()
		while caret.Parent do
			task.wait(0.5)
			if box:IsFocused() and caret.Visible then
				caret.BackgroundTransparency = caret.BackgroundTransparency == 0 and 1 or 0
			end
		end
	end)

	local function renderSelection()
		for _, f in ipairs(selectionFrames) do
			f:Destroy()
		end
		selectionFrames = {}

		local sStart, cPos = box.SelectionStart, box.CursorPosition
		if sStart == -1 or cPos == -1 or sStart == cPos then
			return
		end
		local a = math.min(sStart, cPos)
		local b = math.max(sStart, cPos)

		local text = box.Text
		local pos = 1
		for line in (text .. "\n"):gmatch("(.-)\n") do
			local len = #line
			local lineEndPos = pos + len
			local selA = math.max(a, pos)
			local selB = math.min(b, lineEndPos + 1)
			if selB > selA then
				local col1 = math.min(selA - pos, len)
				local col2 = math.min(selB - pos, len)
				local x1, x2 = 0, 0
				pcall(function()
					x1 = TextService:GetTextSize(line:sub(1, col1), 13, Enum.Font.Code, HUGE).X
					x2 = TextService:GetTextSize(line:sub(1, col2), 13, Enum.Font.Code, HUGE).X
				end)
				local lineIdx = 0
				for _ in text:sub(1, pos - 1):gmatch("\n") do
					lineIdx = lineIdx + 1
				end
				local w = x2 - x1
				if b > lineEndPos then
					w = w + 4
				end
				local sel = Library:Create("Frame", {
					BackgroundColor3 = Library.AccentColor,
					BackgroundTransparency = 0.55,
					BorderSizePixel = 0,
					Position = UDim2.new(0, 2 + x1, 0, lineIdx * LINE_H),
					Size = UDim2.new(0, math.max(w, 2), 0, LINE_H),
					ZIndex = Z,
					Parent = scroll,
				})
				Library:AddToRegistry(sel, { BackgroundColor3 = "AccentColor" })
				selectionFrames[#selectionFrames + 1] = sel
			end
			pos = lineEndPos + 1
		end
	end

	box:GetPropertyChangedSignal("CursorPosition"):Connect(function()
		updateCaret()
		renderSelection()
	end)
	box:GetPropertyChangedSignal("SelectionStart"):Connect(renderSelection)
	box.Focused:Connect(updateCaret)
	box.FocusLost:Connect(function()
		caret.Visible = false
		renderSelection()
	end)

	local function refresh()
		local text = box.Text

		if text == "" then
			highlight.TextColor3 = Color3.fromRGB(150, 150, 162)
			highlight.Text = escapeRich(PLACEHOLDER)
		else
			highlight.TextColor3 = Library.FontColor
			local ok, colored = pcall(highlightLua, text)
			highlight.Text = ok and colored or escapeRich(text)
		end

		local lines = 1
		for _ in text:gmatch("\n") do
			lines = lines + 1
		end
		local nums = {}
		for k = 1, lines do
			nums[k] = tostring(k)
		end
		gutter.Text = table.concat(nums, "\n")
	end

	box:GetPropertyChangedSignal("Text"):Connect(function()
		refresh()
		updateCaret()
		renderSelection()
	end)
	scroll:GetPropertyChangedSignal("CanvasPosition"):Connect(function()
		gutter.Position = UDim2.new(0, 0, 0, -scroll.CanvasPosition.Y)
	end)

	refresh()

	if isContainer and host.Resize then
		pcall(function()
			host:Resize()
		end)
	end

	return {
		get = function()
			return box.Text
		end,
		set = function(text)
			box.Text = text or ""
			refresh()
		end,
		setEditable = function(editable)
			box.TextEditable = editable and true or false
			if not editable then
				caret.Visible = false
			end
		end,
	}
end

return CodeEditor
end)

--- Morph loader. Models live under the executor workspace at luma_morphs/Races and luma_morphs/Ornaments.
---@param groupbox table
function VisualsTab.initMorphLoaderSection(groupbox)
	local Players = game:GetService("Players")
	local lp = Players.LocalPlayer

	-- Executor file API resolution (Synapse Z / generic).
	local readCustomAsset = getcustomasset or getsynasset or (syn and syn.get_custom_asset)
	local listFiles = listfiles

	local FOLDERS = {
		Races = "Luma/luma_morphs/Races",
		Ornaments = "Luma/luma_morphs/Ornaments",
	}

	-- Create the morph folders on first execution, the same way the config / theme folders get made.
	do
		local function ensureFolder(path)
			if type(makefolder) ~= "function" then
				return
			end
			if type(isfolder) == "function" and isfolder(path) then
				return
			end
			pcall(makefolder, path)
		end
		ensureFolder("Luma")
		ensureFolder("Luma/luma_morphs")
		ensureFolder(FOLDERS.Races)
		ensureFolder(FOLDERS.Ornaments)
	end

	local HOLDER_NAME = "LumaMorph_Applied"
	local BACKUP_NAME = "LumaMorph_Backup"
	-- Per-morph baked forward nudge (studs, along facing).
	local MORPH_FORWARD = {
	}
	-- Morphs that ship a HideBody flag but look better worn OVER the body.
	local MORPH_NOHIDE = {
	}
	-- For hide-body morphs with gaps, keep these character limbs visible to plug the holes. morphName -> { limbName = true }.
	local MORPH_KEEP_VISIBLE = {
	}

	local currentCategory = "Races"
	local currentFile = nil
	local lastApplied = nil -- { category = , file = }
	local hideBodyEnabled = true
	local reapplyEnabled = false
	local hideToggle -- forward declared; assigned in UI section below
	local RunService = game:GetService("RunService")
	local hideConn = nil
	local hiddenList = {}

	local function listMorphs(dir)
		if type(listFiles) ~= "function" then return {} end
		local ok, res = pcall(listFiles, dir)
		if not ok or type(res) ~= "table" then return {} end
		local out = {}
		for _, p in ipairs(res) do
			local path = tostring(p):gsub("\\", "/")
			if path:sub(-5):lower() == ".rbxm" then
				local name = path:match("([^/]+)%.rbxm$")
				if name then table.insert(out, name) end
			end
		end
		table.sort(out)
		return out
	end

	local function getChar()
		local c = lp.Character
		if not c then return nil end
		if not c:FindFirstChild("HumanoidRootPart") then
			c:WaitForChild("HumanoidRootPart", 5)
		end
		return c
	end

	-- Stop enforcing invisibility and restore everything we hid.
	local function stopHide()
		if hideConn then hideConn:Disconnect(); hideConn = nil end
		for _, p in ipairs(hiddenList) do
			if p and p.Parent then
				pcall(function()
					if p:IsA("BasePart") then
						p.LocalTransparencyModifier = 0
					end
					if p:GetAttribute("LumaMorphHidden") ~= nil then
						p.Transparency = p:GetAttribute("LumaMorphHidden")
						p:SetAttribute("LumaMorphHidden", nil)
					end
				end)
			end
		end
		hiddenList = {}
	end

	-- Keep the given parts/decals invisible every frame (Deepwoken re-shows them otherwise).
	local function startHide(parts)
		stopHide()
		hiddenList = parts
		hideConn = RunService.RenderStepped:Connect(function()
			for _, p in ipairs(hiddenList) do
				if p and p.Parent then
					if p:IsA("BasePart") then
						p.LocalTransparencyModifier = 1
					else
						p.Transparency = 1
					end
				end
			end
		end)
	end

	local function clearMorphs()
		stopHide()
		local c = lp.Character
		if not c then return end
		-- Remove morph clothing clones we added.
		for _, p in ipairs(c:GetChildren()) do
			if p:GetAttribute("LumaMorphClothing") then p:Destroy() end
		end
		-- Restore any original clothing we backed up.
		local backup = c:FindFirstChild(BACKUP_NAME)
		if backup then
			for _, p in ipairs(backup:GetChildren()) do p.Parent = c end
			backup:Destroy()
		end
		local h = c:FindFirstChild(HOLDER_NAME)
		if h then h:Destroy() end
		-- Restore / remove any face decals we changed.
		for _, p in ipairs(c:GetDescendants()) do
			if p:GetAttribute("LumaMorphFaceClone") then
				p:Destroy()
			elseif p:GetAttribute("LumaMorphFace") ~= nil then
				pcall(function() p.Texture = p:GetAttribute("LumaMorphFace") end)
				p:SetAttribute("LumaMorphFace", nil)
			end
		end
		-- Safety sweep for any leftover hidden decals from a prior run.
		for _, p in ipairs(c:GetDescendants()) do
			if p:GetAttribute("LumaMorphHidden") ~= nil then
				pcall(function() p.Transparency = p:GetAttribute("LumaMorphHidden") end)
				p:SetAttribute("LumaMorphHidden", nil)
			end
		end
		-- Restore any real body limbs we recolored to a morph's skin tone.
		for _, p in ipairs(c:GetDescendants()) do
			if p:GetAttribute("LumaMorphSkin") ~= nil then
				pcall(function() p.Color = p:GetAttribute("LumaMorphSkin") end)
				p:SetAttribute("LumaMorphSkin", nil)
			end
		end
	end

	-- Character limb parts (R15 first, R6 fallback) we can bind morph pieces to.
	local LIMB_NAMES = {
		"Head", "UpperTorso", "LowerTorso",
		"LeftUpperArm", "LeftLowerArm", "LeftHand",
		"RightUpperArm", "RightLowerArm", "RightHand",
		"LeftUpperLeg", "LeftLowerLeg", "LeftFoot",
		"RightUpperLeg", "RightLowerLeg", "RightFoot",
		"Torso", "Left Arm", "Right Arm", "Left Leg", "Right Leg",
	}
	-- Names a morph tends to use for its torso/root anchor.
	local ANCHOR_NAMES = {
		"HumanoidRootPart", "UpperTorso", "Torso", "LowerTorso",
		"tt", "t", "lt", "Chestplate", "Body", "Root", "Middle",
	}

	local function collectLimbs(char)
		local limbs = {}
		for _, name in ipairs(LIMB_NAMES) do
			local p = char:FindFirstChild(name)
			if p and p:IsA("BasePart") then limbs[#limbs + 1] = p end
		end
		return limbs
	end

	local function findCharAnchor(char)
		return char:FindFirstChild("UpperTorso")
			or char:FindFirstChild("Torso")
			or char:FindFirstChild("LowerTorso")
			or char:FindFirstChild("HumanoidRootPart")
	end

	local function findMorphAnchor(model)
		for _, an in ipairs(ANCHOR_NAMES) do
			local p = model:FindFirstChild(an, true)
			if p and p:IsA("BasePart") then return p end
		end
		return model.PrimaryPart or model:FindFirstChildWhichIsA("BasePart", true)
	end

	local function nearestLimb(limbs, pos)
		local best, bestDist
		for _, l in ipairs(limbs) do
			local d = (l.Position - pos).Magnitude
			if not bestDist or d < bestDist then bestDist, best = d, l end
		end
		return best
	end

	-- Map a keyword in a morph part's name to a character limb, so morphs that bind by naming convention land on the intended limb instead of relying on raw nearest-position guessing.
	local LIMB_HINTS = {
		{ "uppertorso", { "UpperTorso" } },
		{ "lowertorso", { "LowerTorso" } },
		{ "torso", { "UpperTorso", "Torso", "LowerTorso" } },
		{ "chest", { "UpperTorso", "Torso" } },
		{ "head", { "Head" } }, { "face", { "Head" } }, { "hair", { "Head" } },
		{ "hat", { "Head" } }, { "horn", { "Head" } }, { "mask", { "Head" } }, { "helm", { "Head" } },
		-- Head/hair overlay meshes (ippo's "Shading") are head pieces; without a hint they'd fall to the torso and render on the chest.
		{ "shading", { "Head" } }, { "shade", { "Head" } },
		{ "lefthand", { "LeftHand", "Left Arm" } }, { "righthand", { "RightHand", "Right Arm" } },
		{ "leftlowerarm", { "LeftLowerArm", "Left Arm" } }, { "rightlowerarm", { "RightLowerArm", "Right Arm" } },
		{ "leftupperarm", { "LeftUpperArm", "Left Arm" } }, { "rightupperarm", { "RightUpperArm", "Right Arm" } },
		{ "leftarm", { "LeftUpperArm", "Left Arm" } }, { "rightarm", { "RightUpperArm", "Right Arm" } },
		{ "leftfoot", { "LeftFoot", "Left Leg" } }, { "rightfoot", { "RightFoot", "Right Leg" } },
		{ "leftlowerleg", { "LeftLowerLeg", "Left Leg" } }, { "rightlowerleg", { "RightLowerLeg", "Right Leg" } },
		{ "leftupperleg", { "LeftUpperLeg", "Left Leg" } }, { "rightupperleg", { "RightUpperLeg", "Right Leg" } },
		{ "leftleg", { "LeftUpperLeg", "Left Leg" } }, { "rightleg", { "RightUpperLeg", "Right Leg" } },
		{ "arm", { "UpperTorso", "Torso" } }, -- generic "arms equipment" binds to the torso anchor
	}
	local function limbFromName(char, name)
		local n = tostring(name):lower():gsub("[^%w]", "")
		for _, pair in ipairs(LIMB_HINTS) do
			if n:find(pair[1], 1, true) then
				for _, cand in ipairs(pair[2]) do
					local limb = char:FindFirstChild(cand)
					if limb and limb:IsA("BasePart") then
						return limb
					end
				end
			end
		end
		return nil
	end

	-- The attachment a morph part welds by.
	local function bindingAttachment(part)
		local a = part:FindFirstChild("Offset") or part:FindFirstChild("JointOffset")
			or part:FindFirstChild("Attach")
		if a and a:IsA("Attachment") then
			return a
		end
		return nil
	end

	-- Per-morph, per-part vertical lift (studs) for pieces a morph anchors too low.
	local MORPH_PART_LIFT = {
		FifthGear = { ArmsEquipment_Torso = 1.3 },
	}
	local function partLift(morphName, partName)
		local base = tostring(morphName):gsub("%s*%(%d+%)%s*$", "")
		local t = MORPH_PART_LIFT[base]
		return (t and t[partName]) or 0
	end

	-- Per-morph skin tone.
	local MORPH_SKIN = {
		FifthGear = Color3.fromRGB(226, 178, 132), -- Gear 5 golden tan
	}
	-- Real body limbs to recolor (R6 + R15 names; whichever the character actually has).
	local SKIN_LIMBS = {
		"Torso", "Left Arm", "Right Arm", "Left Leg", "Right Leg",
		"UpperTorso", "LowerTorso",
		"LeftUpperArm", "LeftLowerArm", "LeftHand",
		"RightUpperArm", "RightLowerArm", "RightHand",
		"LeftUpperLeg", "LeftLowerLeg", "LeftFoot",
		"RightUpperLeg", "RightLowerLeg", "RightFoot",
	}

	-- Full R15 limb names.
	local R15_LIMB_SET = {
		Head = true, UpperTorso = true, LowerTorso = true,
		LeftUpperArm = true, LeftLowerArm = true, LeftHand = true,
		RightUpperArm = true, RightLowerArm = true, RightHand = true,
		LeftUpperLeg = true, LeftLowerLeg = true, LeftFoot = true,
		RightUpperLeg = true, RightLowerLeg = true, RightFoot = true,
	}

	-- ---- Per-morph overrides (keyed by base file name, trailing " (n)" copy-suffix ignored) ----
	local function morphBase(name) return tostring(name):gsub("%s*%(%d+%)%s*$", "") end
	-- Morphs whose "head" is a stiff extension of the body: weld the head region to the TORSO, not the neck, so it can't swing away from the body when a weapon draw / emote animation tilts the head.
	local MORPH_RIGID_HEAD = { TungTung = true }
	-- Redundant / overlapping pieces to drop outright.
	local MORPH_HIDE_PARTS = {
		Mudskipper = { MudskipperHatA = true },
	}
	-- Force a named morph part onto a named character limb, overriding the name-guessing.
	local MORPH_PART_LIMB = {}
	-- Corrective vertical lift (studs) for DRIVEN sub-parts that sit too low -- their joint offset drops them onto the chest.
	local MORPH_SUBPART_LIFT = { ippo = { MeshPart = 1.5 } }

	local function applyModel(model, hideBody, morphName)
		local char = getChar()
		if not char then return false, "no character" end
		if not char:FindFirstChild("HumanoidRootPart") then return false, "no HumanoidRootPart" end

		local limbs = collectLimbs(char)
		if #limbs == 0 then return false, "no limbs on character" end

		local extraHide = {} -- specific parts (e.g. real hair) to hide even when not hiding the whole body
		local morphHasHair = false -- set true if the morph supplies its own hair

		local base = morphBase(morphName)

		-- NOTE: do NOT strip the morph's LocalScripts -- on these Deepwoken race morphs they apply the skin/hair/eye colouring from RaceData at runtime; removing them renders the morph flat black.

		-- Drop pieces this morph is known to double up on (prevents z-fighting overlaps).
		local hideParts = MORPH_HIDE_PARTS[base]
		if hideParts then
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") and hideParts[p.Name] then
					pcall(function() p:Destroy() end)
				end
			end
		end

		-- Per-morph forced part->limb map, consulted before name guessing.
		local partLimbMap = MORPH_PART_LIMB[base]
		local function overrideLimb(partName)
			local nm = partLimbMap and partLimbMap[partName]
			if nm then
				local l = char:FindFirstChild(nm)
				if l and l:IsA("BasePart") then return l end
			end
			return nil
		end
		local rigidHead = MORPH_RIGID_HEAD[base]
		-- Records which character limb each morph part was bound to, so sub-parts of an assembly can inherit their parent's limb instead of falling to a nearest-position guess that can strand them on the chest.
		local partToLimb = {}

		-- Snapshot the real body NOW (before adding morph parts) so we can hide it wholesale, skipping any joint-filler limbs this morph wants kept visible.
		local keepVisible = MORPH_KEEP_VISIBLE[morphName] or {}
		local originalBits = {}
		for _, p in ipairs(char:GetDescendants()) do
			local isPart = p:IsA("BasePart") and p.Name ~= "HumanoidRootPart"
			local isDecal = p:IsA("Decal") or p:IsA("Texture")
			if isPart or isDecal then
				local ownerName = isPart and p.Name or (p.Parent and p.Parent.Name)
				if not (ownerName and keepVisible[ownerName]) then
					if isDecal and p:GetAttribute("LumaMorphHidden") == nil then
						p:SetAttribute("LumaMorphHidden", p.Transparency)
					end
					originalBits[#originalBits + 1] = p
				end
			end
		end

		local cAnchor = findCharAnchor(char)
		local mAnchor = findMorphAnchor(model)
		if not cAnchor or not mAnchor then return false, "could not find torso anchor" end

		-- Does this morph actually replace the body?
		local morphWantsHide = model:FindFirstChild("HideBody", true) ~= nil
		-- Only trust the anchor if it's a genuine torso part; otherwise recenter afterward.
		local TRUE_TORSO = {
			HumanoidRootPart = true, UpperTorso = true, Torso = true,
			LowerTorso = true, tt = true, t = true, lt = true,
		}
		local trustedAnchor = TRUE_TORSO[mAnchor.Name] == true

		-- Transform that maps morph-space onto the character.
		local T = cAnchor.CFrame * mAnchor.CFrame:Inverse()

		-- Parts already driven by the morph's own joints — don't re-weld those.
		local driven = {}
		for _, j in ipairs(model:GetDescendants()) do
			if j:IsA("Weld") or j:IsA("Motor6D") or j:IsA("WeldConstraint")
				or j:IsA("Snap") or j:IsA("ManualWeld") then
				if j.Part1 then driven[j.Part1] = true end
			end
		end

		-- Rigidly slide the whole morph onto the character (keeps internal layout + joints).
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") then
				p.CFrame = T * p.CFrame
			end
		end

		-- If we anchored off a non-torso part (e.g.
		if not trustedAnchor then
			local sum, cnt = Vector3.zero, 0
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") then sum = sum + p.Position; cnt = cnt + 1 end
			end
			if cnt > 0 then
				local center = sum / cnt
				local delta = Vector3.new(cAnchor.Position.X - center.X, 0, cAnchor.Position.Z - center.Z)
				for _, p in ipairs(model:GetDescendants()) do
					if p:IsA("BasePart") then p.CFrame = p.CFrame + delta end
				end
			end
		end

		-- Forward/back nudge along the character's facing: per-morph baked default.
		local fwdAmount = MORPH_FORWARD[morphName] or 0
		if fwdAmount ~= 0 then
			local fwd = cAnchor.CFrame.LookVector * fwdAmount
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") then p.CFrame = p.CFrame + fwd end
			end
		end

		-- Full-body R15 rig morphs (e.g.
		local rigParts = {}
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") and R15_LIMB_SET[p.Name] and char:FindFirstChild(p.Name) then
				rigParts[#rigParts + 1] = p
			end
		end
		if #rigParts >= 6 then
			-- Cut the morph's internal limb welds so each limb is free to follow its own bone.
			for _, w in ipairs(model:GetDescendants()) do
				if (w:IsA("Weld") or w:IsA("Motor6D") or w:IsA("ManualWeld")
					or w:IsA("Snap") or w:IsA("WeldConstraint"))
					and w.Part1 and R15_LIMB_SET[w.Part1.Name] then
					w:Destroy()
				end
			end
			for _, p in ipairs(rigParts) do
				local targetName = p.Name
				local keepPos = false
				-- Rigid-head morphs: pin the head to the torso and KEEP its current offset above the body, so the log head rides the torso as one piece instead of pivoting on the neck when the character animates.
				if rigidHead and p.Name == "Head" then
					if char:FindFirstChild("UpperTorso") then targetName = "UpperTorso"
					elseif char:FindFirstChild("Torso") then targetName = "Torso" end
					keepPos = true
				end
				local real = char:FindFirstChild(targetName)
				if real and real:IsA("BasePart") then
					-- A standard R15 skinned part's CFrame IS its bone frame, so aligning it to the real limb's CFrame fits it exactly; the weld then locks it there.
					if not keepPos then p.CFrame = real.CFrame end
					p.Anchored = false; p.CanCollide = false; p.Massless = true
					local w = Instance.new("Weld")
					w.Name = "LumaMorphWeld"
					w.Part0 = real
					w.Part1 = p
					w.C0 = real.CFrame:Inverse() * p.CFrame
					w.C1 = CFrame.new()
					w.Parent = p
					driven[p] = true
				end
			end
			-- Hide the leftover bone-anchor Parts (la/t/h/...): they only carried the welds we just replaced and would otherwise render as stray blocks around the body.
			for _, p in ipairs(model:GetDescendants()) do
				if p.ClassName == "Part" and not driven[p] then
					pcall(function()
						p.Transparency = 1; p.CanCollide = false; p.Massless = true
					end)
				end
			end
		end

		-- Cosmetic/clothing morphs: place Offset-attachment pieces on the right limb and swap the face decal.
		if not morphWantsHide then
			local headPart = char:FindFirstChild("Head")
			local torsoPart = char:FindFirstChild("UpperTorso") or char:FindFirstChild("Torso")
			for _, p in ipairs(model:GetDescendants()) do
				if p:IsA("BasePart") and not driven[p] then
					local off = bindingAttachment(p)
					if off then
						local n = p.Name:lower()
						local isHair = n:find("hair") ~= nil
						-- Prefer the limb named in the part itself (e.g.
						local limb = overrideLimb(p.Name) or limbFromName(char, p.Name)
						if not limb then
							if isHair or n:find("head") or n:find("face")
								or n:find("hat") or n:find("horn") or n:find("mask") then
								limb = headPart
							else
								limb = torsoPart
							end
						end
						if limb then
							-- The Offset attachment marks where the piece meets the limb; align that attachment onto the limb so it sits right.
							p.CFrame = limb.CFrame * off.CFrame:Inverse()
							-- Per-morph vertical correction for pieces that anchor too low.
							local lift = partLift(morphName, p.Name)
							if lift ~= 0 then
								p.CFrame = p.CFrame + Vector3.new(0, lift, 0)
							end
							p.Anchored = false; p.CanCollide = false; p.Massless = true
							local w = Instance.new("Weld")
							w.Name = "LumaMorphWeld"
							w.Part0 = limb
							w.Part1 = p
							w.C0 = limb.CFrame:Inverse() * p.CFrame
							w.C1 = CFrame.new()
							w.Parent = p
							driven[p] = true
							partToLimb[p] = limb
							if isHair then morphHasHair = true end
						end
					end
				end
			end

			-- Face: Deepwoken renders the face as a Decal sitting on the FaceMount mesh -- NOT the block Head, and NOT the mesh's TextureID.
			local faces = model:FindFirstChild("Faces", true)
			local head = char:FindFirstChild("Head")
			if faces and head then
				local srcFace = faces:FindFirstChild("Base") or faces:FindFirstChildWhichIsA("Decal")
				if srcFace and srcFace:IsA("Decal") then
					local faceTex = srcFace.Texture
					-- The face decal lives under the head (usually on FaceMount).
					local faceDecal
					for _, d in ipairs(head:GetDescendants()) do
						if d:IsA("Decal") then
							if d.Name:lower() == "face" then
								faceDecal = d
								break
							elseif not faceDecal then
								faceDecal = d
							end
						end
					end
					if faceDecal then
						if faceDecal:GetAttribute("LumaMorphFace") == nil then
							faceDecal:SetAttribute("LumaMorphFace", faceDecal.Texture)
						end
						faceDecal.Texture = faceTex
						-- Deepwoken layers separate overlay decals for the eyes/brows on top of the base face; the morph face already includes its own eyes, so hide every OTHER face decal/texture on the head so the real eyes don't show through.
						for _, d in ipairs(head:GetDescendants()) do
							if d ~= faceDecal
								and (d:IsA("Decal") or d:IsA("Texture"))
								and not d:GetAttribute("LumaMorphFaceClone")
							then
								if d:GetAttribute("LumaMorphHidden") == nil then
									d:SetAttribute("LumaMorphHidden", d.Transparency)
								end
								d.Transparency = 1
							end
						end
					else
						-- No existing face decal: mount a fresh one on FaceMount's front (a Decal keeps alpha, so it blends instead of boxing solid).
						local mount = head:FindFirstChild("FaceMount") or head
						local nd = srcFace:Clone()
						nd.Name = "LumaMorphFace"
						nd.Face = Enum.NormalId.Front
						nd:SetAttribute("LumaMorphFaceClone", true)
						nd.Parent = mount
					end
				end
			end
		end

		-- Bind each free (non-driven) part to a limb so it sits right and articulates.
		local bound = 0
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") and not driven[p] then
				local limb = overrideLimb(p.Name) or limbFromName(char, p.Name)
				if not limb then
					-- Inherit the limb an ancestor part was bound to: a strand parented under a hair clump belongs on the head with it, not wherever nearest-position lands.
					local anc = p.Parent
					while anc and anc ~= model do
						if partToLimb[anc] then limb = partToLimb[anc]; break end
						anc = anc.Parent
					end
				end
				limb = limb or nearestLimb(limbs, p.Position)
				if limb then
					p.Anchored = false
					p.CanCollide = false
					p.Massless = true
					local w = Instance.new("Weld")
					w.Name = "LumaMorphWeld"
					w.Part0 = limb
					w.Part1 = p
					w.C0 = limb.CFrame:Inverse() * p.CFrame
					w.C1 = CFrame.new()
					w.Parent = p
					partToLimb[p] = limb
					bound = bound + 1
				end
			end
		end

		-- Safety net: if every part looked driven, at least pin the anchor to the torso.
		if bound == 0 then
			mAnchor.Anchored = false
			local w = Instance.new("Weld")
			w.Name = "LumaMorphWeld"
			w.Part0 = cAnchor
			w.Part1 = mAnchor
			w.C0 = cAnchor.CFrame:Inverse() * mAnchor.CFrame
			w.Parent = mAnchor
		end

		-- Nothing anchored / colliding.
		for _, p in ipairs(model:GetDescendants()) do
			if p:IsA("BasePart") then
				p.Anchored = false
				p.CanCollide = false
			end
		end

		-- Accessories self-weld under the Humanoid.
		for _, acc in ipairs(model:GetDescendants()) do
			if acc:IsA("Accessory") then acc.Parent = char end
		end

		-- Turn effects on (they usually ship disabled).
		for _, fx in ipairs(model:GetDescendants()) do
			if fx:IsA("ParticleEmitter") or fx:IsA("Trail") or fx:IsA("Beam")
				or fx:IsA("PointLight") or fx:IsA("SpotLight") or fx:IsA("SurfaceLight") then
				pcall(function() fx.Enabled = true end)
			end
		end

		-- Apply clothing morphs (Shirt / Pants / graphic) straight onto the character.
		local backup = char:FindFirstChild(BACKUP_NAME)
		for _, cl in ipairs(model:GetDescendants()) do
			if cl:IsA("Shirt") or cl:IsA("Pants") or cl:IsA("ShirtGraphic") then
				if not backup then
					backup = Instance.new("Folder")
					backup.Name = BACKUP_NAME
					backup.Parent = char
				end
				local existing = char:FindFirstChildOfClass(cl.ClassName)
				if existing then existing.Parent = backup end
				local clone = cl:Clone()
				clone:SetAttribute("LumaMorphClothing", true)
				clone.Parent = char
				-- The carrier part (e.g.
				local carrier = cl.Parent
				if carrier and carrier:IsA("BasePart") then
					carrier.Transparency = 1
					carrier.CanCollide = false
				end
			end
		end

		-- Hide the player's real hair whenever the morph replaces the body or brings its own hair.
		if morphWantsHide or morphHasHair then
			for _, acc in ipairs(char:GetChildren()) do
				if acc:IsA("Accessory") then
					local handle = acc:FindFirstChild("Handle")
					local isHairAcc = false
					local ok, at = pcall(function() return acc.AccessoryType end)
					if ok and at == Enum.AccessoryType.Hair then isHairAcc = true end
					if handle and handle:FindFirstChild("HairAttachment") then isHairAcc = true end
					if isHairAcc and handle then extraHide[#extraHide + 1] = handle end
				end
			end
		end

		-- Wear-over morph that brings its OWN head: the real Deepwoken head keeps rendering straight through the morph one -- the "two heads" bug.
		if hideBody and not morphWantsHide then
			local morphReplacesHead = false
			for _, p in ipairs(model:GetDescendants()) do
				-- Exact "Head" only: a piece merely CONTAINING "head" is an accessory, not a head replacement -- hiding the real head for those would leave the character headless.
				if p:IsA("BasePart") and p.Name:lower() == "head" then
					morphReplacesHead = true
					break
				end
			end
			if morphReplacesHead then
				local realHead = char:FindFirstChild("Head")
				if realHead then
					extraHide[#extraHide + 1] = realHead
					for _, d in ipairs(realHead:GetChildren()) do
						if d:IsA("Decal") or d:IsA("Texture") then
							extraHide[#extraHide + 1] = d
						end
					end
				end
			end
		end

		-- Hide the real body ONLY for morphs that truly replace it (e.g.
		if hideBody and morphWantsHide and not MORPH_NOHIDE[morphName] then
			startHide(originalBits)
		elseif #extraHide > 0 then
			startHide(extraHide)
		else
			stopHide()
		end

		-- Skin tone: recolor the real body limbs to the morph's skin so exposed arms/legs/hands match instead of showing the player's own pale skin.
		local skin = MORPH_SKIN[(tostring(morphName):gsub("%s*%(%d+%)%s*$", ""))]
		if skin then
			for _, name in ipairs(SKIN_LIMBS) do
				local part = char:FindFirstChild(name)
				if part and part:IsA("BasePart") then
					if part:GetAttribute("LumaMorphSkin") == nil then
						part:SetAttribute("LumaMorphSkin", part.Color)
					end
					part.Color = skin
				end
			end

			-- The FACE mesh isn't in SKIN_LIMBS: recolor it to skin so the face isn't left white, and clear its Deepwoken face TextureID so the old eyes/skin stop bleeding through the morph face's transparent parts.
			local skinHead = char:FindFirstChild("Head")
			local faceMount = skinHead and skinHead:FindFirstChild("FaceMount")
			if faceMount and faceMount:IsA("MeshPart") then
				if faceMount:GetAttribute("LumaMorphSkin") == nil then
					faceMount:SetAttribute("LumaMorphSkin", faceMount.Color)
				end
				faceMount.Color = skin
				if faceMount:GetAttribute("LumaMorphFaceTex") == nil then
					faceMount:SetAttribute("LumaMorphFaceTex", faceMount.TextureID)
				end
				faceMount.TextureID = ""
			end
		end

		-- Corrective lift for driven sub-parts that hang too low.
		local subLift = MORPH_SUBPART_LIFT[base]
		if subLift then
			for _, j in ipairs(model:GetDescendants()) do
				if (j:IsA("Motor6D") or j:IsA("Weld")) and j.Part1 and subLift[j.Part1.Name] then
					j.C0 = CFrame.new(0, subLift[j.Part1.Name], 0) * j.C0
				end
			end
		end

		local holder = char:FindFirstChild(HOLDER_NAME)
		if not holder then
			holder = Instance.new("Folder")
			holder.Name = HOLDER_NAME
			holder.Parent = char
		end
		model.Parent = holder
		return true
	end

	local function loadMorph(category, fileName)
		if not fileName or fileName == "" then
			return Logger.notify("Pick a morph from the dropdown first.")
		end
		if type(readCustomAsset) ~= "function" then
			return Logger.notify("Executor has no getcustomasset — can't load local models.")
		end
		local path = (FOLDERS[category] or FOLDERS.Races) .. "/" .. fileName .. ".rbxm"
		local ok, asset = pcall(readCustomAsset, path)
		if not ok or not asset then
			return Logger.notify("getcustomasset failed: " .. path)
		end
		local ok2, objs = pcall(function() return game:GetObjects(asset) end)
		if not ok2 or type(objs) ~= "table" or not objs[1] then
			return Logger.notify("Failed to deserialize: " .. tostring(objs))
		end
		local model = objs[1]
		if not model:IsA("Model") then
			local m = Instance.new("Model")
			model.Parent = m
			model = m
		end


		-- Force the morph's meshes/textures to download BEFORE we place anything.
		pcall(function()
			local cp = game:GetService("ContentProvider")
			local assets = { model }
			for _, d in ipairs(model:GetDescendants()) do
				if d:IsA("MeshPart") or d:IsA("SpecialMesh") or d:IsA("Decal") or d:IsA("Texture") then
					assets[#assets + 1] = d
				end
			end
			cp:PreloadAsync(assets)
		end)

		clearMorphs() -- replace any previously applied morph
		local okA, err = applyModel(model, hideBodyEnabled, fileName)
		if okA then
			lastApplied = { category = category, file = fileName }

		else
			if model then model:Destroy() end
			Logger.notify("Apply error: " .. tostring(err))
		end
	end

	-- ===== UI =====
	local catDropdown = groupbox:AddDropdown("LumaMorph_Category", {
		Text = "Category",
		Values = { "Races", "Ornaments" },
		AllowNull = false,
		Default = "Races",
	})

	local morphDropdown = groupbox:AddDropdown("LumaMorph_File", {
		Text = "Morph",
		Values = listMorphs(FOLDERS[currentCategory]),
		AllowNull = true,
		Default = nil,
	})

	catDropdown:OnChanged(function(v)
		currentCategory = v
		morphDropdown:SetValues(listMorphs(FOLDERS[v]))
		morphDropdown:Display()
	end)

	morphDropdown:OnChanged(function(v)
		currentFile = v
	end)

	groupbox:AddButton("Load Morph", function()
		loadMorph(currentCategory, currentFile)
	end)

	groupbox:AddButton("Clear Morph", function()
		clearMorphs()
		lastApplied = nil
		Logger.notify("Cleared applied morphs.")
	end)

	groupbox:AddButton("Refresh List", function()
		morphDropdown:SetValues(listMorphs(FOLDERS[currentCategory]))
		morphDropdown:Display()
	end)

	hideToggle = groupbox:AddToggle("LumaMorph_HideBody", {
		Text = "Hide real body (for HideBody morphs)",
		Default = true,
	})
	hideToggle:OnChanged(function(v)
		hideBodyEnabled = v
	end)

	local reapplyToggle = groupbox:AddToggle("LumaMorph_Reapply", {
		Text = "Reapply on respawn",
		Default = false,
	})
	reapplyToggle:OnChanged(function(v)
		reapplyEnabled = v
	end)

	lp.CharacterAdded:Connect(function()
		if reapplyEnabled and lastApplied then
			task.wait(1.5)
			loadMorph(lastApplied.category, lastApplied.file)
		end
	end)
end

---@param window table
---@param groupbox table
function VisualsTab.initProximitySection(groupbox)
	-- On-screen list of nearby players (name + distance), ported from project-rain's proximity list.
	-- NOTE: distinct flags from GameTab's "PlayerProximity" NOTIFICATION feature -- reusing those collided.
	groupbox:AddToggle("ProximityList", {
		Text = "Player Proximity List",
		Tooltip = "Show an on-screen list of nearby players with their distance (updates live).",
		Default = false,
	})

	local depBox = groupbox:AddDependencyBox()

	depBox:AddSlider("ProximityListRange", {
		Text = "Range",
		Min = 100,
		Max = 5000,
		Default = 1000,
		Rounding = 0,
		Suffix = " studs",
	})

	depBox:AddToggle("ProximityListNotify", {
		Text = "Notify On Enter / Leave",
		Default = true,
	})

	depBox:SetupDependencies({
		{ Toggles.ProximityList, true },
	})
end

function VisualsTab.initTargetHUDSection(groupbox)

		Text = "Target HUD",

		Default = false,

end

function VisualsTab.init(window)
	local tab = window:AddTab("ESP", "Visuals", Icons.Visuals)

	VisualsTab.initESPCustomization(tab:AddDynamicGroupbox("ESP Customization"))
	VisualsTab.initESPOptimizations(tab:AddDynamicGroupbox("ESP Optimizations"))
	VisualsTab.initProximitySection(tab:AddDynamicGroupbox("Player Proximity"))
	VisualsTab.initTargetHUDSection(tab:AddDynamicGroupbox("Target HUD"))

	VisualsTab.addPlayerESP(
		VisualsTab.addEntityESP(VisualsTab.initBaseESPSection("Player", tab:AddDynamicGroupbox("Player ESP")))
	)

	VisualsTab.addFilterESP(
		VisualsTab.addEntityESP(VisualsTab.initBaseESPSection("Mob", tab:AddDynamicGroupbox("Mob ESP")))
	)

	VisualsTab.initBaseESPSection("NPC", tab:AddDynamicGroupbox("NPC ESP"))
