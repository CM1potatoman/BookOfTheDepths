---The chaser fight. Returns its initial location (for the chest). Ports PR's snow-cleanse, bloodjar, waves,
---chaser-down attach-to-head + frost-grab void path.
local function handleChaserFight(chaser)
	if not chaser or not chaser:FindFirstChild("HumanoidRootPart") then
		return nil
	end
	setStage("floor 1: chaser fight")
	local initLocation = chaser.HumanoidRootPart.CFrame
	local voidingMode = cfg("L2VoidChaser", false)
	local time = tick()
	local lastFrostGrab = 0
	local atb, tween

	-- Kill/void mechanism: Fling (constant huge downward velocity), disabled when the fight ends below.
	pcall(function()
		if Toggles and Toggles.Fling then
			Toggles.Fling:SetValue(true)
		end
	end)

	pcall(function()
		repeat
			local hrp = chaser:FindFirstChild("HumanoidRootPart")
			if not hrp then
				break
			end
			local bloodjar = getBloodJar(chaser)
			local chaserDown = hrp:FindFirstChild("REP_SOUND_10463439228")
			local waves = hrp:FindFirstChild("TelegraphAttach") and hrp.TelegraphAttach:FindFirstChild("Ring")
			local foundDroppables = false

			local c = character()
			-- Long-fight snow cleanse: if we've been snowed, go burn it off.
			if tick() - time >= 250 and c and c:FindFirstChild("Torso") and c.Torso:FindFirstChild("NewSnowClump") then
				autoM1(false)
				repeat
					tempTween(CFrame.new(-5898, 479, -5921), 190)
					local closest = nil
					local floor1 = workspace:FindFirstChild("Layer2Floor1")
					for _, v in next, (floor1 and floor1:GetChildren() or {}) do
						if not v:IsA("Model") or not v:FindFirstChild("BurnOff") then
							continue
						end
						local burn = v.BurnOff
						if not burn:FindFirstChild("InteractPrompt") then
							continue
						end
						if not closest or (root().Position - burn.Position).Magnitude < (root().Position - closest.Position).Magnitude then
							closest = burn
						end
					end
					pcall(function()
						fireProx(closest.InteractPrompt)
					end)
					task.wait(1)
				until not (character() and character():FindFirstChild("Torso") and character().Torso:FindFirstChild("NewSnowClump")) or not running
				time = tick()
			end

			local spikeTrap = workspace:FindFirstChild("SpecialSpikeTrap")
			for _, v in next, (spikeTrap and spikeTrap:GetChildren() or {}) do
				if v.Name:match("SpikeMod") and v:FindFirstChild("Stalagmite") and #v.Stalagmite:GetChildren() > 0 then
					foundDroppables = true
				end
			end

			if chaser.Parent ~= nil then
				-- Void the chaser only while it's down, and only if the user asked for it.
				setVoidMobs(voidingMode and chaserDown ~= nil)

				if bloodjar and not waves and not foundDroppables then
					tempTween(CFrame.new(getPosition(bloodjar)), 250)
					if not workspace:FindFirstChild("SpikeStabEff", true) or workspace:FindFirstChild("HitTendril", true) then
						autoM1(true)
						task.wait(0.1)
						autoM1(false)
						task.wait(0.1)
					end
				elseif chaserDown then
					if voidingMode then
						autoM1(tick() - lastFrostGrab > 2 and tick() - lastFrostGrab < 3)
					else
						autoM1(true)
					end

					-- Attach to the chaser's head while it's down.
					atb = task.spawn(function()
						while chaser.Parent ~= nil and running and task.wait() do
							local h = chaser:FindFirstChild("HumanoidRootPart")
							if not h then
								break
							end
							local cf = h.CFrame * CFrame.new(0, -5, 0) * CFrame.Angles(math.rad(90), 0, 0)
							local r = root()
							if r then
								r.AssemblyLinearVelocity = Vector3.zero
							end
							if cf.Y < 20 then
								continue
							end
							if r and (r.Position - cf.Position).Magnitude > 30 then
								if not tween then
									tween = Tweening.goal(MOVE_ID, cf, false)
									setSpeed(240)
									task.delay(0, function()
										Tweening.stop(MOVE_ID)
										tween = nil
									end)
								end
								continue
							else
								if voidingMode and tick() - lastFrostGrab > 5 then
									pcall(function()
										local frost = localPlayer().Backpack:FindFirstChild(FROST_GRAB_NAME)
										if frost then
											character().CharacterHandler.Requests.ActivateMantra:FireServer(frost)
										end
									end)
									lastFrostGrab = tick()
								end
								if r then
									r.CFrame = cf
								end
							end
						end
					end)
					if struct then
						table.insert(struct.threads, atb)
					end
					repeat
						task.wait()
					until not (chaser:FindFirstChild("HumanoidRootPart") and chaser.HumanoidRootPart:FindFirstChild("REP_SOUND_10463439228")) or chaser.Parent == nil or not running
					task.wait(0.25)
					pcall(function()
						task.cancel(atb)
					end)
				else
					autoM1(false)
					tempTween(CFrame.new(-4523, 745, -5069), 125)
				end
			end
			task.wait()
		until chaser.Parent == nil or not running
	end)

	pcall(function()
		task.cancel(atb)
	end)
	Tweening.stop(MOVE_ID)
	autoM1(false)
	setVoidMobs(false)
	pcall(function()
		if Toggles and Toggles.Fling then
			Toggles.Fling:SetValue(false)
		end
	end)
	return initLocation
end

---Trap-door puzzle skip (used when the chaser's already dead and we don't want to re-fight).
local function puzzleSkip()
	setStage("floor 1: puzzle skip")
	tempTween(CFrame.new(-5538.2334, 43.929615, -5194.18164, 0.383664161, 0, -0.923472703, 0, 1, 0, 0.923472703, 0, 0.383664161), 175)

	local floor1 = workspace:FindFirstChild("Layer2Floor1")
	if not floor1 then
		return
	end
	local one, two, three, four
	for _, pp in next, floor1:GetChildren() do
		if not pp.Name:match("PuzzlePart") or not pp:FindFirstChild("Requires") then
			continue
		end
		local req = pp.Requires.Value
		if req:match("AShape") then
			one = pp
		elseif req:match("DShape") then
			two = pp
		elseif req:match("CShape") then
			three = pp
		elseif req:match("BShape") then
			four = pp
		end
		if one and two and three and four then
			break
		end
	end
	if not (one and two and three and four) then
		return
	end

	repeat
		tempTween(CFrame.new(-5579.28174, 6.74480486, -5203.97754, 0.490061522, 0, 0.871687829, 0, 1, 0, -0.871687829, 0, 0.490061522), 175)
		pcall(firetouchinterest, root(), workspace.EncounterTriggers.TrapDoorPuzzle.Spawner, 0)
		pcall(firetouchinterest, root(), workspace.EncounterTriggers.TrapDoorPuzzle.Spawner, 1)
		task.wait(1)
		tempTween(CFrame.new(-5510.97363, 5.74480391, -5181.09619, -0.191880465, 0, -0.981418312, 0, 1, 0, 0.981418312, 0, -0.191880465), 175)
		pcall(firetouchinterest, root(), workspace.EncounterTriggers.TrapDoorPuzzle.Spawner, 0)
		pcall(firetouchinterest, root(), workspace.EncounterTriggers.TrapDoorPuzzle.Spawner, 1)
		task.wait(1)
	until one:FindFirstChild("InteractPrompt", true) or not running

	local function pressPuzzle(part, times)
		tempTween(part:GetPivot() * CFrame.new(0, 4, 0), 175)
		for _ = 1, times do
			repeat
				pcall(function()
					fireProx(part:FindFirstChild("InteractPrompt", true))
				end)
				task.wait()
			until part:FindFirstChild("Shifting", true) or not running
			repeat
				task.wait()
			until not part:FindFirstChild("Shifting", true) or not running
		end
	end
	pressPuzzle(one, 2)
	pressPuzzle(two, 1)
	pressPuzzle(three, 3)
	pressPuzzle(four, 4)

	tempTween(workspace.Layer2Floor1.WindPortal.CFrame, 175)
	for _ = 1, 2 do
		pcall(firetouchinterest, root(), workspace.Layer2Floor1.WindPortal, 0)
		pcall(firetouchinterest, root(), workspace.Layer2Floor1.WindPortal, 1)
	end
end

local function floorOne()
	-- If we don't want to kill the chaser and already did, skip straight to the puzzle/portal.
	if not cfg("L2KillChaser", true) then
		local data = PersistentData.get("l2data")
		if data and data.killedChaser then
			puzzleSkip()
			return
		end
	end

	handleDoor()
	handleBonekeeper()
	handleGenerator()
	setVoidMobs(false)

	local chaser = handleSpawnChaser()
	initChaserFight(chaser)

		local second = workspace:FindFirstChild("SecondStage")
		if second then
			tempTween(second:IsA("BasePart") and second.CFrame or second:GetPivot(), 175)
		end
		return
	end

	-- Escape via the light hook (ends the run; the server teleports us out).
	local hook = workspace:FindFirstChild("InGameLightHook")
	if hook then
		tempTween(hook.CFrame, 175)
	end
end

local function handleKrysanChests()
	setStage("floor 2: kyrsan chests")
	local gui = playerGui()
	local currency = gui and gui:FindFirstChild("CurrencyGui")
	local medallions = currency and currency.CurrencyFrame:FindFirstChild("KyrsanMedallions")
	local amount = medallions and medallions:FindFirstChild("Amount")
	if amount and amount.Text == "250" then
		return
	end

	local chests = {}
	for _, chest in next, thrownChildren() do
		if chest:FindFirstChild("Lid") then
			table.insert(chests, chest)
		end
	end
	while running and #chests > 0 do
		local r = root()
		local closest, idx = nil, nil
		for i, chest in ipairs(chests) do
			if not chest.Parent or not chest:FindFirstChild("Lid") then
				continue
			end
			if not closest or (r.Position - chest.Lid.Position).Magnitude < (r.Position - closest.Lid.Position).Magnitude then
				closest, idx = chest, i
			end
		end
		if not closest then
			break
		end
		tempTween(closest.Lid.CFrame, 175)
		grabChest(closest)
		table.remove(chests, idx)
		task.wait()
	end
end

local function handleEthironFight()
	setStage("floor 2: ethiron gauntlet")
	local ultimateDetected = false
	local boneTweenActive = false
	local ethironStunned = false

	maid:mark(workspace.DescendantAdded:Connect(function(descendant)
		if not running then
			return
		end
		if descendant.Name:lower():find("bombcharge") then
			if not avatarExists() then
				return
			end
			ultimateDetected = true
			while boneTweenActive do
				task.wait()
			end
			tempTween(CFrame.new(787.787598, 1380, 7324.95801), 175)
			detectBombFullyCharged()
			detectBombFired()
			tempTween(CFrame.new(774.600281, 2000, 7465.85791), 175)
			ultimateDetected = false
		end
	end))

	local avatar = nil
	repeat
		for _, v in next, liveChildren() do
			if v.Name:find("%.avatar") then
				avatar = v
				break
			end
		end
		task.wait()
	until avatar or not running
	if not avatar then
		return
	end

	local initHeadCF = nil
	repeat
		pcall(function()
			initHeadCF = avatar:FindFirstChild("Head").CFrame
		end)
		task.wait()
	until initHeadCF or not running

	pcall(function()
		avatar.Humanoid.Animator.AnimationPlayed:Connect(function(anim)
			if anim.Animation.AnimationId:match("rbxassetid://10880826811") then
				ethironStunned = true
			elseif anim.Animation.AnimationId:match("rbxassetid://10978735338") then
				ethironStunned = false
			end
		end)
	end)

	local function hp()
		local h = humanoid()
		return h and h.Health or 0, h and h.MaxHealth or 1
	end

	repeat
		xpcall(function()
			repeat
				local bonespear = getNearestBoneSpear()
				local bonealtar = getNearestAltar()
				local health, maxHealth = hp()

				if health <= maxHealth * 0.25 then
					repeat
						if not ultimateDetected then
							tempTween(CFrame.new(774.600281, 1380, 7465.85791), 175)
						end
						task.wait()
						health = hp()
					until health >= maxHealth * 0.25 or not running
				elseif ultimateDetected then
					task.wait()
				elseif (bonespear and bonealtar and not foundFlyingBoneSpear())
					or (bonealtar and character():FindFirstChild("BoneSpear")) then
					-- Grab a spear.
					if not ultimateDetected and not character():FindFirstChild("BoneSpear") and hp() > maxHealth * 0.40 then
						boneTweenActive = true
						repeat
							pcall(function()
								tempTween(bonespear.CFrame, 175)
							end)
							pcall(function()
								fireProx(bonespear.InteractPrompt)
							end)
							task.wait()
						until bonespear == nil or bonespear.Parent == nil or character():FindFirstChild("BoneSpear") or ultimateDetected or hp() <= maxHealth * 0.40 or not running
					end
					-- Plant it on an altar.
					if not ultimateDetected and character():FindFirstChild("BoneSpear") and hp() > maxHealth * 0.40 then
						boneTweenActive = true
						repeat
							pcall(function()
								tempTween(bonealtar:GetPivot(), 175)
							end)
							pcall(function()
								fireProx(bonealtar.InteractPrompt)
							end)
							task.wait()
						until not character():FindFirstChild("BoneSpear") or ultimateDetected or hp() <= maxHealth * 0.40 or not running
					end
					boneTweenActive = false
				elseif ethironStunned and not foundFlyingBoneSpear() then
					repeat
						boneTweenActive = true
						tempTween(avatar.Head.CFrame * CFrame.new(0, -5, 0), 175)
						boneTweenActive = false
						autoM1(true)
						task.wait()
					until not ethironStunned or hp() <= maxHealth * 0.40 or not running
					autoM1(false)
				else
					boneTweenActive = true
					tempTween(initHeadCF * CFrame.new(0, 150, 0), 175)
					boneTweenActive = false
				end
				task.wait()
			until avatar.Parent == nil or not running
		end, function(err)
			Logger.notify("Auto L2 ethiron error: %s", tostring(err))
		end)
		task.wait()
	until avatar.Parent == nil or not running
	autoM1(false)
end

-- =========================================================================================================
-- Void dive into Layer 2.
-- =========================================================================================================
local function nearestKillPlane()
	local r = root()
	if not r then
		return nil
	end
	local best, bestDist = nil, math.huge
	for _, item in ipairs(collectionService:GetTagged("KillPlane")) do
		if item:IsA("BasePart") and item:IsDescendantOf(workspace) then
			local d = (item.Position - r.Position).Magnitude
			if d < bestDist then
				best, bestDist = item, d
			end
		end
	end
	return best
end

local function inLayer2()
	local d = dungeon()
	if d and tostring(d):match("Layer2") then
		return true
	end
	local a = currentArea()
	return a ~= nil and tostring(a):match("Layer2") ~= nil
end

---Dive into the void to enter Layer 2 (PR go_to_layer_two). Drops kill-brick immunity + fly so we fall.
local function goToLayerTwo()
	setStage("diving into layer 2")
	disableStructFeature("no_kill_bricks")

	local hum = humanoid()
	if hum and hum.Health <= hum.MaxHealth / 2 then
		-- PR kicks here; we just hop to a fresh server + retry instead of booting the user.
		return serverHop()
	end

	if currentArea() and tostring(currentArea()):match("Fragments of Self") then
		-- We got wiped -- stop cleanly rather than loop.
		AutoLayer2.stop()
		return
	end

	disableStructFeature("fly")

	local startedAt = tick()
	local plane = nearestKillPlane()
	repeat
		plane = nearestKillPlane()
		task.wait()
	until plane or tick() - startedAt > 15 or not running or inLayer2()

	maid:mark(task.spawn(function()
		while running do
			if playerCheck(PLAYER_SAFE_RADIUS) then
				serverHop()
				return
			end
			task.wait()
		end
	end))

	while running and tick() - startedAt < 60 do
		if inLayer2() then
			break
		end
		if not plane then
			plane = nearestKillPlane()
		end
		if plane and plane.Parent then
			setSpeed(200)
			Tweening.goal(MOVE_ID, plane.CFrame, false)
		end
		local r = root()
		if r then
			r.AssemblyLinearVelocity = Vector3.new(0, -999, 0)
		end
		task.wait()
	end
	Tweening.stop(MOVE_ID)
end

-- =========================================================================================================
-- State machine.
-- =========================================================================================================
local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "dive", from = "_check_area", to = "_to_layer_two_floor_one" },
		{ name = "floorone", from = "_check_area", to = "_floor_one" },
		{ name = "floortwo", from = "_check_area", to = "_floor_two" },
		{ name = "continuation", from = "_floor_one", to = "_check_area" },
		{ name = "continuation", from = "_floor_two", to = "_check_area" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")

			local deadline = os.clock() + 25
			repeat
				local d = dungeon()
				if d and tostring(d):match("Layer2Floor1") then
					return self:floorone()
				end
				if d and tostring(d):match("Layer2Floor2") then
					return self:floortwo()
				end
				local area = currentArea()
				if area and tostring(area):match("Fragments of Self") then
					-- Wiped out of a run; stop.
					AutoLayer2.stop()
					return
				end
				if area or game.PlaceId == DEPTHS_PLACE_ID or game.PlaceId == EASTERN_PLACE_ID then
					return self:dive()
				end
				task.wait()
			until os.clock() >= deadline or not running
		end,

		onenter_to_layer_two_floor_one = function(self)
			if not running then
				return
			end
			-- First entry always dives in. On a loop re-entry (already cleared once) only continue if the
			-- user wants loops; otherwise finish cleanly.
			local data = PersistentData.get("l2data")
			if data and data.clearedOnce and not cfg("L2AllowLoops", false) then
				Logger.notify("Auto L2: run finished (loops off).")
				AutoLayer2.stop()
				return
			end
			goToLayerTwo()
			self:continuation()
		end,

		onenter_floor_one = function(self)
			if not running then
				return
			end
			floorOne()
			PersistentData.stf("l2data", "clearedOnce", true)
			self:continuation()
		end,

		onenter_floor_two = function(self)
			if not running then
				return
			end
			floorTwo()
			PersistentData.stf("l2data", "clearedOnce", true)
			self:continuation()
		end,
	},
})

-- =========================================================================================================
-- Struct.
-- =========================================================================================================
local function buildStruct()
	return AutomationStruct.construct({
		id = "auto_layer2",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"no_fall",
			"noclip",
			"no_kill_bricks",
			"mod_detector",
			"fly",
			"m1_hold",
			"auto_equip_weapon",
		},
		character_creator_handler_used = false,
	})
end

-- =========================================================================================================
-- Public interface.
-- =========================================================================================================
function AutoLayer2.start()
	local lp = localPlayer()
	if not lp or running then
		return

local function formatDuration(seconds)
	seconds = math.floor(seconds)
	if seconds < 60 then
		return seconds .. "s"
	end
	local minutes = math.floor(seconds / 60)
	if seconds < 3600 then
		return minutes .. "m " .. (seconds % 60) .. "s"
	end
	return math.floor(minutes / 60) .. "h " .. (minutes % 60) .. "m"
end

---Connect a PERSISTENT GetScore.OnClientEvent listener: whenever the server pushes the run score, fold its
---.Echoes into efdata. Idempotent. This is what actually counts echoes -- the fragments wipe itself is
---handled by luma's Wipe module (which bypasses our state machine), so counting can't live only there.
local function connectScoreListener()
	if getScoreConn then
		return
	end
	pcall(function()
		local remote = replicatedStorage:WaitForChild("Requests"):WaitForChild("GetScore")
		getScoreConn = remote.OnClientEvent:Connect(function(v)
			local gained = (typeof(v) == "table" and tonumber(v.Echoes)) or 0
			if gained and gained > 0 then
				local data = PersistentData.get("efdata")
				if data then
					PersistentData.stf("efdata", "echoes", (data.echoes or 0) + gained)
				end
			end
		end)
		maid:mark(getScoreConn)
	end)
end

---End of a cycle: bump the cycle count and request the run score ONCE (the persistent listener folds the
---returned .Echoes into the total). A single fire so we don't double-count the same run.
local function recordCycle()
	local data = PersistentData.get("efdata")
	if data then
		PersistentData.stf("efdata", "cycles", (data.cycles or 0) + 1)
	end
	connectScoreListener()
	pcall(function()
		replicatedStorage:WaitForChild("Requests"):WaitForChild("GetScore"):FireServer()
	end)
end

local function runStats()
	while running do
		-- Keep No Fall Damage forced on for the entire farm (respawns / new characters can drop it).
		if Toggles and Toggles.NoFallDamage and not Toggles.NoFallDamage.Value then
			Toggles.NoFallDamage:SetValue(true)
		end

		local data = PersistentData.get("efdata")
		if data then
			local startedAt = data.startedAt or os.time()
			local elapsed = math.max(0, os.time() - startedAt)
			local echoes = data.echoes or 0
			if echoLabel then
				echoLabel.Text = "Echoes: " .. echoes
			end
			if timeLabel then
				timeLabel.Text = "Elapsed: " .. formatDuration(elapsed)
			end
			if epmLabel then
				epmLabel.Text = string.format("Echoes/min: %.2f", echoes / math.max(1 / 60, elapsed / 60))
			end
		end
		task.wait(0.25)
	end
end

---Per-action watchdog: if a single stage stalls longer than ACTION_TIMEOUT, run the wipe loop to reset the
---whole cycle from a fresh character (Luma's resume brings us back in afterward).
local function runActionWatchdog()
	while running do
		if actionStartedAt > 0 and os.clock() - actionStartedAt > currentActionTimeout then
			actionStartedAt = 0 -- disarm so we only fire once
			-- Titus voiding stage ("fighting titus" is the only stage on VOID_TIMEOUT): do NOT wipe --
			-- wiping nukes the character + the run's echoes. Server-hop to rejoin a fresh server instead
			-- (Luma's resume re-enters and continues), matching the relic farm. Every other stage wipes.
			if currentActionTimeout == VOID_TIMEOUT then
				-- Rejoin instead of wiping. Resolve the slot with the efdata fallback (same as the wipe
				-- path): in the dungeon the live DataSlot attribute is often missing, and serverHop() /
				-- ServerHop.hop no-ops on a nil slot -- which made the timeout look like it did nothing.
				local data = PersistentData.get("efdata")
				local slot = currentSlot() or (data and data.slot)
				setStage("titus voiding timed out -- rejoining")
				pcall(function()
					ServerHop.hop(slot, true)
				end)
				return
			end
			local data = PersistentData.get("efdata")
			local slot = currentSlot() or (data and data.slot)
			if slot then
				setStage("action timed out -- wiping")
				pcall(function()
					Wipe.invoke(slot)
				end)
			end
			return
		end
		task.wait(0.25)
	end
end

-- =========================================================================================================
-- Movement primitives (ported from PR's autotitus table).
-- =========================================================================================================
local function setSpeed(value, force)
	-- Normally the per-automation Tween Speed slider wins; `force` lets a specific move (the dungeon-door
	-- travel) pin an exact speed regardless of the slider.
	if not force then
		value = (Options and Options.EchoFarmTweenSpeed and Options.EchoFarmTweenSpeed.Value) or 100
	end
	if Options and Options.TweenStudsPerSecond and Options.TweenStudsPerSecond.Value ~= value then
		Options.TweenStudsPerSecond:SetValue(value)
	end
end

---Blocking tween to a CFrame at `speed` (PR's Tween.new(cf, true, speed).wait()).
---When `force` is set (the dungeon-door travel), step the root toward the goal at an EXACT studs/second
---ourselves instead of going through the shared Tween Speed slider. That slider is clamped (Min) and shared
---across every automation, so it can't reliably pin a slow speed -- this keeps the door approach at the
---speed we actually ask for (e.g. 10 studs/s) no matter what the slider says.
local function tempTween(cframe, speed, force)
	if force then
		Tweening.stop(MOVE_ID) -- make sure the engine isn't also driving us toward a stale goal
		local targetPos = cframe.Position
		local studsPerSecond = speed or 150
		local dt = task.wait()
		while running do
			local r = root()
			if not r then
				return
			end
			r.AssemblyLinearVelocity = Vector3.zero
			local pos = r.Position
			local toTarget = targetPos - pos
			local dist = toTarget.Magnitude
			if dist <= 1 then
				break
			end
			local stepDist = studsPerSecond * dt
			if stepDist >= dist then
				r.CFrame = CFrame.new(targetPos)
				break
			end
			r.CFrame = CFrame.new(pos + (toTarget / dist) * stepDist)
			dt = task.wait()
		end
		return
	end

	setSpeed(speed or 150, force)
	Tweening.goal(MOVE_ID, cframe, false)
	local goalPos = cframe.Position
	while running do
		local r = root()
		if r and (r.Position - goalPos).Magnitude <= 10 then
			break
		end
		task.wait()
	end
	Tweening.stop(MOVE_ID)
end

---True if another player is within `mag` of `position` (defaults to our root).
local function playerCheck(mag, position)
	local origin = position or (root() and root().Position)
	if not origin then
		return false
	end
	for _, other in next, players:GetPlayers() do
		if other == localPlayer() or not other.Character then
			continue
		end
		local hrp = other.Character:FindFirstChild("HumanoidRootPart")
		if hrp and (origin - hrp.Position).Magnitude <= mag then
			return true
		end
	end
	return false
end

local function serverHop()
	-- currentSlot() reads the live DataSlot attribute, which is often absent inside dungeons (e.g. the
	-- Titus dungeon); fall back to the farm's persisted slot so the hop resolves a real save slot instead
	-- of no-opping on nil. Makes every hop -- auto-log, exit hop, the voiding rejoin -- work in the dungeon.
	local data = PersistentData.get("efdata")
	local slot = currentSlot() or (data and data.slot)
	pcall(function()
		ServerHop.hop(slot, true)
	end)
end

-- Player-proximity "auto log" hop, gated by the per-automation auto-log distance (echo_farm slider). 0 = off,
-- so NOTHING in the farm hops on nearby players -- including the Fragments of Self cash-in.
local function shouldLogHop(position)
	local dist = AutoLog.get("echo_farm")
	if dist <= 0 then
		return false
	end
	return playerCheck(dist, position)
end

---Player-safe tween: hop away if anyone is near us/the target, else fly there at altitude and snap down
---(matches Auto Titus -- travel high above the map so nothing along the way can interrupt us).
---@param groundLevel boolean? when true, travel at ground level (no altitude lift) instead of high up.
local function playerSafeTween(cf, speed, groundLevel, force)
	if shouldLogHop() then
		serverHop()
		while task.wait() do
		end
		return
	end
	local r = root()
	if not r then
		return
	end
	if groundLevel then
		-- Match the dungeon door's Y: snap to the target's height, then tween horizontally across to it at
		-- that same Y level (no altitude lift).
		r.CFrame = CFrame.new(r.Position.X, cf.Position.Y, r.Position.Z)
		tempTween(cf, speed, force)
	else
		local y = math.random(8000, 10000)
		r.CFrame = r.CFrame + Vector3.new(0, y, 0)
		tempTween(cf + Vector3.new(0, y, 0), speed, force)
	end
	if shouldLogHop(cf.Position) then
		serverHop()
		while task.wait() do
		end
		return
	end
	r = root()
	if r then
		r.CFrame = cf
	end
end

-- =========================================================================================================
-- Dungeon flow (ported from PR titus_echofarm).
-- =========================================================================================================
local function enterDungeon()
	setStage("enter dungeon")
	-- Force noclip on for the trip to the dungeon door so geometry can't snag us on the way in.
	pcall(function()
		if Toggles and Toggles.NoClip then
			Toggles.NoClip:SetValue(true)
		end
	end)
	pcall(function()
		replicatedStorage.Requests.UpdateUXSettings:FireServer("FriendlyFire", true)
	end)

	-- Keep noclip enforced for the WHOLE trip to the door (not just once we're there): re-assert the NoClip
	-- toggle and brute-force every character part CanCollide=false each frame while we travel, so geometry
	-- can't snag us on the way in.
	local travelNoclip = runService.Heartbeat:Connect(function()
		if Toggles and Toggles.NoClip and not Toggles.NoClip.Value then
			Toggles.NoClip:SetValue(true)
		end
		local c = character()
		if c then
			for _, part in next, c:GetDescendants() do
				if part:IsA("BasePart") and part.CanCollide then
					part.CanCollide = false
				end
			end
		end
	end)

	playerSafeTween(ENTRANCE, ENTRANCE_SPEED, true, true) -- ground level (y=0) to the door; force 40 studs/s regardless of the Tween Speed slider
	pcall(function()
		travelNoclip:Disconnect()
	end)
	pcall(function()
		replicatedStorage.Requests.UpdateUXSettings:FireServer("FriendlyFire", true)
	end)

	local entry = workspace:WaitForChild("Map", 10)
	entry = entry and entry:WaitForChild("MeritEntry", 5)

	-- Pin us to the entrance while firing the prompt so nothing nudges us off the door.
	local holdConn = runService.Heartbeat:Connect(function()
		local r = root()
		if r then
			r.CFrame = ENTRANCE
		end
		-- Brute-force noclip for the door approach: force every character part CanCollide=false each frame so
		-- geometry can't snag us (doesn't rely on the NoClip toggle / Movement loop).
		local c = character()
		if c then
			for _, part in next, c:GetDescendants() do
				if part:IsA("BasePart") and part.CanCollide then
					part.CanCollide = false
				end
			end
		end
	end)

	local start = tick()
	repeat
		-- Keep noclip ON for the whole door approach (re-assert each pass -- a one-shot set can get cleared by
		-- the fresh-character spawn / feature management, same reason the no-fall loop re-asserts).
		if Toggles and Toggles.NoClip and not Toggles.NoClip.Value then
			Toggles.NoClip:SetValue(true)
		end
		if shouldLogHop(ENTRANCE.Position) then
			if holdConn then
				holdConn:Disconnect()
			end
			serverHop()
			while task.wait() do
			end
			return
		end
		if entry and entry:FindFirstChild("InteractPrompt") then
			Prompt.interact(entry.InteractPrompt)
		end
		task.wait(0.1)
	until tick() - start > 15 or game.PlaceId ~= EASTERN_PLACE_ID or not running

	if holdConn then
		holdConn:Disconnect()
	end
end

-- Old pathfind (mob_ai) breaker, baked straight into the Titus fight (no toggle). Every Heartbeat we
-- snapshot the root's real velocity, then slam it straight down (0, -9e11, 0); we restore the snapshot
-- BEFORE the next physics step so we never actually fall -- the slam lives for one tick, just long enough
-- to replicate the fling. The restore runs on PreRender AND PreSimulation: PreRender covers the focused
-- case, PreSimulation also fires while the window is UNFOCUSED (PreRender doesn't), so an AFK/tabbed-out
-- session can't skip the restore and let the slam rocket us into the void / null zone.
local titusBreakerHB = nil
local titusBreakerPR = nil
local titusBreakerPS = nil
local titusBreakerLastVel = nil

local function stopTitusBreaker()
	if titusBreakerHB then
		pcall(function()
			titusBreakerHB:Disconnect()
		end)
		titusBreakerHB = nil
	end
	if titusBreakerPR then
		pcall(function()
			titusBreakerPR:Disconnect()
		end)
		titusBreakerPR = nil
	end
	if titusBreakerPS then
		pcall(function()
			titusBreakerPS:Disconnect()
		end)
		titusBreakerPS = nil
	end
	if titusBreakerLastVel then
		local r = root()
		if r then
			pcall(function()
				r.AssemblyLinearVelocity = titusBreakerLastVel
			end)
		end
		titusBreakerLastVel = nil
	end
end

local function startTitusBreaker()
	stopTitusBreaker()

	-- Restore the pre-slam velocity. Runs on every pre-physics tick so the slam is always undone before
	-- physics integrates it (otherwise -9e11 would carry us straight into the void).
	local function restoreBreakerVelocity()
		if not titusBreakerLastVel then
			return
		end
		local r = root()
		if r then
			r.AssemblyLinearVelocity = titusBreakerLastVel
		end
		titusBreakerLastVel = nil
	end

	titusBreakerHB = runService.Heartbeat:Connect(function()
		local r = root()
		if not r then
			return
		end
		titusBreakerLastVel = r.AssemblyLinearVelocity
		r.AssemblyLinearVelocity = Vector3.new(0, -9e11, 0)
	end)

	-- PreRender restores while the window is focused; PreSimulation (falling back to Stepped) also fires
	-- when it's unfocused, so the restore can never be skipped and let the slam fling us to the null zone.
	titusBreakerPR = runService.PreRender:Connect(restoreBreakerVelocity)
	local preSim = runService.PreSimulation or runService.Stepped
	if preSim then
		titusBreakerPS = preSim:Connect(restoreBreakerVelocity)
	end
end

---Ram up and down through Titus's torso; the inline pathfind breaker does the actual killing.
local function attachToTitus()
	setStage("fighting titus")

	local titus
	repeat
		for _, ent in next, workspace.Live:GetChildren() do
			if ent.Name:lower():find("titus") and ent.Name:find("%.") then
				titus = ent
				break
			end
		end
		task.wait()
	until titus or not running
	if not titus then
		return
	end

	local torso = titus:WaitForChild("Torso")
	if not torso then
		return
	end

	local playerRoot = root()
	if not playerRoot then
		return
	end

	-- Dodge spam (PR's VIM Q loop -> luma InputClient.dodge).
	maid:mark(task.spawn(function()
		while titus and titus.Parent and running do
			pcall(function()
				InputClient.dodge({ direct = true })
			end)
			task.wait(0.35)
		end
	end))

	local targetCF = CFrame.new(torso.CFrame.X, APPROACH_Y, torso.CFrame.Z)
	playerRoot.AssemblyLinearVelocity = Vector3.zero
	tempTween(targetCF, 250)
	task.wait(0.5)
	tempTween(targetCF, 250)

	-- Kill by ramming up and down THROUGH Titus's torso: oscillate between his torso and a point VOID_SWING
	-- studs above it (position-only target -- a straight vertical lookAt is degenerate and NaNs the CFrame).
	-- The pathfind breaker does the damage; this just drives the motion. Done when he dies, or hop if he
	-- somehow gets flung up off the map.
	startTitusBreaker()
	-- Ram hard and FAST through Titus's torso at a fixed, slider-independent speed (the shared Tween Speed
	-- slider can't be trusted to hold a value at runtime, and the old setSpeed(200) was non-force so it
	-- silently ran at the ~100 slider default). Step the root straight between his torso and VOID_SWING studs
	-- above it, flipping the instant we reach an end, slamming the whole time via the breaker.
	local VOID_SWING = 100
	local RAM_SPEED = 900
	-- If Titus gets flung farther than this from us, stop chasing him. He's on his way out (flung/voiding),
	-- and gliding across the map after his torso is exactly the bug we're fixing. Hold position instead and
	-- keep swinging; he dies or comes back, and the per-action watchdog server-hops if the fight truly stalls.
	local MAX_RAM_DISTANCE = 10000
	local voidGoingIn = true
	local dt = task.wait()
	while running do
		if not titus or not titus.Parent or not torso.Parent then
			break
		end
		if torso.Position.Y > 2000 then
			stopTitusBreaker()
			serverHop()
			return
		end

		playerRoot = root()
		if playerRoot then
			playerRoot.AssemblyLinearVelocity = Vector3.zero

			local ramPos = torso.Position + Vector3.new(0, voidGoingIn and VOID_SWING or 0, 0)
			local toTarget = ramPos - playerRoot.Position
			local dist = toTarget.Magnitude
			-- Only ram/move while he's close. Flung too far -> don't chase; lift up to Y=1000 and wait him out.
			if dist <= MAX_RAM_DISTANCE then
				local stepDist = RAM_SPEED * dt
				if dist <= stepDist or dist < 3 then
					playerRoot.CFrame = CFrame.new(ramPos)
					voidGoingIn = not voidGoingIn
				else
					playerRoot.CFrame = CFrame.new(playerRoot.Position + (toTarget / dist) * stepDist)
				end
			else
				-- Titus flung past MAX_RAM_DISTANCE: sit at Y=1000 (keep our X/Z) instead of gliding after him.
				local pos = playerRoot.Position
				playerRoot.CFrame = CFrame.new(pos.X, 1000, pos.Z)
			end
			pcall(function()
				character().CharacterHandler.Requests.DrawWeapon:FireServer(true)
			end)
		end

		dt = task.wait()
	end
	stopTitusBreaker()
	Tweening.stop(MOVE_ID)
end

---Loot the boss chest (echo: force Loot All).
local function grabChest()
	setStage("looting chest")

	-- Find the chest (wait for it to spawn after the kill).
	local chest
	local findStart = tick()
	repeat
		local thrown = workspace:FindFirstChild("Thrown")
		if thrown then
			for _, object in next, thrown:GetChildren() do
				if object:FindFirstChild("Lid") then
					chest = object
					break
				end
			end
		end
		if not chest then
			task.wait()
		end
	until chest or not running or (tick() - findStart > 15)
	if not chest then
		return
	end

	-- Chest appeared: snap straight onto it and hold there. After the post-kill teleport we're ~1300 studs
	-- up, so don't rely on a slow descend tween -- lock onto the chest position immediately, then the
	-- re-tween loop below keeps us glued to it while we open + loot.
	do
		local rr = root()
		if rr then
			rr.AssemblyLinearVelocity = Vector3.zero
			rr.CFrame = CFrame.new(chest.Lid.CFrame.Position)
		end
	end

	-- Open: same logic as AutoSaramed -- tween onto the lid, then keep firing the prompt (re-tweening each
	-- pass so we stay glued to it) until the ChoicePrompt appears.
	tempTween(chest.Lid.CFrame, 170)
	local promptStart = tick()
	repeat
		task.wait(0.1)
		pcall(function()
			Prompt.interact(chest:FindFirstChildWhichIsA("ProximityPrompt"))
		end)
		pcall(function()
			tempTween(chest.Lid.CFrame, 170)
		end)
	until (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) or not running or (tick() - promptStart > 15)

	local gui = playerGui()
	if not (gui and gui:FindFirstChild("ChoicePrompt")) then
		return
	end

	local time = tick()
	repeat
		task.wait()
		gui = playerGui()
		local cp = gui and gui:FindFirstChild("ChoicePrompt")
		if cp then
			pcall(function()
				cp:FindFirstChild("Choice"):FireServer("LOOT_ALL")
			end)
		end
	until not (gui and gui:FindFirstChild("ChoicePrompt")) or not running or (tick() - time >= 3)

	gui = playerGui()
	local cp = gui and gui:FindFirstChild("ChoicePrompt")
	if cp then
		pcall(function()
			cp:FindFirstChild("Choice"):FireServer("EXIT")
		end)
	end
	-- Intentionally NOT tweening back down to Y=0 -- stay at altitude after looting.
end

---Idol relief-from-flaws + weapon enchant (PR echo getechoes).
local function getEchoes()
	setStage("enchant / idol")

	local lp = localPlayer()
	local backpack = lp and lp:FindFirstChildOfClass("Backpack")
	local char = character()
	if not backpack or not char then
		return
	end
	local hum = humanoid()

	local function equipWeapon(weaponName)
		if not weaponName then
			return false
		end
		for _, tool in ipairs(backpack:GetChildren()) do
			if tool:IsA("Tool") and tool.Name:match(weaponName) then
				hum:EquipTool(tool)
				task.wait(0.1)
				tool:Activate()
				return true
			end
		end
		return false
	end

	-- Idol of Yun'Shul -> relief from flaws.
	local idol = backpack:FindFirstChild("Idol of Yun'Shul")
	if idol and running then
		hum:EquipTool(idol)
		task.wait(0.3)
		for _ = 1, 3 do
			idol:Activate()
			task.wait(0.1)
			if playerGui() and playerGui():FindFirstChild("ChoicePrompt") then
				break
			end
		end
		local prompt = playerGui() and playerGui():WaitForChild("ChoicePrompt", 5)
		if prompt and prompt:FindFirstChild("Choice") then
			prompt.Choice:FireServer("Give me relief from my Flaws.")
			local start = tick()
			repeat
				task.wait(0.1)
			until not (playerGui() and playerGui():FindFirstChild("ChoicePrompt")) or tick() - start > 5
		end
	end

	-- Enchant the main weapon.
	local enchantStone
	for _, tool in ipairs(backpack:GetChildren()) do
		if tool:IsA("Tool") and tool.Name:match("Enchant") then
			enchantStone = tool
			break
		end
	end

	if enchantStone and running then
		local weaponHolder = backpack:FindFirstChild("Weapon")
		local currentWeapon = weaponHolder and weaponHolder:FindFirstChild("Weapon") and weaponHolder.Weapon.Value
		pcall(function()
			replicatedStorage:WaitForChild("Requests"):WaitForChild("Unequip"):InvokeServer("Weapon", true)
		end)
		task.wait(0.5)

		equipWeapon(currentWeapon)

		hum:EquipTool(enchantStone)
		task.wait(0.5)
		enchantStone:Activate()

		local prompt = playerGui() and playerGui():WaitForChild("ChoicePrompt", 5)
		local choice = prompt and prompt:WaitForChild("Choice", 5)
		if choice then
			choice:FireServer(true)
			task.wait(1)
		end
	end
end

local function isCharacterCreation()
	local cc = workspace:FindFirstChild("CharacterCreator")
	local c = character()
	return cc ~= nil and c ~= nil and c.Parent == cc
end

-- Ferryman-style auto log: throughout the entire farm, hop the moment another player comes within the
-- per-automation auto-log distance (adjustable 0-2000 in the menu; 0 = off). Saved per automation.
local function watchForPlayers()
	while running do
		task.wait(1)
		if not running then
			break
		end
		-- Don't hop mid-creation (lobby players would softlock the character creator).
		if isCharacterCreation() then
			continue
		end
		-- Never auto-log in Fragments of Self: it's the shared cash-in hub where other players are ALWAYS
		-- present, so proximity-hopping there just interrupts banking echoes over and over. Skip the area
		-- entirely regardless of the distance setting -- this is the "autologs in Fragments of Self" report.
		local area = currentArea()
		if area and tostring(area):match("Fragments of Self") then
			continue
		end
		local dist = AutoLog.get("echo_farm")
		if dist > 0 and playerCheck(dist) then
			serverHop()
			return
		end
	end
end

---Blow through the game's start / loading screen by firing Requests.StartMenu.Start until our character
---loads into workspace.Live. Skipped in character creation (that's its own flow). PR's framework fires this
---during init; luma has to do it per session since each cycle spans multiple joins.
---Tap a key once via VirtualInputManager (input-only; no remote).
local function pressKey(keyCode)
	pcall(function()
		local vim = Instance.new("VirtualInputManager")
		vim:SendKeyEvent(true, keyCode, false, game)
		vim:SendKeyEvent(false, keyCode, false, game)
	end)
end

local function requestStart()
	-- Blow through the loading screen by spamming A while the LoadingGui is up (input-only -- the old
	-- Requests.StartMenu.Start remote is an anticheat honeypot, firing it = instant ban). Wait for the
	-- character to spawn into workspace.Live. Skipped in character creation.
	local deadline = os.clock() + 25
	repeat
		if isCharacterCreation() then
			return
		end
		local c = character()
		if c and c.Parent and c.Parent.Name == "Live" then
			return
		end
		local pg = players.LocalPlayer and players.LocalPlayer:FindFirstChildOfClass("PlayerGui")
		if pg and pg:FindFirstChild("LoadingGui") then
			pressKey(Enum.KeyCode.A)
		end
		task.wait(0.2)
	until not running or os.clock() >= deadline
end

local stateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "continuation", from = "_create_character", to = "_check_area" },
		{ name = "continuation", from = "_handle_fragments", to = "_check_area" },
		{ name = "die", from = "_check_area", to = "_die" },
		{ name = "die", from = "_run_dungeon", to = "_die" },
		{ name = "fragments", from = "_check_area", to = "_handle_fragments" },
		{ name = "cc", from = "_check_area", to = "_create_character" },
		{ name = "enter", from = "_check_area", to = "_enter_dungeon" },
		{ name = "run", from = "_check_area", to = "_run_dungeon" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")

			-- Merit / dungeon: we always have an existing character here, so it's safe to blow through the
			-- start/loading screen with StartMenu.Start before routing.
			if game.PlaceId == EASTERN_PLACE_ID then
				requestStart()
				return self:enter()
			end
			if game.PlaceId == DUNGEON_PLACE_ID then
				requestStart()
				return self:run()
			end

			-- Anywhere else (lobby / character creator / fragments / depths): DO NOT fire StartMenu.Start --
			-- that would rush the character creator and roll the wrong character. Wait patiently to classify.
			local deadline = os.clock() + 20
			repeat
				if isCharacterCreation() then
					return self:cc()
				end
				local area = currentArea()
				if area and tostring(area):match("Fragments of Self") then
					return self:fragments()
				end
				if area and tostring(area):match("The Depths") then
					return self:die()
				end
				task.wait()
			until os.clock() >= deadline or not running
		end,

		onenter_create_character = function(self)
			if not running then
				return
			end
			setStage("create character")
			struct:create_character()
			self:continuation()
		end,

		onenter_enter_dungeon = function(self)
			if not running then
				return
			end
			enterDungeon()
		end,

		onenter_run_dungeon = function(self)
			if not running then
				return
			end
			setStage("run dungeon")

			local startTime = tick()
			repeat
				task.wait(0.2)
			until (currentArea() and tostring(currentArea()):match("Detainment")) or not running

			task.wait(1)
			attachToTitus()

			-- Titus is down: re-arm the watchdog so the post-kill exit-wait + fly-up + loot/enchant each get
			-- a fresh ACTION_TIMEOUT. The "fighting titus" cap was armed back when the FIGHT started and is
			-- shorter (VOID_TIMEOUT); leaving it armed lets it expire mid-transition and wipe the run before
			-- the idol + enchant stones are used.
			setStage("boss defeated")

			-- Reset the 30s DungeonExitFinal cap to measure from the KILL, not the dungeon-run start, so a
			-- slow fight can't eat the window the run gets to wait for the exit to spawn.
			startTime = tick()

			local hum = humanoid()
			if hum then
				hum.Died:Connect(function()
					serverHop()
				end)
			end

			local finished = false
			local core = workspace:WaitForChild("DetainmentCore", 10)
			local conn
			if core then
				conn = core.ChildAdded:Connect(function(child)
					if child.Name == "DungeonExitFinal" then
						finished = true
					end
				end)
			end

			repeat
				task.wait(0.2)
				if tick() - startTime > 30 then
					if conn then
						conn:Disconnect()
					end
					serverHop()
					while task.wait() do
					end
					return
				end
			until finished or not running

			if conn then
				conn:Disconnect()
			end

			task.wait(0.5)

			-- Tear the pathfind breaker down before looting so velocity is normal for the chest interaction.
			stopTitusBreaker()

			-- Fly up out of the arena, then loot + enchant.
			local r = root()
			if r then
				r.CFrame = r.CFrame * CFrame.new(0, 1300, 0)
			end

			xpcall(function()
				grabChest()
				getEchoes()
			end, function(err)
				Logger.notify("EchoFarm loot error: %s", tostring(err))
			end)

			-- Count this run's echoes (request the score; the persistent listener folds it in) + bump cycle.
			recordCycle()

			task.wait(3.75)

			-- Wait out any lingering danger, then wipe (obliteration) to bank echoes at Fragments of Self.
			local hum2 = humanoid()
			if hum2 and hum2:GetAttribute("DangerExpiration") and hum2:GetAttribute("DangerExpiration") > 0 then
				repeat
					task.wait()
				until not (hum2:GetAttribute("DangerExpiration") and hum2:GetAttribute("DangerExpiration") > 0)
					or not running
			end

			self:die()
		end,

		onenter_handle_fragments = function(self)
			if not running then
				return
			end
			setStage("fragments cash-in")

			local self_npc
			local deadline = os.clock() + 40
			repeat
				local npcs = workspace:FindFirstChild("NPCs")
				self_npc = npcs and npcs:FindFirstChild("Self")
				task.wait()
			until self_npc or os.clock() >= deadline or not running
			if not self_npc then
				serverHop()
				return
			end

			-- Count this run once via the persistent listener, then interact to finalize.
			recordCycle()

			local tpLocation = (self_npc:GetPivot() - Vector3.new(0, 15.7, 0))
				* CFrame.Angles(math.rad(180), math.rad(90), math.rad(180))

			-- No auto-log here: the cash-in happens IN Fragments of Self, a shared hub that's always full of
			-- other players, so a proximity hop would abort the bank-in on practically every run. Just cash in.

			local sendDialogue = replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue")
			local prompt = self_npc:WaitForChild("InteractPrompt", 10)

			local start = tick()
			repeat
				local r = root()
				if r then
					r.CFrame = tpLocation
				end
				if prompt then
					Prompt.interact(prompt)
				end
				pcall(function()
					sendDialogue:FireServer({ choice = "[The End]" })
				end)
				task.wait(0.25)
			until isCharacterCreation() or not (currentArea() and tostring(currentArea()):match("Fragments of Self")) or not running or (tick() - start > 20)

			self:continuation()
		end,

		onenter_die = function(self)
			if not running then
				return
			end
			setStage("wipe")
			local slot = currentSlot() or (PersistentData.get("efdata") and PersistentData.get("efdata").slot)
			pcall(function()
				Wipe.invoke(slot)
			end)
		end,
	},
})

	return AutomationStruct.construct({
		id = "echo_titus",
		state_machine = stateMachine,
		is_running = isRunning,
		features = {
			"no_fall",

			"auto_equip_weapon",
		},

		character_creator_handler_opts = {
			origin = "Merit",
			-- Select the Merit spawn (Authority Ensign origin), verified against DataReplication.Origin. If it
			-- doesn't apply within 10s, server-hop and retry on a fresh server instead of kicking the player.
			origin_button = {
				category = "Authority Ensign",
				spawn = "Merit",
				timeout = 10,
				on_fail = function()
					serverHop()
				end,
			},
			weapon = "Sword",
			-- "All" enables every meta modifier; then toggle "OneBit" (the can't-swim flaw) back off.
			modifiers = { "All", "OneBit" },
		},
	})
end

-- Layer 2 (Chaser) echo farm -- a luma port of project-rain-oss's `auto_echo_layer2.lua`. Reuses every

---Fire Requests.SendDialogue with a choice (or exit), matching PR's send_dialogue.
local function sendDialogue(textOrNil, exit)
	pcall(function()
		local send = replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue")
		if exit then
			send:FireServer({ exit = true })
		else
			send:FireServer({ choice = textOrNil })
		end
	end)
	if not exit then
		task.wait(0.5)
	end
end

---Disable + forget a struct feature mid-run (same pattern EchoFarm uses to drop mob_ai_breaker before loot).
local function disableStructFeature(name)
	pcall(function()
		local feat = struct and struct.active_features and struct.active_features[name]
		if feat then
			feat:disable()
			struct.active_features[name] = nil
		end
	end)
end

local function liveChildren()
	local live = workspace:FindFirstChild("Live")
	return live and live:GetChildren() or {}
end

local function nearestKillPlane()
	local r = root()
	if not r then
		return nil
	end
	local best, bestDist = nil, math.huge
	for _, item in ipairs(game:GetService("CollectionService"):GetTagged("KillPlane")) do
		if item:IsA("BasePart") and item:IsDescendantOf(workspace) then
			local d = (item.Position - r.Position).Magnitude
			if d < bestDist then
				best, bestDist = item, d
			end
		end
	end
	return best
end

local function inLayer2()
	local lp = localPlayer()
	local dungeon = lp and lp:GetAttribute("Dungeon")
	if dungeon and tostring(dungeon):match("Layer2") then
		return true
	end
	local area = currentArea()
	return area ~= nil and tostring(area):match("Layer2") ~= nil
end

---Dive into the void so the server routes us into Layer 2: drop kill-brick immunity + fly so we actually
---fall, then slam downward velocity into the nearest kill plane until the area flips to Layer 2.
local function enterVoid()
	if currentArea() and tostring(currentArea()):match("Fragments of Self") then
		return
	end
	setStage("entering the void")
	disableStructFeature("no_kill_bricks")
	disableStructFeature("fly")

	local startedAt = tick()
	local plane = nearestKillPlane()
	repeat
		plane = nearestKillPlane()
		task.wait()
	until plane or tick() - startedAt > 15 or not running or inLayer2()

	while running and tick() - startedAt < 60 do
		if inLayer2() then
			break
		end
		if playerCheck(300) then
			serverHop()
			while task.wait() do
			end
			return
		end
		local r = root()
		if r then
			r.AssemblyLinearVelocity = Vector3.new(0, -999, 0)
		end
		if plane and plane.Parent then
			-- Void physics need the full dive speed (slowing this broke the dive); keep it at 200 even though
			-- the L2 travel tweens are slower.
			setSpeed(200)
			Tweening.goal(MOVE_ID, plane.CFrame, false)
		else
			plane = nearestKillPlane()
		end
		task.wait()
	end
	Tweening.stop(MOVE_ID)
end

---Tween to an NPC under workspace.NPCs, fire its prompt until the dialogue opens, then pick `choice` (if any)
---and/or exit. Shared by the key/door/generator steps.
local function tweenToNpcAndTalk(npcName, choice, exitAfter)
	local npc
	local findStart = tick()
	repeat
		local npcs = workspace:FindFirstChild("NPCs")
		npc = npcs and npcs:FindFirstChild(npcName)
		task.wait()
	until npc or tick() - findStart > 20 or not running
	if not npc then
		return false
	end

	local start = tick()
	repeat
		pcall(function()
			tempTween(npc:GetPivot(), 150)
		end)
		local prompt = npc:FindFirstChild("InteractPrompt")
		if prompt then
			Prompt.interact(prompt)
		end
		task.wait(0.5)
		local gui = playerGui()
		if gui and gui:FindFirstChild("DialogueGui") and gui.DialogueGui.Enabled then
			break
		end
	until tick() - start > 20 or not running

	if choice then
		sendDialogue(choice, false)
	end
	if exitAfter then
		sendDialogue(nil, true)
	end
	return true
end

---Bonekeeper encounter: enable VoidMobs (clears trash mobs), wait for the keeper to spawn + its bridge
---barrier to drop (PR spawn_keeper).
local function spawnKeeper()
	setStage("bonekeeper")
	tempTween(L2_BONEKEEPER, 150)
	pcall(function()
		if Toggles and Toggles.VoidMobs then
			Toggles.VoidMobs:SetValue(true)
		end
	end)

	local keeper
	local startedAt = tick()
	repeat
		if not keeper then
			pcall(function()
				tempTween(L2_BONEKEEPER_APPROACH, 100)
			end)
		end
		for _, ent in next, liveChildren() do
			if ent ~= character() and ent.Name:find("%.boneboy") then
				keeper = ent
			end
		end
		task.wait()
	until keeper or tick() - startedAt > 40 or not running

	-- Wait for the boneboy to disappear (bridge barrier drops), hovering 100 studs directly above it the whole
	-- time. Tweened toward keeper.Position + 100Y every frame (tweening both axes is anticheat-safe); once the
	-- keeper is gone we just hold and keep watching the barrier.
	local bridgeStart = tick()
	repeat
		if keeper and keeper.Parent then
			local kp = keeper:GetPivot().Position
			setSpeed(150)
			Tweening.goal(MOVE_ID, CFrame.new(kp.X, kp.Y + 100, kp.Z), false)
		end
		task.wait()
		local triggers = workspace:FindFirstChild("EncounterTriggers")
		local bridge = triggers and (triggers:FindFirstChild("BonekeeperBridge") or triggers:FindFirstChild("BoneKeeperBridge"))
		if not bridge or not bridge:FindFirstChild("Barrier") then
			break
		end
	until tick() - bridgeStart > 40 or not running
	Tweening.stop(MOVE_ID)
end

---Find + engage the chaser (PR chaser_route). Returns the chaser model (or nil).
local function chaserRoute()
	setStage("finding chaser")
	tempTween(L2_CHASER_HIGH, 150)

	local chaser
	local startedAt = tick()
	repeat
		pcall(function()
			tempTween(L2_CHASER_DOWN, 150)
		end)
		for _, ent in next, liveChildren() do
			if ent ~= character() and ent.Name:find("chaser") then
				chaser = ent
				break
			end
		end
		if not chaser then
			pcall(function()
				tempTween(L2_CHASER_ANGLE, 150)
			end)
		end
		task.wait()
	until chaser or tick() - startedAt > 40 or not running
	if not chaser then
		return nil
	end

	-- Last approach before the fight. The anticheat flags HORIZONTAL (X/Z) position snaps but not VERTICAL
	-- ones, so horizontal travel must be tweened and only Y may be snapped. Snap UP to Y=20000, TWEEN
	-- horizontally to directly above the chaser at that altitude, then snap straight DOWN onto it.
	do
		local r = root()
		if r then
			r.CFrame = CFrame.new(r.Position.X, 20000, r.Position.Z)
		end
		local cp = chaser:GetPivot().Position
		tempTween(CFrame.new(cp.X, 20000, cp.Z), 150)
		r = root()
		if r then
			-- Vertical-only snap down onto the chaser (keep our tweened X/Z, drop Y to the chaser's).
			r.CFrame = CFrame.new(r.Position.X, chaser:GetPivot().Position.Y, r.Position.Z)
		end
	end

	local talkStart = tick()
	repeat
		-- Hold on the chaser by TWEENING (never a horizontal snap) while we fire the prompt.
		pcall(function()
			tempTween(chaser:GetPivot(), 150)
		end)
		local prompt = chaser:FindFirstChild("InteractPrompt")
		if prompt then
			Prompt.interact(prompt)
		end
		task.wait(0.5)
		local gui = playerGui()
		if gui and gui:FindFirstChild("DialogueGui") and gui.DialogueGui.Enabled then
			break
		end
	until tick() - talkStart > 20 or not running

	drawWeapon()
	-- PR's dialogue chain that starts the fight.
	sendDialogue("Ethiron's Wake?", false)
	sendDialogue("What happened here?", false)
	sendDialogue("They mutinied?", false)
	sendDialogue("The City?", false)
	return chaser
end

---Void the chaser: EchoFarm.voiding engages luma's chaser hooks (own-not-drop + Fling), we extend
---our simulation radius + fling its owned root up, and orbit it to keep it owned. Done when it's gone or
---flung past L2_VOID_Y.
local function voidChaser(chaser)
	if not chaser then
		return
	end
	setStage("voiding chaser")
	EchoFarm.voiding = true
	pcall(function()
		sethiddenproperty(localPlayer(), "MaxSimulationRadius", 5000)
		sethiddenproperty(localPlayer(), "SimulationRadius", 5000)
	end)

	local startedAt = tick()
	while running and chaser.Parent do
		drawWeapon()
		local hrp = chaser:FindFirstChild("HumanoidRootPart")
		if hrp then
			if hrp.Position.Y > L2_VOID_Y then
				break
			end
			-- Orbit just above its root to keep it owned + in range (void physics -- keep full speed)...
			setSpeed(200)
			Tweening.goal(MOVE_ID, CFrame.new(hrp.Position) * CFrame.new(0, 6, 0), false)
			-- ...and fling it straight up (works once we're its network owner; harmless otherwise).
			pcall(function()
				hrp.AssemblyLinearVelocity = Vector3.new(0, 12500, 0)
			end)
		end
		if tick() - startedAt > ACTION_TIMEOUT then
			break
		end
		task.wait()
	end

	Tweening.stop(MOVE_ID)
	EchoFarm.voiding = false
end

-- Layer 2 state machine. Mirrors the Titus one's shared states (check_area / create_character / fragments /
-- die), but routes by Layer 2 geography and runs the void + chaser route instead of the Titus dungeon.
local l2StateMachine = StateMachine.create({
	initial = "idle",
	events = {
		{ name = "start", from = "idle", to = "_check_area" },
		{ name = "continuation", from = "_create_character", to = "_check_area" },
		{ name = "continuation", from = "_handle_fragments", to = "_check_area" },
		{ name = "continuation", from = "_enter_void", to = "_check_area" },
		{ name = "die", from = "_check_area", to = "_die" },
		{ name = "die", from = "_run_void", to = "_die" },
		{ name = "fragments", from = "_check_area", to = "_handle_fragments" },
		{ name = "cc", from = "_check_area", to = "_create_character" },
		{ name = "enter_void", from = "_check_area", to = "_enter_void" },
		{ name = "run_void", from = "_check_area", to = "_run_void" },
	},
	callbacks = {
		onenter_check_area = function(self)
			if not running then
				return
			end
			setStage("check area")

			-- Blow through the start/loading screen so the character actually spawns before we classify
			-- (skipped automatically during character creation so we don't rush the creator).
			if not isCharacterCreation() then
				requestStart()
			end

			local deadline = os.clock() + 25
			repeat
				if isCharacterCreation() then
					return self:cc()
				end
				if inLayer2() then
					return self:run_void()
				end
				local area = currentArea()
				if area and tostring(area):match("Fragments of Self") then
					return self:fragments()
				end
				-- Have a spawned character in the overworld (not creating, not in Layer 2, not at Fragments)
				-- -> dive into the void. Gate on the character being in Live, NOT on CurrentArea being set:
				-- right after creation the area attribute is often still nil, which previously left this
				-- stuck here (never diving) until it timed out.
				local c = character()
				if c and c.Parent and c.Parent.Name == "Live" then
					return self:enter_void()
				end
				task.wait()
			until os.clock() >= deadline or not running
		end,

		onenter_create_character = function(self)
			if not running then
				return
			end
			setStage("create character")
			struct:create_character()
			self:continuation()
		end,

		onenter_enter_void = function(self)
			if not running then
				return
			end
			enterVoid()
			self:continuation()
		end,

		onenter_run_void = function(self)
			if not running then
				return
			end
			setStage("run layer 2")

			local startTime = tick()
			repeat
				task.wait(0.2)
			until inLayer2() or not running or tick() - startTime > 30

			getKey()
			spawnKeeper()
			goToGenerator()
			local chaser = chaserRoute()
			voidChaser(chaser)

			recordCycle()
			task.wait(2)
			self:die()
		end,

		onenter_handle_fragments = function(self)
			if not running then
				return
			end
			setStage("fragments cash-in")

			local self_npc
			local deadline = os.clock() + 40
			repeat
				local npcs = workspace:FindFirstChild("NPCs")
				self_npc = npcs and npcs:FindFirstChild("Self")
				task.wait()
			until self_npc or os.clock() >= deadline or not running
			if not self_npc then
				serverHop()
				return
			end

			recordCycle()

			local tpLocation = (self_npc:GetPivot() - Vector3.new(0, 15.7, 0))
				* CFrame.Angles(math.rad(180), math.rad(90), math.rad(180))

			if shouldLogHop(tpLocation.Position) then
				serverHop()
				while task.wait() do
				end
				return
			end

			local sendDlg = replicatedStorage:WaitForChild("Requests"):WaitForChild("SendDialogue")
			local prompt = self_npc:WaitForChild("InteractPrompt", 10)

			local start = tick()
			repeat
				local r = root()
				if r then
					r.CFrame = tpLocation
				end
				if prompt then
					Prompt.interact(prompt)
				end
				pcall(function()
					sendDlg:FireServer({ choice = "[The End]" })
				end)
				task.wait(0.25)
			until isCharacterCreation() or not (currentArea() and tostring(currentArea()):match("Fragments of Self")) or not running or (tick() - start > 20)

			self:continuation()
		end,

		onenter_die = function(self)
			if not running then
				return
			end
			setStage("wipe")
			local slot = currentSlot() or (PersistentData.get("efdata") and PersistentData.get("efdata").slot)
			pcall(function()
				Wipe.invoke(slot)
			end)
		end,
	},
})

-- =========================================================================================================
-- Public interface.
-- =========================================================================================================
function EchoFarm.start()
	local lp = localPlayer()
	if not lp then
		return
	end
	local data = PersistentData.get("efdata")
	if not data or not data.slot then
		return
	end
	if running then
		return
	end

	running = true
	buildOverlay()
	setStage("starting")
	AntiAFK.start("EchoFarm")

	maid:mark(task.spawn(runStats))

	-- Per-action 60s watchdog: wipe-reset the cycle if a stage stalls.
	maid:mark(task.spawn(runActionWatchdog))

	-- Ferryman-style auto log: hop if any player comes within 1000 studs.
	maid:mark(task.spawn(watchForPlayers))

	-- Persistent echo counter for this session.
	connectScoreListener()

	-- Layer 2 echo farm was removed -- only the Titus farm runs now (force it even if stale efdata says Layer2).
	local sm = stateMachine
	sm.current = "idle"
	sm.asyncState = StateMachine.NONE
	sm.currentTransitioningEvent = nil

	struct = buildStruct()
	struct:run()
end

function EchoFarm.invoke(mode)
	local lp = localPlayer()
	if not lp then
		return
	end
	if running then
		return
	end

	local slot = currentSlot()
	if not slot then
		return Logger.notify("EchoFarm: no data slot found.")
	end

	-- Fresh run stats. Layer 2 echo farm was removed -- always Titus regardless of the passed mode.
	PersistentData.set("efdata", {
		slot = slot,
		mode = "Titus",
		startedAt = os.time(),
		echoes = 0,
		cycles = 0,
	})

	-- Wipe the current character on start; the wipe teleports us out, and Luma's resume (efdata) brings us
	-- back into a fresh farm cycle (character creation -> enter -> fight -> ...).
	pcall(function()
		Wipe.invoke(slot)
	end)
end

function EchoFarm.stop()
	if not running then
		return
	end
	running = false

	-- Clear the chaser-void flag so the Exploits / OwnershipWatcher hooks don't keep running after a stop.
	EchoFarm.voiding = false
	stopTitusBreaker()

	if struct then
		pcall(function()
			struct:stop()
		end)
		struct = nil
	end

	Tweening.stop(MOVE_ID)

	pcall(function()
		AntiAFK.stop("EchoFarm")
	end)

	maid:clean()
	getScoreConn = nil
	destroyOverlay()

	PersistentData.set("efdata", nil)
end

return EchoFarm
end)

-- Isolate each feature's init/detach: one feature erroring can no longer abort loading/unloading the rest.
-- This is the core of the re-execution fix -- before, a single throwing detach left every later module's
-- connections/threads/hooks leaked, and they piled up across teleports until the client died.
---@param module table
---@param method string
local function isolated(module, method)
	local fn = module[method]
	if type(fn) ~= "function" then
		return
	end

	pcall(fn)
end

---@note: Careful with features that have entire return LPH_NO_VIRTUALIZE(function() blocks. We assume that we don't care about what's placed in there.
function Features.init()
	for _, module in ipairs(INIT_ORDER) do
		isolated(module, "init")
	end

	if not armorshield or armorshield.current_role == "builder" then
		isolated(AnimationVisualizer, "init")
	end

	if not LPH_OBFUSCATED or shared.Luma.verbose then
		Logger.warn("Features initialized.")
	end
end

function Features.detach()
	-- Admin-panel-driven exploits: stop them no matter the game branch so their render loops
	-- (and the inspector's namecall hook) never survive an unload.
	isolated(AnimationFlooder, "detach")
	isolated(AnimationInspector, "detach")

	if not armorshield or armorshield.current_role == "builder" then
		isolated(AnimationVisualizer, "detach")
	end

	for _, module in ipairs(DETACH_ORDER) do
		isolated(module, "detach")
	end

	Logger.warn("Features detached.")
end

return Features

-- Small helpers
local function mk(class, props, parent)
	local inst = Instance.new(class)
	for key, value in pairs(props) do
		inst[key] = value
	end
	if parent then
		inst.Parent = parent
	end
	return inst
end

local function fmtNum(value, digits)
	if type(value) ~= "number" then
		return "?"
	end
	return string.format("%." .. (digits or 2) .. "f", value)
end

local function safeGet(fn, fallback)
	local ok, value = pcall(fn)
	if ok and value ~= nil then
		return value
	end
	return fallback
end

-- The real asset behind an AnimationId.
local function assetIdOf(animationId)
	local text = tostring(animationId or "")
	local viaQuery = text:match("[?&]id=(%d+)")
	if viaQuery then
		return tonumber(viaQuery)
	end
	return tonumber(text:match("(%d+)"))
end

-- Query junk (a `?` AND an `&`) beyond a plain asset reference = id-spoof tell.
local function isMutatedId(animationId)
	local text = tostring(animationId or "")
	return text:find("?", 1, true) ~= nil and text:find("&", 1, true) ~= nil
end

local function lookupName(assetId)
	local ok, info = pcall(function()
		return MarketplaceService:GetProductInfo(assetId, Enum.InfoType.Asset)
	end)
	local resolved = (ok and info and info.Name) or false
	nameCache[assetId] = resolved
	namePending[assetId] = nil
	return resolved
end

local function queueName(assetId)
	if not assetId or nameCache[assetId] ~= nil or namePending[assetId] then
		return
	end
	namePending[assetId] = true
	nameQueue[#nameQueue + 1] = assetId
end

-- One background worker chews through lookups (GetProductInfo yields + rate-limits).
local function startResolver()
	if resolverRunning then
		return
	end
	resolverRunning = true
	task.spawn(function()
		while running do
			local assetId = table.remove(nameQueue, 1)
			if assetId then
				lookupName(assetId)
				task.wait(0.3)
			else
				task.wait(0.25)
			end
		end
		resolverRunning = false
	end)
end

-- File logging -- name / id / priority, and any change to them, appended to a txt in the executor's workspace folder for later reading.
local function loggingEnabled()
	return Configuration.expectToggleValue("AnimInspectLogToFile") == true
end

local function writeLog(text)
	pcall(function()
		if appendfile then
			appendfile(LOG_FILE, text)
		elseif writefile and readfile and isfile then
			local existing = isfile(LOG_FILE) and readfile(LOG_FILE) or ""
			if #existing < 5 * 1024 * 1024 then
				writefile(LOG_FILE, existing .. text)
			end
		end
	end)
end

local function flushLog()
	if #logBuffer == 0 then
		return
	end
	local text = table.concat(logBuffer, "\n") .. "\n"
	logBuffer = {}
	writeLog(text)
end

local function logLine(text)
	if not loggingEnabled() then
		return
	end
	logBuffer[#logBuffer + 1] = string.format("[%8.2f] %s", os.clock() - logSessionStart, text)
	if #logBuffer >= 400 then
		flushLog()
	end
end

local function updateRow(row)
	local track = row.track
	local anim = safeGet(function() return track.Animation end)
	local animationId = anim and tostring(anim.AnimationId) or nil
	local mutated = isMutatedId(animationId)

	local assetId = row.assetId
	local resolved = assetId and nameCache[assetId] or nil
	local displayName = resolved or (anim and anim.Name ~= "" and anim.Name) or "?"
	if displayName == "Animation" and assetId then
		displayName = "id:" .. tostring(assetId)
	end

	local priorityName = tostring(safeGet(function() return track.Priority.Name end, "?"))

	row.nameLabel.Text = string.format("%s   ·   %s", displayName, priorityName)
	row.nameLabel.TextColor3 = mutated and RISK or FG

	local args = playArgs[track]
	local fade = args and fmtNum(args.fade) or "?"

	row.propLabel.Text = string.format(
		"spd %s · wt %s/%s · len %s · t %s · fade %s%s",
		fmtNum(safeGet(function() return track.Speed end)),
		fmtNum(safeGet(function() return track.WeightCurrent end)),
		fmtNum(safeGet(function() return track.WeightTarget end)),
		fmtNum(safeGet(function() return track.Length end)),
		fmtNum(safeGet(function() return track.TimePosition end)),
		fade,
		mutated and "  · MUTATED" or ""
	)

	row.idLabel.Text = "id " .. (animationId or "?")
	row.idLabel.TextColor3 = mutated and RISK or ACCENT

	-- Peak WeightCurrent across this track's life. WeightCurrent is ~0 for EVERY
	-- animation the instant it starts (they all ramp up from 0), so a single play-time
	-- read tells you nothing -- the tell for a 0-weight flood bait is that it NEVER
	-- ramps. Sample the peak here every update and log it on stop.
	row.peakWC = math.max(row.peakWC or 0, safeGet(function() return track.WeightCurrent end, 0) or 0)

	-- File log: one line on first sight, then any change to name / id / priority.
	local idStr = animationId or "?"
	if row.loggedId == nil then
		row.loggedName = displayName
		row.loggedId = idStr
		row.loggedPriority = priorityName
		logLine(string.format('[+] name="%s" id=%s priority=%s wt=%s spd=%s', displayName, idStr, priorityName,
			fmtNum(safeGet(function() return track.WeightTarget end)),
			fmtNum(safeGet(function() return track.Speed end))))
	else
		if idStr ~= row.loggedId then
			logLine(string.format("[~] id=%s  id: %s -> %s", row.loggedId, row.loggedId, idStr))
			row.loggedId = idStr
		end
		if displayName ~= row.loggedName then
			logLine(string.format('[~] id=%s  name: "%s" -> "%s"', row.loggedId, row.loggedName, displayName))
			row.loggedName = displayName
		end
		if priorityName ~= row.loggedPriority then
			logLine(string.format("[~] id=%s  priority: %s -> %s", row.loggedId, row.loggedPriority, priorityName))
			row.loggedPriority = priorityName
		end
	end
end

local function removeRow(track)
	local row = rows[track]
	if not row then
		return
	end
	if row.loggedId then
		logLine(string.format("[-] id=%s  (stopped) peakwc=%s", row.loggedId, fmtNum(row.peakWC or 0)))
	end
	rows[track] = nil
	rowCount = rowCount - 1
	if row.stoppedConn then
		pcall(function() row.stoppedConn:Disconnect() end)
	end
	if row.frame then
		row.frame:Destroy()
	end
	updateEmptyState()
end

	queueName(assetId)

	local row = { frame = frame, nameLabel = nameLabel, propLabel = propLabel, idLabel = idLabel, assetId = assetId, track = track }
	rows[track] = row
	rowCount = rowCount + 1

	-- DELETE the row the instant the animation ends.
	row.stoppedConn = track.Stopped:Connect(function()
		removeRow(track)
	end)

	updateRow(row)
	updateEmptyState()
end

local function clearRows()
	for track in pairs(rows) do
		removeRow(track)
	end
	rowCount = 0
	rows = {}
	updateEmptyState()
end

-- Live refresh loop: repaints visible rows and prunes tracks that stopped without firing .Stopped.
local function startUpdateLoop()
	if updateConn then
		return
	end
	updateConn = RunService.RenderStepped:Connect(function(dt)
		if not running then
			return
		end
		updateAccum = updateAccum + dt
		if updateAccum < UPDATE_INTERVAL then
			return
		end
		updateAccum = 0

		local dead = nil
		for track, row in pairs(rows) do
			if not safeGet(function() return track.IsPlaying end, false) then
				dead = dead or {}
				dead[#dead + 1] = track
			else
				updateRow(row)
			end
		end
		if dead then
			for _, track in ipairs(dead) do
				removeRow(track)
			end
		end

		logFlushAccum = logFlushAccum + UPDATE_INTERVAL
		if logFlushAccum >= LOG_FLUSH_INTERVAL then
			logFlushAccum = 0
			flushLog()
		end
	end)
end

-- Parallel-hook guard (see Game/Hooking.lua): on executors without parallel_hooks_allowed, install via a
-- read-only metatable swap so this __namecall hook can't be dispatched from an Actor parallel phase, which
-- is an illegal cross-phase op the engine Kernel panics on (RSL - panic: ...Kernel).
local parallelHooksAllowed = parallel_hooks_allowed
local hookmetamethod = (not parallelHooksAllowed)
	and function(target, method, hook)
		local metatable = getrawmetatable(target)
		setreadonly(metatable, false)
		local original = metatable[method]
		metatable[method] = hook
		setreadonly(metatable, true)
		return original
	end
	or hookmetamethod

-- Namecall hook -- capture the caller's Play(fade, weight, speed) args.
local function installHook()
	if hookActive then
		return
	end
	if not (hookmetamethod and getnamecallmethod and checkcaller) then
		return -- fade time simply stays "?" without exploit hook support
	end
	local ok = pcall(function()
		oldNamecall = hookmetamethod(game, "__namecall", function(self, ...)
			-- NOTE: deliberately NOT skipping checkcaller() calls.
			if not hookActive then
				return oldNamecall(self, ...)
			end
			if getnamecallmethod() == "Play" then
				local okClass, isTrack = pcall(game.IsA, self, "AnimationTrack")
				if okClass and isTrack then
					local a, b, c = ...
					playArgs[self] = {
						fade = type(a) == "number" and a or 0.1,
						weight = type(b) == "number" and b or 1,
						speed = type(c) == "number" and c or 1,
					}
				end
			end
			return oldNamecall(self, ...)
		end)
	end)
	hookActive = ok
end

local function removeHook()
	if not hookActive then
		return
	end
	hookActive = false
	if oldNamecall and hookmetamethod then
		pcall(function()
			hookmetamethod(game, "__namecall", oldNamecall)
		end)
		oldNamecall = nil
	end
end

-- Target binding
local function bindAnimator(animator, character)
	if animatorConn then
		pcall(function() animatorConn:Disconnect() end)
		animatorConn = nil
	end
	if not animator then
		return
	end

	animatorConn = animator.AnimationPlayed:Connect(function(track)
		if running then
			addRow(track)
		end
	end)

	-- Seed with whatever is already playing.
	for _, track in ipairs(safeGet(function() return animator:GetPlayingAnimationTracks() end, {})) do
		addRow(track)
	end
end

local function hookCharacter(character)
	clearRows()
	if not character then
		return
	end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
	if animator then
		bindAnimator(animator, character)
		return
	end
	-- Character just spawned; its Animator may not exist yet.
	task.spawn(function()
		local hum = character:WaitForChild("Humanoid", 5)
		local anim = hum and hum:WaitForChild("Animator", 5)
		if anim and running and targetPlayer and targetPlayer.Character == character then
			bindAnimator(anim, character)
		end
	end)
end

local function bindPlayer(player)
	if charAddedConn then
		pcall(function() charAddedConn:Disconnect() end)
		charAddedConn = nil
	end
	if charRemovingConn then
		pcall(function() charRemovingConn:Disconnect() end)
		charRemovingConn = nil
	end

	targetPlayer = player
	targetName = player and player.Name or nil

	if titleLabel then
		titleLabel.Text = player and ("Animation Inspector — " .. player.Name) or "Animation Inspector"
	end
	updateEmptyState()

	if not player then
		clearRows()
		return
	end

	charAddedConn = player.CharacterAdded:Connect(function(character)
		if running and targetPlayer == player then
			hookCharacter(character)
		end
	end)
	charRemovingConn = player.CharacterRemoving:Connect(function()
		if running and targetPlayer == player then
			clearRows()
		end
	end)

	hookCharacter(player.Character)
end

local function resolveByName(name)
	if not name or name == "" then
		return nil
	end
	for _, p in ipairs(Players:GetPlayers()) do
		if p.Name == name then
			return p
		end
	end
	local lower = string.lower(name)
	for _, p in ipairs(Players:GetPlayers()) do
		if string.lower(p.Name) == lower or string.lower(p.DisplayName) == lower then
			return p
		end
	end
	return nil
end

-- Public API
function AnimationInspector.startByName(name)
	local player = resolveByName(name)
	if not player then
		pcall(function()
			Library:Notify("Animation Inspector: pick a player first.", 4)
		end)
		pcall(function()
			local toggles = getgenv and getgenv().Toggles
			if toggles and toggles.AnimationInspector then
				toggles.AnimationInspector:SetValue(false)
			end
		end)
		return
	end

	running = true
	if loggingEnabled() then
		logBuffer = {}
		logFlushAccum = 0
		logSessionStart = os.clock()
		pcall(function()
			if writefile then
				writefile(LOG_FILE, string.format(
					"=== Animation Inspector log | target=%s | %s ===\n",
					player.Name, os.date("%Y-%m-%d %H:%M:%S")
				))
			end
		end)
	end
	buildWindow()
	if screenGui then
		screenGui.Enabled = true
	end
	installHook()
	startResolver()
	startUpdateLoop()
	bindPlayer(player)
end

function AnimationInspector.setTargetByName(name)
	if not running then
		return
	end
	local player = resolveByName(name)
	if player then
		bindPlayer(player)
	end
end

function AnimationInspector.stop()
	if not running then
		return
	end
	running = false

	if updateConn then
		pcall(function() updateConn:Disconnect() end)
		updateConn = nil
	end
	if animatorConn then
		pcall(function() animatorConn:Disconnect() end)
		animatorConn = nil
	end
	if charAddedConn then
		pcall(function() charAddedConn:Disconnect() end)
		charAddedConn = nil
	end
	if charRemovingConn then
		pcall(function() charRemovingConn:Disconnect() end)
		charRemovingConn = nil
	end

	clearRows()
	removeHook()
	flushLog()

	if screenGui then
		screenGui:Destroy()
		screenGui = nil
		mainFrame, titleLabel, subLabel, listFrame, emptyLabel = nil, nil, nil, nil, nil
	end

	targetPlayer = nil
	targetName = nil
end

-- Full teardown (script detach): tear everything down for good.
function AnimationInspector.detach()
	AnimationInspector.stop()
end

return AnimationInspector

	"rbxassetid://9726608174",
	"rbxassetid://111647546998749",
	"rbxassetid://5064195992",
	"rbxassetid://6607519294",
	"rbxassetid://7600160919",
}
local MOVE_TYPES = { M1 = true, Mantra = true, Critical = true }
local FLOOD_PRIORITY = Enum.AnimationPriority.Action
local MAX_POOL = 200

-- State.
local running = false
local targetPlayer = nil
local targetName = nil
local currentAnimator = nil
local trackPool = {}
local accumulator = 0
local poolCursor = 0
local renderConn = nil
local charAddedConn = nil
local charRemovingConn = nil

-- Helpers
local function getIntensity()
	local value = Configuration.expectOptionValue("AnimationFlooderIntensity")
	if type(value) ~= "number" then
		return 8
	end
	return value
end

-- Attack animation ids straight out of the timing store; fall back to the hard list.
local function collectAttackIds()
	local ids, seen = {}, {}
	pcall(function()
		local TimingStore = require("Game/Timings/SaveManager")
		local list = TimingStore.as:list()
		for _, entry in ipairs(list) do
			if MOVE_TYPES[entry.tag] then
				local id = entry:id()
				if type(id) == "string" and id:match("%d") and not seen[id] then
					seen[id] = true
					ids[#ids + 1] = id
				end
			end
		end
	end)
	if #ids == 0 then
		ids = FALLBACK_IDS
	end
	return ids
end

local function destroyPool()
	for _, track in ipairs(trackPool) do
		pcall(function() track:Stop(0); track:Destroy() end)
	end
	trackPool = {}
	poolCursor = 0
end

-- Load our decoy pool onto the TARGET's animator.
local function loadPool(animator, idList)
	destroyPool()
	if type(idList) ~= "table" or #idList == 0 then
		return
	end

	local targetSize = math.min(MAX_POOL, math.max(#idList, 20))
	local cursor, attempts, maxAttempts = 0, 0, targetSize + #idList + 8
	while #trackPool < targetSize and attempts < maxAttempts do
		attempts = attempts + 1
		cursor = cursor % #idList + 1
		local id = idList[cursor]
		if not string.find(id, "://", 1, true) then
			id = "rbxassetid://" .. id
		end
		local animation = Instance.new("Animation")
		animation.AnimationId = id
		local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
		if ok and track then
			track.Looped = false
			pcall(function() track.Priority = FLOOD_PRIORITY end)
			trackPool[#trackPool + 1] = track
		end
	end
end

-- Target binding
local function bindCharacter(character)
	currentAnimator = nil
	destroyPool()
	if not character then
		return
	end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	local animator = humanoid and humanoid:FindFirstChildOfClass("Animator")
	if animator then
		currentAnimator = animator
		loadPool(animator, collectAttackIds())
		return
	end
	-- Character just spawned; wait for its Animator.
	task.spawn(function()
		local hum = character:WaitForChild("Humanoid", 5)
		local anim = hum and hum:WaitForChild("Animator", 5)
		if anim and running and targetPlayer and targetPlayer.Character == character then
			currentAnimator = anim
			loadPool(anim, collectAttackIds())
		end
	end)
end

local function bindPlayer(player)
	if charAddedConn then
		pcall(function() charAddedConn:Disconnect() end)
		charAddedConn = nil
	end
	if charRemovingConn then
		pcall(function() charRemovingConn:Disconnect() end)
		charRemovingConn = nil
	end

	targetPlayer = player
	targetName = player and player.Name or nil

	if not player then
		currentAnimator = nil
		destroyPool()
		return
	end

	charAddedConn = player.CharacterAdded:Connect(function(character)
		if running and targetPlayer == player then
			bindCharacter(character)
		end
	end)
	charRemovingConn = player.CharacterRemoving:Connect(function()
		if running and targetPlayer == player then
			currentAnimator = nil
			destroyPool()
		end
	end)

	bindCharacter(player.Character)
end

local function resolveByName(name)
	if not name or name == "" then
		return nil
	end
	for _, p in ipairs(Players:GetPlayers()) do
		if p.Name == name then
			return p
		end
	end
	local lower = string.lower(name)
	for _, p in ipairs(Players:GetPlayers()) do
		if string.lower(p.Name) == lower or string.lower(p.DisplayName) == lower then
			return p
		end
	end
	return nil
end

-- Flood loop
local function startLoop()
	if renderConn then
		return
	end
	renderConn = RunService.RenderStepped:Connect(function(deltaTime)
		if not running then
			return
		end
		local animator = currentAnimator
		if not animator or #trackPool == 0 then
			return
		end

		local interval = 1 / math.max(1, getIntensity())
		accumulator = math.min(accumulator + deltaTime, interval * 2)

		local burstsThisFrame = 0
		while accumulator >= interval and burstsThisFrame < 50 do
			accumulator = accumulator - interval
			burstsThisFrame = burstsThisFrame + 1

			-- Prefer an idle track; if all are mid-play, hard-restart the next one so a fresh AnimationPlayed still fires.
			local chosen = nil
			for _ = 1, #trackPool do
				poolCursor = poolCursor % #trackPool + 1
				local candidate = trackPool[poolCursor]
				if candidate and not candidate.IsPlaying then
					chosen = candidate
					break
				end
			end
			local recycled = false
			if not chosen then
				poolCursor = poolCursor % #trackPool + 1
				chosen = trackPool[poolCursor]
				recycled = chosen ~= nil
			end
			if chosen then
				pcall(function()
					if recycled then
						chosen:Stop(0)
					end
					chosen.Priority = FLOOD_PRIORITY
					chosen:Play(0.1, 1, 1)
				end)
			end
		end
	end)
end

-- Public API
function AnimationFlooder.startByName(name)
	local player = resolveByName(name)
	if not player then
		pcall(function() Library:Notify("Animation Flooder: pick a target player first.", 4) end)
		pcall(function()
			local toggles = getgenv and getgenv().Toggles
			if toggles and toggles.AnimationFlooder then
				toggles.AnimationFlooder:SetValue(false)
			end
		end)
		return
	end
	running = true
	startLoop()
	bindPlayer(player)
end

function AnimationFlooder.setTargetByName(name)
	if not running then
		return
	end
	local player = resolveByName(name)
	if player then
		bindPlayer(player)
	end
end

function AnimationFlooder.stop()
	if not running then
		return
	end
	running = false
	if renderConn then
		pcall(function() renderConn:Disconnect() end)
		renderConn = nil
	end
	if charAddedConn then
		pcall(function() charAddedConn:Disconnect() end)
		charAddedConn = nil
	end
	if charRemovingConn then
		pcall(function() charRemovingConn:Disconnect() end)
		charRemovingConn = nil
	end
	destroyPool()
	currentAnimator = nil
	targetPlayer = nil
	targetName = nil
	accumulator = 0
end

function AnimationFlooder.detach()
	AnimationFlooder.stop()
end

return AnimationFlooder
end)()
end)
__bundle_register("Features/Game/RevealAnimations", function(require, _LOADED, __bundle_register, __bundle_modules)
return LPH_NO_VIRTUALIZE(function()
	local Maid = require("Utility/Maid")

	local Signal = require("Utility/Signal")

	local Configuration = require("Utility/Configuration")

	local Logger = require("Utility/Logger")

	local SaveManager = require("Game/Timings/SaveManager")

	local RevealAnimations = {}

	local runService = game:GetService("RunService")
	local players = game:GetService("Players")
	local collectionService = game:GetService("CollectionService")

	local revealAnimationsMaid = Maid.new()

	local renderStepped = Signal.new(runService.RenderStepped)

	local lastUpdateTime = 0
	local updateThrottleTime = 0.03
	local weightValue = 1
	local modifiedAnimations = {}

	local function checkAnimationTrack(animationTrack)
		local priority = animationTrack.Priority

		if
			priority == Enum.AnimationPriority.Action2
			or priority == Enum.AnimationPriority.Action3
			or priority == Enum.AnimationPriority.Action4
		then
			return 0
		end

		if priority == Enum.AnimationPriority.Core or priority == Enum.AnimationPriority.Idle then
			return weightValue
		end

		if animationTrack.WeightCurrent <= 0.01 then
			return weightValue
		end

		local threshold = tonumber(Configuration.expectOptionValue("RevealWeightThreshold")) or 0.5

		if animationTrack.WeightCurrent < threshold then
			return weightValue
		end

		return nil
	end

	local function updateRevealAnimations()
		if not Configuration.expectToggleValue("RevealAnimations") then
			for animationTrack in pairs(modifiedAnimations) do
				if animationTrack.Parent then
					pcall(function()
						collectionService:RemoveTag(animationTrack, "RevealAnimationsModified")
					end)
				end
			end
			modifiedAnimations = {}
			return
		end

		local currentTime = os.clock()

		if currentTime - lastUpdateTime < updateThrottleTime then
			return
		end

		lastUpdateTime = currentTime

		local localPlayer = players.LocalPlayer
		if not localPlayer then
			return
		end

		for _, player in ipairs(players:GetPlayers()) do
			if player == localPlayer then
				continue
			end

			local character = player.Character

			if not character then
				continue
			end

			local humanoid = character:FindFirstChildOfClass("Humanoid")
			if not humanoid then
				continue
			end

			local animator = humanoid:FindFirstChildOfClass("Animator")
			if not animator then
				continue
			end

			local success, animationTracks = pcall(function()
				return animator:GetPlayingAnimationTracks()
			end)

			if not success then
				continue
			end

			for _, animationTrack in ipairs(animationTracks) do
				if SaveManager.as:index(tostring(animationTrack.Animation.AnimationId)) ~= nil then
					continue
				end

				local adjustWeight = checkAnimationTrack(animationTrack)

				if adjustWeight ~= nil then
					pcall(function()
						animationTrack:AdjustWeight(adjustWeight, 0)
						if not collectionService:HasTag(animationTrack, "RevealAnimationsModified") then
							collectionService:AddTag(animationTrack, "RevealAnimationsModified")
							modifiedAnimations[animationTrack] = true
						end
					end)
				end
			end
		end
	end

	function RevealAnimations.init()
		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("RevealAnimations initialized.")
		end
		revealAnimationsMaid:add(renderStepped:connect("RevealAnimations_RenderStepped", updateRevealAnimations))
	end

	function RevealAnimations.detach()
		Logger.warn("RevealAnimations detached.")
		for animationTrack in pairs(modifiedAnimations) do
			if animationTrack.Parent then
				pcall(function()
					collectionService:RemoveTag(animationTrack, "RevealAnimationsModified")
				end)
			end
		end
		modifiedAnimations = {}
		revealAnimationsMaid:clean()
	end

	return RevealAnimations
end)
end)
__bundle_register("Features/Game/OptimizeGame", function(require, _LOADED, __bundle_register, __bundle_modules)
return LPH_NO_VIRTUALIZE(function()
	local OptimizeGame = {}

	---@module Utility.Signal
	local Signal = require("Utility/Signal")

	---@module Utility.Logger
	local Logger = require("Utility/Logger")

	---@module Utility.Configuration
	local Configuration = require("Utility/Configuration")

	---@module Utility.Maid
	local Maid = require("Utility/Maid")

	local optimizeGameMaid = Maid.new()

	local getinfo = getinfo or debug.getinfo
	local getconstants = getconstants or debug.getconstants

	local alive = false
	local seaHookedFn = nil
	local seaOldFn = nil
	local replicationRestores = {}

	-- Only fire a retry while the module is still attached, so retry loops can't keep running (and re-driving
	-- render connections) after a detach/re-exec.
	local function retry(callback)
		task.delay(1, function()
			if alive then
				callback()
			end
		end)
	end

	local LOW_TEXTURE = "rbxassetid://130143779"

	local initialized = false
	local ranVisuals = false

	---@param obj Instance
	local function downgradeObject(obj)
		if obj:IsA("BasePart") then
			obj.Material = Enum.Material.Plastic
			obj.Reflectance = 0
			obj.CastShadow = false
		end

		if obj:IsA("Texture") or obj:IsA("Decal") then
			obj.Texture = LOW_TEXTURE
		elseif obj:IsA("SurfaceAppearance") then
			obj.ColorMap = LOW_TEXTURE
			obj.MetalnessMap = ""
			obj.RoughnessMap = ""
			obj.NormalMap = ""
		end

		if obj:IsA("ParticleEmitter") then
			obj.Enabled = false
			obj.Rate = 0
			obj.Speed = NumberRange.new(0)
		end

		if obj:IsA("PointLight") or obj:IsA("SpotLight") or obj:IsA("SurfaceLight") then
			obj.Enabled = false
			obj.Brightness = 0
			obj.Range = 2
		end
	end

	function OptimizeGame.runAggressive()
		if ranVisuals then
			return
		end
		ranVisuals = true
		task.spawn(function()
			local count = 0
			for _, obj in next, game:GetDescendants() do
				pcall(downgradeObject, obj)
				count = count + 1
				if count % 1000 == 0 then
					task.wait()
				end
			end
			Logger.notify("Aggressively optimized %d objects.", count)
		end)
	end

	local function setupSeaClient()
		local sea_client_func

		local function scan(objects)
			for _, object in next, objects do
				if typeof(object) ~= "function" or not getinfo(object).source:find("SeaClient") then
					continue
				end
				if not table.find(getconstants(object), "getWaveAt") then
					continue
				end

				sea_client_func = object
				break
			end
		end
		
		if filtergc then
			pcall(function()
				scan(filtergc("function", {
					IgnoreExecutor = false,
					Constants = { "getWaveAt" },
				}))
			end)
		end

		if not sea_client_func then
			scan(getgc(false))
		end

		if not sea_client_func then
			return Logger.warn("Failed to get SeaClient function.")
		end

		-- When throttled, hand back the LAST real result instead of nil. The old hook returned nothing
		-- on a skipped frame, so anything sampling the sea (floating objects, the local player while
		-- swimming) got nil and jittered -- that was the "lag". Returning the previous value is never
		-- worse than nil. os.clock + an inline Toggles read (no expectToggleValue call) keep the
		-- per-invocation cost near zero on this very hot function.
		local SEA_INTERVAL = 1 / 20
		local last_allowed_call = 0
		local cached = { n = 0 }
		local cachedArgs = { n = 0 }

		---@note: The cached result is only valid for the exact arguments it was computed with.
		local function sameArgs(...)
			local count = select("#", ...)
			if cachedArgs.n ~= count then
				return false
			end

			for index = 1, count do
				if cachedArgs[index] ~= (select(index, ...)) then
					return false
				end
			end

			return true
		end

		local old
		old = hookfunction(sea_client_func, function(...)
			local optimizeToggle = Toggles and Toggles.OptimizeGame
			if
				optimizeToggle
				and optimizeToggle.Value
				and os.clock() - last_allowed_call < SEA_INTERVAL
				and sameArgs(...)
			then
				return table.unpack(cached, 1, cached.n)
			end

			last_allowed_call = os.clock()
			cachedArgs = table.pack(...)
			cached = table.pack(old(...))
			return table.unpack(cached, 1, cached.n)
		end)

		seaHookedFn = sea_client_func
		seaOldFn = old
	end

	-- LowGraphics attribute state.
	local forcingLowGraphics = false
	local originalLowGraphics = nil

	---Force the LowGraphics player attribute while optimizing. WorldClient skips ambient VFX
	---(wind trails, glitches, bubbles, meteors, sail bones, despawn tweens) when it's set.
	---@param enabled boolean
	local function updateLowGraphics(enabled)
		local localPlayer = game:GetService("Players").LocalPlayer
		if not localPlayer then
			return
		end

		if enabled then
			if not forcingLowGraphics then
				originalLowGraphics = localPlayer:GetAttribute("LowGraphics")
				forcingLowGraphics = true
			end

			if localPlayer:GetAttribute("LowGraphics") ~= true then
				localPlayer:SetAttribute("LowGraphics", true)
			end
		elseif forcingLowGraphics then
			forcingLowGraphics = false
			localPlayer:SetAttribute("LowGraphics", originalLowGraphics)
		end
	end

	local function setupWorldClient()
		local last_world_client_func
		local function apply()
			task.wait(1)

			local hi = "WorldClient"
			local cns = getconnections(game:GetService("RunService").PreRender)

			local conn
			for _, object in next, cns do
				local func = object.Function
				if not func or not getinfo(func).source:find(hi) then
					continue
				end

				if last_world_client_func == func then
					return task.delay(1, apply)
				end

				last_world_client_func = func

				-- Crash fix (macOS/macsploit): getconnections() objects on macsploit expose only
				-- .Function and :Disconnect() - the Synapse-style :Disable() method is absent, so the
				-- original object:Disable() raised "attempt to call a nil value (method 'Disable')".
				-- On a worker-thread VM that error unwinds through the executor's native trampoline into
				-- the same NULL+0x28 SIGSEGV documented for conn:Fire(). Neutralize the connection through
				-- whatever the executor actually supports and record whether it stuck: if we cannot silence
				-- the game's own render connection we must NOT install the manual re-fire below, or every
				-- frame renders twice.
				local disabled = false
				if type(object.Disable) == "function" then
					disabled = pcall(function()
						object:Disable()
					end)
				end
				if not disabled then
					-- macsploit path: no :Disable(); try the writable .Enabled flag it exposes on
					-- getconnections objects, and confirm the write actually took.
					disabled = pcall(function()
						object.Enabled = false
					end) and object.Enabled == false
				end
				if not disabled and type(object.Disconnect) == "function" then
					-- Last resort: drop the native listener entirely. We re-fire .Function ourselves
					-- below, so a hard disconnect still yields exactly one render per throttled frame.
					disabled = pcall(function()
						object:Disconnect()
					end)
				end
				if not disabled then
					-- Could not neutralize the render connection on this executor; skip the manual
					-- optimization for this UI rather than double-firing WorldClient. Retry later in
					-- case the connection set changes.
					last_world_client_func = nil
					return task.delay(1, apply)
				end

				conn = object
			end

			if not conn then
				return
			end

			-- Crash fix (macOS/macsploit): conn:Fire() on a getconnections() object dereferences the
			-- executor's cached connection_data inside its Fire path with no liveness check - a
			-- freed/stale pointer after reconnections or respawns produces the NULL+0x28 read we
			-- SIGSEGV'd on. conn.Function is null-safe by comparison, so invoke the listener directly
			-- and wrap in pcall - a defunct connection degrades to a skipped frame, not a dead process.
			local function fireConnection(delta)
				return pcall(function()
					conn.Function(delta)
				end)
			end

			---@note: Re-enable the original connection on cleanup, otherwise WorldClient stays dead after detach.
			optimizeGameMaid["WorldClientRestore"] = function()
				pcall(function()
					if type(conn.Enable) == "function" then
						conn:Enable()
					else
						conn.Enabled = true
					end
				end)
			end

			local last_update = tick()
			optimizeGameMaid["WorldClientRender"] = game:GetService("RunService").PreRender:Connect(function()
				local now = tick()
				local enabled = Configuration.expectToggleValue("OptimizeGame")

				updateLowGraphics(enabled)

				if enabled and now - last_update < 1 / 15 then
					return
				end

				---@note: WorldClient feeds this into SmoothDamp - it must be the real elapsed time.
				local dt = now - last_update
				last_update = now

				return fireConnection(dt)
			end)
		end

		apply()

		local players = game:GetService("Players")
		local localPlayer = players.LocalPlayer
		if localPlayer then
			optimizeGameMaid:add(localPlayer.PlayerGui.ChildAdded:Connect(function(object)
				if not Configuration.expectToggleValue("OptimizeGame") then
					return
				end
				if object.Name ~= "WorldClient" then
					return
				end

				apply()
			end))
		end
	end

	-- PR's opti_clouds: the game drives cloud/entity replication rendering from a RenderStepped
	-- connection inside PlayerScripts.Replication. We neutralize the game's own connection and
	-- re-drive its function at 10Hz, so replication rendering costs a fraction of the frames without
	-- disappearing. Mirrors setupWorldClient's crash-safe handling for macsploit.
	local function setupReplication()
		local last_replication_func

		local function apply()
			local rep_func

			for _, object in next, getconnections(game:GetService("RunService").RenderStepped) do
				local func = object.Function
				if not func or not getinfo(func).source:find("PlayerScripts.Replication") then
					continue
				end

				if last_replication_func == func then
					return retry(apply)
				end

				last_replication_func = func

				-- Same macsploit-safe neutralization as setupWorldClient: the getconnections object may
				-- lack :Disable(); fall back to the writable .Enabled flag, then a hard :Disconnect(). Each
				-- path records a `restore` closure so detach can hand the game's own render connection back
				-- instead of leaving it neutralized across a re-exec.
				local restore = nil
				if type(object.Disable) == "function" and pcall(object.Disable, object) then
					restore = function()
						if type(object.Enable) == "function" then
							object:Enable()
						else
							object.Enabled = true
						end
					end
				end
				if not restore and pcall(function()
					object.Enabled = false
				end) and object.Enabled == false then
					restore = function()
						object.Enabled = true
					end
				end
				if not restore and type(object.Disconnect) == "function" and pcall(object.Disconnect, object) then
					restore = function()
						game:GetService("RunService").RenderStepped:Connect(func)
					end
				end
				if not restore then
					last_replication_func = nil
					return retry(apply)
				end

				replicationRestores[#replicationRestores + 1] = restore
				rep_func = func
				break
			end

			if not rep_func then
				-- Replication connection not present yet (early join); retry.
				return retry(apply)
			end

			-- Re-drive at 10Hz. Invoke rep_func directly (null-safe) rather than conn:Fire() to avoid
			-- the macsploit SIGSEGV, wrapped in pcall so a defunct connection degrades to a skipped frame.
			local last_update = tick()
			optimizeGameMaid["ReplicationRender"] = game:GetService("RunService").RenderStepped:Connect(function()
				local delta = tick() - last_update

				if not Configuration.expectToggleValue("OptimizeGame") then
					last_update = tick()
					return pcall(rep_func, delta)
				end

				if delta < 1 / 10 then
					return
				end
				last_update = tick()

				return pcall(rep_func, delta)
			end)
		end

		apply()
	end

	function OptimizeGame.init()
		if initialized then
			return
		end
		initialized = true
		alive = true

		-- Gate the render-connection manipulation on the toggle. setupSeaClient / setupReplication use
		-- getconnections to neutralize the game's OWN render connections and re-drive the game's render
		-- functions each frame for higher FPS; only run them when OptimizeGame is enabled.
		--
		-- setupWorldClient is DELIBERATELY NOT CALLED. It re-drove Deepwoken's WorldClient PreRender function
		-- from our own main-VM PreRender connection; WorldClient runs in the game's parallel Actor render
		-- phase, so calling its function from our serial connection ran parallel-context Lua in the wrong
		-- phase -- the illegal cross-phase op the engine Kernel-panics on, on the RENDER thread
		-- (RBXCRASH: FatalRuntimeError (RSL - panic: ...Kernel)), randomly ~40 min in, independent of every

__bundle_register("Features/Automation/AutoProg", function(require, _LOADED, __bundle_register, __bundle_modules)
-- Auto Prog: PR's auto progression, luma-native. Ties the Auto Builder (invest attribute points +
-- pick talent/mantra cards toward the pre/post spread) together with a farm (Auto Ferryman or Auto
-- Saramed) that earns the points, with a small state machine:
--
--   DOING_PRESHRINE  -- farm + build the pre spread; when the pre attributes are met, KICK to shrine.
--   DOING_POSTSHRINE -- farm + build the post spread; when the final attributes are met, KICK (done).
--
-- No persistence: after the kick, rejoin, re-execute luma, and press "Start Postshrine" (or "Start
-- Preshrine") to continue. The in-progress build + farm resume across the farm's own server hops via
-- their own MemStorage state.
--
-- Omits PR's Trial of One / Brutus fortitude / depths escape / rejoin-slot machinery (PR-only infra).

---@module Features.Automation.AutoBuilder
local AutoBuilder = require("Features/Automation/AutoBuilder")

---@module Features.Automation.AutoFerryman
local AutoFerryman = require("Features/Automation/AutoFerryman")

---@module Features.Automation.AutoSaramed
local AutoSaramed = require("Features/Automation/AutoSaramed")

---@module Utility.Configuration
local Configuration = require("Utility/Configuration")

---@module Utility.Logger
local Logger = require("Utility/Logger")

---@module Utility.Maid
local Maid = require("Utility/Maid")

---@module Utility.Signal
local Signal = require("Utility/Signal")

local players = game:GetService("Players")
local runService = game:GetService("RunService")

local AutoProg = {}

local CHECK_INTERVAL = 1.0

-- In-memory only: "NULL" | "DOING_PRESHRINE" | "DOING_POSTSHRINE".
local state = "NULL"

local progMaid = Maid.new()
local lastCheck = 0

local function farmMethod()
	return Configuration.expectOptionValue("AutoProgFarmMethod") or "Ferryman"
end

local function startFarm()
	-- Each farm self-guards on location (teleports/hops if we're not there) and is idempotent.
	if farmMethod() == "Saramed" then
		AutoSaramed.start()
	else
		AutoFerryman.start()
	end
end

local function stopFarms()
	pcall(AutoFerryman.stop)
	pcall(AutoSaramed.stop)
end

---Aim the builder at a spread, start building (invest + pick), and run the farm.
---@param phase "pre"|"post"
local function runPhase(phase)
	AutoBuilder.setTarget(phase)
	AutoBuilder.startPicking()
	startFarm()
end

---Stop everything and kick with a message.
---@param message string
local function endRun(message)
	state = "NULL"
	AutoBuilder.stopPicking()
	stopFarms()

	pcall(function()
		players.LocalPlayer:Kick(message)
	end)
end

---Start a phase from the UI.
---@param phase "pre"|"post"
---@return boolean
function AutoProg.start(phase)
	if phase ~= "pre" and phase ~= "post" then
		return false
	end

	if not AutoBuilder.current then
		Logger.notify("Auto Prog: load a build first.")
		return false
	end

	state = phase == "post" and "DOING_POSTSHRINE" or "DOING_PRESHRINE"
	runPhase(phase)

	Logger.notify(
		string.format(
			"Auto Prog: %s started (farming %s).",
			phase == "post" and "postshrine" or "preshrine",
			farmMethod()
		)
	)

	return true
end

---@return boolean
function AutoProg.startPreshrine()
	return AutoProg.start("pre")
end

---@return boolean
function AutoProg.startPostshrine()
	return AutoProg.start("post")
end

function AutoProg.stop()
	state = "NULL"
	AutoBuilder.stopPicking()
	stopFarms()
	Logger.notify("Auto Prog: stopped.")
end

---@return boolean
function AutoProg.isRunning()
	return state == "DOING_PRESHRINE" or state == "DOING_POSTSHRINE"
end

-- The state-machine watcher: kicks to shrine when preshrine is done, and kicks when the build is
-- finished. Armed once at init.
function AutoProg.init()
	local renderStepped = Signal.new(runService.RenderStepped)

	progMaid:add(renderStepped:connect("AutoProg_Watch", function()
		if os.clock() - lastCheck < CHECK_INTERVAL then
			return
		end
		lastCheck = os.clock()

		if state == "DOING_PRESHRINE" then
			if AutoBuilder.shrineReady() then
				endRun(
					'Auto Prog: preshrine done. shrine, then rejoin and press "Start Postshrine".'
				)
			end
		elseif state == "DOING_POSTSHRINE" then
			if AutoBuilder.finished() then
				endRun("Auto Prog: build finished. :)")
			end
		end
	end))

	if not LPH_OBFUSCATED or (shared.Luma and shared.Luma.verbose) then
		Logger.warn("Auto Prog initialized.")
	end
end

function AutoProg.detach()
	progMaid:clean()
end

return AutoProg
end)

	---@param timePosition number
	---@param animationLength number
	---@return number?
	local function getTimeElapsedFromTp(timePosition, animationLength)
		if not currentPlaybackData then
			return nil
		end

		if timePosition <= 0 then
			return 0.0
		end

		local currentPos = 0
		local elapsed = 0
		local dt = 0.01
		local iterations = 0

		while currentPos < timePosition do
			local speed = currentPlaybackData:last(elapsed) or 1
			local stepSize = speed * dt

			iterations = iterations + 1

			---@note: We can calculate how many iterations this should likely take based on the length (e.g 4 seconds) of the animation.
			if iterations >= math.max(((animationLength * 100) * 10), 1000) then
				break
			end

			if currentPos + stepSize > timePosition then
				local remainingTime = (timePosition - currentPos) / speed
				elapsed = elapsed + remainingTime
				break
			end

			currentPos = currentPos + stepSize
			elapsed = elapsed + dt
		end

		return elapsed
	end

local PRIORITY_ORDER = {
	Enum.AnimationPriority.Core,
	Enum.AnimationPriority.Idle,
	Enum.AnimationPriority.Movement,
	Enum.AnimationPriority.Action,
	Enum.AnimationPriority.Action2,
	Enum.AnimationPriority.Action3,
	Enum.AnimationPriority.Action4,
}
local PRIORITY_INDEX = {}
for index, priority in ipairs(PRIORITY_ORDER) do
	PRIORITY_INDEX[priority] = index
end
local ACTION_INDEX = PRIORITY_INDEX[Enum.AnimationPriority.Action]
local BOOST_OFFSET = PRIORITY_INDEX[Enum.AnimationPriority.Action4] - PRIORITY_INDEX[Enum.AnimationPriority.Action]
local ACTION3_INDEX = PRIORITY_INDEX[Enum.AnimationPriority.Action3]
local ACTION4_INDEX = PRIORITY_INDEX[Enum.AnimationPriority.Action4]

local function boostPriority(priority)
	local index = PRIORITY_INDEX[priority] or ACTION4_INDEX
	return PRIORITY_ORDER[math.clamp(index + BOOST_OFFSET, ACTION3_INDEX, ACTION4_INDEX)]
end

local loadedTrackRegistry = setmetatable({}, { __mode = "k" })

local ApBreaker = { force = false }

local function getOption(name, fallback)
	local value = Configuration.expectOptionValue(name)
	if value == nil then
		return fallback
	end
	return value
end

local ANIM_LIMIT = 60
local function underAnimationLimit(animator)
	if Configuration.expectToggleValue("ApBreakerDontHitAnimLimit") == false then
		return true
	end
	if not animator then
		return true
	end
	local ok, playing = pcall(function() return animator:GetPlayingAnimationTracks() end)
	if not ok or type(playing) ~= "table" then
		return true
	end
	return #playing < ANIM_LIMIT
end

local FALLBACK_AGGRESSIVE2_ANIMATION_IDS = {
	"rbxassetid://9726608174",
	"rbxassetid://111647546998749",
	"rbxassetid://5064195992",
	"rbxassetid://6607519294",
	"rbxassetid://7600160919",
}
local AGGRESSIVE2_MOVE_TYPES = { M1 = true, Mantra = true, Critical = true }
local AGGRESSIVE2_PRIORITY = Enum.AnimationPriority.Action

local function collectKnownAttackAnimationIds()
	local ids, seen = {}, {}
	pcall(function()
		local list = TimingStore.as:list()
		for _, entry in ipairs(list) do
			if AGGRESSIVE2_MOVE_TYPES[entry.tag] then
				local id = entry:id()
				if type(id) == "string" and id:match("%d") and not seen[id] then
					seen[id] = true
					ids[#ids + 1] = id
				end
			end
		end
	end)
	return ids
end

local ULTIMATE_IDLE_COVER_ID = "rbxassetid://180435571"
local ULTIMATE_IDLE_FADE = 0.2
local ULTIMATE_COVER_REPLAY = 0.3

local function startUltimateMode(character)
	if not character then return function() end end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return function() end end
	local animator = humanoid:FindFirstChildOfClass("Animator")
	if not animator then return function() end end

	local trackPool = {}
	local trackPoolSet = {}
	local poolSize = 0

	local stopped = false

	local function loadDecoyTracks(idList, spread)
		for _, track in next, trackPool do
			pcall(function() track:Stop(0); track:Destroy() end)
		end
		trackPool = {}
		trackPoolSet = {}
		poolSize = 0
		if type(idList) ~= "table" or #idList == 0 then return end

		local targetPoolSize = math.min(200, math.max(#idList, 20))
		local cursor, attempts, maxAttempts = 0, 0, targetPoolSize + #idList + 8
		while #trackPool < targetPoolSize and attempts < maxAttempts and not stopped do
			attempts = attempts + 1
			cursor = cursor % #idList + 1
			local id = idList[cursor]
			if not string.find(id, "://", 1, true) then
				id = "rbxassetid://" .. id
			end
			local animation = Instance.new("Animation")
			animation.AnimationId = id
			local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
			if ok and track then
				loadedTrackRegistry[track] = true
				track.Looped = false
				trackPool[#trackPool + 1] = track
				trackPoolSet[track] = true
			end

			if spread and attempts % 20 == 0 then
				task.wait()
			end
		end
		poolSize = #trackPool

		if stopped then
			for _, track in next, trackPool do
				pcall(function() track:Stop(0); track:Destroy() end)
			end
		end
	end

	local knownIds = collectKnownAttackAnimationIds()
	local usingFallbackIds = false
	if #knownIds == 0 then
		knownIds = FALLBACK_AGGRESSIVE2_ANIMATION_IDS
		usingFallbackIds = true
	end
	loadDecoyTracks(knownIds)

	local maskTrack = nil
	local maskSet = {}
	do
		local animation = Instance.new("Animation")
		animation.AnimationId = ULTIMATE_IDLE_COVER_ID
		local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
		if ok and track then
			loadedTrackRegistry[track] = true
			track.Looped = true
			maskTrack = track
			maskSet[track] = true
		end
	end
	local lastMaskPlay = 0

	local running = true
	local aggressive2Accumulator = 0
	local poolCursor = 0
	local framesSinceCheck = 0
	local fallbackRefreshTimer = 0
	local lastPlayTime = os.clock()
	local originalPriorities = setmetatable({}, { __mode = "k" })
	local function detectRealAttack(maskVisualEnabled)
		local realAttackPlaying = false
		for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
			if trackPoolSet[track] then
				if track.Priority ~= AGGRESSIVE2_PRIORITY then
					pcall(function() track.Priority = AGGRESSIVE2_PRIORITY end)
				end
			elseif not maskSet[track] and not loadedTrackRegistry[track] and track.Animation and track.Animation.AnimationId then
				local savedPriority = originalPriorities[track]
				if savedPriority == nil then
					savedPriority = track.Priority
					originalPriorities[track] = savedPriority
				end
				local boosted = boostPriority(savedPriority)
				pcall(function() track.Priority = boosted end)
				if (PRIORITY_INDEX[boosted] or 0) > ACTION_INDEX and track.WeightCurrent > 0.5 then
					realAttackPlaying = true
				end
			end
		end
		if maskTrack and maskVisualEnabled then
			realAttackPlaying = true
		end
		return realAttackPlaying
	end

	local function restoreForeignPriorities()
		for track, priority in pairs(originalPriorities) do
			pcall(function() track.Priority = priority end)
		end
		originalPriorities = setmetatable({}, { __mode = "k" })
	end

	local renderSteppedConn
	renderSteppedConn = RunService.RenderStepped:Connect(function(deltaTime)
		if not running then return end
		if not character.Parent then
			running = false
			return
		end

		if usingFallbackIds then
			fallbackRefreshTimer = fallbackRefreshTimer + deltaTime
			if fallbackRefreshTimer >= 1.5 then
				fallbackRefreshTimer = 0
				local refreshed = collectKnownAttackAnimationIds()
				if #refreshed > 0 then
					usingFallbackIds = false
					task.spawn(loadDecoyTracks, refreshed, true)
				end
			end
		end

		if poolSize == 0 then return end

		local maskVisualEnabled = Configuration.expectToggleValue("ExpApBreakerMaskVisual") ~= false

		local shouldRunAggressive2 = true
		if maskVisualEnabled then
			shouldRunAggressive2 = detectRealAttack(maskVisualEnabled)
		elseif next(originalPriorities) ~= nil then
			restoreForeignPriorities()
		end

		if maskTrack then
			if maskVisualEnabled then
				local now = os.clock()
				if not maskTrack.IsPlaying then
					pcall(function()
						maskTrack.Priority = Enum.AnimationPriority.Action2
						maskTrack:Play(0, 1, 1)
					end)
					lastMaskPlay = now
				elseif now - lastMaskPlay >= ULTIMATE_COVER_REPLAY then
					lastMaskPlay = now
					pcall(function()
						local position = maskTrack.TimePosition
						maskTrack:Stop(0)
						maskTrack.Priority = Enum.AnimationPriority.Action2
						maskTrack:Play(0, 1, 1)
						maskTrack.TimePosition = position
					end)
				end
			elseif maskTrack.IsPlaying then
				pcall(function() maskTrack:Stop(ULTIMATE_IDLE_FADE) end)
			end
		end

		if maskVisualEnabled and not shouldRunAggressive2 then
			for _, track in next, trackPool do
				if track.IsPlaying then
					pcall(function() track:Stop(0) end)
				end
			end
			framesSinceCheck = 0
			aggressive2Accumulator = 0
			return
		end

		framesSinceCheck = framesSinceCheck + 1
		if maskVisualEnabled and framesSinceCheck < 3 then return end

		local intensity = getOption("ExpApBreakerAggressive2Intensity", 200)
		local interval = 1 / math.max(1, intensity)
		aggressive2Accumulator = math.min(aggressive2Accumulator + deltaTime, interval * 2)

		if not underAnimationLimit(animator) then
			return
		end

		local burstsThisFrame = 0
		while aggressive2Accumulator >= interval and burstsThisFrame < 50 do
			aggressive2Accumulator = aggressive2Accumulator - interval
			burstsThisFrame = burstsThisFrame + 1

			local chosenTrack = nil
			local fallbackZeroLenTrack = nil
			for _ = 1, poolSize do
				poolCursor = poolCursor % poolSize + 1
				local candidate = trackPool[poolCursor]
				if candidate and not candidate.IsPlaying then
					if candidate.Length > 0 then
						chosenTrack = candidate
						break
					elseif not fallbackZeroLenTrack then
						fallbackZeroLenTrack = candidate
					end
				end
			end
			if not chosenTrack and os.clock() - lastPlayTime > 8 then
				chosenTrack = fallbackZeroLenTrack
			end
			if chosenTrack then
				pcall(function()
					chosenTrack.Priority = AGGRESSIVE2_PRIORITY
					chosenTrack:Play(0.1, 1, 1)
				end)
				lastPlayTime = os.clock()
			end
		end
	end)

	return function()
		running = false
		stopped = true
		if renderSteppedConn then
			renderSteppedConn:Disconnect()
			renderSteppedConn = nil
		end
		restoreForeignPriorities()
		if maskTrack then
			pcall(function() maskTrack:Stop(0); maskTrack:Destroy() end)
			maskTrack = nil
		end
		for _, track in next, trackPool do
			pcall(function() track:Stop(0); track:Destroy() end)
		end
	end
end

local FADE_TIME_FLOOD_FADE = 99999
local FADE_TIME_FLOOD_PRIORITY = Enum.AnimationPriority.Action

	if not character then return function() end end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return function() end end
	local animator = humanoid:FindFirstChildOfClass("Animator")

			pcall(function() track:Stop(0); track:Destroy() end)
		end
		trackPool = {}
	end

	local function loadPool(idList)
		destroyPool()
		if type(idList) ~= "table" or #idList == 0 then return end
		local targetSize = math.min(200, math.max(#idList, 20))
		local cursor, attempts, maxAttempts = 0, 0, targetSize + #idList + 8
		while #trackPool < targetSize and attempts < maxAttempts do
			attempts = attempts + 1
			cursor = cursor % #idList + 1
			local id = idList[cursor]
			if not string.find(id, "://", 1, true) then
				id = "rbxassetid://" .. id
			end
			local animation = Instance.new("Animation")
			animation.AnimationId = id
			local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
			if ok and track then
				loadedTrackRegistry[track] = true
				track.Looped = false
				trackPool[#trackPool + 1] = track
			end
		end
	end

	local usingFallback = false
	if #ids == 0 then
		ids = FALLBACK_AGGRESSIVE2_ANIMATION_IDS
		usingFallback = true
	end

	local running = true
	local accumulator = 0
	local poolCursor = 0
	local fallbackRefreshTimer = 0
	local renderConn
	renderConn = RunService.RenderStepped:Connect(function(deltaTime)
		if not running then return end
		if not character.Parent then
			running = false
			return
		end

		if usingFallback then
			fallbackRefreshTimer = fallbackRefreshTimer + deltaTime
			if fallbackRefreshTimer >= 1.5 then
				fallbackRefreshTimer = 0
				local refreshed = collectKnownAttackAnimationIds()
				if #refreshed > 0 then
					loadPool(refreshed)
					usingFallback = false
				end
			end
		end

		local poolSize = #trackPool
		if poolSize == 0 then return end

		local intensity = getOption("ExpApBreakerAggressive2Intensity", 200)
		local interval = 1 / math.max(1, intensity)
		accumulator = math.min(accumulator + deltaTime, interval * 2)

		if not underAnimationLimit(animator) then
			return
		end

		local burstsThisFrame = 0
		while accumulator >= interval and burstsThisFrame < 50 do
			accumulator = accumulator - interval
			burstsThisFrame = burstsThisFrame + 1

			local chosen = nil
			for _ = 1, poolSize do
				poolCursor = poolCursor % poolSize + 1
				local candidate = trackPool[poolCursor]
				if candidate and not candidate.IsPlaying then
					chosen = candidate
					break
				end
			end
			if chosen then
				pcall(function()
					chosen.Priority = FADE_TIME_FLOOD_PRIORITY
					chosen:Play(FADE_TIME_FLOOD_FADE, 1, 1)
				end)
			end
		end
	end)

	return function()
		running = false
		if renderConn then
			pcall(function() renderConn:Disconnect() end)
			renderConn = nil
		end
		destroyPool()
	end

				end)
			end
		end
	end)

	return function()
		running = false
		if renderConn then
			pcall(function() renderConn:Disconnect() end)
			renderConn = nil
		end
		destroyPool()
	end
end

local function randomStringFrom(pool, n)
	local poolLen = utf8.len(pool)
	local cs = {}
	for i = 1, n do
		local poolIdx = math.random(1, poolLen)
		local startByte = utf8.offset(pool, poolIdx)
		local endByte = utf8.offset(pool, poolIdx + 1) - 1
		cs[i] = string.sub(pool, startByte, endByte)
	end
	return table.concat(cs)
end

local function obfuscateAnimId(id)
	local junk = ""
	local percentEscapes = {}
	for i = 1, 25 do
		percentEscapes[i] = "%" .. i
	end
	if math.random() > 0.5 then
		for _ = 1, 2 do
			junk = junk .. percentEscapes[math.ceil(math.random(1, 20))]
		end
	else
		junk = junk .. randomStringFrom(GOD_OBF_CHARS, 5)
	end

	local urlSchemes = { "rbxhttp://", "rbxasset://" }
	local made = urlSchemes[math.floor(math.random(1, #urlSchemes))]
		.. "?id="
		.. randomStringFrom(GOD_OBF_CHARS, math.random(2, 7))
		.. "?id="
		.. tostring(GOD_OBF_FAKES[math.floor(math.random(1, #GOD_OBF_FAKES))])
		.. junk

	local suffix = "?id=" .. id .. "&"
	local remaining = math.max(0, 100 - #made - #suffix) - 1
	local left = math.random(math.floor(remaining * 0.4), math.floor(remaining * 0.6))
	local right = remaining - left

	made = made
		.. randomStringFrom(GOD_OBF_CHARS, left)
		.. suffix
		.. randomStringFrom("ABCDEFGHIJKMNOLPQRSTUVWXYZabcdefghijkmnolpqrstuvwxyz%", 1)
		.. randomStringFrom(GOD_OBF_CHARS, right)

	return made
end

local function startMutateMode(character)
	if not character then return function() end end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return function() end end
	local animator = humanoid:FindFirstChildOfClass("Animator")
	if not animator then return function() end end

	local trackPool = {}
	local function destroyPool()
		for _, track in next, trackPool do
			pcall(function() track:Stop(0); track:Destroy() end)
		end
		trackPool = {}
	end

	local attempts = 0
	while #trackPool < MUTATE_POOL and attempts < MUTATE_POOL * 2 do
		attempts = attempts + 1
		local realId = GOD_ANIMS[math.random(1, #GOD_ANIMS)]
		local animation = Instance.new("Animation")
		animation.AnimationId = obfuscateAnimId(realId)
		local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
		if ok and track then
			loadedTrackRegistry[track] = true
			track.Looped = false
			trackPool[#trackPool + 1] = track
		end
	end

	local running = true
	local accumulator = 0
	local poolCursor = 0
	local renderConn
	renderConn = RunService.RenderStepped:Connect(function(deltaTime)
		if not running then return end
		if not character.Parent then
			running = false
			return
		end
		local poolSize = #trackPool
		if poolSize == 0 then return end

		local intensity = getOption("ExpApBreakerAggressive2Intensity", 200)
		local interval = 1 / math.max(1, intensity)
		accumulator = math.min(accumulator + deltaTime, interval * 2)

		if not underAnimationLimit(animator) then
			return
		end

		local burstsThisFrame = 0
		while accumulator >= interval and burstsThisFrame < 50 do
			accumulator = accumulator - interval
			burstsThisFrame = burstsThisFrame + 1

			local chosen = nil
			for _ = 1, poolSize do
				poolCursor = poolCursor % poolSize + 1
				local candidate = trackPool[poolCursor]
				if candidate and not candidate.IsPlaying then
					chosen = candidate
					break
				end
			end
			local recycled = false
			if not chosen then
				poolCursor = poolCursor % poolSize + 1
				chosen = trackPool[poolCursor]
				recycled = chosen ~= nil
			end
			if chosen then
				pcall(function()
					if recycled then chosen:Stop(0) end
					chosen.Priority = MUTATE_PRIORITY
					chosen:Play(0.1, 1, 1)
				end)
			end
		end
	end)

	return function()
		running = false
		if renderConn then
			pcall(function() renderConn:Disconnect() end)
			renderConn = nil
		end
		destroyPool()
	end
end

-- ---- Blatant Flood tuning --------------------------------------------------
local FULL_BLATANT_POOL_TARGET = 150
local FULL_BLATANT_FADE_HIGH = 99999

-- All known animation ids (every tag), not just attack tags.
local function collectAllKnownAnimationIds()
	local ids, seen = {}, {}
	pcall(function()
		local list = TimingStore.as:list()
		for _, entry in ipairs(list) do
			local id = entry:id()
			if type(id) == "string" and id:match("%d") and not seen[id] then
				seen[id] = true
				ids[#ids + 1] = id
			end
		end
	end)
	return ids
end

local function numericAnimId(id)
	if type(id) ~= "string" then
		return nil
	end
	return id:match("id=(%d+)") or id:match("(%d+)%s*$") or id:match("(%d+)")
end

local function randomPriority()
	return PRIORITY_ORDER[math.random(1, #PRIORITY_ORDER)]
end
local function randomWeight()
	if math.random() < 0.5 then
		return math.random() * 0.15
	end
	return math.random() * 9.5 + 0.5
end
local function randomFade()
	if math.random() < 0.5 then
		return math.random() * 0.15
	end
	return FULL_BLATANT_FADE_HIGH
end
local function randomSpeed()
	if math.random() < 0.5 then
		return math.random() * 0.9 + 0.1
	end
	return math.random() * 24 + 1
end

-- ---- Blatant Flood ---------------------------------------------------------
-- Floods every known timing at a high rate with randomized properties AND blanks the animation id
-- of your real attacks to nothing -- so inside the storm the enemy's auto-parry can't even read
-- what you threw, let alone parry it. It does NOT stop your real attacks: they still land, just
-- with an empty (unresolvable) id. Ids are blanked the instant the attack plays (AnimationPlayed)
-- with a per-frame backstop.
local function startBlatantFloodMode(character)
	if not character then return function() end end
	local humanoid = character:FindFirstChildOfClass("Humanoid")
	if not humanoid then return function() end end
	local animator = humanoid:FindFirstChildOfClass("Animator")
	if not animator then return function() end end

	local trackPool = {}
	local function destroyPool()
		for _, track in next, trackPool do
			pcall(function() track:Stop(0); track:Destroy() end)
		end
		trackPool = {}
	end

	-- Numeric ids of the moves you actually throw (M1 / Mantra / Critical) -- these are the ones an
	-- enemy has timings for, so blanking exactly these is what defeats their parry.
	local attackIdSet = {}
	local function rebuildAttackIdSet()
		attackIdSet = {}
		for _, id in ipairs(collectKnownAttackAnimationIds()) do
			local n = numericAnimId(id)
			if n then
				attackIdSet[n] = true
			end
		end
	end

	-- Tracks whose id we've already blanked, so we don't touch them every frame.
	local obfuscated = setmetatable({}, { __mode = "k" })
	local function obfuscateRealAttack(track)
		if loadedTrackRegistry[track] or obfuscated[track] or not track.Animation then
			return
		end
		local n = numericAnimId(track.Animation.AnimationId)
		if n and attackIdSet[n] then
			obfuscated[track] = true
			-- Blank the id to nothing: the track keeps playing (your attack still lands) but the
			-- replicated id becomes empty, so the enemy can't resolve a timing for it.
			pcall(function()
				track.Animation.AnimationId = ""
			end)
		end
	end

	local function loadPool(idList)
		destroyPool()
		if type(idList) ~= "table" or #idList == 0 then return end
		local target = math.min(FULL_BLATANT_POOL_TARGET, math.max(#idList, 40))
		local cursor, attempts, maxAttempts = 0, 0, target + #idList + 16
		while #trackPool < target and attempts < maxAttempts do
			attempts = attempts + 1
			cursor = cursor % #idList + 1
			local id = idList[cursor]
			if not string.find(id, "://", 1, true) then
				id = "rbxassetid://" .. id
			end
			local animation = Instance.new("Animation")
			animation.AnimationId = id
			local ok, track = pcall(function() return animator:LoadAnimation(animation) end)
			if ok and track then
				loadedTrackRegistry[track] = true
				track.Looped = false
				trackPool[#trackPool + 1] = track
			end
		end
	end

	local ids = collectAllKnownAnimationIds()
	local usingFallback = false
	if #ids == 0 then
		ids = FALLBACK_AGGRESSIVE2_ANIMATION_IDS
		usingFallback = true
	end
	loadPool(ids)
	rebuildAttackIdSet()

	local running = true
	local accumulator = 0
	local poolCursor = 0
	local fallbackRefreshTimer = 0

	-- Blank real attacks the instant they start (before the next frame can leak the id).
	local animPlayedConn = animator.AnimationPlayed:Connect(function(track)
		if running then
			obfuscateRealAttack(track)
		end
	end)

	local renderConn
	renderConn = RunService.RenderStepped:Connect(function(deltaTime)
		if not running then return end
		if not character.Parent then
			running = false
			return
		end

		-- Per-frame backstop for any real attack the AnimationPlayed hook missed.
		for _, track in ipairs(animator:GetPlayingAnimationTracks()) do
			obfuscateRealAttack(track)
		end

		if usingFallback then
			fallbackRefreshTimer = fallbackRefreshTimer + deltaTime
			if fallbackRefreshTimer >= 1.5 then
				fallbackRefreshTimer = 0
				local refreshed = collectAllKnownAnimationIds()
				if #refreshed > 0 then
					loadPool(refreshed)
					rebuildAttackIdSet()
					usingFallback = false
				end
			end
		end

		local poolSize = #trackPool
		if poolSize == 0 then return end

		-- Rate is user-tunable (default 60/s). Each Play() replicates to the server, so a very high
		-- rate saturates your uplink and spikes YOUR ping -- keep this modest; 40-80/s already buries
		-- the real swing in noise. Burst cap scales with rate so a low rate can't batch a huge frame.
		local rate = getOption("BlatantFloodRate", 60)
		rate = math.clamp(tonumber(rate) or 60, 1, 300)
		local burstCap = math.max(4, math.ceil(rate / 10))
		local interval = 1 / rate
		accumulator = math.min(accumulator + deltaTime, interval * burstCap)

		local burstsThisFrame = 0
		while accumulator >= interval and burstsThisFrame < burstCap do
			accumulator = accumulator - interval
			burstsThisFrame = burstsThisFrame + 1

			poolCursor = poolCursor % poolSize + 1
			local track = trackPool[poolCursor]
			if track then
				pcall(function()
					if track.IsPlaying then
						track:Stop(0)
					end
					track.Priority = randomPriority()
					track:Play(randomFade(), randomWeight(), randomSpeed())
				end)
			end
		end
	end)

	return function()
		running = false
		if animPlayedConn then
			pcall(function() animPlayedConn:Disconnect() end)
			animPlayedConn = nil
		end
		if renderConn then
			pcall(function() renderConn:Disconnect() end)
			renderConn = nil
		end
		destroyPool()
	end
end

local blatantFloodActive = false

local blatantFloodCharConn = nil

function ApBreaker:StartBlatantFlood()

	blatantFloodActive = true

	local player = Players.LocalPlayer
	local function apply(character)
		if blatantFloodCleanup then
			pcall(blatantFloodCleanup)
			blatantFloodCleanup = nil
		end
		if blatantFloodActive and character then
			blatantFloodCleanup = startBlatantFloodMode(character)
		end
	end

	if not blatantFloodCharConn then
		blatantFloodCharConn = player.CharacterAdded:Connect(apply)
	end
	apply(player.Character or player.CharacterAdded:Wait())
	Logger.warn("Blatant Flood started.")
end

	blatantFloodActive = false

	if blatantFloodCleanup then
		pcall(blatantFloodCleanup)
		blatantFloodCleanup = nil
	end
	if blatantFloodCharConn then
		blatantFloodCharConn:Disconnect()
		blatantFloodCharConn = nil
	end
	Logger.warn("Blatant Flood stopped.")
end

local MODE_STARTERS = {
	["Priority Mask"]   = startUltimateMode,
	["Fade Time Flood"] = startFadeTimeFloodMode,
	["Monster Flood"]   = startMonsterFloodMode,
	["Mutate"]          = startMutateMode,
}

local apBreakerActive = false
local apBreakerCleanups = {}
local characterAddedConn = nil

local function getSelectedModes()
	local selected = Configuration.expectOptionValue("ExpApBreakerModes")
	local result = {}
	if type(selected) == "table" then
		for key, value in pairs(selected) do
			local name = (value == true) and key or value
			if type(name) == "string" and MODE_STARTERS[name] then
				result[name] = true
			end
		end
	end
	return result
end

local function applyModes(character, forceRestart)
	if forceRestart then
		for _, cleanup in pairs(apBreakerCleanups) do
			pcall(cleanup)
		end
		apBreakerCleanups = {}
	end

	if not apBreakerActive or not character then
		for _, cleanup in pairs(apBreakerCleanups) do
			pcall(cleanup)
		end
		apBreakerCleanups = {}
		return
	end

	local selected = getSelectedModes()

	for mode, cleanup in pairs(apBreakerCleanups) do
		if not selected[mode] then
			pcall(cleanup)
			apBreakerCleanups[mode] = nil
		end
	end

	for mode in pairs(selected) do
		if not apBreakerCleanups[mode] then
			apBreakerCleanups[mode] = MODE_STARTERS[mode](character)
		end
	end
end

function ApBreaker:Start()
	if apBreakerActive then return end
	apBreakerActive = true

	local player = Players.LocalPlayer
	if not characterAddedConn then
		characterAddedConn = player.CharacterAdded:Connect(function(character)
			applyModes(character, true)
		end)
	end

	applyModes(player.Character or player.CharacterAdded:Wait(), true)
	Logger.warn("AP Breaker started.")
end

function ApBreaker:Stop()
	if not apBreakerActive then return end
	apBreakerActive = false

	for _, cleanup in pairs(apBreakerCleanups) do
		pcall(cleanup)
	end
	apBreakerCleanups = {}

	if characterAddedConn then
		characterAddedConn:Disconnect()
		characterAddedConn = nil
	end
	Logger.warn("AP Breaker stopped.")
end

function ApBreaker.refresh()
	if not apBreakerActive then return end
	applyModes(Players.LocalPlayer.Character, false)
end

function ApBreaker.init()
	if Configuration.expectToggleValue("ExpApBreaker") then
		ApBreaker:Start()
	end
	if Configuration.expectToggleValue("BlatantFlood") then
		ApBreaker:StartBlatantFlood()
	end
end

function ApBreaker.detach()
	ApBreaker.force = true
	ApBreaker:Stop()
	ApBreaker:StopBlatantFlood()
end

end

return ApBreaker

__bundle_register("Utility/Throttle", function(require, _LOADED, __bundle_register, __bundle_modules)
---@class Throttle
--- Wrap a per-frame callback so it runs at most once per `interval` seconds, forwarding all
--- arguments (e.g. RenderStepped's dt) straight through. Uses os.clock, so it works regardless
--- of whether the driving signal passes a delta.
---
--- Intended for PASSIVE features that don't need full framerate (attribute farms, ownership
--- scan, ...). Do NOT use it on combat/UI/movement loops -- those want every frame.
---
---@param interval number Minimum seconds between calls.
---@param fn fun(...): ...
---@return fun(...): ...?
local function throttle(interval, fn)
	local last = 0

	return function(...)
		local now = os.clock()
		if now - last < interval then
			return
		end
		last = now
		return fn(...)
	end
end

return throttle
end)

	function Spoofing.rics()
		for _, player in next, players:GetPlayers() do
			Spoofing.facs(player, "Guild")
			Spoofing.facs(player, "FirstName")
			Spoofing.facs(player, "LastName")

			local character = player.Character
			if not character then
				continue
			end

			Spoofing.facs(character, "GuildRich")

			local humanoid = character:FindFirstChild("Humanoid")
			if not humanoid then
				continue
			end

			Spoofing.facs(humanoid, "CharacterName")
		end

		local serverRegion = replicatedStorage:FindFirstChild("SERVER_REGION")
		local serverName = replicatedStorage:FindFirstChild("SERVER_NAME")
		local serverAge = replicatedStorage:FindFirstChild("SERVER_AGE")

		if not serverRegion or not serverName or not serverAge then
			return
		end

		Spoofing.fvcs(serverRegion)
		Spoofing.fvcs(serverName)

		local original = serverAge.Value
		serverAge.Value = "L"
		serverAge.Value = original
	end

	-- PR streamer_mode's server-info spoof: WRITE the values directly and re-apply them whenever the game
	-- changes them (.Changed), instead of intercepting the .Value READ through the now-removed __index hook.
	local function hookServerValue(valueInstance, getSpoofed)
		if not valueInstance then
			return
		end
		local function apply()
			if Spoofing.force or not Configuration.expectToggleValue("InfoSpoofing") then
				return
			end
			local spoofed = getSpoofed()
			if spoofed ~= nil and valueInstance.Value ~= spoofed then
				valueInstance.Value = spoofed
			end
		end
		spoofingMaid:add(valueInstance.Changed:Connect(apply))
		apply()
	end

	function Spoofing.init()
		local localPlayer = players.LocalPlayer
		local playerGui = localPlayer:WaitForChild("PlayerGui")
		local pgDescendantAdded = Signal.new(playerGui.DescendantAdded)

		spoofingMaid:add(renderStepped:connect("Spoofing_OnRenderStepped", updateSpoofing))
		spoofingMaid:add(pgDescendantAdded:connect("Spoofing_OnPGDescendantAdded", onPgDescendantAdded))

		-- Server-info spoof the PR way (direct writes + .Changed re-apply), replacing the dropped __index reads.
		pcall(function()
			hookServerValue(replicatedStorage:WaitForChild("SERVER_NAME", 10), function()
				return Configuration.expectOptionValue("SpoofedServerName")
			end)
			hookServerValue(replicatedStorage:WaitForChild("SERVER_REGION", 10), function()
				return Configuration.expectOptionValue("SpoofedServerRegion")
			end)
			hookServerValue(replicatedStorage:WaitForChild("SERVER_AGE", 10), function()
				return Configuration.expectOptionValue("SpoofedServerAge")
			end)
		end)

		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("Spoofing initialized.")
		end
	end

	function Spoofing.detach()
		Spoofing.force = true

		spoofingMaid:clean()

		Spoofing.rics()

		fakeFreestylerBand.Parent = nil
		fakeKongaClutchRing.Parent = nil
		resetLightweightSpoof()
		resetCaptainEtreaSpoof()

		Logger.warn("Spoofing detached.")
	end

	return Spoofing
end)()
end)
__bundle_register("Utility/OriginalStoreManager", function(require, _LOADED, __bundle_register, __bundle_modules)
---@module Utility.OriginalStore
local OriginalStore = require("Utility/OriginalStore")

---@class OriginalStoreManager
---@param inner OriginalStore[]
local OriginalStoreManager = {}
OriginalStoreManager.__index = OriginalStoreManager

---@param data table|Instance
OriginalStoreManager.forget = LPH_NO_VIRTUALIZE(function(self, data)
	self.inner[data] = nil
end)

---@param data table|Instance
---@param index any
OriginalStoreManager.mark = LPH_NO_VIRTUALIZE(function(self, data, index)
	local object = self.inner[data] or OriginalStore.new()

	object:mark(data, index)

	self.inner[data] = object
end)

---@param data table|Instance
---@param index any
---@param value any
OriginalStoreManager.add = LPH_NO_VIRTUALIZE(function(self, data, index, value)
	local object = self.inner[data] or OriginalStore.new()

	object:set(data, index, value)

	self.inner[data] = object
end)

---@return OriginalStore[]
OriginalStoreManager.data = LPH_NO_VIRTUALIZE(function(self)
	return self.inner
end)

---@param data table|Instance
---@return OriginalStore
OriginalStoreManager.get = LPH_NO_VIRTUALIZE(function(self, data)
	return self.inner[data]
end)

OriginalStoreManager.restore = LPH_NO_VIRTUALIZE(function(self)
	for _, store in next, self.inner do
		store:restore()
	end
end)

OriginalStoreManager.detach = LPH_NO_VIRTUALIZE(function(self)
	for _, store in next, self.inner do
		store:detach()
	end

	self.inner = {}
end)

---@return OriginalStoreManager
OriginalStoreManager.new = LPH_NO_VIRTUALIZE(function()
	local self = setmetatable({}, OriginalStoreManager)
	self.inner = {}
	return self
end)

return OriginalStoreManager
end)

			Url = "https://new.deepwoken.co/api/proxy/builds",
			Method = "POST",
			Headers = {
				["Content-Type"] = "application/json",
				["Accept"] = "application/json",
				["Origin"] = "https://deepwoken.co",
				["Referer"] = "https://deepwoken.co/builder",
				["Sec-Fetch-Site"] = "same-origin",
				["Sec-Fetch-Mode"] = "cors",
				["Sec-Fetch-Dest"] = "empty",
			},
			Body = JSON.encode(data),
		})

		if not response then
			return Logger.notify("Failed to steal build from '%s' due to no response.", fetchName(player))
		end

		if not response.Success then
			return Logger.notify(
				"(Status %i) Failed to steal build from '%s' due to build creation failure.",
				response.StatusCode,
				fetchName(player)
			)
		end

		if not response.Body then
			return Logger.notify("Failed to steal build from '%s' due to no response body.", fetchName(player))
		end

		local decoded = JSON.decode(response.Body)

		if not decoded or not decoded.id then
			return Logger.notify(
				"(Status %i) Failed to steal build from '%s' due to invalid response body.",
				response.StatusCode,
				fetchName(player)
			)
		end

		local buildUrl = string.format("https://deepwoken.co/builder?id=%s", decoded.id)
		setclipboard(buildUrl)

		local path = string.format("%s %s Stolen Build", os.date("%B %d %Y"), player.Name)

		end

stolen on %s
build url: %s
%s]], os.date("%B %d %Y"), buildUrl, data.content.notes))
	end

	local function onHairStealPlayer(player)

		if not character then
			return Logger.notify(
				"Failed to steal hair IDs from '%s' because their character does not exist.",

			)
		end

			if not instance:IsA("Accessory") then

			end

			local isHairSlot = instance.AccessoryType == Enum.AccessoryType.Hair
			if not isHairSlot then

				isHairSlot = handle ~= nil and handle:FindFirstChild("HairAttachment") ~= nil
			end

			if not isHairSlot then
				continue
			end

			local id = instance.Name:match("^Asset:(%d+)$") or instance.Name:match("(%d+)")
			if id and not seen[id] then
				seen[id] = true
				table.insert(hairIds, id)
			end
		end

		if #hairIds == 0 then
			return Logger.notify("Failed to find any hair IDs for '%s'.", fetchName(player))
		end

	local function updatePlayerProximity()
		local proximityRange = Configuration.expectOptionValue("PlayerProximityRange") or 350
		local playersInRange = Finder.gpir(proximityRange)
		if not playersInRange then
			return
		end

		local localPlayer = players.LocalPlayer
		if not localPlayer then
			return
		end

		local backpack = localPlayer:FindFirstChild("Backpack")
		if not backpack then
			return
		end

		local now = os.clock()

		-- Refresh the last-in-range stamp for everyone currently in range.
		for _, player in next, playersInRange do
			Monitoring.proximityLastSeen[player] = now
		end

		-- Dismiss a player's notification only after they've been out of range for the full grace window.
		-- A one-cycle drop-out keeps the notification up, so returning doesn't re-fire a fresh "entered".
		for player, removeNotification in next, Monitoring.seen do
			local isInPlayerRange = table.find(playersInRange, player)
			if isInPlayerRange then
				continue
			end

			if now - (Monitoring.proximityLastSeen[player] or 0) <= PROXIMITY_GRACE then
				continue
			end

			removeNotification()

			Monitoring.seen[player] = nil
			Monitoring.proximityLastSeen[player] = nil
		end

		for _, player in next, playersInRange do
			if Monitoring.seen[player] ~= nil then
				continue
			end

			if
				Configuration.expectToggleValue("PlayerProximityVW")
				and not backpack:FindFirstChild("Talent:Voidwalker Contract")
			then
				continue
			end

			Monitoring.seen[player] =
				Logger.mnnotify("%s entered your proximity radius of %i studs.", fetchName(player), proximityRange)

			if Configuration.expectToggleValue("PlayerProximityBeep") then
				beepSound.SoundId = "rbxassetid://100849623977896"
				beepSound.PlaybackSpeed = 1
				beepSound.Volume = Configuration.expectOptionValue("PlayerProximityBeepVolume") or 0.1
				beepSound:Play()
			end
		end
	end

	local function updateSubjectMonitoring()
		cameraSubject:set(workspace.CurrentCamera, "CameraSubject", Monitoring.subject)

		if Monitoring.subject ~= nil then
			players.LocalPlayer:AddTag("ForcedSubject")
		else
			players.LocalPlayer:RemoveTag("ForcedSubject")
		end
		-- The spectate stream request is NOT done here any more -- it runs on its own throttled connection in
		-- Monitoring.init, exactly like Project Rain's leaderboard_spectate (inline, never a per-frame spawn).
	end

		if Monitoring.subject then
			updateSubjectMonitoring()
		else
			cameraSubject:restore()
		end

		if os.clock() - lastUpdateTime <= 2.0 then
			return
		end

		lastUpdateTime = os.clock()

		if Configuration.expectToggleValue("PlayerListWhitelisting") then
			updatePlayerListWhitelisting()
		else
			plwMaid:clean()
		end

		if Configuration.expectToggleValue("BuildStealer") then
			updateBuildStealing()
		else
			buildStealMaid:clean()
		end

		if Configuration.expectToggleValue("HairStealer") then
			updateHairStealing()
		else
			hairStealMaid:clean()
		end

		if Configuration.expectToggleValue("PlayerSpectating") then
			updateSpectating()
		else
			spectateMaid:clean()
		end

		if Configuration.expectToggleValue("PlayerProximity") then
			updatePlayerProximity()
		end
	end

	function Monitoring.init()
		monitoringMaid:add(renderStepped:connect("Monitoring_OnRenderStepped", updateMonitoring))

		-- Project Rain's leaderboard_spectate streaming, verbatim in structure: a dedicated RenderStepped loop
		-- that, while a subject is set, calls RequestStreamAroundAsync INLINE in a pcall at most every 0.25s.
		-- PR issues one request per 0.25s on this one connection and lets it return before the next -- so at
		-- most ~one is ever in flight. (luma's old code wrapped it in TaskSpawner.spawn + maid:add EVERY frame,
		-- which piled thousands of parked requests into the fiber pool -> "RSL - panic: failed to spawn a fiber".)
		local lastStreamRequest = tick()
		monitoringMaid:add(renderStepped:connect("Monitoring_StreamSpectate", function()
			if tick() - lastStreamRequest < 0.25 or Monitoring.subject == nil then
				return
			end
			lastStreamRequest = tick()
			pcall(function()
				players.LocalPlayer:RequestStreamAroundAsync(Monitoring.subject.Position, 1000)
			end)
		end))

		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("Monitoring initialized.")
		end
	end

	function Monitoring.detach()
		monitoringMaid:clean()
		buildStealMaid:clean()
		hairStealMaid:clean()
		spectateMaid:clean()
		plwMaid:clean()
		showHiddenMap:restore()

		local refreshLeaderboard = LeaderboardClient.glrf()

		if refreshLeaderboard then
			refreshLeaderboard()
		end

		Logger.warn("Monitoring detached.")
	end

	return Monitoring
end)()
end)
__bundle_register("Game/LeaderboardClient", function(require, _LOADED, __bundle_register, __bundle_modules)
local LeaderboardClient = { calling = false }

local players = game:GetService("Players")

local cachedUpvaluesTable = nil
local cachedUpvaluesFunction = nil
local lastDataCacheTime = os.clock()
local lastFunctionCacheTime = os.clock()

---@param func function?
---@return boolean
local function filterFunction(func)
	if not func or not islclosure(func) then
		return false
	end

	local info = debug.getinfo(func)
	if info.name ~= nil and info.name ~= "" then
		return false
	end

	local constants = debug.getconstants(func)
	if #constants ~= 1 or constants[1] ~= "Destroy" then
		return false
	end

	local upvalues = debug.getupvalues(func)
	if not upvalues then
		return false
	end

	return true
end

---@return function?
function LeaderboardClient.glrf()
	if cachedUpvaluesFunction and os.clock() - lastFunctionCacheTime <= 2.0 then
		return cachedUpvaluesFunction
	end

	for _, con in next, getconnections(players.PlayerRemoving) do
		local func = con.Function
		if not filterFunction(func) then
			continue
		end

		local upvalues = debug.getupvalues(func)
		if not upvalues then
			continue
		end

		local upvaluesFunction = upvalues[2]

		cachedUpvaluesFunction = upvaluesFunction
		lastFunctionCacheTime = os.clock()

		return function()
			if LeaderboardClient.calling then
				return
			end

			LeaderboardClient.calling = true

			upvaluesFunction()

			LeaderboardClient.calling = false
		end
	end

	return nil
end

---@return table?
function LeaderboardClient.gld()
	if cachedUpvaluesTable and os.clock() - lastDataCacheTime <= 2.0 then
		return cachedUpvaluesTable
	end

	for _, con in next, getconnections(players.PlayerRemoving) do
		local func = con.Function
		if not filterFunction(func) then
			continue
		end

		local upvalues = debug.getupvalues(func)
		if not upvalues then
			continue
		end

		local upvaluesTable = upvalues[1]

		cachedUpvaluesTable = upvaluesTable
		lastDataCacheTime = os.clock()

		return upvaluesTable
	end

	return nil
end

return LeaderboardClient
end)

			hideEffect(effect)
		end

		if
			(effect.Class == "BeingWinded" or effect.Class == "StrongWind")
			and Configuration.expectToggleValue("NoWind")
		then
			lastWindEffectTimestamp = os.clock()
			return hideEffect(effect)
		end

		if Configuration.expectToggleValue("NoWind") and os.clock() - lastWindEffectTimestamp <= 30.0 then
			if
				effect.Class == "Speed"
				or effect.Class == "NoJump"
				or effect.Class == "NoStun"
				or effect.Class == "NoSprint"
				or effect.Class == "NoRoll"
			then
				return hideEffect(effect)
			end
		end

		if
			(effect.Class == "NoJump" or effect.Class == "NoJumpAlt")
			and Configuration.expectToggleValue("AlwaysAllowJump")
		then
			return hideEffect(effect)
		end

		if
			effect.Class == "SpeedOverride"
			and effect.Value < 14
			and Configuration.expectToggleValue("NoSpeedDebuff")
		then
			return hideEffect(effect)
		end

		if effect.Class == "Speed" and effect.Value < 0 and Configuration.expectToggleValue("NoSpeedDebuff") then
			rawset(effect, "Value", 0)
		end

		if effect.Class == "Burning" and Configuration.expectToggleValue("AutoExtinguishFire") then
			local serverSlide = KeyHandling.getRemote("ServerSlide")
			local serverSlideStop = KeyHandling.getRemote("ServerSlideStop")

			if not serverSlide or not serverSlideStop then
				return Logger.warn("ServerSlide or ServerSlideStop not found.")
			end

			serverSlide:FireServer(true)
			serverSlideStop:FireServer()
		end

		if effect.Class == "Blinded" and Configuration.expectToggleValue("NoBlind") then
			return hideEffect(effect)
		end
	end

	---@param descendant Instance
	local function onWorkspaceDescendantAdded(descendant)
		if descendant:IsA("Model") and descendant.Name == "ResonanceDoor" then
			yunShulResonanceDoorMap:mark(descendant, "Parent")
		end

		if descendant:IsA("Model") and descendant.Name == "HiveGate" then
			hiveGateMap:mark(descendant, "Parent")
		end

		if descendant.Name == "WindPusher" and Configuration.expectToggleValue("NoWind") then
			debris:AddItem(descendant, 0.01)
		end

		if not descendant:IsA("BasePart") then
			return
		end

		if descendant.Name == "LifeField" then
			lightBarrierMap:mark(descendant, "CFrame")
		end

		local killInstance = descendant.Name == "KillBrick" or descendant.Name == "KillPlane"
		local killChasm = descendant.Name:match("Chasm")
		local superWall = descendant.Name == "SuperWall"
		local damagePart = descendant.Name == "Lava" and descendant:FindFirstChild("Hazardous") and not descendant:HasTag("DamageBrick")

		if killInstance or killChasm or superWall then
			killBricksMap:mark(descendant, "CFrame")
		end

		if descendant.Name == "DeepPassage_Yun" then
			yunShulBarrierMap:mark(descendant, "CFrame")
		end

		if damagePart then
			damageBricksMap:mark(descendant, "Parent")
		end
	end

	---@param descendant Instance
	local function onWorkspaceDescendantRemoving(descendant)
		killBricksMap:forget(descendant)
		lightBarrierMap:forget(descendant)
	end

	-- Effect removal rides the shared Game/EffectDispatcher as an "added" hook (registered in Removal.init) --
	-- one EffectReplicator connection for the whole script, deferred + identity-8 so updateRemovalEffects'
	-- FireServer (AutoExtinguishFire) can't dispatch from the parallel Actor phase.

	function Removal.init()
		removalMaid:add(
			workspaceDescendantAdded:connect("Removal_WorkspaceDescendantAdded", onWorkspaceDescendantAdded)
		)
		removalMaid:add(
			workspaceDescendantRemoving:connect("Removal_WorkspaceDescendantRemoving", onWorkspaceDescendantRemoving)
		)

		removalMaid:add(renderStepped:connect("Removal_RenderStepped", updateRemoval))
		removalMaid:add(renderStepped:connect("Removal_NoBlind", updateNoBlindFast))

		EffectDispatcher.hook("added", updateRemovalEffects)

		for _, descendant in pairs(workspace:GetDescendants()) do
			onWorkspaceDescendantAdded(descendant)
		end

		if not LPH_OBFUSCATED or shared.Luma.verbose then
			Logger.warn("Removal initialized.")
		end
	end

	function Removal.detach()
		removalMaid:clean()

		resetNoBlind()
		restoreNoAcidWater()

		Logger.warn("Removal detached.")
	end

---@param bar table
---@param value number
---@param maxValue number
local function setBar(bar, value, maxValue)
	local ratio = 0
	if maxValue and maxValue > 0 then
		ratio = math.clamp(value / maxValue, 0, 1)
	end
	bar.fill.Size = UDim2.fromScale(ratio, 1)
	bar.text.Text = string.format("%d / %d", math.floor(value + 0.5), math.floor((maxValue or 0) + 0.5))
end

-- Reads a child value-store (.Value / .MaxValue). Returns nil if the entity doesn't expose it.
---@param entity Instance
---@param storeName string
---@return number?, number?
local function readStore(entity, storeName)
	local store = entity:FindFirstChild(storeName)
	if not store then
		return nil
	end
	local ok, value, maxValue = pcall(function()
		return store.Value, store.MaxValue
	end)
	if ok and type(value) == "number" and type(maxValue) == "number" then
		return value, maxValue
	end
	return nil
end

local function update()

		if frame then
			frame.Visible = false
		end
		return
	end

	build()

	local best = Targeting.best()[1]
	if not best or not best.character or not best.humanoid or best.humanoid.Health <= 0 then
		frame.Visible = false
		return
	end

	frame.Visible = true

	local entity = best.character
	local player = players:GetPlayerFromCharacter(entity)
	local name = (player and (player:GetAttribute("CharacterName") or player.Name))
		or entity:GetAttribute("MOB_rich_name")
		or entity.Name
	local level = entity:GetAttribute("Level")
	nameLabel.Text = (level and string.format("%s  [Lv.%d]", tostring(name), level)) or tostring(name)
	distLabel.Text = string.format("%.0fst", best.du or 0)

	for _, spec in ipairs(BAR_SPECS) do
		local bar = bars[spec.id]
		if spec.id == "health" then
			bar.row.Visible = true
			setBar(bar, best.humanoid.Health, best.humanoid.MaxHealth)
		else
			local value, maxValue = readStore(entity, spec.store)
			if value then
				bar.row.Visible = true
				setBar(bar, value, maxValue)
			else
				bar.row.Visible = false
			end
		end
	end
end

__bundle_register("Features/Visuals/ProximityList", function(require, _LOADED, __bundle_register, __bundle_modules)
-- ProximityList: an on-screen list of nearby players (name + distance), ported from project-rain's
-- proximity_list. Draggable frame, updates ~0.15s, notifies on enter/leave, highlights voidwalkers.
-- Gated on the ProximityList toggle, with ProximityListRange + ProximityListNotify options.
local ProximityList = {}

---@module GUI.Library
local Library = require("GUI/Library")

---@module Utility.Logger
local Logger = require("Utility/Logger")

---@module Utility.Configuration
local Configuration = require("Utility/Configuration")

---@module Utility.CoreGuiManager
local CoreGuiManager = require("Utility/CoreGuiManager")

local players = game:GetService("Players")
local runService = game:GetService("RunService")

local screenGui, outer, listFrame
local labelMap = {} -- UserId -> TextLabel
local notified = {} -- UserId -> true while we've announced this player's presence (NOT tied to the label)
local lastInRange = {} -- UserId -> os.clock() last time actually in range (for departure debounce)
local active = false

-- A player must be out of range for this long before we announce they "left" -- stops the enter/leave
-- notifications from spamming when a still player briefly drops out of range (HRP streaming, health read,
-- or jitter right at the range boundary).
local PROXIMITY_GRACE = 2

local function localPlayer()
	return players.LocalPlayer
end

local function root()
	local char = localPlayer() and localPlayer().Character
	return char and char:FindFirstChild("HumanoidRootPart")
end

local function guiParent()
	return (gethui and gethui()) or game:GetService("CoreGui")
end

local function mk(class, props, parent)
	local instance = Instance.new(class)
	for key, value in pairs(props) do
		instance[key] = value
	end
	instance.Parent = parent
	return instance
end

---Studs -> "123m" / "1.2km".
local function formatDistance(d)
	if d >= 1000 then
		return string.format("%.1fkm", d / 1000)
	end
	return string.format("%im", d)
end

---Voidwalker check (PR's): their Backpack holds the Voideye / Grasp of Eylis talent. NOTE: other players'
---Backpacks usually aren't replicated to us, so this rarely triggers for remote players -- kept for parity.
local function isVoidwalker(plr)
	local bp = plr:FindFirstChildOfClass("Backpack")
	if not bp then
		return false
	end
	return bp:FindFirstChild("Talent:Voideye") ~= nil or bp:FindFirstChild("Talent:Grasp of Eylis") ~= nil
end

local function build()

		return

	screenGui.ResetOnSpawn = false
	screenGui.ZIndexBehavior = Enum.ZIndexBehavior.Global
	screenGui.DisplayOrder = 9998

---Players within `range` of us (alive), sorted nearest-first.
local function getInRange(range)
	local inRange = {}
	local r = root()
	if not r then
		return inRange
	end
	local lp = localPlayer()
	for _, plr in ipairs(players:GetPlayers()) do
		if plr == lp then
			continue
		end
		local char = plr.Character
		local hrp = char and char:FindFirstChild("HumanoidRootPart")
		local hum = char and char:FindFirstChildOfClass("Humanoid")
		if hrp and hum and hum.Health > 0 then
			local dist = (r.Position - hrp.Position).Magnitude
			if dist <= range then
				table.insert(inRange, { player = plr, distance = dist })
			end
		end
	end
	table.sort(inRange, function(a, b)
		return a.distance < b.distance
	end)
	return inRange
end

local function addLabel(voidwalker)
	local label = mk("TextLabel", {
		BackgroundTransparency = 1,
		Size = UDim2.new(1, 0, 0, 18),
		FontFace = Library.Font,
		TextSize = 14,
		TextColor3 = voidwalker and Library.AccentColor or (Library.FontColor or Color3.new(1, 1, 1)),
		Text = "",
		ZIndex = 202,
	}, listFrame)
	Library:AddToRegistry(label, { TextColor3 = voidwalker and "AccentColor" or "FontColor" })
	return label
end

local function clearAll()
	for key, label in pairs(labelMap) do
		pcall(function()
			label:Destroy()
		end)
		labelMap[key] = nil
	end
	for key in pairs(notified) do
		notified[key] = nil
	end
	for key in pairs(lastInRange) do
		lastInRange[key] = nil
	end
end

local function sync(inRange, notify)
	local seen = {}
	local order = 0
	local now = os.clock()
	for _, data in ipairs(inRange) do
		local plr = data.player
		local key = plr.UserId
		seen[key] = true
		lastInRange[key] = now
		local name = plr:GetAttribute("CharacterName") or plr.Name
		local vw = isVoidwalker(plr)

		-- Announce entry ONCE per sustained presence -- keyed off `notified`, not the label, so a one-cycle
		-- drop-out (which destroys/recreates the label below) can't re-fire the notification.
		if notify and not notified[key] then
			Logger.notify(
				"%s [%s] entered your proximity range.%s",
				tostring(plr:GetAttribute("CharacterName") or ""),
				plr.Name,
				vw and " (voidwalker)" or ""
			)
		end
		notified[key] = true

		local label = labelMap[key]
		if not label or not label.Parent then
			label = addLabel(vw)
			labelMap[key] = label
		end
		label.Text = name .. " (" .. formatDistance(data.distance) .. ")"
		label.LayoutOrder = order
		order = order + 1
	end

	-- Labels track the instantaneous range (remove when not seen this cycle).
	for key, label in pairs(labelMap) do
		if not seen[key] then
			pcall(function()
				label:Destroy()
			end)
			labelMap[key] = nil
		end
	end

	-- Departures are debounced: only announce "left" once a player has been out of range for the full
	-- grace window. A brief flicker keeps `notified` set, so returning doesn't re-announce an entry.
	for key in pairs(notified) do
		if not seen[key] and (now - (lastInRange[key] or 0)) > PROXIMITY_GRACE then
			if notify then
				local plr = players:GetPlayerByUserId(key)
				Logger.notify(
					"%s left your proximity range.",
					plr and ((plr:GetAttribute("CharacterName") or "") .. " [" .. plr.Name .. "]") or "A player"
				)
			end
			notified[key] = nil
			lastInRange[key] = nil
		end
	end

	-- Manual sizing (AutomaticSize was unreliable here): grow the box to fit the title + one row per player.
	local count = 0
	for _ in pairs(labelMap) do
		count = count + 1
	end
	if listFrame then
		listFrame.Size = UDim2.new(1, 0, 0, count * 18)
	end
	if outer then
		outer.Size = UDim2.new(0, 190, 0, 22 + count * 18)
	end
end

function ProximityList.init()
	if active then
		return
	end
	active = true

	-- Build the window ONCE, up front (same as the automation overlays' buildOverlay), then drive it from a
	-- task.spawn loop -- not a lazy build inside a render connection.
	build()
	if screenGui then
		screenGui.Enabled = false
	end

	task.spawn(function()
		while active do
			task.wait(0.15)
			pcall(function()
				if not active then
					return
				end
				if not Configuration.expectToggleValue("ProximityList") then
					if screenGui and screenGui.Enabled then
						screenGui.Enabled = false
						clearAll()
					end
					return
				end

				if not screenGui then
					build()
				end

				local range = Configuration.expectOptionValue("ProximityListRange") or 1000
				local notify = Configuration.expectToggleValue("ProximityListNotify")
				sync(getInRange(range), notify)

				-- Show the box whenever the feature is on (just the "Proximity" title when nobody's in range).
				screenGui.Enabled = true
			end)
		end
	end)
end

function ProximityList.detach()
	active = false
	clearAll()
	if screenGui then
		pcall(function()
			screenGui:Destroy()
		end)
		screenGui = nil
	end
	outer, listFrame = nil, nil
end

return ProximityList
end)

local stripTags = nil
local correctBuilderName = nil
local mantraDisplayToInternal = nil

local cachedTalentLookup = {}
local cachedMantraLookup = {}
local lastBdata = nil

local updateCardFrames = LPH_NO_VIRTUALIZE(function()
	local drinfo = Visuals.drinfo
	if not drinfo then
		return
	end

	---@type BuilderData
	local bdata = Visuals.bdata
	if not bdata then
		return
	end

	if lastBdata ~= bdata then
		lastBdata = bdata
		table.clear(cachedTalentLookup)
		table.clear(cachedMantraLookup)

		for _, talent in next, bdata.talents do
			local talentClean = stripTags(talent)
			cachedTalentLookup[talentClean] = talent
			cachedTalentLookup[talent] = talent
		end

		for _, mantra in next, bdata.mantras do
			local mantraClean = correctBuilderName(stripTags(mantra))
			cachedMantraLookup[mantraClean] = mantra
			cachedMantraLookup[mantra] = mantra
		end
	end

	local talentLookup = cachedTalentLookup
	local mantraLookup = cachedMantraLookup

	for frame in next, cardFrames do
		local title = frame:FindFirstChild("Title")
		if not title then
			continue
		end

		local border = frame:FindFirstChild("Border")
		if not border then
			continue
		end

		local trimmedName = string.gsub(title.Text, "^%s*(.-)%s*$", "%1")
		local fullTalentName = talentLookup[trimmedName] or mantraLookup[trimmedName]
		local cardInData = fullTalentName ~= nil

		buildAssistanceMap:add(border, "ImageColor3", cardInData and Color3.new(0, 255, 0) or Color3.new(255, 0, 0))

		if
			cardInData
			and fullTalentName
			and bdata.ddata:possible(fullTalentName, bdata.pre)
			and not bdata.ddata:possible(fullTalentName, bdata.post)
		then
			buildAssistanceMap:add(border, "ImageColor3", Color3.new(255, 0, 255))
		end

		if cardInData then
			continue
		end

		local mappingMatch = {
			["Vitality"] = { expectedValue = bdata.traits["Vitality"], value = drinfo["TraitHealth"] },
			["Erudition"] = { expectedValue = bdata.traits["Erudition"], value = drinfo["TraitEther"] },
			["Proficiency"] = { expectedValue = bdata.traits["Proficiency"], value = drinfo["TraitWeaponDamage"] },
			["Songchant"] = { expectedValue = bdata.traits["Songchant"], value = drinfo["TraitMantraDamage"] },
		}

		for idx, data in next, mappingMatch do
			if not title.Text:match(idx) then
				continue
			end

			buildAssistanceMap:add(
				border,
				"ImageColor3",
				data.expectedValue ~= data.value and Color3.new(0, 255, 0) or Color3.new(255, 0, 0)
			)
		end
	end
end)

---@param jframe Frame
local updatePowerBackground = LPH_NO_VIRTUALIZE(function(jframe)
	local panels = jframe and jframe:FindFirstChild("Panels")
	local infoFrame = panels and panels:FindFirstChild("InfoFrame")
	local sheets = infoFrame and infoFrame:FindFirstChild("Sheets")
	local power = sheets and sheets:FindFirstChild("Power")
	local background = power and power:FindFirstChild("Background")
	if not background then
		return
	end

	local drinfo = Visuals.drinfo
	if not drinfo then
		return
	end

	---@type BuilderData
	local bdata = Visuals.bdata
	if not bdata then
		return
	end

	---@note: We do not care if there is no pre-shrine state at all.
	if not bdata:dshrine() then
		return
	end

	local color = Color3.fromRGB(245, 137, 5)
	local pstate = bdata:ipre(drinfo)

	if pstate == 0 then
		color = Color3.fromRGB(97, 4, 113)
	end

	if pstate == 1 then
		color = Color3.fromRGB(37, 129, 236)
	end

	buildAssistanceMap:add(background, "BackgroundColor3", color)
end)

---@param jframe Frame
local updateAttributeFrame = LPH_NO_VIRTUALIZE(function(jframe)
	local panels = jframe:FindFirstChild("Panels")
	local attributeFrame = panels and panels:FindFirstChild("AttributeFrame")
	local sheets = attributeFrame and attributeFrame:FindFirstChild("Sheets")
	if not sheets then
		return
	end

	local drinfo = Visuals.drinfo
	if not drinfo then
		return
	end

	---@type BuilderData
	local bdata = Visuals.bdata
	if not bdata then
		return
	end

	local attributes = bdata:attributes(drinfo)
	local mapping = {
		["Agility"] = attributes.base["Agility"],
		["Strength"] = attributes.base["Strength"],
		["Fortitude"] = attributes.base["Fortitude"],
		["Intelligence"] = attributes.base["Intelligence"],
		["Willpower"] = attributes.base["Willpower"],
		["Charisma"] = attributes.base["Charisma"],
		["ElementBlood"] = attributes.attunement["Bloodrend"],
		["ElementFire"] = attributes.attunement["Flamecharm"],
		["ElementIce"] = attributes.attunement["Frostdraw"],
		["ElementLightning"] = attributes.attunement["Thundercall"],
		["ElementWind"] = attributes.attunement["Galebreathe"],
		["ElementShadow"] = attributes.attunement["Shadowcast"],
		["ElementMetal"] = attributes.attunement["Ironsing"],
		["WeaponHeavy"] = attributes.weapon["Heavy Wep."],
		["WeaponMedium"] = attributes.weapon["Medium Wep."],
		["WeaponLight"] = attributes.weapon["Light Wep."],
	}

	for _, instance in next, sheets:GetDescendants() do
		if not instance:IsA("TextButton") then
			continue
		end

		local expectedValue = mapping[instance.Name]
		if not expectedValue then
			continue
		end

		local background = instance:FindFirstChild("Background")
		if not background then
			continue
		end

		local valueLabel = instance:FindFirstChild("Value")
		if not valueLabel then
			continue
		end

		local statInvested = tonumber(drinfo["Stat" .. instance.Name])
		if not statInvested then
			continue
		end

		local value = (bdata:ipre(drinfo) == 0 and statInvested <= 0) and statInvested or tonumber(valueLabel.Text)
		if not value then
			continue
		end

		local abbrevLabel = instance:FindFirstChild("Abbrev")
		if not abbrevLabel then
